```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna_think/R-mpl-brief/20_r1/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def find_column(columns, terms):
    """Return the first column whose name contains any of the supplied terms."""
    for column in columns:
        name = str(column).strip().lower()
        if any(term in name for term in terms):
            return column
    return None


sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df = df.dropna(how="all")
    if df.empty:
        continue

    x_col = find_column(df.columns, ["temperature", "temp", "온도"])
    weight_col = find_column(df.columns, ["weight", "mass", "중량", "무게"])
    dtg_col = find_column(df.columns, ["d.w", "dtg", "derivative", "미분"])

    # Fall back to the first three columns when headers do not identify TGA fields.
    if x_col is None and len(df.columns) >= 1:
        x_col = df.columns[0]
    if weight_col is None and len(df.columns) >= 3:
        weight_col = df.columns[2]
    if dtg_col is None and len(df.columns) >= 4:
        dtg_col = df.columns[3]
    if x_col is None or weight_col is None:
        continue

    x = pd.to_numeric(df[x_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    valid = x.notna() & weight.notna()
    x = x[valid].to_numpy()
    weight = weight[valid].to_numpy()

    if not len(x):
        continue

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax1.axvspan(300, 450, color="lightgray", alpha=0.55, zorder=0)
    weight_line, = ax1.plot(x, weight, color="#1f4e9c", label="Weight (%)", zorder=2)
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="#1f4e9c")
    ax1.tick_params(axis="y", labelcolor="#1f4e9c")
    ax1.set_title(str(sheet_name))

    # Mark the measured weight nearest 800 °C.
    residue_idx = int(np.argmin(np.abs(x - 800)))
    residue_x = x[residue_idx]
    residue_y = weight[residue_idx]
    ax1.plot(residue_x, residue_y, "o", color="#1f4e9c", zorder=4)
    ax1.annotate(
        f"Residue {residue_y:.1f}%",
        (residue_x, residue_y),
        xytext=(8, 9),
        textcoords="offset points",
        color="#1f4e9c",
    )

    handles = [weight_line]
    if dtg_col is not None:
        dtg_all = pd.to_numeric(df[dtg_col], errors="coerce")
        dtg = dtg_all[valid].to_numpy()
        dtg_valid = np.isfinite(dtg)
        if dtg_valid.any():
            x_dtg = x[dtg_valid]
            dtg = dtg[dtg_valid]
            dtg_up = -dtg
            ax2 = ax1.twinx()
            dtg_line, = ax2.plot(
                x_dtg, dtg_up, color="#c0392b", label="DTG", zorder=2
            )
            ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
            ax2.tick_params(axis="y", labelcolor="#c0392b")
            handles.append(dtg_line)

            peak_candidates = np.flatnonzero(x_dtg > 150)
            if len(peak_candidates):
                peak_idx = peak_candidates[np.argmax(dtg_up[peak_candidates])]
                peak_x = x_dtg[peak_idx]
                peak_y = dtg_up[peak_idx]
                ax2.plot(peak_x, peak_y, "o", color="#c0392b", zorder=4)
                ax2.annotate(
                    f"{peak_x:.0f} °C",
                    (peak_x, peak_y),
                    xytext=(8, 9),
                    textcoords="offset points",
                    color="#c0392b",
                )

    ax1.legend(handles, [line.get_label() for line in handles], loc="center left")
    ax1.grid(False)
    fig.tight_layout()

    safe_name = "".join(
        char if char.isalnum() or char in "-_." else "_" for char in str(sheet_name)
    )
    fig.savefig(OUT_DIR / f"{safe_name}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
