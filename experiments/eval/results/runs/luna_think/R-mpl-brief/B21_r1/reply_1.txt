```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna_think/R-mpl-brief/B21_r1/out")
SHEET_NAME = "Raw"

OUT_DIR.mkdir(parents=True, exist_ok=True)
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)

# Select TGA columns by their headers.
columns = {str(col).strip().lower(): col for col in df.columns}
temp_col = next(col for key, col in columns.items() if "temperature" in key)
weight_col = next(col for key, col in columns.items() if "weight" in key and "deriv" not in key)
dtg_col = next(col for key, col in columns.items() if "deriv" in key and "weight" in key)

data = df[[temp_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[temp_col].to_numpy()
weight = data[weight_col].to_numpy()
dtg = data[dtg_col].to_numpy()

fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx()

line1, = ax1.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
line2, = ax2.plot(temperature, -dtg, color="#c0392b", linewidth=1.5, label="-DTG (%/°C)")

# Residue: measured weight at the point nearest 800 °C.
residue_idx = int(np.argmin(np.abs(temperature - 800)))
ax1.plot(temperature[residue_idx], weight[residue_idx], marker="o", color="#1f4e9c")
ax1.annotate(
    f"Residue {weight[residue_idx]:.1f}%",
    (temperature[residue_idx], weight[residue_idx]),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
)

# DTG peak: largest upward peak above 150 °C.
eligible = np.flatnonzero(temperature > 150)
if eligible.size:
    peak_idx = int(eligible[np.argmax(-dtg[eligible])])
    ax2.plot(temperature[peak_idx], -dtg[peak_idx], marker="o", color="#c0392b")
    ax2.annotate(
        f"{temperature[peak_idx]:.0f} °C",
        (temperature[peak_idx], -dtg[peak_idx]),
        xytext=(8, 10),
        textcoords="offset points",
        color="#c0392b",
    )

ax1.set_title(SHEET_NAME)
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.grid(True, alpha=0.25)
ax1.legend([line1, line2], [line1.get_label(), line2.get_label()], loc="center left")

# Reverse the temperature axis so higher temperatures appear on the left.
ax1.set_xlim(float(np.nanmax(temperature)), float(np.nanmin(temperature)))

fig.tight_layout()
fig.savefig(OUT_DIR / "Raw.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
