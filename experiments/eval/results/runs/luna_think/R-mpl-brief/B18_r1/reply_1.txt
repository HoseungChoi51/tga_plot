```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna_think/R-mpl-brief/B18_r1/out")
SHEET_NAME = "Annealed"

OUT_DIR.mkdir(parents=True, exist_ok=True)
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)

# TGA column selection by header name.
def find_column(keywords):
    for col in df.columns:
        if any(word in str(col).lower() for word in keywords):
            return col
    raise KeyError(f"Could not find a column matching {keywords}")

temp_col = find_column(["temperature", "temp"])
weight_col = find_column(["weight"])
dtg_col = find_column(["deriv", "dtg"])

data = df[[temp_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
data = data.sort_values(temp_col)
temperature = data[temp_col].to_numpy(dtype=float)
weight = data[weight_col].to_numpy(dtype=float)
dtg = data[dtg_col].to_numpy(dtype=float)

# Locate the strongest mass-loss event above 150 °C.
eligible = np.flatnonzero(temperature > 150)
peak_idx = eligible[np.argmin(dtg[eligible])]
peak_temp = temperature[peak_idx]
peak_weight = weight[peak_idx]

# Tangent-intersection onset: tangent at the steepest mass-loss point
# intersects a pre-event horizontal baseline (median weight over the
# preceding 30–100 °C interval).
pre_event = (temperature >= peak_temp - 100) & (temperature <= peak_temp - 30)
if pre_event.sum() < 3:
    pre_event = (temperature < peak_temp) & (temperature >= peak_temp - 150)
if pre_event.sum() < 2:
    raise ValueError("Not enough pre-event data to estimate an extrapolated onset.")

baseline_weight = float(np.median(weight[pre_event]))
slope = float(dtg[peak_idx])
if not np.isfinite(slope) or abs(slope) < 1e-12:
    raise ValueError("Unable to estimate onset: DTG slope is near zero.")

onset_temp = peak_temp + (baseline_weight - peak_weight) / slope

# Residue at the measured point nearest 800 °C.
residue_idx = int(np.argmin(np.abs(temperature - 800)))
residue_temp = temperature[residue_idx]
residue = weight[residue_idx]

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

line1, = ax1.plot(temperature, weight, color="#1f4e9c", lw=1.8, label="Weight (%)")
line2, = ax2.plot(temperature, -dtg, color="#c0392b", lw=1.5, label="−DTG (%/°C)")

# Standard residue and DTG peak annotations.
ax1.plot(residue_temp, residue, "o", color="#1f4e9c", ms=6)
ax1.annotate(
    f"Residue {residue:.1f}%",
    xy=(residue_temp, residue),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
    fontsize=9,
)
ax2.plot(peak_temp, -dtg[peak_idx], "o", color="#c0392b", ms=6)
ax2.annotate(
    f"{peak_temp:.0f} °C",
    xy=(peak_temp, -dtg[peak_idx]),
    xytext=(8, 10),
    textcoords="offset points",
    color="#c0392b",
    fontsize=9,
)

# Extrapolated onset: mark the tangent/baseline intersection and show the tangent.
ax1.axvline(onset_temp, color="#238b45", ls="--", lw=1.2)
ax1.plot(onset_temp, baseline_weight, "D", color="#238b45", ms=5)
tangent_x = np.linspace(onset_temp, peak_temp, 100)
tangent_y = peak_weight + slope * (tangent_x - peak_temp)
ax1.plot(tangent_x, tangent_y, color="#238b45", ls=":", lw=1.4, label="Onset tangent")
ax1.annotate(
    f"Onset {onset_temp:.1f} °C",
    xy=(onset_temp, baseline_weight),
    xytext=(8, -22),
    textcoords="offset points",
    color="#238b45",
    fontsize=10,
    arrowprops=dict(arrowstyle="->", color="#238b45", lw=1),
)

ax1.set_title(SHEET_NAME)
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("−DTG (%/°C)")
ax1.grid(True, alpha=0.25)
ax1.legend([line1, line2, ax1.lines[-2]], ["Weight (%)", "−DTG (%/°C)", "Onset tangent"], loc="center left")

fig.tight_layout()
fig.savefig(OUT_DIR / "Annealed_onset.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
