```python
import json
import os
import numpy as np
import pandas as pd
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

df = pd.read_csv(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

def find_column(fragment):
    for col in df.columns:
        if fragment.lower() in col.lower():
            return col
    raise ValueError(f"Required column not found: {fragment}")

x_col = find_column("Temperature")
weight_col = find_column("Weight")
heat_col = find_column("Heat Flow")

x = pd.to_numeric(df[x_col], errors="coerce").to_numpy(dtype=float)
weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy(dtype=float)
heat = pd.to_numeric(df[heat_col], errors="coerce").to_numpy(dtype=float)

valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(heat)
x, weight, heat = x[valid], weight[valid], heat[valid]
order = np.argsort(x)
x, weight, heat = x[order], weight[order], heat[order]

# With the usual DSC convention, endothermic peaks are negative and
# exothermic peaks are positive. Select the two most prominent of the
# former and the most prominent of the latter, above 150 °C.
def selected_peaks(values, polarity, count):
    signal = polarity * values
    candidates, props = find_peaks(signal, prominence=0)
    eligible = candidates[x[candidates] > 150]
    if len(eligible) == 0:
        eligible = candidates
    if len(eligible) == 0:
        idx = int(np.argmax(signal))
        return [idx]
    prominences = props["prominences"]
    ranked = sorted(eligible, key=lambda i: prominences[np.where(candidates == i)[0][0]], reverse=True)
    return sorted(ranked[:count], key=lambda i: x[i])

endo_indices = selected_peaks(heat, -1, 2)
exo_indices = selected_peaks(heat, 1, 1)

peak_points = []
for idx in endo_indices:
    peak_points.append({
        "value": [float(x[idx]), float(heat[idx])],
        "name": f"Endothermic {x[idx]:.0f} °C",
        "itemStyle": {"color": "#1f77b4"},
        "label": {"show": True, "formatter": f"흡열 {x[idx]:.0f} °C", "position": "top"}
    })
for idx in exo_indices:
    peak_points.append({
        "value": [float(x[idx]), float(heat[idx])],
        "name": f"Exothermic {x[idx]:.0f} °C",
        "itemStyle": {"color": "#c0392b"},
        "label": {"show": True, "formatter": f"발열 {x[idx]:.0f} °C", "position": "top"}
    })

residue_idx = int(np.argmin(np.abs(x - 800)))
residue_point = {
    "value": [float(x[residue_idx]), float(weight[residue_idx])],
    "name": f"Residue {weight[residue_idx]:.1f}%",
    "itemStyle": {"color": "#1f4e9c"},
    "label": {
        "show": True,
        "formatter": f"Residue {weight[residue_idx]:.1f}%",
        "position": "top"
    }
}

option = {
    "animation": False,
    "title": {"text": "caox", "left": "center"},
    "tooltip": {"trigger": "axis"},
    "legend": {"data": ["Weight (%)", "Heat Flow (W/g)"], "left": "center", "top": 30},
    "grid": {"left": 75, "right": 75, "top": 85, "bottom": 65},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 38,
        "axisLine": {"show": True},
        "axisTick": {"show": True},
        "axisLabel": {"show": True}
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 50,
            "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
            "axisLabel": {"color": "#1f4e9c"},
            "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}}
        },
        {
            "type": "value",
            "name": "Heat Flow (W/g)",
            "nameLocation": "middle",
            "nameGap": 55,
            "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
            "axisLabel": {"color": "#c0392b"},
            "splitLine": {"show": False}
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "showSymbol": False,
            "smooth": False,
            "yAxisIndex": 0,
            "lineStyle": {"width": 2, "color": "#1f4e9c"},
            "data": [[float(a), float(b)] for a, b in zip(x, weight)],
            "markPoint": {"symbol": "circle", "symbolSize": 9, "data": [residue_point]}
        },
        {
            "name": "Heat Flow (W/g)",
            "type": "line",
            "showSymbol": False,
            "smooth": False,
            "yAxisIndex": 1,
            "lineStyle": {"width": 2, "color": "#c0392b"},
            "data": [[float(a), float(b)] for a, b in zip(x, heat)]
        },
        {
            "name": "Heat-flow peak temperatures",
            "type": "scatter",
            "yAxisIndex": 1,
            "symbol": "circle",
            "symbolSize": 9,
            "data": peak_points,
            "tooltip": {"show": True}
        }
    ]
}

result = {
    "figures": [{
        "name": "caox",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }]
}
print(json.dumps(result, ensure_ascii=False, allow_nan=False))
```
