def plot_loaded(loaded, out_path):
    fig, axes = plt.subplots(2, 1, figsize=(11, 8), sharex=True)
    n = len(loaded)
    cmap = plt.cm.tab10 if n <= 10 else plt.cm.tab20
    colors = cmap(np.linspace(0, 1, n)) if n > 1 else [cmap(0.0)]
    has_dtg = False
    for i, (label, df) in enumerate(loaded):
        temp_col = df.columns[0]
        weight_col = df.columns[1]
        color = colors[i]
        axes[0].plot(df[temp_col], df[weight_col], label=label, color=color, linewidth=1.5)
        if len(df.columns) > 2:
            dtg_col = df.columns[2]
            axes[1].plot(df[temp_col], df[dtg_col], label=label, color=color, linewidth=1.1)
            has_dtg = True
    axes[0].set_ylabel("무게 (%)")
    axes[1].set_ylabel("DTG (D.W, %/°C)")
    axes[1].set_xlabel("온도 (°C)")
    axes[0].set_title("TGA/DTG 측정 결과")
    axes[0].grid(True, linestyle="--", alpha=0.45)
    axes[1].grid(True, linestyle="--", alpha=0.45)
    axes[1].axhline(0, color="gray", linewidth=0.8)
    axes[0].legend(loc="best", frameon=True, fontsize=9)
    if has_dtg:
        axes[1].legend(loc="best", frameon=True, fontsize=9)
    ymin, ymax = axes[0].get_ylim()
    axes[0].set_ylim(0, max(105.0, ymax))
    fig.tight_layout()
    fig.savefig(out_path, dpi=300, bbox_inches="tight")
    plt.close(fig)
