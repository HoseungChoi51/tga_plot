```python
import os
import re

import numpy as np
import pandas as pd

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib import font_manager

from scipy.signal import find_peaks, savgol_filter

DATA_FILE = "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/CNT/Plot/eval/results/runs/t06/R-mpl-plain/03_r1/out"

SHEET = "1006R1-V acid"
X_MIN = 100.0
X_MAX = 800.0


def set_korean_font():
    candidates = [
        "Noto Sans CJK KR",
        "NanumGothic",
        "Malgun Gothic",
        "DejaVu Sans",
    ]
    available = {f.name for f in font_manager.fontManager.ttflist}
    chosen = next((name for name in candidates if name in available), "DejaVu Sans")

    plt.rcParams["font.family"] = chosen
    plt.rcParams["axes.unicode_minus"] = False


def read_sheet(path, sheet_name):
    xl = pd.ExcelFile(path)

    if sheet_name in xl.sheet_names:
        target = sheet_name
    else:
        target = next(
            (s for s in xl.sheet_names if "acid" in s.lower()),
            xl.sheet_names[0],
        )

    df = pd.read_excel(xl, sheet_name=target, header=0)
    return target, df


def choose_columns(df):
    cols = list(df.columns)

    def find(keys, fallback_pos):
        for col in cols:
            s = str(col).lower().replace(" ", "")
            if any(key in s for key in keys):
                return col
        return cols[fallback_pos] if len(cols) > fallback_pos else None

    x_col = find(["temperature", "temp", "x"], 1)
    y1_col = find(["weight", "tg", "mass"], 2)
    y2_col = find(["d.w", "dw", "dtg", "derivative", "der"], 3)

    return x_col, y1_col, y2_col


def smooth(y, window=25):
    y = np.asarray(y, dtype=float)
    n = len(y)

    if n < 3:
        return y

    window = min(window, n)
    if window % 2 == 0:
        window -= 1
    if window < 3:
        return y

    polyorder = min(2, window - 1)

    try:
        return savgol_filter(y, window_length=window, polyorder=polyorder)
    except Exception:
        return (
            pd.Series(y)
            .rolling(window=window, center=True, min_periods=1)
            .mean()
            .to_numpy()
        )


def refine_peak_indices(x, y, idxs, width_frac=0.01):
    n = len(x)
    half = max(3, int(n * width_frac))
    refined = []

    for i in idxs:
        lo = max(0, i - half)
        hi = min(n, i + half + 1)
        j = lo + int(np.argmin(y[lo:hi]))

        if j not in refined:
            refined.append(j)

    return np.array(refined, dtype=int)


def find_two_peaks(x, y):
    n = len(x)

    if n == 0:
        return np.array([], dtype=int)
    if n == 1:
        return np.array([0], dtype=int)
    if n == 2:
        return np.array([0, 1], dtype=int)

    y_smooth = smooth(y, window=25)
    inv = -y_smooth
    distance = max(5, n // 50)

    for prominence in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0]:
        idx, props = find_peaks(inv, prominence=prominence, distance=distance)

        if len(idx) >= 2:
            if "prominences" in props and len(props["prominences"]) == len(idx):
                order = np.argsort(props["prominences"])[::-1]
            else:
                order = np.argsort(-inv[idx])

            top = idx[order[:2]]
            refined = refine_peak_indices(
                x,
                y,
                top,
                width_frac=max(0.01, 1.5 * distance / n),
            )

            if len(refined) >= 2:
                return np.sort(refined)

    order = np.argsort(y)
    chosen = []
    min_dist = max(10, n // 100)

    for i in order:
        if all(abs(int(i) - int(c)) >= min_dist for c in chosen):
            chosen.append(int(i))

        if len(chosen) >= 2:
            break

    if len(chosen) < 2:
        chosen = list(range(min(2, n)))

    return np.sort(np.array(chosen, dtype=int))


def main():
    os.makedirs(OUT_DIR, exist_ok=True)
    set_korean_font()

    target, df = read_sheet(DATA_FILE, SHEET)

    x_col, y1_col, y2_col = choose_columns(df)

    if x_col is None or y1_col is None or y2_col is None:
        raise ValueError("Temperature / Weight(%) / D.W(%/C) 컬럼을 찾지 못했습니다.")

    df = df[[x_col, y1_col, y2_col]].copy()
    df.columns = ["x", "y1", "y2"]

    for col in df.columns:
        df[col] = pd.to_numeric(df[col], errors="coerce")

    df = df.dropna().sort_values("x").reset_index(drop=True)

    mask = (df["x"] >= X_MIN) & (df["x"] <= X_MAX)
    sub = df[mask]

    if sub.empty:
        sub = df

    x = sub["x"].to_numpy()
    y1 = sub["y1"].to_numpy()
    y2 = sub["y2"].to_numpy()

    peaks = find_two_peaks(x, y2)

    fig, ax = plt.subplots(figsize=(8, 5))

    line1, = ax.plot(
        x,
        y1,
        color="tab:blue",
        linewidth=1.8,
        label="Weight (%)",
    )

    ax.set_xlabel("Temperature (°C)")
    ax.set_ylabel("Weight (%)")
    ax.set_xlim(X_MIN, X_MAX)
    ax.grid(True, linestyle=":", alpha=0.5)

    ax2 = ax.twinx()

    line2, = ax2.plot(
        x,
        y2,
        color="tab:red",
        linewidth=1.5,
        label="D.W (%/°C)",
    )

    ax2.set_ylabel("D.W (%/°C)")
    ax2.axhline(0, color="gray", linewidth=0.8, alpha=0.6)

    marker, = ax2.plot(
        [],
        [],
        "o",
        color="crimson",
        markersize=6,
        label="DTG peak",
    )

    if len(peaks) > 0:
        for i in peaks:
            ax.axvline(x[i], color="gray", linestyle=":", linewidth=1)
            ax2.plot(x[i], y2[i], "o", color="crimson", markersize=6)

        xmin, xmax = ax.get_xlim()
        ymin, ymax = ax2.get_ylim()

        dx = (xmax - xmin) * 0.025
        dy = (ymax - ymin) * 0.12

        for k, i in enumerate(peaks, start=1):
            tx = x[i] + dx
            ha = "left"

            if tx > xmax - (xmax - xmin) * 0.18:
                tx = x[i] - dx
                ha = "right"

            ty = y2[i] + dy

            if ty > ymax:
                ty = y2[i] - dy

            va = "bottom" if ty > y2[i] else "top"

            ax2.annotate(
                f"Peak {k}\n{x[i]:.1f} °C\n{y2[i]:.3f}",
                xy=(x[i], y2[i]),
                xytext=(tx, ty),
                ha=ha,
                va=va,
                fontsize=8,
                color="black",
                bbox=dict(
                    boxstyle="round,pad=0.25",
                    facecolor="white",
                    edgecolor="crimson",
                    alpha=0.85,
                ),
                arrowprops=dict(
                    arrowstyle="->",
                    color="crimson",
                    linewidth=1,
                ),
            )

    ax.set_title(f"{target} TGA/DTG")

    ax.legend(
        handles=[line1, line2, marker],
        loc="best",
        framealpha=0.9,
    )

    fig.tight_layout()

    safe_name = re.sub(r"[^A-Za-z0-9_.-]+", "_", target)
    out_path = os.path.join(OUT_DIR, f"{safe_name}.png")

    fig.savefig(out_path, dpi=300)
    plt.close(fig)


if __name__ == "__main__":
    main()
```
