def find_col(df, candidates):
    for c in df.columns:
        if str(c).strip() in candidates:
            return c
    for c in df.columns:
        cs = str(c).strip().lower()
        for cand in candidates:
            if str(cand).strip().lower() in cs:
                return c
    return None
