def compute_onset(x,w,d):
    n=len(x)
    if n < 10:
       return None
    w_s = smooth(w, 51)
    d_s = smooth(d, 51)
    # replace nan? 
    if not np.all(np.isfinite(w_s)):
        w_s = np.nan_to_num(w_s, nan=np.nanmean(w) if np.any(np.isfinite(w)) else 0.0)
    if not np.all(np.isfinite(d_s)):
        d_s = np.nan_to_num(d_s, nan=0.0)
    mask = x > 150.0
    if np.any(mask):
        idxs = np.where(mask)[0]
        # use raw d if finite; fallback d_s
        d_for_peak = np.where(np.isfinite(d), d, d_s)
        peak_idx = int(idxs[np.argmin(d_for_peak[idxs])])
    else:
        d_for_peak = np.where(np.isfinite(d), d, d_s)
        peak_idx = int(np.argmin(d_for_peak))
    peak_slope = float(d_s[peak_idx]) if np.isfinite(d_s[peak_idx]) else float(d[peak_idx])
    if not np.isfinite(peak_slope) or peak_slope >= 0:
        # most negative smoothed derivative
        finite = d_s[np.isfinite(d_s)]
        if len(finite)==0:
            return None
        peak_slope = float(np.min(finite))
    if not np.isfinite(peak_slope) or peak_slope >= 0:
        return None
    thresh = 0.1 * abs(peak_slope)
    start_idx = 0
    for i in range(peak_idx - 1, -1, -1):
        if d_s[i] >= -thresh:
            start_idx = i
            break
    b_end = max(0, start_idx)
    b_start = max(0, b_end - 150)
    if b_end > b_start:
        baseline = float(np.median(w_s[b_start:b_end]))
    else:
        m = max(1, int(0.1*n))
        baseline = float(np.median(w_s[:m]))
    if not np.isfinite(baseline):
        baseline = float(w[0])
    w_peak = float(w[peak_idx])
    if baseline <= w_peak + 0.5:
        pre = w_s[:max(1, peak_idx)]
        if len(pre) > 0:
            baseline = float(np.max(pre))
    onset = None
    if peak_slope < 0 and baseline > w_peak:
        onset = float(x[peak_idx] + (baseline - w_peak) / peak_slope)
    if onset is None or not np.isfinite(onset) or onset >= x[peak_idx] or onset < x[0]:
        target = baseline - 5.0
        below = np.where(w_s[:peak_idx] < target)[0]
        if len(below) > 0:
            onset = float(x[below[0]])
        else:
            onset = float(x[max(0, peak_idx - max(5, n//50))])
    onset = float(np.clip(onset, x[0], x[peak_idx]))
    return {"onset": onset, "baseline": float(baseline), "peak_idx": peak_idx, "slope": peak_slope}
