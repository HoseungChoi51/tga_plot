def find_col(df, patterns, exclude=None):
    for col in df.columns:
        c = str(col).lower()
        if exclude:
            if any(e in c for e in exclude):
                continue
        if any(p in c for p in patterns):
            return col
    return None

def get_numeric_columns(df):
    cols=[]
    for col in df.columns:
        s = pd.to_numeric(df[col], errors="coerce")
        if s.notna().sum() > 1:
            cols.append(col)
    return cols
