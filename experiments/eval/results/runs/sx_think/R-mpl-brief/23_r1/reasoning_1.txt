def _corr(a,b):
   if len(a)<2: return 0.0
   a=np.asarray(a,float); b=np.asarray(b,float)
   m=np.isfinite(a)&np.isfinite(b)
   if m.sum()<2 or np.std(a[m])==0 or np.std(b[m])==0: return 0.0
   return float(np.corrcoef(a[m], b[m])[0,1])

def _valid_xy(x,y1):
   x=np.asarray(x,float); y1=np.asarray(y1,float)
   m=np.isfinite(x)&np.isfinite(y1)
   if m.sum()<10: return False
   x=x[m]; y1=y1[m]
   if x.max() <= x.min(): return False
   if np.nanmax(y1) <= 0: return False
   corr=_corr(x, np.arange(len(x)))
   if corr < 0.8: return False
   return True

def _valid_triplet(x,y1,y2):
   if not _valid_xy(x,y1): return False
   y2=np.asarray(y2,float); x=np.asarray(x,float)
   m=np.isfinite(x)&np.isfinite(y2)
   if m.sum()<10: return False
   y2=y2[m]
   if not np.any(np.isfinite(y2)): return False
   if np.nanmax(np.abs(y2)) > 1000: return False # maybe not derivative
   return True

   def make(i,j,k=None): ...
   def get(i): return df[cols[i]].to_numpy(dtype=float)
