def _find_col(df, keywords, exclude=None):
    for col in df.columns:
        s = str(col).lower()
        if exclude and any(e in s for e in exclude):
            continue
        if any(k in s for k in keywords):
            return col
    return None

def _select_columns(df):
    x = _find_col(df, ['temperature', 'temp', '°c'])

    y1 = _find_col(df, ['weight', 'wt'], exclude=['deriv', 'dtg', 'derivative'])

        y1 = _find_col(df, ['%'])
