def compute_onset(temp, weight, dtg, peak_idx):
    peak_T = float(temp[peak_idx])
    peak_w = float(weight[peak_idx])
    peak_dtg = float(dtg[peak_idx])
    m = peak_dtg
    if not np.isfinite(m) or m >= 0:
       span = float(temp.max()-temp.min()) if len(temp)>1 else 1.0
       win = np.abs(temp-peak_T) <= max(5.0, span*0.02)
       if win.sum() >= 3:
          try:
             m_fit = float(np.polyfit(temp[win], weight[win], 1)[0])
             if np.isfinite(m_fit) and m_fit < 0: m = m_fit
          except Exception: pass
    if not np.isfinite(m) or m >= 0:
       pre = temp < peak_T
       if pre.sum() >= 3:
          try:
             m_fit = float(np.polyfit(temp[pre], weight[pre], 1)[0])
             if np.isfinite(m_fit) and m_fit < 0: m = m_fit
          except Exception: pass
    if not np.isfinite(m) or m >= 0:
       m = -1e-6
    abs_peak = abs(peak_dtg) if np.isfinite(peak_dtg) and peak_dtg != 0 else abs(m)
    pre = temp < peak_T
    baseline = np.nan
    for win in (100.0, 50.0, 200.0, 300.0, 500.0):
        cand = pre & (temp >= peak_T - win)
        if cand.sum() >= 5:
            stable = cand & (np.abs(dtg) <= max(1e-6, 0.1*abs_peak))
            if stable.sum() >= 5:
                baseline = float(np.median(weight[stable]))
            else:
                # use upper part of pre-peak weights to approximate plateau
                baseline = float(np.percentile(weight[cand], 75))
            break
    if not np.isfinite(baseline):
        init = temp <= min(150.0, peak_T)
        if init.sum() >= 5:
            baseline = float(np.median(weight[init]))
        elif len(weight) > 0:
            baseline = float(np.median(weight[:max(5, len(temp)//10)]))
        else:
            baseline = peak_w + 1.0
    if not np.isfinite(baseline) or baseline <= peak_w:
        init = temp <= min(150.0, peak_T)
        if init.sum() > 0:
            baseline = max(float(np.median(weight[init])), peak_w + 0.1)
        elif len(weight) > 0:
            baseline = max(float(np.mean(weight[:max(5, len(temp)//10)])), peak_w + 0.1)
        else:
            baseline = peak_w + 1.0
    onset = peak_T + (baseline - peak_w) / m
    if not np.isfinite(onset) or onset > peak_T or onset < float(temp.min()) or onset > float(temp.max()):
        pre = temp < peak_T
        if pre.any():
            drop = abs(baseline - peak_w)
            thresh = baseline - max(0.5, 0.01*drop)
            below = pre & (weight <= thresh)
            if below.any():
                onset = float(temp[below][0])
            else:
                span = peak_T - float(temp.min())
                onset = peak_T - min(20.0, max(5.0, span*0.05))
        else:
            onset = peak_T
    onset = float(np.clip(onset, float(temp.min()), peak_T))
    return onset, float(baseline), float(m)
