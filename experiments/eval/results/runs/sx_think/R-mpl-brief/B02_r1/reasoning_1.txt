def pad_indices(indices, x, y, sign, n, exclude=None):
    indices = [int(i) for i in indices]
    exclude = set(exclude or [])
    if len(indices) >= n:
       return indices[:n]
    candidates = find_peaks_top(x, y, sign, max(n*2, n))[0]
    for idx in candidates:
       idx=int(idx)
       if idx not in indices and idx not in exclude:
          indices.append(idx)
       if len(indices) >= n: break
    if len(indices) < n:
       ys = sign * y
       order = np.argsort(ys)[::-1]
       for idx in order:
          idx=int(idx)
          if idx not in indices and idx not in exclude and np.isfinite(y[idx]):
             indices.append(idx)
          if len(indices) >= n: break
    return indices[:n]
