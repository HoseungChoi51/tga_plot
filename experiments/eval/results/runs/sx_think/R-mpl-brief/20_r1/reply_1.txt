EXCEL_EXTS = {".xlsx", ".xls", ".xlsm"}


def read_raw(path, sheet=None):
    ext = Path(path).suffix.lower()
    if ext in EXCEL_EXTS:
        return pd.read_excel(path, sheet_name=sheet, header=None)
    return pd.read_csv(path, header=None)


def get_sheets(path):
    ext = Path(path).suffix.lower()
    if ext in EXCEL_EXTS:
        return pd.ExcelFile(path).sheet_names
    return [Path(path).stem]


def find_header_row(raw):
    n = min(10, len(raw))
    for i in range(n):
        if i + 1 >= len(raw):
            break

        row = raw.iloc[i]
        if row.notna().sum() < 2:
            continue

        nxt = raw.iloc[i + 1]
        numeric_count = 0

        for j in range(len(row)):
            if pd.isna(row[j]) or pd.isna(nxt[j]):
                continue
            try:
                float(nxt[j])
                numeric_count += 1
            except (TypeError, ValueError):
                pass

        if numeric_count >= 2:
            return i

    return 0


def load_sheet(path, sheet):
    raw = read_raw(path, sheet)
    if raw is None or raw.empty:
        return pd.DataFrame()

    h = find_header_row(raw)
    header = raw.iloc[h].astype(str).tolist()
    df = raw.iloc[h + 1:].copy()
    df.columns = header

    for c in df.columns:
        df[c] = pd.to_numeric(df[c], errors="coerce")

    df = df.dropna(how="all")
    return df.reset_index(drop=True)


def find_tga_columns(df):
    if df.empty:
        return None, None, None

    cols = list(df.columns)
    lower = [str(c).lower() for c in cols]

    x_idx = y1_idx = y2_idx = None

    for i, name in enumerate(lower):
        if any(k in name for k in ("dtg", "d.w", "d w", "derivative", "dw")):
            y2_idx = i
            break

    for i, name in enumerate(lower):
        if "temp" in name:
            x_idx = i
            break

    for i, name in enumerate(lower):
        if "weight" in name and i != y2_idx:
            y1_idx = i
            break

    numeric_cols = [i for i in range(df.shape[1]) if df.iloc[:, i].notna().any()]

    def col_range(i):
        s = df.iloc[:, i].dropna()
        return float(s.max() - s.min()) if not s.empty else -np.inf

    def weight_score(i):
        s = df.iloc[:, i].dropna()
        if s.empty:
            return (np.inf, np.inf)
        good = 0 if (50 <= s.max() <= 110 and s.min() >= 0) else 1
        return (good, abs(float(s.max()) - 100.0))

    def derivative_score(i):
        s = df.iloc[:, i].dropna()
        if s.empty:
            return (np.inf, np.inf)
        has_neg = 0 if s.min() < 0 else 1
        return (has_neg, float(s.abs().max()))

    chosen = []
    for pos, idx in enumerate((x_idx, y1_idx, y2_idx)):
        if idx is None or idx in chosen:
            candidates = [i for i in numeric_cols if i not in chosen]
            if not candidates:
                continue

            if pos == 0:
                idx = max(candidates, key=col_range)
            elif pos == 1:
                idx = min(candidates, key=weight_score)
            else:
                idx = min(candidates, key=derivative_score)

        chosen.append(idx)

    if len(chosen) < 3:
        return None, None, None

    return chosen[0], chosen[1], chosen[2]


def safe_filename(name, fallback):
    s = re.sub(r"[^\w\-]+", "_", str(name), flags=re.UNICODE).strip("._")
    return s or fallback
