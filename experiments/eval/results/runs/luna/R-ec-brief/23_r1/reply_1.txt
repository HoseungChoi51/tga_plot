def read_numeric_data(path):
    # Instrument exports commonly have a metadata preamble and tab-, comma-,
    # or whitespace-separated numeric data.
    lines = Path(path).read_text(encoding="utf-8", errors="replace").splitlines()
    rows = []
    for line in lines:
        line = line.strip()
        if not line:
            continue
        parts = re.split(r"[,\t; ]+", line)
        if len(parts) < 3:
            continue
        try:
            vals = [float(x) for x in parts]
        except ValueError:
            continue
        rows.append(vals)

    if not rows:
        raise ValueError(f"No numeric rows found in {path}")

    ncols = min(map(len, rows))
    data = np.asarray([row[:ncols] for row in rows if len(row) >= ncols], dtype=float)
    if data.shape[1] < 4:
        raise ValueError("Expected at least four numeric columns (time, temperature, weight, DTG).")
    return data
