def parse_numeric_line(line):
    """Parse a delimited row, tolerating decimal commas and instrument metadata."""
    text = line.strip().lstrip("\ufeff")
    if not text:
        return None

    # First try common explicit separators. Decimal commas are only supported
    # when fields are separated by tabs or semicolons.
    for delimiter in ("\t", ";", "|"):
        if delimiter in text:
            fields = next(csv.reader([text], delimiter=delimiter))
            values = []
            for field in fields:
                field = field.strip().strip('"')
                if not field:
                    continue
                try:
                    values.append(float(field.replace(",", ".")))
                except ValueError:
                    return None
            return values if len(values) >= 3 else None

    # Comma-separated exports with dot decimals.
    if "," in text:
        fields = next(csv.reader([text], delimiter=","))
        values = []
        for field in fields:
            field = field.strip().strip('"')
            if not field:
                continue
            try:
                values.append(float(field))
            except ValueError:
                return None
        if len(values) >= 3:
            return values

    # Whitespace-separated numeric records; permits scientific notation.
    tokens = re.findall(r"[-+]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][-+]?\d+)?", text)
    if len(tokens) >= 3:
        # Avoid mistaking metadata containing numbers for tabular data:
        # numeric row must consist almost entirely of numeric tokens.
        residue = re.sub(
            r"[-+]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][-+]?\d+)?", "", text
        )
        if not residue.strip(" \t,;|"):
            return [float(token) for token in tokens]
    return None


def read_numeric_data(path):
    lines = Path(path).read_text(encoding="utf-8", errors="replace").splitlines()
    rows = []
    for line in lines:
        row = parse_numeric_line(line)
        if row is not None:
            rows.append(row)

    if not rows:
        # Fallback: locate the numeric table block using pandas' delimiter
        # inference, skipping arbitrary instrument header lines.
        import pandas as pd

        for sep in (r"\s+", "\t", ";", ","):
            try:
                frame = pd.read_csv(
                    path, sep=sep, engine="python", header=None, comment="#",
                    on_bad_lines="skip"
                )
                frame = frame.apply(pd.to_numeric, errors="coerce")
                frame = frame.dropna(axis=0, how="all").dropna(axis=1, how="all")
                if frame.shape[0] and frame.shape[1] >= 3:
                    rows = frame.to_numpy(dtype=float).tolist()
                    break
            except Exception:
                continue

    if not rows:
        raise ValueError(f"No numeric data table could be read from {path}")

    width = min(len(row) for row in rows)
    values = np.asarray([row[:width] for row in rows], dtype=float)
    if values.shape[1] < 4:
        raise ValueError(
            f"Expected at least 4 numeric columns; found {values.shape[1]}"
        )
    return values
