```python
import json
import os
import numpy as np
import pandas as pd
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

df = pd.read_csv(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

def find_column(hint):
    for col in df.columns:
        if hint.lower() in col.lower():
            return col
    raise ValueError(f"Missing column containing {hint!r}")

x_col = find_column("Temperature")
weight_col = find_column("Weight")
heat_col = find_column("Heat Flow")

x = pd.to_numeric(df[x_col], errors="coerce").to_numpy(dtype=float)
weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy(dtype=float)
heat = pd.to_numeric(df[heat_col], errors="coerce").to_numpy(dtype=float)
valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(heat)
x, weight, heat = x[valid], weight[valid], heat[valid]

order = np.argsort(x)
x, weight, heat = x[order], weight[order], heat[order]

# Select the two strongest endothermic peaks and the strongest exothermic peak.
# In conventional heat-flow data, endothermic peaks point downward and
# exothermic peaks point upward.
endothermic, _ = find_peaks(-heat, distance=max(1, len(heat) // 100))
exothermic, _ = find_peaks(heat, distance=max(1, len(heat) // 100))

if len(endothermic) < 2:
    endothermic = np.argsort(heat)[:2]
endothermic = sorted(endothermic, key=lambda i: heat[i])[:2]
exothermic_idx = int(exothermic[np.argmax(heat[exothermic])]) if len(exothermic) else int(np.argmax(heat))

peak_indices = endothermic + [exothermic_idx]
peak_points = [{"coord": [float(x[i]), float(heat[i])]} for i in peak_indices]
peak_labels = [
    {"coord": [float(x[i]), float(heat[i])],
     "value": f"{x[i]:.0f} °C"}
    for i in peak_indices
]

option = {
    "animation": False,
    "title": {
        "text": "caox",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR"}
    },
    "tooltip": {"trigger": "axis"},
    "legend": {"data": ["Weight", "Heat Flow"], "left": "center"},
    "grid": {"left": 75, "right": 75, "top": 75, "bottom": 65},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 38,
        "axisLine": {"show": True},
        "axisTick": {"show": True},
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"}
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 55,
            "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
            "axisLabel": {"color": "#1f4e9c"},
            "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}}
        },
        {
            "type": "value",
            "name": "Heat Flow (W/g)",
            "nameLocation": "middle",
            "nameGap": 55,
            "position": "right",
            "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
            "axisLabel": {"color": "#c0392b"},
            "splitLine": {"show": False}
        }
    ],
    "series": [
        {
            "name": "Weight",
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 0,
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "data": [[float(a), float(b)] for a, b in zip(x, weight)]
        },
        {
            "name": "Heat Flow",
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 1,
            "lineStyle": {"color": "#c0392b", "width": 2},
            "data": [[float(a), float(b)] for a, b in zip(x, heat)],
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 9,
                "itemStyle": {"color": "#c0392b", "borderColor": "#ffffff", "borderWidth": 1},
                "label": {
                    "show": True,
                    "formatter": "{c}",
                    "position": "top",
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 12
                },
                "data": peak_labels
            }
        }
    ]
}

result = {
    "figures": [{
        "name": "caox",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }]
}
print(json.dumps(result, ensure_ascii=False, allow_nan=False))
```
