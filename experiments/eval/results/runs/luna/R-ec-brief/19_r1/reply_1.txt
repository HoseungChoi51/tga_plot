```python
import json
import re
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

TEXT_COLOR = "#808080"
TEXT_SIZE = 13


def clean_name(value):
    name = re.sub(r"[^\w가-힣-]+", "_", str(value).strip()).strip("_")
    return name or "plot"


def as_float_array(series):
    return pd.to_numeric(series, errors="coerce").to_numpy(dtype=float)


def make_option(sheet_name, temperature, weight, dtg):
    valid = np.isfinite(temperature) & np.isfinite(weight) & np.isfinite(dtg)
    temperature, weight, dtg = temperature[valid], weight[valid], dtg[valid]
    if not len(temperature):
        return None

    # DTG is negative during mass loss, so negate it to display peaks upward.
    dtg_up = -dtg
    residue_idx = int(np.argmin(np.abs(temperature - 800.0)))
    peak_candidates = np.flatnonzero((temperature > 150.0) & np.isfinite(dtg_up))
    peak_idx = int(peak_candidates[np.argmax(dtg_up[peak_candidates])]) if len(peak_candidates) else None

    series = [
        {
            "name": "Weight",
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 0,
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "data": [[float(x), float(y)] for x, y in zip(temperature, weight)],
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#1f4e9c"},
                "label": {
                    "show": True,
                    "position": "right",
                    "color": TEXT_COLOR,
                    "fontSize": TEXT_SIZE,
                    "formatter": f"Residue {weight[residue_idx]:.1f}%",
                },
                "data": [{"coord": [float(temperature[residue_idx]), float(weight[residue_idx])]}],
            },
        },
        {
            "name": "DTG",
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 1,
            "lineStyle": {"color": "#c0392b", "width": 2},
            "data": [[float(x), float(y)] for x, y in zip(temperature, dtg_up)],
        },
    ]

    if peak_idx is not None:
        series[1]["markPoint"] = {
            "symbol": "circle",
            "symbolSize": 8,
            "itemStyle": {"color": "#c0392b"},
            "label": {
                "show": True,
                "position": "top",
                "color": TEXT_COLOR,
                "fontSize": TEXT_SIZE,
                "formatter": f"{temperature[peak_idx]:.0f} °C",
            },
            "data": [{"coord": [float(temperature[peak_idx]), float(dtg_up[peak_idx])]}],
        }

    return {
        "title": {
            "text": str(sheet_name),
            "left": "center",
            "textStyle": {"color": TEXT_COLOR, "fontSize": TEXT_SIZE},
        },
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": TEXT_SIZE,
            "color": TEXT_COLOR,
        },
        "tooltip": {"trigger": "axis"},
        "legend": {
            "data": ["Weight", "DTG"],
            "left": "center",
            "top": 36,
            "textStyle": {"color": TEXT_COLOR, "fontSize": TEXT_SIZE},
        },
        "grid": {"left": 72, "right": 72, "top": 82, "bottom": 62},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 38,
            "axisLabel": {"color": TEXT_COLOR, "fontSize": TEXT_SIZE},
            "nameTextStyle": {"color": TEXT_COLOR, "fontSize": TEXT_SIZE},
            "axisLine": {"lineStyle": {"color": TEXT_COLOR}},
            "axisTick": {"lineStyle": {"color": TEXT_COLOR}},
            "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 52,
                "axisLabel": {"color": TEXT_COLOR, "fontSize": TEXT_SIZE},
                "nameTextStyle": {"color": TEXT_COLOR, "fontSize": TEXT_SIZE},
                "axisLine": {"show": True, "lineStyle": {"color": TEXT_COLOR}},
                "axisTick": {"lineStyle": {"color": TEXT_COLOR}},
                "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
            },
            {
                "type": "value",
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 52,
                "axisLabel": {"color": TEXT_COLOR, "fontSize": TEXT_SIZE},
                "nameTextStyle": {"color": TEXT_COLOR, "fontSize": TEXT_SIZE},
                "axisLine": {"show": True, "lineStyle": {"color": TEXT_COLOR}},
                "axisTick": {"lineStyle": {"color": TEXT_COLOR}},
                "splitLine": {"show": False},
            },
        ],
        "series": series,
    }


workbook = pd.ExcelFile(DATA_FILE)
figures = []

for sheet in workbook.sheet_names:
    frame = pd.read_excel(DATA_FILE, sheet_name=sheet)
    if frame.empty:
        continue

    normalized = {str(column).strip().lower(): column for column in frame.columns}

    def find_column(terms, fallback_index):
        for key, column in normalized.items():
            if any(term in key for term in terms):
                return column
        return frame.columns[fallback_index] if len(frame.columns) > fallback_index else None

    x_col = find_column(["temperature", "temp"], 1)
    weight_col = find_column(["weight"], 2)
    dtg_col = find_column(["d.w", "dtg", "derivative"], 3)

    if x_col is None or weight_col is None or dtg_col is None:
        continue

    option = make_option(
        sheet,
        as_float_array(frame[x_col]),
        as_float_array(frame[weight_col]),
        as_float_array(frame[dtg_col]),
    )
    if option is not None:
        figures.append(
            {
                "name": clean_name(sheet),
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        )

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
```
