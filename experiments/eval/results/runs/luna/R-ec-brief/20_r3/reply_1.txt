import json
import os
import pandas as pd
import numpy as np

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def clean_num(value):
    if pd.isna(value):
        return None
    return float(value)

def load_sheets(path):
    book = pd.ExcelFile(path)
    result = []
    for sheet in book.sheet_names:
        df = pd.read_excel(path, sheet_name=sheet)
        temp_col = next((c for c in df.columns if "temperature" in str(c).lower() or "temp" in str(c).lower()), None)
        weight_col = next((c for c in df.columns if "weight" in str(c).lower()), None)
        dtg_col = next((c for c in df.columns if "d.w" in str(c).lower() or "dtg" in str(c).lower()), None)
        if temp_col is None or weight_col is None:
            numeric = [c for c in df.columns if pd.api.types.is_numeric_dtype(df[c])]
            if len(numeric) < 2:
                continue
            temp_col, weight_col = numeric[0], numeric[1]
            dtg_col = numeric[2] if len(numeric) > 2 else None

        x = pd.to_numeric(df[temp_col], errors="coerce")
        weight = pd.to_numeric(df[weight_col], errors="coerce")
        valid = x.notna() & weight.notna()
        x = x[valid].to_numpy(dtype=float)
        weight = weight[valid].to_numpy(dtype=float)

        if dtg_col is not None:
            dtg_series = pd.to_numeric(df[dtg_col], errors="coerce")[valid]
            dtg = dtg_series.to_numpy(dtype=float)
        else:
            dtg = np.gradient(weight, x) if len(x) > 1 else np.zeros_like(x)

        finite = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
        x, weight, dtg = x[finite], weight[finite], dtg[finite]
        if len(x):
            order = np.argsort(x)
            result.append((sheet, x[order], weight[order], dtg[order]))
    return result

def make_option(sheet, x, weight, dtg):
    residue_index = int(np.argmin(np.abs(x - 800)))
    residue_x = float(x[residue_index])
    residue_y = float(weight[residue_index])

    peak_candidates = np.where(x > 150)[0]
    if len(peak_candidates):
        peak_index = int(peak_candidates[np.argmax(-dtg[peak_candidates])])
    else:
        peak_index = int(np.argmax(-dtg))
    peak_x = float(x[peak_index])
    peak_y = float(-dtg[peak_index])

    xmin, xmax = float(np.min(x)), float(np.max(x))
    ymin, ymax = float(np.min(weight)), float(np.max(weight))
    dtg_plot = -dtg
    dtg_max = float(np.max(dtg_plot)) if len(dtg_plot) else 1.0
    dtg_min = float(np.min(dtg_plot)) if len(dtg_plot) else 0.0
    pad = max((dtg_max - dtg_min) * 0.08, 0.01)

    return {
        "animation": False,
        "title": {
            "text": str(sheet),
            "left": "center",
            "top": 12,
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16}
        },
        "legend": {
            "data": ["Weight (%)", "DTG (-D.W)"],
            "left": 70,
            "top": "middle",
            "orient": "vertical",
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11}
        },
        "grid": {"left": 82, "right": 75, "top": 65, "bottom": 65, "containLabel": False},
        "tooltip": {"trigger": "axis"},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 35,
            "min": xmin,
            "max": xmax,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#dddddd", "type": "dashed"}},
            "axisLine": {"lineStyle": {"color": "#333333"}}
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 52,
                "min": min(0, ymin - (ymax - ymin) * 0.05),
                "max": ymax + (ymax - ymin) * 0.05,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": True, "lineStyle": {"color": "#dddddd", "type": "dashed"}}
            },
            {
                "type": "value",
                "name": "DTG (-D.W, %/°C)",
                "nameLocation": "middle",
                "nameGap": 55,
                "min": dtg_min - pad,
                "max": dtg_max + pad,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "showSymbol": False,
                "lineStyle": {"width": 2, "color": "#1f4e9c"},
                "data": [[clean_num(a), clean_num(b)] for a, b in zip(x, weight)],
                "markArea": {
                    "silent": True,
                    "itemStyle": {"color": "rgba(190, 190, 190, 0.30)"},
                    "data": [[{"xAxis": 300}, {"xAxis": 450}]]
                },
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "itemStyle": {"color": "#1f4e9c"},
                    "label": {
                        "show": True,
                        "formatter": f"Residue {residue_y:.1f}%",
                        "position": "top",
                        "fontFamily": "Noto Sans CJK KR",
                        "fontSize": 11
                    },
                    "data": [{"coord": [residue_x, residue_y]}]
                }
            },
            {
                "name": "DTG (-D.W)",
                "type": "line",
                "yAxisIndex": 1,
                "showSymbol": False,
                "lineStyle": {"width": 1.8, "color": "#c0392b"},
                "data": [[clean_num(a), clean_num(b)] for a, b in zip(x, dtg_plot)],
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "itemStyle": {"color": "#c0392b"},
                    "label": {
                        "show": True,
                        "formatter": f"{peak_x:.0f} °C",
                        "position": "top",
                        "fontFamily": "Noto Sans CJK KR",
                        "fontSize": 11
                    },
                    "data": [{"coord": [peak_x, peak_y]}]
                }
            }
        ]
    }

figures = []
for sheet, x, weight, dtg in load_sheets(DATA_FILE):
    figures.append({
        "name": str(sheet),
        "option": make_option(sheet, x, weight, dtg),
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    })

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
