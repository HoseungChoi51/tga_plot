```python
import json
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/blank_rows_on_top.xlsx"

def clean_number(value):
    try:
        number = float(value)
        return number if np.isfinite(number) else None
    except (TypeError, ValueError):
        return None

def main():
    workbook = pd.ExcelFile(DATA_FILE)
    figures = []

    for sheet_name in workbook.sheet_names:
        raw = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=None, engine="openpyxl")

        # Locate the header row containing the requested Excel columns' labels.
        header_row = None
        for idx in range(min(len(raw), 30)):
            row = raw.iloc[idx].astype(str).str.strip().str.lower()
            if "temperature" in row.values and any("weight" in cell for cell in row.values):
                header_row = idx
                break

        if header_row is None:
            continue

        headers = raw.iloc[header_row].astype(str).str.strip()
        col_positions = {}
        for excel_col, candidates in {
            "C": ("temperature",),
            "D": ("weight",),
            "E": ("d.w", "dw", "derivative"),
        }.items():
            for pos, label in enumerate(headers):
                label_lower = label.lower()
                if any(candidate in label_lower for candidate in candidates):
                    col_positions[excel_col] = pos
                    break

        if not all(col in col_positions for col in ("C", "D", "E")):
            continue

        data = raw.iloc[header_row + 1 :]
        x = pd.to_numeric(data.iloc[:, col_positions["C"]], errors="coerce")
        y1 = pd.to_numeric(data.iloc[:, col_positions["D"]], errors="coerce")
        y2_raw = pd.to_numeric(data.iloc[:, col_positions["E"]], errors="coerce")

        valid = x.notna() & y1.notna() & y2_raw.notna()
        x = x[valid].to_numpy(dtype=float)
        y1 = y1[valid].to_numpy(dtype=float)
        y2 = -y2_raw[valid].to_numpy(dtype=float)

        if len(x) == 0:
            continue

        weight_label = str(headers.iloc[col_positions["D"]])
        dtg_label = str(headers.iloc[col_positions["E"]])

        series = [
            {
                "name": weight_label,
                "type": "line",
                "showSymbol": False,
                "yAxisIndex": 0,
                "data": [[clean_number(a), clean_number(b)] for a, b in zip(x, y1)],
                "lineStyle": {"color": "#1f4e9c", "width": 2},
            },
            {
                "name": f"-{dtg_label}",
                "type": "line",
                "showSymbol": False,
                "yAxisIndex": 1,
                "data": [[clean_number(a), clean_number(b)] for a, b in zip(x, y2)],
                "lineStyle": {"color": "#c0392b", "width": 2},
            },
        ]

        option = {
            "animation": False,
            "title": {
                "text": str(sheet_name),
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
            "tooltip": {"trigger": "axis"},
            "legend": {"data": [weight_label, f"-{dtg_label}", "Residue", "DTG peak"], "left": 12, "top": "middle"},
            "grid": {"left": 78, "right": 78, "top": 58, "bottom": 65},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 38,
                "axisLine": {"show": True},
                "axisTick": {"show": True},
                "splitLine": {"show": False},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": weight_label,
                    "nameLocation": "middle",
                    "nameGap": 55,
                    "axisLine": {"show": True},
                    "axisTick": {"show": True},
                    "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
                },
                {
                    "type": "value",
                    "name": f"-{dtg_label}",
                    "nameLocation": "middle",
                    "nameGap": 55,
                    "axisLine": {"show": True},
                    "axisTick": {"show": True},
                    "splitLine": {"show": False},
                },
            ],
            "series": series,
        }

        # Residue marker at the measured point nearest 800 °C.
        residue_index = int(np.argmin(np.abs(x - 800.0)))
        option["series"][0]["markPoint"] = {
            "symbol": "circle",
            "symbolSize": 9,
            "itemStyle": {"color": "#1f4e9c"},
            "label": {
                "show": True,
                "formatter": f"Residue {y1[residue_index]:.1f}%",
                "position": "right",
            },
            "data": [{"coord": [clean_number(x[residue_index]), clean_number(y1[residue_index])]}],
        }

        # Highest upward DTG peak above 150 °C.
        eligible = np.where(x > 150.0)[0]
        if len(eligible):
            peak_index = int(eligible[np.argmax(y2[eligible])])
            option["series"][1]["markPoint"] = {
                "symbol": "circle",
                "symbolSize": 9,
                "itemStyle": {"color": "#c0392b"},
                "label": {
                    "show": True,
                    "formatter": f"{x[peak_index]:.0f} °C",
                    "position": "right",
                },
                "data": [{"coord": [clean_number(x[peak_index]), clean_number(y2[peak_index])]}],
            }

        figures.append(
            {
                "name": f"plot{len(figures) + 1}",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        )

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))

if __name__ == "__main__":
    main()
```
