```python
import json
import re
import sys

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"


def clean_name(value):
    name = re.sub(r"[^\w가-힣.-]+", "_", str(value).strip())
    return name.strip("_") or "plot"


def finite_pairs(x, y):
    mask = np.isfinite(x) & np.isfinite(y)
    return [[float(a), float(b)] for a, b in zip(x[mask], y[mask])]


def make_option(sheet_name, frame):
    frame.columns = [str(c).strip() for c in frame.columns]
    lower = {c: c.lower() for c in frame.columns}

    def choose(predicate, fallback):
        return next((c for c in frame.columns if predicate(lower[c])), fallback)

    x_col = choose(lambda c: "temp" in c or "temperature" in c or "온도" in c, frame.columns[1])
    weight_col = choose(lambda c: "weight" in c or "mass" in c or "중량" in c or "무게" in c, frame.columns[2])
    dtg_col = choose(lambda c: "dtg" in c or "d.w" in c or "derivative" in c, frame.columns[3])

    x = pd.to_numeric(frame[x_col], errors="coerce").to_numpy(dtype=float)
    weight = pd.to_numeric(frame[weight_col], errors="coerce").to_numpy(dtype=float)
    dtg_raw = pd.to_numeric(frame[dtg_col], errors="coerce").to_numpy(dtype=float)
    dtg = -dtg_raw

    valid_x = np.isfinite(x)
    if not valid_x.any():
        return None

    residue_idx = int(np.nanargmin(np.where(valid_x, np.abs(x - 800), np.inf)))
    peak_candidates = np.flatnonzero(valid_x & np.isfinite(dtg) & (x > 150))
    peak_idx = int(peak_candidates[np.argmax(dtg[peak_candidates])]) if peak_candidates.size else None

    series = [
        {
            "name": weight_col,
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 0,
            "data": finite_pairs(x, weight),
            "lineStyle": {"width": 2, "color": "#1f4e9c"},
            "itemStyle": {"color": "#1f4e9c"},
        },
        {
            "name": "DTG",
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 1,
            "data": finite_pairs(x, dtg),
            "lineStyle": {"width": 2, "color": "#c0392b"},
            "itemStyle": {"color": "#c0392b"},
        },
    ]

    if np.isfinite(weight[residue_idx]):
        residue_label = f"Residue {weight[residue_idx]:.1f}%"
        series[0]["markPoint"] = {
            "symbol": "circle",
            "symbolSize": 9,
            "data": [{
                "name": residue_label,
                "coord": [float(x[residue_idx]), float(weight[residue_idx])],
                "value": residue_label,
                "itemStyle": {"color": "#1f4e9c"},
                "label": {"show": True, "position": "top", "formatter": residue_label},
            }],
        }

    if peak_idx is not None:
        peak_label = f"{x[peak_idx]:.0f} °C"
        series[1]["markPoint"] = {
            "symbol": "circle",
            "symbolSize": 9,
            "data": [{
                "name": peak_label,
                "coord": [float(x[peak_idx]), float(dtg[peak_idx])],
                "value": peak_label,
                "itemStyle": {"color": "#c0392b"},
                "label": {"show": True, "position": "top", "formatter": peak_label},
            }],
        }

    option = {
        "animation": False,
        "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        "title": {"text": str(sheet_name), "left": "center"},
        "tooltip": {"trigger": "axis"},
        "legend": {"data": [weight_col, "DTG"], "left": "center", "top": 30},
        "grid": {"left": 75, "right": 75, "top": 80, "bottom": 60},
        "xAxis": {
            "type": "value",
            "name": x_col,
            "nameLocation": "middle",
            "nameGap": 35,
            "min": float(np.nanmin(x[valid_x])),
            "max": float(np.nanmax(x[valid_x])),
            "axisTick": {"show": True},
            "axisLabel": {"show": True},
            "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
        },
        "yAxis": [
            {
                "type": "value",
                "name": weight_col,
                "nameLocation": "middle",
                "nameGap": 52,
                "axisTick": {"show": True},
                "axisLabel": {"show": True},
                "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
            },
            {
                "type": "value",
                "name": "DTG (-D.W, %/°C)",
                "nameLocation": "middle",
                "nameGap": 55,
                "axisTick": {"show": True},
                "axisLabel": {"show": True},
                "splitLine": {"show": False},
            },
        ],
        "series": series,
    }
    return option


def main():
    figures = []
    book = pd.ExcelFile(DATA_FILE)
    for sheet in book.sheet_names:
        frame = pd.read_excel(book, sheet_name=sheet)
        frame = frame.dropna(how="all")
        if frame.shape[1] < 4 or frame.empty:
            continue
        option = make_option(sheet, frame)
        if option is not None:
            figures.append({
                "name": clean_name(sheet),
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            })
    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
