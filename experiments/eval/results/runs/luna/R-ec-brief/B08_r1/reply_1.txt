```python
import json
import os
import re

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"


def clean_number(value):
    try:
        number = float(value)
        return number if np.isfinite(number) else None
    except (TypeError, ValueError):
        return None


def main():
    workbook = pd.ExcelFile(DATA_FILE)
    sheet_name = "Raw"
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)

    def find_column(pattern, fallback_index):
        for column in df.columns:
            if re.search(pattern, str(column), re.IGNORECASE):
                return column
        return df.columns[fallback_index]

    x_col = find_column(r"temperature|temp", 1)
    weight_col = find_column(r"weight", 2)
    dtg_col = find_column(r"deriv|dtg", 3)

    x = pd.to_numeric(df[x_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    dtg = pd.to_numeric(df[dtg_col], errors="coerce")

    valid = x.notna() & weight.notna() & dtg.notna()
    x = x[valid].to_numpy(dtype=float)
    weight = weight[valid].to_numpy(dtype=float)
    dtg = dtg[valid].to_numpy(dtype=float)

    if len(x) == 0:
        raise ValueError("Raw 시트에 유효한 TGA 데이터가 없습니다.")

    residue_idx = int(np.argmin(np.abs(x - 800)))
    peak_candidates = np.flatnonzero(x > 150)
    peak_idx = int(peak_candidates[np.argmax(-dtg[peak_candidates])]) if len(peak_candidates) else int(np.argmax(-dtg))

    residue_x = float(x[residue_idx])
    residue_y = float(weight[residue_idx])
    peak_x = float(x[peak_idx])
    peak_y = float(-dtg[peak_idx])

    x_min, x_max = float(np.min(x)), float(np.max(x))
    y1_min = min(0.0, float(np.min(weight)))
    y1_max = max(105.0, float(np.max(weight)))
    y2_max = max(0.1, float(np.max(-dtg)) * 1.12)

    option = {
        "animation": False,
        "backgroundColor": "#ffffff",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "color": "#222222"},
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": 12,
            "textStyle": {"fontSize": 17, "fontWeight": "normal"},
        },
        "legend": {
            "data": ["Weight (%)", "-DTG (%/°C)"],
            "left": 62,
            "top": "middle",
            "orient": "vertical",
            "selectedMode": False,
            "backgroundColor": "rgba(255,255,255,0.78)",
        },
        "grid": {"left": 72, "right": 72, "top": 66, "bottom": 60},
        "tooltip": {"trigger": "axis"},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 34,
            "min": x_min,
            "max": x_max,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"fontSize": 11},
            "splitLine": {"show": False},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 45,
                "min": y1_min,
                "max": y1_max,
                "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                "axisLabel": {"fontSize": 11},
                "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}},
            },
            {
                "type": "value",
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 48,
                "min": 0,
                "max": y2_max,
                "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                "axisLabel": {"fontSize": 11},
                "splitLine": {"show": False},
            },
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "showSymbol": False,
                "data": [[clean_number(a), clean_number(b)] for a, b in zip(x, weight)],
                "lineStyle": {"width": 2, "color": "#1f4e9c"},
                "itemStyle": {"color": "#1f4e9c"},
                "markLine": {
                    "silent": True,
                    "symbol": ["none", "none"],
                    "lineStyle": {"color": "#666666", "type": "dashed", "width": 1.4},
                    "label": {
                        "show": True,
                        "formatter": "50%",
                        "position": "insideEndTop",
                        "color": "#555555",
                    },
                    "data": [{"yAxis": 50}],
                },
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "itemStyle": {"color": "#1f4e9c", "borderColor": "#ffffff", "borderWidth": 1},
                    "label": {
                        "show": True,
                        "formatter": f"Residue {residue_y:.1f}%",
                        "position": "top",
                        "color": "#1f4e9c",
                    },
                    "data": [{"coord": [residue_x, residue_y]}],
                },
            },
            {
                "name": "-DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "showSymbol": False,
                "data": [[clean_number(a), clean_number(-b)] for a, b in zip(x, dtg)],
                "lineStyle": {"width": 2, "color": "#c0392b"},
                "itemStyle": {"color": "#c0392b"},
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "itemStyle": {"color": "#c0392b", "borderColor": "#ffffff", "borderWidth": 1},
                    "label": {
                        "show": True,
                        "formatter": f"{peak_x:.0f} °C",
                        "position": "top",
                        "color": "#c0392b",
                    },
                    "data": [{"coord": [peak_x, peak_y]}],
                },
            },
        ],
        "graphic": [
            {
                "type": "text",
                "right": 22,
                "top": 20,
                "style": {
                    "text": "공기, 10 °C/min",
                    "font": "14px Noto Sans CJK KR",
                    "fill": "#333333",
                    "textAlign": "right",
                },
            }
        ],
    }

    result = {
        "figures": [
            {
                "name": "raw",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        ]
    }
    print(json.dumps(result, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
