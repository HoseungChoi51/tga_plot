```python
import json
import re
from pathlib import Path

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt"

def read_data(path):
    p = Path(path)
    if p.suffix.lower() in {".xls", ".xlsx"}:
        return pd.read_excel(p, sheet_name=None, header=None)
    if p.suffix.lower() == ".csv":
        return {"instrument_export": pd.read_csv(p, header=None)}
    try:
        return {"instrument_export": pd.read_csv(p, sep=None, engine="python", header=None, comment="#")}
    except Exception:
        return {"instrument_export": pd.read_csv(p, delim_whitespace=True, header=None, comment="#")}

def numeric_data(df):
    out = df.apply(pd.to_numeric, errors="coerce")
    out = out.dropna(how="all").dropna(axis=1, how="all")
    return out

def choose_columns(df):
    arr = numeric_data(df)
    if arr.shape[1] < 3:
        return None
    vals = [arr.iloc[:, i].dropna().to_numpy(dtype=float) for i in range(arr.shape[1])]
    # Typical export: time, temperature, weight, derivative weight.
    if len(vals) >= 4:
        candidates = []
        for i in range(len(vals)):
            v = vals[i]
            if len(v) and np.nanmin(v) >= 0 and np.nanmax(v) <= 100.5 and np.nanmedian(v) > 5:
                candidates.append(i)
        weight_i = candidates[-1] if candidates else 2
        # Derivative is most likely the final column.
        return arr.iloc[:, 1].to_numpy(float), arr.iloc[:, weight_i].to_numpy(float), arr.iloc[:, -1].to_numpy(float)
    return arr.iloc[:, 0].to_numpy(float), arr.iloc[:, 1].to_numpy(float), arr.iloc[:, 2].to_numpy(float)

def clean_xy(x, y):
    mask = np.isfinite(x) & np.isfinite(y)
    return x[mask], y[mask]

def main():
    sheets = read_data(DATA_FILE)
    figures = []
    for sheet_name, raw in sheets.items():
        selected = choose_columns(raw)
        if selected is None:
            continue
        x, weight, dtg = selected
        x, weight = clean_xy(x, weight)
        xd, dtg = clean_xy(selected[0], selected[2])
        if len(x) < 2 or len(xd) < 2:
            continue

        residue_idx = int(np.nanargmin(np.abs(x - 800.0)))
        residue_temp = float(x[residue_idx])
        residue = float(weight[residue_idx])

        peak_mask = xd > 150
        if np.any(peak_mask):
            peak_indices = np.flatnonzero(peak_mask)
            peak_idx = int(peak_indices[np.nanargmax(-dtg[peak_mask])])
            peak_temp = float(xd[peak_idx])
            peak_y = float(-dtg[peak_idx])
            peak_label = f"{peak_temp:.0f} °C"
        else:
            peak_temp = None
            peak_y = None
            peak_label = ""

        option = {
            "animation": False,
            "title": {
                "text": str(sheet_name),
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
            },
            "tooltip": {"trigger": "axis"},
            "legend": {"data": ["Weight", "DTG"], "left": "center", "bottom": 8},
            "grid": {"left": 72, "right": 72, "top": 58, "bottom": 58},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 32,
                "axisLine": {"show": True},
                "axisTick": {"show": True},
                "splitLine": {"show": False},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "axisLine": {"show": True},
                    "axisTick": {"show": True},
                    "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
                },
                {
                    "type": "value",
                    "name": "-DTG (%/°C)",
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "axisLine": {"show": True},
                    "axisTick": {"show": True},
                    "splitLine": {"show": False},
                },
            ],
            "series": [
                {
                    "name": "Weight",
                    "type": "line",
                    "showSymbol": False,
                    "data": [[float(a), float(b)] for a, b in zip(x, weight)],
                    "lineStyle": {"color": "#1f4e9c", "width": 2},
                    "itemStyle": {"color": "#1f4e9c"},
                    "markPoint": {
                        "symbol": "circle",
                        "symbolSize": 9,
                        "data": [
                            {
                                "coord": [residue_temp, residue],
                                "value": residue,
                                "itemStyle": {"color": "#1f4e9c"},
                                "label": {
                                    "show": True,
                                    "formatter": f"Residue: {residue:.1f}%",
                                    "position": "right",
                                    "color": "#1f4e9c",
                                    "fontFamily": "Noto Sans CJK KR",
                                },
                            }
                        ],
                    },
                },
                {
                    "name": "DTG",
                    "type": "line",
                    "yAxisIndex": 1,
                    "showSymbol": False,
                    "data": [[float(a), float(-b)] for a, b in zip(xd, dtg)],
                    "lineStyle": {"color": "#c0392b", "width": 2},
                    "itemStyle": {"color": "#c0392b"},
                    "markPoint": {
                        "symbol": "circle",
                        "symbolSize": 9,
                        "data": (
                            [
                                {
                                    "coord": [peak_temp, peak_y],
                                    "value": peak_y,
                                    "itemStyle": {"color": "#c0392b"},
                                    "label": {
                                        "show": True,
                                        "formatter": peak_label,
                                        "position": "top",
                                        "color": "#c0392b",
                                        "fontFamily": "Noto Sans CJK KR",
                                    },
                                }
                            ]
                            if peak_temp is not None
                            else []
                        ),
                    },
                },
            ],
        }
        name = re.sub(r"[^A-Za-z0-9_-]+", "_", str(sheet_name)).strip("_") or "plot1"
        figures.append({"name": name, "option": option, "width": 614, "height": 442, "dpi": 300, "formats": ["png"]})

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))

if __name__ == "__main__":
    main()
```
