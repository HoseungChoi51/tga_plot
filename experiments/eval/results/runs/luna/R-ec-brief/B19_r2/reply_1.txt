import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

df = pd.read_excel(DATA_FILE, sheet_name="PVC")
temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce").to_numpy()
weight = pd.to_numeric(df["Weight (%)"], errors="coerce").to_numpy()

valid = np.isfinite(temperature) & np.isfinite(weight)
temperature = temperature[valid]
weight = weight[valid]

def weight_at(target):
    order = np.argsort(temperature)
    return float(np.interp(target, temperature[order], weight[order]))

weight_200 = weight_at(200)
weight_400 = weight_at(400)
mass_loss = weight_200 - weight_400

# Standard TGA annotations: residue nearest 800 °C and highest DTG peak above 150 °C.
dtg = pd.to_numeric(df["Deriv. Weight (%/°C)"], errors="coerce").to_numpy()[valid]

peak_candidates = np.where(temperature > 150)[0]
peak_idx = int(peak_candidates[np.nanargmax(-dtg[peak_candidates])])

x = temperature.tolist()
weight_data = weight.tolist()
dtg_data = (-dtg).tolist()
