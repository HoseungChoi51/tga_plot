import json
import math
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
SHEET = "Raw"

df = pd.read_excel(DATA_FILE, sheet_name=SHEET)
cols = list(df.columns)


def find_col(predicate, fallback):
    for col in cols:
        if predicate(str(col).lower()):
            return col
    return cols[fallback]


xcol = find_col(lambda s: "temperature" in s or "temp" in s, 1)
weight_col = find_col(lambda s: "weight" in s and ("deriv" not in s and "dtg" not in s), 2)
dtg_col = find_col(lambda s: "deriv" in s or "dtg" in s, 3)

x = pd.to_numeric(df[xcol], errors="coerce").to_numpy(dtype=float)
weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy(dtype=float)
dtg = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy(dtype=float)

valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
x, weight, dtg = x[valid], weight[valid], dtg[valid]
order = np.argsort(x)
x, weight, dtg = x[order], weight[order], dtg[order]

if len(x) == 0:
    raise ValueError("Raw 시트에 유효한 온도, 무게, DTG 데이터가 없습니다.")

# The DTG curve is inverted so mass-loss peaks point upward.
dtg_plot = -dtg

residue_idx = int(np.argmin(np.abs(x - 800.0)))
peak_candidates = np.flatnonzero(x > 150.0)
peak_idx = int(peak_candidates[np.argmax(dtg_plot[peak_candidates])]) if len(peak_candidates) else int(np.argmax(dtg_plot))

xmin, xmax = float(np.min(x)), float(np.max(x))
y1min, y1max = float(np.min(weight)), float(np.max(weight))
y2min, y2max = float(np.min(dtg_plot)), float(np.max(dtg_plot))
y1pad = max((y1max - y1min) * 0.08, 1.0)
y2pad = max((y2max - y2min) * 0.12, 0.02)

    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 48,
            "min": y1min - y1pad,
            "max": y1max + y1pad,
            "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
            "axisTick": {"show": True},
            "axisLabel": {"fontSize": 11},
            "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}},
        },
        {
            "type": "value",

}
