```python
import json
import os
import re

import numpy as np
import pandas as pd
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/xrd.csv"

df = pd.read_csv(DATA_FILE)
if df.shape[1] < 2:
    raise ValueError("XRD 패턴에는 2개 이상의 열이 필요합니다.")

def numeric_column(series):
    return pd.to_numeric(
        series.astype(str).str.replace(",", "", regex=False).str.strip(),
        errors="coerce",
    )

x = numeric_column(df.iloc[:, 0]).to_numpy(dtype=float)
y = numeric_column(df.iloc[:, 1]).to_numpy(dtype=float)
valid = np.isfinite(x) & np.isfinite(y)
x, y = x[valid], y[valid]

order = np.argsort(x)
x, y = x[order], y[order]

if len(x) == 0:
    raise ValueError("유효한 XRD 데이터가 없습니다.")

# Find local maxima, then select the five strongest peaks. If fewer than five
# local maxima are found, fill the remaining slots with the strongest points.
peaks, _ = find_peaks(y, prominence=0)
ranked = sorted(peaks.tolist(), key=lambda i: y[i], reverse=True)
selected = ranked[:5]

if len(selected) < min(5, len(y)):
    used = set(selected)
    for i in np.argsort(y)[::-1]:
        i = int(i)
        if i not in used:
            selected.append(i)
            used.add(i)
        if len(selected) >= min(5, len(y)):
            break

selected = sorted(selected, key=lambda i: x[i])
peak_data = [[float(x[i]), float(y[i])] for i in selected]

# Place labels above each selected peak, with a small vertical offset.
y_range = float(np.nanmax(y) - np.nanmin(y))
offset = 0.045 * y_range if y_range > 0 else 1.0
label_data = [
    {
        "coord": [float(x[i]), float(y[i] + offset)],
        "value": f"{x[i]:.2f}°",
    }
    for i in selected
]

option = {
    "animation": False,
    "title": {
        "text": "XRD Pattern",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR"},
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "cross"},
    },
    "grid": {"left": 75, "right": 35, "top": 75, "bottom": 65},
    "xAxis": {
        "type": "value",
        "name": "2θ (deg)",
        "nameLocation": "middle",
        "nameGap": 38,
        "min": float(np.min(x)),
        "max": float(np.max(x)),
        "axisLine": {"show": True},
        "axisTick": {"show": True},
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
    },
    "yAxis": {
        "type": "value",
        "name": "Intensity (counts)",
        "nameLocation": "middle",
        "nameGap": 55,
        "axisLine": {"show": True},
        "axisTick": {"show": True},
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
    },
    "series": [
        {
            "name": "Intensity",
            "type": "line",
            "data": [[float(a), float(b)] for a, b in zip(x, y)],
            "showSymbol": False,
            "symbol": "none",
            "lineStyle": {"color": "#1f4e9c", "width": 1.5},
            "itemStyle": {"color": "#1f4e9c"},
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#c0392b"},
                "label": {
                    "show": True,
                    "position": "top",
                    "distance": 7,
                    "color": "#c0392b",
                    "fontSize": 11,
                    "fontFamily": "Noto Sans CJK KR",
                    "formatter": "{b}",
                },
                "data": [
                    {
                        "name": label["value"],
                        "coord": label["coord"],
                        "value": label["value"],
                    }
                    for label in label_data
                ],
            },
        }
    ],
}

print(json.dumps({
    "figures": [{
        "name": "xrd_pattern",
        "option": option,
        "width": 900,
        "height": 600,
        "dpi": 300,
        "formats": ["png"],
    }]
}, ensure_ascii=False, allow_nan=False))
```
