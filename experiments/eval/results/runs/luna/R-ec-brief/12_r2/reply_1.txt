```python
import json
import re
import sys
from pathlib import Path

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def clean_name(value):
    return re.sub(r'[\\/*?:"<>|]+', "_", str(value)).strip() or "plot"

def find_col(columns, pattern, fallback):
    for col in columns:
        if re.search(pattern, str(col), re.I):
            return col
    return columns[fallback] if len(columns) > fallback else None

def numeric_series(series):
    return pd.to_numeric(series, errors="coerce")

figures = []
book = pd.ExcelFile(DATA_FILE)

for sheet in book.sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    df = df.dropna(how="all")
    if df.empty:
        continue

    cols = list(df.columns)
    x_col = find_col(cols, r"temperature|temp|온도", 1)
    weight_col = find_col(cols, r"weight|mass|중량|무게", 2)
    dtg_col = find_col(cols, r"d\.?\s*w|dtg|derivative|미분", 3)
    if x_col is None or weight_col is None:
        continue

    x = numeric_series(df[x_col])
    weight = numeric_series(df[weight_col])
    valid = x.notna() & weight.notna()
    x = x[valid].to_numpy(dtype=float)
    weight = weight[valid].to_numpy(dtype=float)

    dtg = None
    if dtg_col is not None:
        dtg_series = numeric_series(df.loc[valid, dtg_col])
        if dtg_series.notna().any():
            dtg = dtg_series.fillna(0).to_numpy(dtype=float)

    order = np.argsort(x)
    x, weight = x[order], weight[order]
    if dtg is not None:
        dtg = dtg[order]

    residue_idx = int(np.argmin(np.abs(x - 800)))
    residue = float(weight[residue_idx])

    peak_idx = None
    if dtg is not None:
        eligible = np.flatnonzero(x > 150)
        if eligible.size:
            peak_idx = int(eligible[np.argmax(-dtg[eligible])])

    series = [
        {
            "name": str(weight_col),
            "type": "line",
            "yAxisIndex": 0,
            "showSymbol": False,
            "lineStyle": {"width": 2, "color": "#1f4e9c"},
            "data": [[float(a), float(b)] for a, b in zip(x, weight)],
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 9,
                "itemStyle": {"color": "#1f4e9c"},
                "label": {
                    "show": True,
                    "formatter": f"Residue {residue:.1f}%",
                    "position": "right",
                    "color": "#1f4e9c",
                    "fontFamily": "Noto Sans CJK KR",
                },
                "data": [{"coord": [float(x[residue_idx]), residue]}],
            },
        }
    ]

    if dtg is not None:
        dtg_data = -dtg
        dtg_series = {
            "name": str(dtg_col) + " (-DTG)",
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "lineStyle": {"width": 2, "color": "#c0392b"},
            "data": [[float(a), float(b)] for a, b in zip(x, dtg_data)],
        }
        if peak_idx is not None:
            dtg_series["markPoint"] = {
                "symbol": "circle",
                "symbolSize": 9,
                "itemStyle": {"color": "#c0392b"},
                "label": {
                    "show": True,
                    "formatter": f"{x[peak_idx]:.0f} °C",
                    "position": "top",
                    "color": "#c0392b",
                    "fontFamily": "Noto Sans CJK KR",
                },
                "data": [{"coord": [float(x[peak_idx]), float(dtg_data[peak_idx])]}],
            }
        series.append(dtg_series)

    series[0]["markLine"] = {
        "silent": True,
        "symbol": ["none", "none"],
        "lineStyle": {"color": "#666666", "type": "dashed", "width": 1.5},
        "label": {
            "show": True,
            "formatter": "산화 시작",
            "position": "insideEndTop",
            "color": "#555555",
            "fontFamily": "Noto Sans CJK KR",
        },
        "data": [{"xAxis": 450}],
    }

    option = {
        "animation": False,
        "title": {
            "text": str(sheet),
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "tooltip": {"trigger": "axis"},
        "legend": {"data": [s["name"] for s in series], "left": "center"},
        "grid": {"left": 72, "right": 72, "top": 70, "bottom": 60},
        "xAxis": {
            "type": "value",
            "name": str(x_col),
            "nameLocation": "middle",
            "nameGap": 35,
            "min": float(np.nanmin(x)),
            "max": float(np.nanmax(x)),
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "yAxis": [
            {
                "type": "value",
                "name": str(weight_col),
                "nameLocation": "middle",
                "nameGap": 50,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
            },
            {
                "type": "value",
                "name": "- " + str(dtg_col) if dtg is not None else "DTG",
                "nameLocation": "middle",
                "nameGap": 50,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": False},
            },
        ],
        "series": series,
    }

    figures.append(
        {
            "name": clean_name(sheet),
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"],
        }
    )

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
```
