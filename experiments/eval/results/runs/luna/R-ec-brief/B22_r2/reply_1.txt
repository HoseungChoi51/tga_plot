```python
import json
import os
import re

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
SHEET = "Raw"
WIDTH = 614
HEIGHT = 442
DPI = 300
FORMATS = ["png"]


def find_column(columns, pattern):
    for col in columns:
        if re.search(pattern, str(col), flags=re.IGNORECASE):
            return col
    return None


df = pd.read_excel(DATA_FILE, sheet_name=SHEET)
df.columns = [str(c).strip() for c in df.columns]

x_col = find_column(df.columns, r"temperature|temp")
weight_col = find_column(df.columns, r"weight")
dtg_col = find_column(df.columns, r"deriv|dtg")

if x_col is None or weight_col is None or dtg_col is None:
    raise ValueError("Raw 시트에서 온도, 무게, DTG 열을 찾을 수 없습니다.")

for col in (x_col, weight_col, dtg_col):
    df[col] = pd.to_numeric(df[col], errors="coerce")

df = df.dropna(subset=[x_col, weight_col, dtg_col]).sort_values(x_col)
temperature = df[x_col].to_numpy(dtype=float)
weight = df[weight_col].to_numpy(dtype=float)
dtg = df[dtg_col].to_numpy(dtype=float)

# 150 °C에 가장 가까운 측정점의 무게를 기준으로 건조 기준 무게를 계산합니다.
baseline_index = int(np.nanargmin(np.abs(temperature - 150.0)))
baseline_weight = weight[baseline_index]
if not np.isfinite(baseline_weight) or baseline_weight == 0:
    raise ValueError("150 °C 부근의 기준 무게가 유효하지 않습니다.")

dry_weight = weight / baseline_weight * 100.0

residue_index = int(np.nanargmin(np.abs(temperature - 800.0)))
peak_candidates = np.flatnonzero(temperature > 150.0)
if len(peak_candidates) == 0:
    raise ValueError("150 °C보다 높은 온도의 DTG 데이터가 없습니다.")
peak_index = int(peak_candidates[np.nanargmax(-dtg[peak_candidates])])

x_data = temperature.tolist()
weight_data = dry_weight.tolist()
dtg_data = (-dtg).tolist()

option = {
    "animation": False,
    "title": {
        "text": SHEET,
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
    },
    "tooltip": {"trigger": "axis"},
    "legend": {
        "data": ["Weight (dry basis)", "-DTG"],
        "left": "center",
        "top": 38,
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11},
    },
    "grid": {"left": 62, "right": 62, "top": 80, "bottom": 55},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 32,
        "min": float(np.nanmin(temperature)),
        "max": float(np.nanmax(temperature)),
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        "axisLine": {"show": True},
        "axisTick": {"show": True},
        "splitLine": {"show": False},
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 42,
            "min": 0,
            "max": max(100, float(np.nanmax(dry_weight)) * 1.05),
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
            "axisTick": {"show": True},
            "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}},
        },
        {
            "type": "value",
            "name": "-DTG (%/°C)",
            "nameLocation": "middle",
            "nameGap": 42,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
            "axisTick": {"show": True},
            "splitLine": {"show": False},
        },
    ],
    "series": [
        {
            "name": "Weight (dry basis)",
            "type": "line",
            "yAxisIndex": 0,
            "showSymbol": False,
            "smooth": False,
            "lineStyle": {"width": 2, "color": "#1f4e9c"},
            "data": [[x, y] for x, y in zip(x_data, weight_data)],
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#1f4e9c"},
                "label": {
                    "show": True,
                    "formatter": f"Residue {dry_weight[residue_index]:.1f}%",
                    "position": "right",
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 11,
                },
                "data": [
                    {
                        "coord": [
                            float(temperature[residue_index]),
                            float(dry_weight[residue_index]),
                        ]
                    }
                ],
            },
        },
        {
            "name": "-DTG",
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "smooth": False,
            "lineStyle": {"width": 2, "color": "#c0392b"},
            "data": [[x, y] for x, y in zip(x_data, dtg_data)],
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#c0392b"},
                "label": {
                    "show": True,
                    "formatter": f"{temperature[peak_index]:.0f} °C",
                    "position": "top",
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 11,
                },
                "data": [
                    {
                        "coord": [
                            float(temperature[peak_index]),
                            float(-dtg[peak_index]),
                        ]
                    }
                ],
            },
        },
    ],
}

result = {
    "figures": [
        {
            "name": "raw_dry_basis",
            "option": option,
            "width": WIDTH,
            "height": HEIGHT,
            "dpi": DPI,
            "formats": FORMATS,
        }
    ]
}
print(json.dumps(result, ensure_ascii=False, allow_nan=False))
```
