```python
import json
import os
import re
import sys

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"


def clean_name(value):
    value = re.sub(r"[^\w.-]+", "_", str(value).strip(), flags=re.UNICODE)
    return value.strip("_.") or "plot"


def main():
    excel = pd.ExcelFile(DATA_FILE)
    colors = ["#1f4e9c", "#c0392b", "#2e8b57", "#8e44ad", "#d68910", "#117a8b"]
    series = []
    legend = []
    all_x = []
    all_weight = []
    all_dtg = []

    for index, sheet_name in enumerate(excel.sheet_names):
        df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        if df.empty or len(df.columns) < 4:
            continue

        cols = list(df.columns)
        def find_col(words, fallback):
            for col in cols:
                if any(word in str(col).lower() for word in words):
                    return col
            return cols[fallback]

        x_col = find_col(["temperature", "temp"], 1)
        weight_col = find_col(["weight", "mass"], 2)
        dtg_col = find_col(["d.w", "dtg", "derivative"], 3)

        x = pd.to_numeric(df[x_col], errors="coerce").to_numpy(dtype=float)
        weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy(dtype=float)
        dtg_raw = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy(dtype=float)
        valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg_raw)
        x, weight, dtg_raw = x[valid], weight[valid], dtg_raw[valid]
        if not len(x):
            continue

        order = np.argsort(x)
        x, weight, dtg_raw = x[order], weight[order], dtg_raw[order]
        color = colors[index % len(colors)]
        name = str(sheet_name)
        legend.append(name)
        all_x.extend(x.tolist())
        all_weight.extend(weight.tolist())
        all_dtg.extend((-dtg_raw).tolist())

        series.append({
            "name": name,
            "type": "line",
            "yAxisIndex": 0,
            "showSymbol": False,
            "smooth": False,
            "lineStyle": {"width": 2, "color": color},
            "itemStyle": {"color": color},
            "data": [[float(a), float(b)] for a, b in zip(x, weight)]
        })
        series.append({
            "name": f"{name} DTG",
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "smooth": False,
            "lineStyle": {"width": 1.6, "color": color, "type": "dashed"},
            "itemStyle": {"color": color},
            "data": [[float(a), float(-b)] for a, b in zip(x, dtg_raw)]
        })

    if not series:
        raise ValueError("유효한 TGA 데이터가 없습니다.")

    xmin, xmax = float(min(all_x)), float(max(all_x))
    ymin, ymax = float(min(all_weight)), float(max(all_weight))
    dmin, dmax = float(min(all_dtg)), float(max(all_dtg))
    if ymin == ymax:
        ymin, ymax = ymin - 1, ymax + 1
    if dmin == dmax:
        dmin, dmax = dmin - 0.1, dmax + 0.1
    ypad = (ymax - ymin) * 0.06
    dpad = (dmax - dmin) * 0.08

    option = {
        "animation": False,
        "backgroundColor": "#ffffff",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "color": "#222222"},
        "title": {
            "text": "TGA 시트 비교",
            "left": "center",
            "top": 12,
            "textStyle": {"fontSize": 17, "fontWeight": "normal"}
        },
        "legend": {
            "type": "scroll",
            "orient": "vertical",
            "left": 74,
            "top": "middle",
            "data": legend,
            "backgroundColor": "rgba(255,255,255,0.85)",
            "padding": [6, 8],
            "textStyle": {"fontSize": 11}
        },
        "grid": {"left": 78, "right": 76, "top": 62, "bottom": 62},
        "tooltip": {"trigger": "axis"},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 36,
            "min": xmin,
            "max": xmax,
            "axisLine": {"lineStyle": {"color": "#555555"}},
            "axisTick": {"show": True},
            "axisLabel": {"fontSize": 11},
            "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}}
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 52,
                "min": ymin - ypad,
                "max": ymax + ypad,
                "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                "axisLabel": {"fontSize": 11},
                "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}}
            },
            {
                "type": "value",
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 52,
                "min": dmin - dpad,
                "max": dmax + dpad,
                "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                "axisLabel": {"fontSize": 11},
                "splitLine": {"show": False}
            }
        ],
        "series": series
    }

    result = {
        "figures": [{
            "name": clean_name(os.path.splitext(os.path.basename(DATA_FILE))[0] + "_overlay"),
            "option": option,
            "width": 960,
            "height": 620,
            "dpi": 300,
            "formats": ["png"]
        }]
    }
    print(json.dumps(result, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
