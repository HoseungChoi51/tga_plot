```python
import json
import re
import sys
from pathlib import Path

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"

WEIGHT_COLOR = "#1f4e9c"
DTG_COLOR = "#c0392b"


def numeric_column(df, name):
    matches = [col for col in df.columns if str(col).strip().lower() == name.lower()]
    if not matches:
        return None
    return pd.to_numeric(df[matches[0]], errors="coerce").to_numpy(dtype=float)


def clean_number(value):
    return float(value) if np.isfinite(value) else None


def make_option(sheet_name, temperature, weight, dtg):
    valid = np.isfinite(temperature) & np.isfinite(weight)
    if not np.any(valid):
        return None

    temperature = temperature[valid]
    weight = weight[valid]

    dtg_valid = np.isfinite(dtg)
    if len(dtg) == len(valid):
        dtg = dtg[valid]
        dtg_valid = np.isfinite(dtg)
    else:
        dtg = np.full(len(temperature), np.nan)
        dtg_valid = np.zeros(len(temperature), dtype=bool)

    weight_data = [
        [clean_number(x), clean_number(y)]
        for x, y in zip(temperature, weight)
    ]
    dtg_data = [
        [clean_number(x), clean_number(-y)]
        for x, y in zip(temperature[dtg_valid], dtg[dtg_valid])
    ]

    # Residue: measured weight at the point nearest 800 °C.
    residue_index = int(np.argmin(np.abs(temperature - 800)))
    residue_x = float(temperature[residue_index])
    residue_y = float(weight[residue_index])

    # Peak: highest upward DTG peak above 150 °C (DTG is inverted for display).
    peak_x = None
    peak_y = None
    candidates = dtg_valid & (temperature > 150)
    if np.any(candidates):
        candidate_indices = np.flatnonzero(candidates)
        peak_index = int(candidate_indices[np.argmax(-dtg[candidates])])
        peak_x = float(temperature[peak_index])
        peak_y = float(-dtg[peak_index])

    series = [
        {
            "name": "Weight (mg)",
            "type": "line",
            "yAxisIndex": 0,
            "showSymbol": False,
            "connectNulls": False,
            "lineStyle": {"color": WEIGHT_COLOR, "width": 2},
            "itemStyle": {"color": WEIGHT_COLOR},
            "data": weight_data,
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 9,
                "itemStyle": {"color": WEIGHT_COLOR},
                "label": {
                    "show": True,
                    "formatter": f"Residue {residue_y:.1f}%",
                    "position": "top",
                    "color": WEIGHT_COLOR,
                },
                "data": [{"coord": [residue_x, residue_y]}],
            },
        },
        {
            "name": "DTG",
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "connectNulls": False,
            "lineStyle": {"color": DTG_COLOR, "width": 2},
            "itemStyle": {"color": DTG_COLOR},
            "data": dtg_data,
        },
    ]

    if peak_x is not None:
        series[1]["markPoint"] = {
            "symbol": "circle",
            "symbolSize": 9,
            "itemStyle": {"color": DTG_COLOR},
            "label": {
                "show": True,
                "formatter": f"{peak_x:.0f} °C",
                "position": "top",
                "color": DTG_COLOR,
            },
            "data": [{"coord": [peak_x, peak_y]}],
        }

    return {
        "title": {
            "text": str(sheet_name),
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "tooltip": {"trigger": "axis"},
        "legend": {"data": ["Weight (mg)", "DTG"], "left": "center", "top": 36},
        "grid": {"left": 72, "right": 72, "top": 82, "bottom": 58},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 34,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"show": True},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (mg)",
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True, "lineStyle": {"color": WEIGHT_COLOR}},
                "axisLabel": {"color": WEIGHT_COLOR},
                "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
            },
            {
                "type": "value",
                "name": "DTG (-%/°C)",
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True, "lineStyle": {"color": DTG_COLOR}},
                "axisLabel": {"color": DTG_COLOR},
                "splitLine": {"show": False},
            },
        ],
        "series": series,
    }


def main():
    path = Path(DATA_FILE)
    figures = []

    workbook = pd.ExcelFile(path)
    for sheet_name in workbook.sheet_names:
        df = pd.read_excel(path, sheet_name=sheet_name)

        temperature = numeric_column(df, "Temperature (°C)")
        weight = numeric_column(df, "Weight (mg)")
        dtg = numeric_column(df, "Deriv. Weight (%/°C)")

        if temperature is None or weight is None:
            continue
        if dtg is None:
            dtg = np.full(len(df), np.nan)

        option = make_option(sheet_name, temperature, weight, dtg)
        if option is None:
            continue

        safe_name = re.sub(r"[^A-Za-z0-9_-]+", "_", str(sheet_name)).strip("_") or "plot"
        figures.append(
            {
                "name": safe_name,
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        )

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
