import json
import os
import sys

import numpy as np
import pandas as pd
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/xrd.csv"


def main():
    df = pd.read_csv(DATA_FILE)
    x = pd.to_numeric(df.iloc[:, 0], errors="coerce").to_numpy(dtype=float)
    y = pd.to_numeric(df.iloc[:, 1], errors="coerce").to_numpy(dtype=float)

    valid = np.isfinite(x) & np.isfinite(y)
    x, y = x[valid], y[valid]
    order = np.argsort(x)
    x, y = x[order], y[order]

    # Find local maxima, then keep the five strongest distinct peaks.
    peaks, _ = find_peaks(y, distance=max(1, len(y) // 300))
    if len(peaks) < 5:
        peaks = np.argsort(y)[::-1]

    selected = []
    min_separation = max(1, len(y) // 100)
    for idx in sorted(peaks, key=lambda i: y[i], reverse=True):
        if all(abs(int(idx) - prev) >= min_separation for prev in selected):
            selected.append(int(idx))
        if len(selected) == 5:
            break

    # Fill remaining slots from strongest points if there are fewer than five local maxima.
    if len(selected) < min(5, len(y)):
        for idx in np.argsort(y)[::-1]:
            idx = int(idx)
            if all(abs(idx - prev) >= min_separation for prev in selected):
                selected.append(idx)
            if len(selected) == 5:
                break

    selected.sort(key=lambda i: x[i])
    peak_data = [
        {
            "coord": [float(x[i]), float(y[i])],
            "value": f"{x[i]:.2f}°",
            "itemStyle": {"color": "#c0392b"},
            "symbolSize": 7,
            "label": {
                "show": True,
                "position": "top",
                "formatter": f"{x[i]:.2f}°",
                "color": "#c0392b",
                "fontSize": 11,
                "fontFamily": "Noto Sans CJK KR",
            },
        }
        for i in selected
    ]

    option = {
        "animation": False,
        "title": {
            "text": "XRD Pattern",
            "left": "center",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16,
                "fontWeight": "normal",
            },
        },
        "tooltip": {"trigger": "axis"},
        "grid": {"left": 78, "right": 32, "top": 58, "bottom": 68},
        "xAxis": {
            "type": "value",
            "name": "2Theta (deg)",
            "nameLocation": "middle",
            "nameGap": 38,
            "min": float(np.min(x)),
            "max": float(np.max(x)),
            "axisLine": {"show": True, "lineStyle": {"color": "#444"}},
            "axisTick": {"show": True},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 12},
            "splitLine": {"show": False},
        },
        "yAxis": {
            "type": "value",
            "name": "Intensity (counts)",
            "nameLocation": "middle",
            "nameGap": 55,
            "min": 0,
            "axisLine": {"show": True, "lineStyle": {"color": "#444"}},
            "axisTick": {"show": True},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 12},
            "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
        },
        "series": [
            {
                "name": "Intensity",
                "type": "line",
                "data": [[float(a), float(b)] for a, b in zip(x, y)],
                "showSymbol": False,
                "symbol": "none",
                "lineStyle": {"color": "#1f4e9c", "width": 1.5},
                "itemStyle": {"color": "#1f4e9c"},
                "markPoint": {
                    "symbol": "circle",
                    "data": peak_data,
                },
            }
        ],
    }

    result = {
        "figures": [
            {
                "name": "xrd_pattern",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        ]
    }
    print(json.dumps(result, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
