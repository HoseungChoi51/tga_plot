import json
import math
import os
import pandas as pd
import numpy as np

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

df = pd.read_excel(DATA_FILE, sheet_name="PVC")
df.columns = [str(c).strip() for c in df.columns]

def find_col(tokens):
    for c in df.columns:
        if all(token.lower() in c.lower() for token in tokens):
            return c
    return None

xcol = find_col(["temperature"]) or df.columns[1]
wcol = find_col(["weight"]) or df.columns[2]
dcol = find_col(["deriv"]) or df.columns[3]

data = df[[xcol, wcol, dcol]].apply(pd.to_numeric, errors="coerce").dropna()
x = data[xcol].to_numpy(dtype=float)
weight = data[wcol].to_numpy(dtype=float)
dtg = data[dcol].to_numpy(dtype=float)

# TGA convention: invert the negative derivative so mass-loss peaks point upward.
dtg_plot = -dtg

residue_idx = int(np.nanargmin(np.abs(x - 800.0)))
residue_temp = float(x[residue_idx])
residue_value = float(weight[residue_idx])

peak_mask = x > 150.0
if np.any(peak_mask):
    candidate_indices = np.flatnonzero(peak_mask)
    peak_idx = int(candidate_indices[np.nanargmax(dtg_plot[peak_mask])])
else:
    peak_idx = int(np.nanargmax(dtg_plot))
peak_temp = float(x[peak_idx])
peak_value = float(dtg_plot[peak_idx])

def clean_num(v):
    return None if not math.isfinite(float(v)) else float(v)

weight_points = [[clean_num(a), clean_num(b)] for a, b in zip(x, weight)]
dtg_points = [[clean_num(a), clean_num(b)] for a, b in zip(x, dtg_plot)]
