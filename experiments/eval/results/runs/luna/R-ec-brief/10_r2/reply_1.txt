```python
import json
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

figures = []

sheets = pd.read_excel(DATA_FILE, sheet_name=None)
for sheet_name, df in sheets.items():
    df.columns = [str(c).strip() for c in df.columns]

    def find_column(candidates):
        for col in df.columns:
            if any(candidate.lower() in col.lower() for candidate in candidates):
                return col
        return None

    x_col = find_column(["temperature", "temp"])
    weight_col = find_column(["weight"])
    dtg_col = find_column(["d.w", "dtg", "derivative"])

    if x_col is None or weight_col is None or dtg_col is None:
        continue

    x = pd.to_numeric(df[x_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    dtg = pd.to_numeric(df[dtg_col], errors="coerce")
    valid = x.notna() & weight.notna() & dtg.notna()
    x, weight, dtg = x[valid], weight[valid], dtg[valid]

    if len(x) == 0:
        continue

    residue_idx = (x - 700).abs().idxmin()
    residue_x = float(x.loc[residue_idx])
    residue_y = float(weight.loc[residue_idx])

    peak_mask = x > 150
    if peak_mask.any():
        peak_idx = dtg[peak_mask].idxmin()
        peak_x = float(x.loc[peak_idx])
        peak_y = float(-dtg.loc[peak_idx])
    else:
        peak_x = float(x.iloc[0])
        peak_y = float(-dtg.iloc[0])

    x_values = x.astype(float).tolist()
    weight_values = weight.astype(float).tolist()
    dtg_values = (-dtg).astype(float).tolist()

    option = {
        "animation": False,
        "title": {
            "text": sheet_name,
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR"}
        },
        "tooltip": {"trigger": "axis"},
        "legend": {
            "data": ["Weight(%)", "-DTG(%/°C)"],
            "left": "center",
            "top": 36,
            "textStyle": {"fontFamily": "Noto Sans CJK KR"}
        },
        "grid": {"left": 72, "right": 72, "top": 82, "bottom": 62},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 36,
            "min": float(np.nanmin(x_values)),
            "max": float(np.nanmax(x_values)),
            "axisLine": {"lineStyle": {"color": "#666666"}},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}}
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 48,
                "min": 0,
                "max": 105,
                "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}}
            },
            {
                "type": "value",
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 50,
                "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight(%)",
                "type": "line",
                "yAxisIndex": 0,
                "showSymbol": False,
                "data": [[xx, yy] for xx, yy in zip(x_values, weight_values)],
                "lineStyle": {"color": "#1f4e9c", "width": 2},
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "itemStyle": {"color": "#1f4e9c"},
                    "label": {
                        "show": True,
                        "formatter": f"Residue {residue_y:.2f}%",
                        "position": "top",
                        "fontFamily": "Noto Sans CJK KR"
                    },
                    "data": [{"coord": [residue_x, residue_y]}]
                }
            },
            {
                "name": "-DTG(%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "showSymbol": False,
                "data": [[xx, yy] for xx, yy in zip(x_values, dtg_values)],
                "lineStyle": {"color": "#c0392b", "width": 2},
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "itemStyle": {"color": "#c0392b"},
                    "label": {
                        "show": True,
                        "formatter": f"{peak_x:.0f} °C",
                        "position": "top",
                        "fontFamily": "Noto Sans CJK KR"
                    },
                    "data": [{"coord": [peak_x, peak_y]}]
                }
            }
        ]
    }

    figures.append({
        "name": sheet_name.replace("/", "-").replace("\\", "-"),
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    })

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
```
