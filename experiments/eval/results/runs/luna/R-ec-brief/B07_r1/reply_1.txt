```python
import json
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

def clean_number(value):
    value = float(value)
    return None if not np.isfinite(value) else value

def main():
    xls = pd.ExcelFile(DATA_FILE)
    sheet = xls.sheet_names[1]
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)

    temperature = pd.to_numeric(df.iloc[:, 1], errors="coerce").to_numpy(dtype=float)
    weight = pd.to_numeric(df.iloc[:, 2], errors="coerce").to_numpy(dtype=float)
    valid = np.isfinite(temperature) & np.isfinite(weight)
    temperature, weight = temperature[valid], weight[valid]

    points = []
    for target in (300, 600):
        idx = int(np.argmin(np.abs(temperature - target)))
        points.append({
            "temperature": clean_number(temperature[idx]),
            "weight": clean_number(weight[idx]),
            "target": target,
        })

    option = {
        "title": {
            "text": sheet,
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
        },
        "tooltip": {"trigger": "axis"},
        "legend": {"data": ["Weight"], "left": "center", "top": 35},
        "grid": {"left": 70, "right": 30, "top": 85, "bottom": 60},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 38,
            "axisLine": {"lineStyle": {"color": "#555"}},
            "splitLine": {"show": False},
        },
        "yAxis": {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 48,
            "axisLine": {"show": True, "lineStyle": {"color": "#555"}},
            "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
        },
        "series": [
            {
                "name": "Weight",
                "type": "line",
                "showSymbol": False,
                "data": [
                    [clean_number(x), clean_number(y)]
                    for x, y in zip(temperature, weight)
                ],
                "lineStyle": {"color": "#1f4e9c", "width": 2},
                "itemStyle": {"color": "#1f4e9c"},
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "itemStyle": {"color": "#c0392b"},
                    "label": {
                        "show": True,
                        "position": "top",
                        "formatter": "{b}: {c}%",
                        "fontFamily": "Noto Sans CJK KR",
                        "fontSize": 11,
                    },
                    "data": [
                        {
                            "name": f"{point['target']} °C",
                            "coord": [point["temperature"], point["weight"]],
                            "value": point["weight"],
                        }
                        for point in points
                    ],
                },
            }
        ],
        "textStyle": {"fontFamily": "Noto Sans CJK KR"},
    }

    print(json.dumps({
        "figures": [{
            "name": "PVC",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"],
        }]
    }, ensure_ascii=False, allow_nan=False))

if __name__ == "__main__":
    main()
```
