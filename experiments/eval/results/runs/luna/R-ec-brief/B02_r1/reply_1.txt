```python
import json
import os
import re
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

df = pd.read_csv(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

def find_col(patterns):
    for pattern in patterns:
        for col in df.columns:
            if re.search(pattern, col, re.I):
                return col
    return None

x_col = find_col([r"temperature", r"temp", r"온도"]) or df.columns[1]
weight_col = find_col([r"weight\s*\(%", r"weight", r"중량", r"무게"]) or df.columns[2]
heat_col = find_col([r"heat\s*flow", r"heat", r"열흐름"]) or df.columns[4]

data = pd.DataFrame({
    "x": pd.to_numeric(df[x_col], errors="coerce"),
    "weight": pd.to_numeric(df[weight_col], errors="coerce"),
    "heat": pd.to_numeric(df[heat_col], errors="coerce"),
}).dropna().sort_values("x")

x = data["x"].to_numpy(dtype=float)
weight = data["weight"].to_numpy(dtype=float)
heat = data["heat"].to_numpy(dtype=float)

# Identify heat-flow extrema above 150 °C. Endothermic peaks are positive
# heat-flow maxima; exothermic peaks are negative minima.
try:
    from scipy.signal import find_peaks
    positive_peaks, _ = find_peaks(heat, prominence=max(np.ptp(heat) * 0.015, 1e-12))
    negative_peaks, _ = find_peaks(-heat, prominence=max(np.ptp(heat) * 0.015, 1e-12))
except Exception:
    positive_peaks = np.array([], dtype=int)
    negative_peaks = np.array([], dtype=int)

valid = x >= 150
endothermic = [i for i in positive_peaks if valid[i]]
exothermic = [i for i in negative_peaks if valid[i]]

# Ensure requested labels are produced even if peak-finding finds too few
# local extrema: select strongest positive and negative points in the range.
valid_indices = np.flatnonzero(valid)
if len(endothermic) < 2 and len(valid_indices):
    candidates = sorted(valid_indices, key=lambda i: heat[i], reverse=True)
    for i in candidates:
        if i not in endothermic and heat[i] > 0:
            endothermic.append(int(i))
        if len(endothermic) >= 2:
            break
if len(exothermic) < 1 and len(valid_indices):
    i = int(valid_indices[np.argmin(heat[valid_indices])])
    if heat[i] < 0:
        exothermic.append(i)

# Keep the two strongest endothermic peaks and strongest exothermic peak.
endothermic = sorted(endothermic, key=lambda i: heat[i], reverse=True)[:2]
exothermic = sorted(exothermic, key=lambda i: heat[i])[:1]
peak_points = [(i, "흡열") for i in endothermic] + [(i, "발열") for i in exothermic]
peak_points.sort(key=lambda item: x[item[0]])

residue_idx = int(np.argmin(np.abs(x - 800)))
residue_x = float(x[residue_idx])
residue_y = float(weight[residue_idx])

heat_min, heat_max = float(np.min(heat)), float(np.max(heat))
heat_pad = max((heat_max - heat_min) * 0.08, 0.02)
weight_min, weight_max = float(np.min(weight)), float(np.max(weight))
weight_pad = max((weight_max - weight_min) * 0.06, 1.0)

series = [
    {
        "name": "Weight (%)",
        "type": "line",
        "yAxisIndex": 0,
        "showSymbol": False,
        "lineStyle": {"width": 2, "color": "#1f4e9c"},
        "data": [[float(a), float(b)] for a, b in zip(x, weight)],
        "markPoint": {
            "symbol": "circle",
            "symbolSize": 9,
            "itemStyle": {"color": "#1f4e9c"},
            "label": {
                "show": True,
                "formatter": f"Residue {residue_y:.1f}%",
                "position": "top",
                "color": "#1f4e9c",
                "fontSize": 12,
            },
            "data": [{"coord": [residue_x, residue_y]}],
        },
    },
    {
        "name": "Heat Flow (W/g)",
        "type": "line",
        "yAxisIndex": 1,
        "showSymbol": False,
        "lineStyle": {"width": 2, "color": "#c0392b"},
        "data": [[float(a), float(b)] for a, b in zip(x, heat)],
    },
]

for idx, kind in peak_points:
    series[1].setdefault("markPoint", {
        "symbol": "circle",
        "symbolSize": 9,
        "itemStyle": {"color": "#c0392b"},
        "label": {
            "show": True,
            "position": "top",
            "color": "#c0392b",
            "fontSize": 12,
            "formatter": "{b}",
        },
        "data": [],
    })
    series[1]["markPoint"]["data"].append({
        "name": f"{kind} {x[idx]:.0f} °C",
        "coord": [float(x[idx]), float(heat[idx])],
    })

option = {
    "animation": False,
    "title": {
        "text": "caox",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
    },
    "tooltip": {"trigger": "axis"},
    "legend": {
        "data": ["Weight (%)", "Heat Flow (W/g)"],
        "left": "center",
        "bottom": 8,
        "textStyle": {"fontFamily": "Noto Sans CJK KR"},
    },
    "grid": {"left": 75, "right": 75, "top": 65, "bottom": 60},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 32,
        "min": float(np.min(x)),
        "max": float(np.max(x)),
        "axisLine": {"show": True},
        "axisTick": {"show": True},
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}},
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 52,
            "min": float(weight_min - weight_pad),
            "max": float(weight_max + weight_pad),
            "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR", "color": "#1f4e9c"},
            "splitLine": {"show": False},
        },
        {
            "type": "value",
            "name": "Heat Flow (W/g)",
            "nameLocation": "middle",
            "nameGap": 52,
            "min": float(heat_min - heat_pad),
            "max": float(heat_max + heat_pad),
            "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR", "color": "#c0392b"},
            "splitLine": {"show": False},
        },
    ],
    "series": series,
}

result = {
    "figures": [{
        "name": "caox",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"],
    }]
}
print(json.dumps(result, ensure_ascii=False, allow_nan=False))
```
