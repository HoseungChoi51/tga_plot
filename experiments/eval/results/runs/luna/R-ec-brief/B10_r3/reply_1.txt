import pandas as pd

# Identify the standard TGA columns by their header names.
cols = {str(c).strip().lower(): c for c in df.columns}
x_col = next(c for k, c in cols.items() if "temperature" in k)
weight_col = next(c for k, c in cols.items() if "weight" in k and "deriv" not in k)
dtg_col = next(c for k, c in cols.items() if "deriv" in k and "weight" in k)

x = pd.to_numeric(df[x_col], errors="coerce").to_numpy(dtype=float)
weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy(dtype=float)
dtg = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy(dtype=float)
valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
x, weight, dtg = x[valid], weight[valid], dtg[valid]

# TGA derivative is negative during mass loss; invert it so the DTG peaks point upward.
dtg_plot = -dtg

residue_idx = int(np.argmin(np.abs(x - 800.0)))
residue_x = float(x[residue_idx])
residue_y = float(weight[residue_idx])

peak_candidates = np.flatnonzero(x > 150.0)
peak_idx = int(peak_candidates[np.argmax(dtg_plot[peak_candidates])])
peak_x = float(x[peak_idx])
peak_y = float(dtg_plot[peak_idx])
