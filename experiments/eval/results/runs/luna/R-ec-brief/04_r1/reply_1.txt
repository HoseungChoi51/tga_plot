import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
SHEET_NAME = "1006R1-V normal"

def read_sheet(path, sheet_name):
    df = pd.read_excel(path, sheet_name=sheet_name)
    df.columns = [str(c).strip() for c in df.columns]
    return df

def find_column(df, candidates, fallback_index):
    for col in df.columns:
        if any(token.lower() in str(col).lower() for token in candidates):
            return col
    return df.columns[fallback_index]

df = read_sheet(DATA_FILE, SHEET_NAME)
x_col = find_column(df, ["temperature", "temp"], 1)
weight_col = find_column(df, ["weight", "mass"], 2)
dtg_col = find_column(df, ["d.w", "dtg", "derivative"], 3)

for col in (x_col, weight_col, dtg_col):
    df[col] = pd.to_numeric(df[col], errors="coerce")
df = df.dropna(subset=[x_col, weight_col, dtg_col]).sort_values(x_col)

x = df[x_col].to_numpy(dtype=float)
weight = df[weight_col].to_numpy(dtype=float)
dtg = df[dtg_col].to_numpy(dtype=float)

residue_index = int(np.argmin(np.abs(x - 800.0)))
residue_x = float(x[residue_index])
residue_y = float(weight[residue_index])
peak_mask = x > 150.0
if np.any(peak_mask):
    peak_index = int(np.flatnonzero(peak_mask)[np.argmax(-dtg[peak_mask])])
    peak_x = float(x[peak_index])
    peak_y = float(-dtg[peak_index])
else:
    peak_index = int(np.argmax(-dtg))
    peak_x = float(x[peak_index])
    peak_y = float(-dtg[peak_index])
