```python
import json
import math
import os
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"

def clean_number(value):
    try:
        number = float(value)
        return number if math.isfinite(number) else None
    except (TypeError, ValueError):
        return None

def main():
    workbook = pd.ExcelFile(DATA_FILE)
    figures = []

    for sheet_name in workbook.sheet_names:
        df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        df.columns = [str(c).strip() for c in df.columns]

        x_col = next((c for c in df.columns if "temperature" in c.lower()), None)
        y1_col = next((c for c in df.columns if "weight (mg)" in c.lower()), None)
        y2_col = next((c for c in df.columns if "deriv" in c.lower() and "weight" in c.lower()), None)
        if x_col is None or y1_col is None:
            continue

        x = pd.to_numeric(df[x_col], errors="coerce")
        y1 = pd.to_numeric(df[y1_col], errors="coerce")
        valid = x.notna() & y1.notna()
        x_vals = x[valid].to_numpy(dtype=float)
        y1_vals = y1[valid].to_numpy(dtype=float)
        if not len(x_vals):
            continue

        y1_data = [[clean_number(a), clean_number(b)] for a, b in zip(x_vals, y1_vals)]
        series = [{
            "name": y1_col,
            "type": "line",
            "showSymbol": False,
            "connectNulls": False,
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "itemStyle": {"color": "#1f4e9c"},
            "data": y1_data,
        }]
        y2_axis = []
        legend_data = [y1_col]

        if y2_col is not None:
            dy = pd.to_numeric(df[y2_col], errors="coerce")
            valid2 = x.notna() & dy.notna()
            x2 = x[valid2].to_numpy(dtype=float)
            dy2 = dy[valid2].to_numpy(dtype=float)
            peak = None
            peak_indices = np.where(x2 > 150)[0]
            if len(peak_indices):
                idx = peak_indices[np.argmax(-dy2[peak_indices])]
                peak = (float(x2[idx]), float(-dy2[idx]))
            y2_data = [[clean_number(a), clean_number(-b)] for a, b in zip(x2, dy2)]
            series.append({
                "name": y2_col + " (-DTG)",
                "type": "line",
                "yAxisIndex": 1,
                "showSymbol": False,
                "connectNulls": False,
                "lineStyle": {"color": "#c0392b", "width": 2},
                "itemStyle": {"color": "#c0392b"},
                "data": y2_data,
            })
            legend_data.append(y2_col + " (-DTG)")
            if peak is not None:
                series[1]["markPoint"] = {
                    "symbol": "circle",
                    "symbolSize": 8,
                    "itemStyle": {"color": "#c0392b"},
                    "label": {
                        "show": True,
                        "formatter": f"{peak[0]:.0f} °C",
                        "position": "top",
                        "color": "#c0392b",
                    },
                    "data": [{"coord": [peak[0], peak[1]]}],
                }
            y2_axis = [{
                "type": "value",
                "name": "DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                "axisLabel": {"color": "#444"},
                "splitLine": {"show": False},
            }]

        target_x = 800.0
        residue_idx = int(np.argmin(np.abs(x_vals - target_x)))
        residue_x = float(x_vals[residue_idx])
        residue_y = float(y1_vals[residue_idx])
        series[0]["markPoint"] = {
            "symbol": "circle",
            "symbolSize": 8,
            "itemStyle": {"color": "#1f4e9c"},
            "label": {
                "show": True,
                "formatter": f"Residue {residue_y:.1f} mg",
                "position": "right",
                "color": "#1f4e9c",
            },
            "data": [{"coord": [residue_x, residue_y]}],
        }

        option = {
            "animation": False,
            "title": {
                "text": str(sheet_name),
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
            },
            "tooltip": {"trigger": "axis"},
            "legend": {"data": legend_data, "left": "center", "top": 34},
            "grid": {"left": 72, "right": 72 if y2_axis else 28, "top": 82, "bottom": 58},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 34,
                "axisLine": {"show": True},
                "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (mg)",
                    "nameLocation": "middle",
                    "nameGap": 52,
                    "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                    "axisLabel": {"color": "#444"},
                    "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
                }
            ] + y2_axis,
            "series": series,
        }

        figures.append({
            "name": str(sheet_name),
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"],
        })

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))

if __name__ == "__main__":
    main()
```
