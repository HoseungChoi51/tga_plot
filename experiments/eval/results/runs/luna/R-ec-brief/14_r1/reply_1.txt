```python
import json
import os
import re
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/korean_excel.csv"

def clean(value):
    if pd.isna(value):
        return None
    try:
        return float(value)
    except (TypeError, ValueError):
        return str(value)

def main():
    df = pd.read_csv(DATA_FILE)
    df = df.dropna(how="all")

    def find_column(pattern, fallback):
        for col in df.columns:
            if re.search(pattern, str(col), re.IGNORECASE):
                return col
        return df.columns[fallback]

    x_col = find_column(r"온도|temp", 1)
    weight_col = find_column(r"무게|weight", 2)
    dtg_col = find_column(r"미분|dtg|derivative", 3)

    x = pd.to_numeric(df[x_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    dtg = pd.to_numeric(df[dtg_col], errors="coerce")
    valid = x.notna() & weight.notna() & dtg.notna()
    x = x[valid].to_numpy(dtype=float)
    weight = weight[valid].to_numpy(dtype=float)
    dtg = dtg[valid].to_numpy(dtype=float)

    if not len(x):
        raise ValueError("No valid numeric TGA data found.")

    residue_index = int(np.argmin(np.abs(x - 800.0)))
    peak_candidates = np.flatnonzero(x > 150.0)
    peak_index = int(peak_candidates[np.argmax(-dtg[peak_candidates])]) if len(peak_candidates) else int(np.argmax(-dtg))

    option = {
        "animation": False,
        "title": {
            "text": os.path.splitext(os.path.basename(DATA_FILE))[0],
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR"}
        },
        "tooltip": {"trigger": "axis"},
        "legend": {"data": ["무게(%)", "-미분무게(%/°C)"], "left": 20, "top": "middle"},
        "grid": {"left": 80, "right": 80, "top": 65, "bottom": 65},
        "xAxis": {
            "type": "value",
            "name": str(x_col),
            "nameLocation": "middle",
            "nameGap": 38,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"}
        },
        "yAxis": [
            {
                "type": "value",
                "name": str(weight_col),
                "nameLocation": "middle",
                "nameGap": 55,
                "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}}
            },
            {
                "type": "value",
                "name": "-미분무게(%/°C)",
                "nameLocation": "middle",
                "nameGap": 55,
                "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "무게(%)",
                "type": "line",
                "showSymbol": False,
                "data": [[clean(a), clean(b)] for a, b in zip(x, weight)],
                "lineStyle": {"width": 2, "color": "#1f4e9c"},
                "itemStyle": {"color": "#1f4e9c"},
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "label": {
                        "show": True,
                        "formatter": f"Residue {weight[residue_index]:.1f}%",
                        "position": "top",
                        "fontFamily": "Noto Sans CJK KR"
                    },
                    "data": [{"coord": [clean(x[residue_index]), clean(weight[residue_index])]}]
                }
            },
            {
                "name": "-미분무게(%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "showSymbol": False,
                "data": [[clean(a), clean(-b)] for a, b in zip(x, dtg)],
                "lineStyle": {"width": 2, "color": "#c0392b"},
                "itemStyle": {"color": "#c0392b"},
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "label": {
                        "show": True,
                        "formatter": f"{x[peak_index]:.0f} °C",
                        "position": "top",
                        "fontFamily": "Noto Sans CJK KR"
                    },
                    "data": [{"coord": [clean(x[peak_index]), clean(-dtg[peak_index])]}]
                }
            }
        ]
    }

    print(json.dumps({
        "figures": [{
            "name": "korean_excel",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["pdf", "png"]
        }]
    }, ensure_ascii=False, allow_nan=False))

if __name__ == "__main__":
    main()
```
