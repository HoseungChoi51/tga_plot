```python
import json
import re
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def clean(value):
    if pd.isna(value):
        return None
    try:
        number = float(value)
        return number if np.isfinite(number) else None
    except (TypeError, ValueError):
        return None

def label_name(name, fallback):
    return str(name).strip() if name is not None else fallback

figures = []
book = pd.ExcelFile(DATA_FILE)

for sheet_name in book.sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    if df.empty:
        continue

    names = [str(c).strip() for c in df.columns]
    def find_col(pattern, fallback):
        for i, name in enumerate(names):
            if re.search(pattern, name, re.I):
                return df.columns[i]
        return df.columns[fallback] if len(df.columns) > fallback else None

    x_col = find_col(r"temperature|temp", 1)
    y1_col = find_col(r"weight", 2)
    y2_col = find_col(r"d\.?\s*w|dtg|derivative", 3)
    if x_col is None or y1_col is None:
        continue

    x = pd.to_numeric(df[x_col], errors="coerce")
    y1 = pd.to_numeric(df[y1_col], errors="coerce")
    valid = x.notna() & y1.notna()
    x_values = x[valid].to_numpy(dtype=float)
    weight_values = y1[valid].to_numpy(dtype=float)
    if len(x_values) == 0:
        continue

    y2_values = None
    if y2_col is not None:
        dtg = pd.to_numeric(df[y2_col], errors="coerce")
        y2_values = (-dtg[valid]).to_numpy(dtype=float)

    weight_data = [[clean(a), clean(b)] for a, b in zip(x_values, weight_values)]
    dtg_data = (
        [[clean(a), clean(b)] for a, b in zip(x_values, y2_values)]
        if y2_values is not None else []
    )

    residue_idx = int(np.argmin(np.abs(x_values - 800)))
    residue_x = clean(x_values[residue_idx])
    residue_y = clean(weight_values[residue_idx])

    peak_idx = None
    if y2_values is not None:
        eligible = np.where((x_values > 150) & np.isfinite(y2_values))[0]
        if len(eligible):
            peak_idx = int(eligible[np.argmax(y2_values[eligible])])

    series = [
        {
            "name": label_name(y1_col, "Weight"),
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 0,
            "data": weight_data,
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#1f4e9c"},
                "label": {
                    "show": True,
                    "formatter": f"Residue {residue_y:.1f}%",
                    "position": "top"
                },
                "data": [{"coord": [residue_x, residue_y]}]
            }
        }
    ]

    if y2_values is not None:
        dtg_series = {
            "name": "DTG (-)",
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 1,
            "data": dtg_data,
            "lineStyle": {"color": "#c0392b", "width": 2}
        }
        if peak_idx is not None:
            peak_x = clean(x_values[peak_idx])
            peak_y = clean(y2_values[peak_idx])
            dtg_series["markPoint"] = {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#c0392b"},
                "label": {
                    "show": True,
                    "formatter": f"{peak_x:.0f} °C",
                    "position": "top"
                },
                "data": [{"coord": [peak_x, peak_y]}]
            }
        series.append(dtg_series)

    option = {
        "animation": False,
        "title": {
            "text": str(sheet_name),
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR"}
        },
        "legend": {
            "top": 32,
            "right": 12,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 9
            }
        },
        "tooltip": {"trigger": "axis"},
        "grid": {"left": 72, "right": 72, "top": 78, "bottom": 58},
        "xAxis": {
            "type": "value",
            "name": label_name(x_col, "Temperature"),
            "nameLocation": "middle",
            "nameGap": 34,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"}
        },
        "yAxis": [
            {
                "type": "value",
                "name": label_name(y1_col, "Weight (%)"),
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True},
                "axisTick": {"show": True},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}}
            },
            {
                "type": "value",
                "name": "DTG (-)",
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True},
                "axisTick": {"show": True},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": False}
            }
        ],
        "series": series
    }

    safe_name = re.sub(r"[^A-Za-z0-9_-]+", "_", str(sheet_name)).strip("_") or "plot"
    figures.append({
        "name": safe_name,
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    })

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
```
