```python
import json
import os
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

COLORS = ["#1f4e9c", "#c0392b", "#2e8b57"]


def main():
    sheets = pd.read_excel(DATA_FILE, sheet_name=None)
    series = []
    all_x = []

    for i, (name, df) in enumerate(sheets.items()):
        if df.shape[1] < 3:
            continue

        x = pd.to_numeric(df.iloc[:, 1], errors="coerce")
        weight = pd.to_numeric(df.iloc[:, 2], errors="coerce")
        valid = x.notna() & weight.notna()
        x_values = x[valid].to_numpy(dtype=float)
        y_values = weight[valid].to_numpy(dtype=float)

        if not len(x_values):
            continue

        # Stack curves at 20 percentage-point intervals while preserving each
        # curve's measured weight values.
        offset = i * 20.0
        all_x.extend(x_values.tolist())
        series.append({
            "name": str(name),
            "type": "line",
            "showSymbol": False,
            "connectNulls": False,
            "lineStyle": {"width": 2, "color": COLORS[i % len(COLORS)]},
            "data": [[float(xv), float(yv + offset)] for xv, yv in zip(x_values, y_values)],
        })

    if not series:
        raise ValueError("No usable temperature/weight data found.")

    x_min = min(all_x)
    x_max = max(all_x)
    option = {
        "animation": False,
        "title": {
            "text": "무게 곡선 (20% 오프셋)",
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
        },
        "tooltip": {"trigger": "axis"},
        "legend": {"data": [s["name"] for s in series], "left": "center", "bottom": 8},
        "grid": {"left": 72, "right": 30, "top": 58, "bottom": 72},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 34,
            "min": float(x_min),
            "max": float(x_max),
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "yAxis": {
            "type": "value",
            "name": "Weight (%) + offset",
            "nameLocation": "middle",
            "nameGap": 52,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
        },
        "series": series,
    }

    print(json.dumps({
        "figures": [{
            "name": "polymer_weight_offset",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"],
        }]
    }, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
