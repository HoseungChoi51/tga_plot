```python
import json
import os
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

COLORS = ["#1f4e9c", "#c0392b", "#2e8b57"]
OFFSET_STEP = 20.0


def clean_number(value):
    try:
        number = float(value)
        return number if np.isfinite(number) else None
    except (TypeError, ValueError):
        return None


def main():
    workbook = pd.ExcelFile(DATA_FILE)
    series = []
    legend = []

    for index, sheet_name in enumerate(workbook.sheet_names):
        df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        if df.shape[1] < 3:
            continue

        # TGA data: temperature in column B, weight (%) in column C.
        temperature = pd.to_numeric(df.iloc[:, 1], errors="coerce")
        weight = pd.to_numeric(df.iloc[:, 2], errors="coerce")
        valid = temperature.notna() & weight.notna()
        temperature = temperature[valid].to_numpy(dtype=float)
        weight = weight[valid].to_numpy(dtype=float)
        if len(temperature) == 0:
            continue

        offset = index * OFFSET_STEP
        data = [
            [clean_number(x), clean_number(y + offset)]
            for x, y in zip(temperature, weight)
        ]
        series.append({
            "name": str(sheet_name),
            "type": "line",
            "showSymbol": False,
            "connectNulls": False,
            "lineStyle": {"width": 2, "color": COLORS[index % len(COLORS)]},
            "itemStyle": {"color": COLORS[index % len(COLORS)]},
            "data": data,
        })
        legend.append(str(sheet_name))

    option = {
        "animation": False,
        "title": {
            "text": "세 시트 무게 곡선 (20% 간격 offset)",
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
        },
        "tooltip": {"trigger": "axis"},
        "legend": {
            "data": legend,
            "left": "center",
            "bottom": 8,
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11},
        },
        "grid": {"left": 72, "right": 32, "top": 64, "bottom": 82},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 34,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "yAxis": {
            "type": "value",
            "name": "Weight (%) + offset",
            "nameLocation": "middle",
            "nameGap": 52,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
        },
        "series": series,
    }

    result = {
        "figures": [{
            "name": "overlaid_weight_curves",
            "option": option,
            "width": 800,
            "height": 560,
            "dpi": 300,
            "formats": ["png"],
        }]
    }
    print(json.dumps(result, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
