```python
import json
import os
import re

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
SHEET_NAME = "1006R1-V normal"

def clean_name(value):
    return re.sub(r"\s+", " ", str(value)).strip()

def main():
    df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
    df.columns = [clean_name(c) for c in df.columns]

    def find_column(candidates):
        for col in df.columns:
            normalized = col.lower().replace(" ", "")
            if any(candidate in normalized for candidate in candidates):
                return col
        return None

    x_col = find_column(["temperature", "temp", "온도"])
    weight_col = find_column(["weight(%)", "weight%", "weight", "중량"])
    dtg_col = find_column(["d.w", "dtg", "derivative", "미분"])

    if x_col is None or weight_col is None or dtg_col is None:
        raise ValueError("Could not identify temperature, weight, and DTG columns.")

    x = pd.to_numeric(df[x_col], errors="coerce").to_numpy(dtype=float)
    weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy(dtype=float)
    dtg = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy(dtype=float)
    valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
    x, weight, dtg = x[valid], weight[valid], dtg[valid]

    if not len(x):
        raise ValueError("No valid numeric TGA data found.")

    weight_data = [[float(a), float(b)] for a, b in zip(x, weight)]
    dtg_data = [[float(a), float(-b)] for a, b in zip(x, dtg)]

    residue_index = int(np.argmin(np.abs(x - 800.0)))
    residue_x = float(x[residue_index])
    residue_y = float(weight[residue_index])

    peak_candidates = np.flatnonzero((x > 150.0) & (dtg < 0.0))
    if len(peak_candidates):
        peak_index = int(peak_candidates[np.argmax(-dtg[peak_candidates])])
    else:
        peak_candidates = np.flatnonzero(x > 150.0)
        peak_index = int(peak_candidates[np.argmax(-dtg[peak_candidates])]) if len(peak_candidates) else int(np.argmax(-dtg))
    peak_x = float(x[peak_index])
    peak_y = float(-dtg[peak_index])

    option = {
        "animation": False,
        "title": {
            "text": "CNT 산화 거동",
            "left": "center",
            "top": 12,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 18,
                "fontWeight": "normal",
                "color": "navy",
            },
        },
        "tooltip": {"trigger": "axis"},
        "legend": {
            "data": ["Weight(%)", "-DTG"],
            "left": 82,
            "top": "middle",
            "orient": "vertical",
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11},
        },
        "grid": {"left": 72, "right": 72, "top": 66, "bottom": 60},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 34,
            "min": float(np.nanmin(x)),
            "max": float(np.nanmax(x)),
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"show": True, "fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 48,
                "min": 0,
                "max": 105,
                "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                "axisTick": {"show": True},
                "axisLabel": {"show": True, "fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
            },
            {
                "type": "value",
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 48,
                "inverse": False,
                "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                "axisTick": {"show": True},
                "axisLabel": {"show": True, "fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": False},
            },
        ],
        "series": [
            {
                "name": "Weight(%)",
                "type": "line",
                "data": weight_data,
                "showSymbol": False,
                "smooth": False,
                "lineStyle": {"width": 2, "color": "#1f4e9c"},
                "yAxisIndex": 0,
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "data": [
                        {
                            "name": f"Residue {residue_y:.1f}%",
                            "coord": [residue_x, residue_y],
                            "itemStyle": {"color": "#1f4e9c"},
                            "label": {
                                "show": True,
                                "formatter": f"Residue {residue_y:.1f}%",
                                "position": "right",
                                "fontFamily": "Noto Sans CJK KR",
                                "fontSize": 11,
                            },
                        }
                    ],
                },
            },
            {
                "name": "-DTG",
                "type": "line",
                "data": dtg_data,
                "showSymbol": False,
                "smooth": False,
                "lineStyle": {"width": 2, "color": "#c0392b"},
                "yAxisIndex": 1,
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "data": [
                        {
                            "name": f"{peak_x:.0f} °C",
                            "coord": [peak_x, peak_y],
                            "itemStyle": {"color": "#c0392b"},
                            "label": {
                                "show": True,
                                "formatter": f"{peak_x:.0f} °C",
                                "position": "top",
                                "fontFamily": "Noto Sans CJK KR",
                                "fontSize": 11,
                            },
                        }
                    ],
                },
            },
        ],
    }

    result = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        ]
    }
    print(json.dumps(result, ensure_ascii=False, allow_nan=False))

if __name__ == "__main__":
    main()
```
