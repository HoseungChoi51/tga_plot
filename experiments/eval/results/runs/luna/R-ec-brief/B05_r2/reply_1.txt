```python
import json
import os
import re
import sys

import numpy as np
import pandas as pd

DATA_FILES = [
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_A.csv",
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_B.txt",
]

COLORS = ["#1f4e9c", "#287a4b"]
DTG_COLORS = ["#c0392b", "#e67e22"]


def read_data(path):
    ext = os.path.splitext(path)[1].lower()
    if ext == ".csv":
        df = pd.read_csv(path)
    elif ext in (".txt", ".tsv"):
        try:
            df = pd.read_csv(path, sep=None, engine="python")
            if df.shape[1] < 2:
                raise ValueError("Could not detect delimiters")
        except Exception:
            df = pd.read_csv(path, sep=r"\s+", engine="python")
    elif ext in (".xlsx", ".xls"):
        df = pd.read_excel(path)
    else:
        raise ValueError(f"Unsupported file type: {path}")

    if df.shape[1] < 4:
        raise ValueError(f"Expected at least four columns in {path}")

    # TGA columns: temperature, weight, derivative weight.
    return (
        pd.to_numeric(df.iloc[:, 1], errors="coerce").to_numpy(),
        pd.to_numeric(df.iloc[:, 2], errors="coerce").to_numpy(),
        pd.to_numeric(df.iloc[:, 3], errors="coerce").to_numpy(),
    )


series = []
legend = []
all_temps = []

for idx, path in enumerate(DATA_FILES):
    temperature, weight, dtg = read_data(path)
    valid = np.isfinite(temperature) & np.isfinite(weight) & np.isfinite(dtg)
    temperature, weight, dtg = temperature[valid], weight[valid], dtg[valid]
    if not len(temperature):
        continue

    label = f"Lot {'A' if idx == 0 else 'B'}"
    color = COLORS[idx % len(COLORS)]
    dtg_color = DTG_COLORS[idx % len(DTG_COLORS)]
    legend.extend([label, f"{label} DTG"])

    series.append({
        "name": label,
        "type": "line",
        "yAxisIndex": 0,
        "showSymbol": False,
        "connectNulls": False,
        "lineStyle": {"width": 2, "color": color},
        "itemStyle": {"color": color},
        "data": [[float(x), float(y)] for x, y in zip(temperature, weight)],
    })
    series.append({
        "name": f"{label} DTG",
        "type": "line",
        "yAxisIndex": 1,
        "showSymbol": False,
        "connectNulls": False,
        "lineStyle": {"width": 1.5, "color": dtg_color},
        "itemStyle": {"color": dtg_color},
        "data": [[float(x), float(-y)] for x, y in zip(temperature, dtg)],
    })

    all_temps.extend(temperature.tolist())

option = {
    "animation": False,
    "title": {
        "text": "TGA — Lot A / Lot B",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
    },
    "tooltip": {"trigger": "axis", "axisPointer": {"type": "cross"}},
    "legend": {
        "data": legend,
        "left": "center",
        "bottom": 4,
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11},
    },
    "grid": {"left": 70, "right": 70, "top": 65, "bottom": 80},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 32,
        "min": float(min(all_temps)) if all_temps else 0,
        "max": float(max(all_temps)) if all_temps else 900,
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 48,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
        },
        {
            "type": "value",
            "name": "−DTG (%/°C)",
            "nameLocation": "middle",
            "nameGap": 48,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": False},
        },
    ],
    "series": series,
}

print(json.dumps({
    "figures": [{
        "name": "tga_lot_comparison",
        "option": option,
        "width": 800,
        "height": 500,
        "dpi": 300,
        "formats": ["png"],
    }]
}, ensure_ascii=False, allow_nan=False))
```
