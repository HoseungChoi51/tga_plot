```python
import json
import re
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"


def safe_name(value):
    name = re.sub(r"[^A-Za-z0-9_-]+", "_", str(value)).strip("_")
    return name or "plot"


def main():
    workbook = pd.ExcelFile(DATA_FILE)
    figures = []

    for sheet in workbook.sheet_names:
        df = pd.read_excel(DATA_FILE, sheet_name=sheet)
        df.columns = [str(c).strip() for c in df.columns]

        def find_column(patterns, fallback):
            for pattern in patterns:
                for col in df.columns:
                    if re.search(pattern, col, re.I):
                        return col
            return df.columns[fallback] if len(df.columns) > fallback else None

        xcol = find_column([r"temperature", r"temp", r"온도"], 1)
        y1col = find_column([r"weight", r"mass", r"중량", r"무게"], 2)
        y2col = find_column([r"\bd\.?\s*w\b", r"dtg", r"derivative", r"미분"], 3)

        if xcol is None or y1col is None:
            continue

        cols = [xcol, y1col] + ([y2col] if y2col else [])
        data = df[cols].apply(pd.to_numeric, errors="coerce").dropna(subset=[xcol, y1col])
        if data.empty:
            continue

        x = data[xcol].to_numpy(dtype=float)
        weight = data[y1col].to_numpy(dtype=float)
        dtg = data[y2col].to_numpy(dtype=float) if y2col else np.zeros_like(x)

        valid_residue = np.isfinite(x) & np.isfinite(weight)
        if valid_residue.any():
            idx800 = np.flatnonzero(valid_residue)[np.argmin(np.abs(x[valid_residue] - 800))]
            residue_x = float(x[idx800])
            residue_y = float(weight[idx800])
        else:
            residue_x, residue_y = None, None

        valid_peak = np.isfinite(x) & np.isfinite(dtg) & (x > 150)
        if valid_peak.any():
            eligible = np.flatnonzero(valid_peak)
            peak_idx = eligible[np.argmax(-dtg[eligible])]
            peak_x = float(x[peak_idx])
            peak_y = float(-dtg[peak_idx])
        else:
            peak_x, peak_y = None, None

        weight_data = [[float(a), float(b)] for a, b in zip(x, weight)]
        dtg_data = [[float(a), float(-b)] for a, b in zip(x, dtg) if np.isfinite(a) and np.isfinite(b)]

        series = [
            {
                "name": str(y1col),
                "type": "line",
                "yAxisIndex": 0,
                "showSymbol": False,
                "smooth": False,
                "lineStyle": {"width": 2, "color": "#1f4e9c"},
                "data": weight_data,
            }
        ]
        if y2col:
            series.append({
                "name": str(y2col),
                "type": "line",
                "yAxisIndex": 1,
                "showSymbol": False,
                "smooth": False,
                "lineStyle": {"width": 2, "color": "#c0392b"},
                "data": dtg_data,
            })

        if residue_x is not None:
            series[0]["markPoint"] = {
                "symbol": "circle",
                "symbolSize": 9,
                "itemStyle": {"color": "#1f4e9c"},
                "label": {
                    "show": True,
                    "formatter": f"Residue {residue_y:.1f}%",
                    "position": "top",
                    "color": "#1f4e9c",
                },
                "data": [{"coord": [residue_x, residue_y]}],
            }
        if peak_x is not None and y2col:
            series[1]["markPoint"] = {
                "symbol": "circle",
                "symbolSize": 9,
                "itemStyle": {"color": "#c0392b"},
                "label": {
                    "show": True,
                    "formatter": f"{peak_x:.0f} °C",
                    "position": "top",
                    "color": "#c0392b",
                },
                "data": [{"coord": [peak_x, peak_y]}],
            }

        option = {
            "animation": False,
            "title": {
                "text": str(sheet),
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
            },
            "tooltip": {"trigger": "axis"},
            "legend": {"show": False},
            "grid": {"left": 75, "right": 75, "top": 65, "bottom": 65},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 38,
                "scale": False,
                "splitLine": {"show": True, "lineStyle": {"color": "#d9d9d9", "type": "solid"}},
                "axisLine": {"show": True},
                "axisTick": {"show": True},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "nameLocation": "middle",
                    "nameGap": 50,
                    "splitLine": {"show": True, "lineStyle": {"color": "#d9d9d9", "type": "solid"}},
                    "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                    "axisLabel": {"color": "#1f4e9c"},
                },
                {
                    "type": "value",
                    "name": "-DTG (%/°C)",
                    "nameLocation": "middle",
                    "nameGap": 50,
                    "splitLine": {"show": False},
                    "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                    "axisLabel": {"color": "#c0392b"},
                },
            ],
            "series": series,
        }

        figures.append({
            "name": safe_name(sheet),
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"],
        })

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
