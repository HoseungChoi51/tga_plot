def clean_name(value):
    return re.sub(r"\s+", " ", str(value).strip().lower())


def find_column(columns, patterns):
    normalized = [clean_name(c) for c in columns]
    for pattern in patterns:
        for i, name in enumerate(normalized):
            if re.search(pattern, name):
                return columns[i]
    return None


def nearest_index(values, target):
    return int(np.nanargmin(np.abs(np.asarray(values, dtype=float) - target)))


def calculate_onset(temperature, weight, dtg):
    """
    Estimate the extrapolated onset of the dominant mass-loss event above 150 °C.
    Use the tangent at the steepest point on the descending weight curve and
    intersect it with the pre-event weight baseline.
    """
    t = np.asarray(temperature, dtype=float)
    w = np.asarray(weight, dtype=float)
    d = np.asarray(dtg, dtype=float)

    valid = np.isfinite(t) & np.isfinite(w) & np.isfinite(d)
    t, w, d = t[valid], w[valid], d[valid]
    order = np.argsort(t)
    t, w, d = t[order], w[order], d[order]

    # The strongest negative DTG event above 150 °C defines the main loss step.
    eligible = np.flatnonzero(t > 150)
    if len(eligible) == 0:
        return None, None

    peak_i = eligible[np.argmin(d[eligible])]
    peak_t = t[peak_i]

    # Estimate a pre-event baseline from the preceding 20–60 °C window.
    baseline_mask = (t >= max(t[0], peak_t - 100)) & (t <= peak_t - 30)
    if baseline_mask.sum() < 3:
        baseline_mask = (t >= t[0]) & (t <= peak_t - 10)
    if baseline_mask.sum() == 0:
        return None, None
    baseline = float(np.nanmedian(w[baseline_mask]))

    # Find the steepest descending segment in the dominant event neighborhood.
    event_mask = (t >= max(150, peak_t - 100)) & (t <= peak_t + 30)
    event_indices = np.flatnonzero(event_mask)
    if len(event_indices) < 3:
        event_indices = np.arange(max(0, peak_i - 3), min(len(t), peak_i + 4))

    # DTG is reported per °C, so the most negative value approximates d(weight)/dT.
    slope_i = event_indices[np.argmin(d[event_indices])]
    slope = float(d[slope_i])
    point_t = float(t[slope_i])
    point_w = float(w[slope_i])

    if not np.isfinite(slope) or slope >= 0:
        return None, None

    onset_t = point_t + (baseline - point_w) / slope
    if not np.isfinite(onset_t) or onset_t > point_t or onset_t < t[0]:
        return None, None

    return float(onset_t), baseline
