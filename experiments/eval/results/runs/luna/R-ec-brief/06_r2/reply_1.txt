```python
import json
import re
import sys
from pathlib import Path

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"


def clean_name(value):
    name = re.sub(r"[^\w.-]+", "_", str(value).strip(), flags=re.UNICODE)
    return name.strip("_") or "plot"


def finite_pairs(x, y):
    result = []
    for xv, yv in zip(x, y):
        if pd.notna(xv) and pd.notna(yv) and np.isfinite(float(xv)) and np.isfinite(float(yv)):
            result.append([float(xv), float(yv)])
    return result


def make_option(sheet_name, frame):
    columns = {str(col).strip().lower(): col for col in frame.columns}

    def find_column(patterns, fallback):
        for col in frame.columns:
            normalized = str(col).strip().lower()
            if any(re.search(pattern, normalized) for pattern in patterns):
                return col
        return fallback

    x_col = find_column([r"temperature", r"temp"], frame.columns[1] if len(frame.columns) > 1 else frame.columns[0])
    weight_col = find_column([r"weight", r"mass"], frame.columns[2] if len(frame.columns) > 2 else frame.columns[-1])
    dtg_col = find_column([r"dtg", r"d\.?w", r"derivative"], frame.columns[3] if len(frame.columns) > 3 else None)

    x = pd.to_numeric(frame[x_col], errors="coerce")
    weight = pd.to_numeric(frame[weight_col], errors="coerce")
    dtg = pd.to_numeric(frame[dtg_col], errors="coerce") if dtg_col is not None else pd.Series(np.nan, index=frame.index)

    weight_data = finite_pairs(x, weight)
    dtg_data = finite_pairs(x, -dtg)

    residue_label = None
    residue_marker = None
    valid_weight = pd.DataFrame({"x": x, "y": weight}).dropna()
    if not valid_weight.empty:
        idx = (valid_weight["x"] - 800).abs().idxmin()
        residue_x = float(valid_weight.loc[idx, "x"])
        residue_y = float(valid_weight.loc[idx, "y"])
        residue_label = f"Residue {residue_y:.1f}%"
        residue_marker = [residue_x, residue_y]

    peak_label = None
    peak_marker = None
    valid_dtg = pd.DataFrame({"x": x, "y": -dtg}).dropna()
    valid_dtg = valid_dtg[valid_dtg["x"] > 150]
    if not valid_dtg.empty:
        idx = valid_dtg["y"].idxmax()
        peak_x = float(valid_dtg.loc[idx, "x"])
        peak_y = float(valid_dtg.loc[idx, "y"])
        peak_label = f"{peak_x:.0f} °C"
        peak_marker = [peak_x, peak_y]

    series = [
        {
            "name": str(weight_col),
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 0,
            "data": weight_data,
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "itemStyle": {"color": "#1f4e9c"},
        }
    ]
    if dtg_col is not None:
        series.append(
            {
                "name": str(dtg_col),
                "type": "line",
                "showSymbol": False,
                "yAxisIndex": 1,
                "data": dtg_data,
                "lineStyle": {"color": "#c0392b", "width": 2},
                "itemStyle": {"color": "#c0392b"},
            }
        )

    if residue_marker is not None:
        series[0]["markPoint"] = {
            "symbol": "circle",
            "symbolSize": 9,
            "itemStyle": {"color": "#1f4e9c"},
            "label": {"show": True, "formatter": residue_label, "position": "top"},
            "data": [{"coord": residue_marker}],
        }
    if peak_marker is not None and dtg_col is not None:
        series[1]["markPoint"] = {
            "symbol": "circle",
            "symbolSize": 9,
            "itemStyle": {"color": "#c0392b"},
            "label": {"show": True, "formatter": peak_label, "position": "top"},
            "data": [{"coord": peak_marker}],
        }

    return {
        "title": {"text": str(sheet_name), "left": "center"},
        "tooltip": {"trigger": "axis"},
        "legend": {"show": False},
        "grid": {"left": 70, "right": 70, "top": 60, "bottom": 65},
        "xAxis": {
            "type": "value",
            "name": str(x_col),
            "nameLocation": "middle",
            "nameGap": 35,
            "splitLine": {"show": True, "lineStyle": {"color": "#d9d9d9"}},
        },
        "yAxis": [
            {
                "type": "value",
                "name": str(weight_col),
                "nameLocation": "middle",
                "nameGap": 48,
                "splitLine": {"show": True, "lineStyle": {"color": "#d9d9d9"}},
            },
            {
                "type": "value",
                "name": str(dtg_col) if dtg_col is not None else "DTG",
                "nameLocation": "middle",
                "nameGap": 48,
                "position": "right",
                "splitLine": {"show": False},
            },
        ],
        "series": series,
    }


def main():
    path = Path(DATA_FILE)
    workbook = pd.ExcelFile(path)
    figures = []

    for sheet_name in workbook.sheet_names:
        frame = pd.read_excel(path, sheet_name=sheet_name)
        frame = frame.dropna(how="all")
        figures.append(
            {
                "name": clean_name(sheet_name),
                "option": make_option(sheet_name, frame),
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        )

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
