```python
import json
import os
import re
import numpy as np
import pandas as pd
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

df = pd.read_csv(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

def find_column(patterns):
    for pattern in patterns:
        for col in df.columns:
            if re.search(pattern, col, re.IGNORECASE):
                return col
    return None

x_col = find_column([r"temperature", r"temp"])
weight_col = find_column([r"weight\s*\(%"])
heat_col = find_column([r"heat\s*flow", r"heatflow"])

if x_col is None or weight_col is None or heat_col is None:
    raise ValueError("CSV에 Temperature, Weight, Heat Flow 열이 필요합니다.")

data = df[[x_col, weight_col, heat_col]].apply(pd.to_numeric, errors="coerce").dropna()
data = data.sort_values(x_col)
x = data[x_col].to_numpy(dtype=float)
weight = data[weight_col].to_numpy(dtype=float)
heat = data[heat_col].to_numpy(dtype=float)

# Heat-flow extrema: positive peaks are exothermic, negative peaks endothermic.
# Select two strongest endothermic peaks and the strongest exothermic peak.
prominence = max(float(np.nanstd(heat)) * 0.08, 1e-9)
positive_peaks, _ = find_peaks(heat, prominence=prominence)
negative_peaks, _ = find_peaks(-heat, prominence=prominence)

# Ignore endpoints and weak local fluctuations by ranking peak prominence/height.
endothermic = sorted(negative_peaks, key=lambda i: heat[i])[:2]
exothermic = sorted(positive_peaks, key=lambda i: heat[i], reverse=True)[:1]

# If smoothing/noise means fewer local maxima were found, use ranked interior extrema.
interior = np.arange(1, max(len(heat) - 1, 1))
if len(endothermic) < 2 and len(heat) > 2:
    candidates = [int(i) for i in interior if i not in endothermic]
    endothermic += sorted(candidates, key=lambda i: heat[i])[: 2 - len(endothermic)]
if len(exothermic) < 1 and len(heat) > 2:
    candidates = [int(i) for i in interior if i not in exothermic]
    exothermic += sorted(candidates, key=lambda i: heat[i], reverse=True)[:1 - len(exothermic)]

peak_specs = [
    (i, "흡열 피크", "#c0392b") for i in endothermic[:2]
] + [
    (i, "발열 피크", "#1f77b4") for i in exothermic[:1]
]
peak_specs.sort(key=lambda item: x[item[0]])

residue_idx = int(np.argmin(np.abs(x - 800.0)))
residue_x = float(x[residue_idx])
residue_y = float(weight[residue_idx])

series = [
    {
        "name": "Weight (%)",
        "type": "line",
        "yAxisIndex": 0,
        "showSymbol": False,
        "lineStyle": {"width": 2, "color": "#1f4e9c"},
        "data": [[float(a), float(b)] for a, b in zip(x, weight)],
        "markPoint": {
            "symbol": "circle",
            "symbolSize": 9,
            "itemStyle": {"color": "#1f4e9c"},
            "label": {
                "show": True,
                "formatter": f"Residue {residue_y:.1f}%",
                "position": "top",
                "color": "#1f4e9c",
                "fontSize": 12,
            },
            "data": [{"coord": [residue_x, residue_y]}],
        },
    },
    {
        "name": "Heat Flow (W/g)",
        "type": "line",
        "yAxisIndex": 1,
        "showSymbol": False,
        "lineStyle": {"width": 2, "color": "#c0392b"},
        "data": [[float(a), float(b)] for a, b in zip(x, heat)],
        "markPoint": {
            "symbol": "circle",
            "symbolSize": 9,
            "data": [
                {
                    "coord": [float(x[i]), float(heat[i])],
                    "name": label,
                    "itemStyle": {"color": color},
                    "label": {
                        "show": True,
                        "formatter": f"{float(x[i]):.0f} °C",
                        "position": "top" if heat[i] >= 0 else "bottom",
                        "color": color,
                        "fontSize": 12,
                    },
                }
                for i, label, color in peak_specs
            ],
        },
    },
]

option = {
    "animation": False,
    "title": {
        "text": "caox",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
    },
    "tooltip": {"trigger": "axis"},
    "legend": {"data": ["Weight (%)", "Heat Flow (W/g)"], "left": "center"},
    "grid": {"left": 76, "right": 76, "top": 76, "bottom": 62},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 36,
        "axisLine": {"show": True},
        "axisTick": {"show": True},
        "axisLabel": {"show": True, "fontFamily": "Noto Sans CJK KR"},
        "splitLine": {"show": False},
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 52,
            "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
            "axisTick": {"show": True},
            "axisLabel": {"show": True, "fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
        },
        {
            "type": "value",
            "name": "Heat Flow (W/g)",
            "nameLocation": "middle",
            "nameGap": 56,
            "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
            "axisTick": {"show": True},
            "axisLabel": {"show": True, "fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": False},
        },
    ],
    "series": series,
}

result = {
    "figures": [
        {
            "name": "caox",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"],
        }
    ]
}
print(json.dumps(result, ensure_ascii=False, allow_nan=False))
```
