```python
import json
import math
import os
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

def clean(value):
    if pd.isna(value):
        return None
    value = float(value)
    return value if math.isfinite(value) else None

df = pd.read_excel(DATA_FILE, sheet_name="PET", engine="openpyxl")
df.columns = [str(c).strip() for c in df.columns]

temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
dtg = pd.to_numeric(df["Deriv. Weight (%/°C)"], errors="coerce")

valid = temperature.notna() & weight.notna() & dtg.notna()
temperature = temperature[valid].reset_index(drop=True)
weight = weight[valid].reset_index(drop=True)
dtg = dtg[valid].reset_index(drop=True)

residue_idx = (temperature - 800).abs().idxmin()
residue_temp = float(temperature.iloc[residue_idx])
residue_weight = float(weight.iloc[residue_idx])

peak_candidates = (temperature > 150) & dtg.notna()
if peak_candidates.any():
    peak_idx = dtg[peak_candidates].idxmin()
    peak_temp = float(temperature.iloc[peak_idx])
    peak_value = float(-dtg.iloc[peak_idx])
else:
    peak_temp = None
    peak_value = None

series = [
    {
        "name": "Weight",
        "type": "line",
        "yAxisIndex": 0,
        "showSymbol": False,
        "connectNulls": False,
        "lineStyle": {"color": "#1f4e9c", "width": 2},
        "data": [[clean(x), clean(y)] for x, y in zip(temperature, weight)],
        "markPoint": {
            "symbol": "circle",
            "symbolSize": 9,
            "itemStyle": {"color": "#1f4e9c"},
            "data": [{
                "coord": [residue_temp, residue_weight],
                "name": "Residue",
                "label": {
                    "show": True,
                    "formatter": f"Residue {residue_weight:.2f}%",
                    "position": "top",
                    "color": "#1f4e9c"
                }
            }]
        }
    },
    {
        "name": "-DTG",
        "type": "line",
        "yAxisIndex": 1,
        "showSymbol": False,
        "connectNulls": False,
        "lineStyle": {"color": "#c0392b", "width": 2},
        "data": [[clean(x), clean(-y)] for x, y in zip(temperature, dtg)],
        "markPoint": {
            "symbol": "circle",
            "symbolSize": 9,
            "itemStyle": {"color": "#c0392b"},
            "data": ([{
                "coord": [peak_temp, peak_value],
                "name": "DTG peak",
                "label": {
                    "show": True,
                    "formatter": f"{peak_temp:.0f} °C",
                    "position": "top",
                    "color": "#c0392b"
                }
            }] if peak_temp is not None else [])
        }
    }
]

option = {
    "title": {
        "text": "PET 열분해",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR"}
    },
    "tooltip": {"trigger": "axis"},
    "legend": {"data": ["Weight", "-DTG"], "left": 12, "bottom": 10},
    "grid": {"left": 70, "right": 70, "top": 65, "bottom": 65},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 35,
        "splitLine": {"show": True}
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 48,
            "splitLine": {"show": True}
        },
        {
            "type": "value",
            "name": "-DTG (%/°C)",
            "nameLocation": "middle",
            "nameGap": 48,
            "position": "right",
            "splitLine": {"show": False}
        }
    ],
    "series": series
}

print(json.dumps({
    "figures": [{
        "name": "PET",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }]
}, ensure_ascii=False, allow_nan=False))
```
