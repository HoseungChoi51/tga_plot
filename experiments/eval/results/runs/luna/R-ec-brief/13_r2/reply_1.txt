```python
import json
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"

def clean_number(value):
    return None if pd.isna(value) or not np.isfinite(float(value)) else float(value)

workbook = pd.ExcelFile(DATA_FILE)
figures = []

for sheet in workbook.sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    df.columns = [str(c).strip() for c in df.columns]

    x_col = next((c for c in df.columns if "temperature" in c.lower()), None)
    y1_col = next((c for c in df.columns if "weight (mg)" in c.lower()), None)
    y2_col = next((c for c in df.columns if "deriv" in c.lower() and "weight" in c.lower()), None)
    if not (x_col and y1_col and y2_col):
        continue

    data = df[[x_col, y1_col, y2_col]].apply(pd.to_numeric, errors="coerce").dropna()
    if data.empty:
        continue

    x = data[x_col].to_numpy(dtype=float)
    y1 = data[y1_col].to_numpy(dtype=float)
    dtg = data[y2_col].to_numpy(dtype=float)
    y2 = -dtg

    residue_idx = int(np.argmin(np.abs(x - 800.0)))
    peak_candidates = np.flatnonzero(x > 150.0)
    peak_idx = int(peak_candidates[np.argmax(y2[peak_candidates])]) if len(peak_candidates) else None

    series = [
        {
            "name": "Weight (mg)",
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 0,
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "data": [[clean_number(a), clean_number(b)] for a, b in zip(x, y1)],
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#1f4e9c"},
                "label": {
                    "show": True,
                    "formatter": f"Residue {y1[residue_idx]:.1f} mg",
                    "position": "bottom",
                    "color": "#1f4e9c"
                },
                "data": [{"coord": [clean_number(x[residue_idx]), clean_number(y1[residue_idx])]}]
            }
        },
        {
            "name": "DTG (-)",
            "type": "line",
            "showSymbol": False,
            "yAxisIndex": 1,
            "lineStyle": {"color": "#c0392b", "width": 2},
            "data": [[clean_number(a), clean_number(b)] for a, b in zip(x, y2)]
        }
    ]

    if peak_idx is not None:
        series[1]["markPoint"] = {
            "symbol": "circle",
            "symbolSize": 8,
            "itemStyle": {"color": "#c0392b"},
            "label": {
                "show": True,
                "formatter": f"{x[peak_idx]:.0f} °C",
                "position": "top",
                "color": "#c0392b"
            },
            "data": [{"coord": [clean_number(x[peak_idx]), clean_number(y2[peak_idx])]}]
        }

    option = {
        "animation": False,
        "title": {"text": sheet, "left": "center"},
        "tooltip": {"trigger": "axis"},
        "legend": {"left": 12, "top": "middle", "orient": "vertical"},
        "grid": {"left": 90, "right": 90, "top": 65, "bottom": 65},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 35,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"show": True}
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (mg)",
                "nameLocation": "middle",
                "nameGap": 55,
                "axisLine": {"show": True},
                "axisTick": {"show": True},
                "axisLabel": {"show": True},
                "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}}
            },
            {
                "type": "value",
                "name": "DTG (-%/°C)",
                "nameLocation": "middle",
                "nameGap": 55,
                "axisLine": {"show": True},
                "axisTick": {"show": True},
                "axisLabel": {"show": True},
                "splitLine": {"show": False}
            }
        ],
        "series": series
    }

    figures.append({
        "name": f"tga_{sheet}",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    })

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
```
