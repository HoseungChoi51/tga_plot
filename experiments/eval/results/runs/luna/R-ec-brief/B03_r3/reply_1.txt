```python
import json
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/rheology.csv"

df = pd.read_csv(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

x_col = "Angular frequency (rad/s)"
y1_col = "Storage modulus G' (Pa)"
y2_col = "Loss modulus G'' (Pa)"

data1 = []
data2 = []
for _, row in df.iterrows():
    try:
        x = float(row[x_col])
        y1 = float(row[y1_col])
        y2 = float(row[y2_col])
    except (TypeError, ValueError, KeyError):
        continue
    if x > 0 and y1 > 0:
        data1.append([x, y1])
    if x > 0 and y2 > 0:
        data2.append([x, y2])

option = {
    "title": {
        "text": "Rheology",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR"}
    },
    "tooltip": {"trigger": "axis"},
    "legend": {"data": ["G'", "G''"], "left": "center", "bottom": 8},
    "grid": {"left": 76, "right": 76, "top": 60, "bottom": 76},
    "xAxis": {
        "type": "log",
        "name": "Angular frequency (rad/s)",
        "nameLocation": "middle",
        "nameGap": 34,
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"}
    },
    "yAxis": [
        {
            "type": "log",
            "name": "Storage modulus G' (Pa)",
            "nameLocation": "middle",
            "nameGap": 56,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"}
        },
        {
            "type": "log",
            "name": "Loss modulus G'' (Pa)",
            "nameLocation": "middle",
            "nameGap": 56,
            "position": "right",
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"}
        }
    ],
    "series": [
        {
            "name": "G'",
            "type": "line",
            "data": data1,
            "yAxisIndex": 0,
            "showSymbol": False,
            "lineStyle": {"color": "#1f4e9c", "width": 2}
        },
        {
            "name": "G''",
            "type": "line",
            "data": data2,
            "yAxisIndex": 1,
            "showSymbol": False,
            "lineStyle": {"color": "#c0392b", "width": 2}
        }
    ]
}

print(json.dumps({
    "figures": [{
        "name": "rheology",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }]
}, ensure_ascii=False, allow_nan=False))
```
