weight_col = next((c for c in df.columns if "weight" in c.lower() and "deriv" not in c.lower()), df.columns[2])
dtg_col = next((c for c in df.columns if "deriv" in c.lower()), df.columns[3])

dtg = pd.to_numeric(df[dtg_col], errors="coerce")
valid = temperature.notna() & weight.notna() & dtg.notna()
temperature = temperature[valid].to_numpy()
weight = weight[valid].to_numpy()
dtg = dtg[valid].to_numpy()

order = np.argsort(temperature)
temperature, weight, dtg = temperature[order], weight[order], dtg[order]

residue_idx = int(np.argmin(np.abs(temperature - 800)))

residue_y = float(weight[residue_idx])
