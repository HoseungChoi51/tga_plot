```python
import json
import os
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
SHEET = "Annealed"

def main():
    df = pd.read_excel(DATA_FILE, sheet_name=SHEET)
    df.columns = [str(c).strip() for c in df.columns]

    temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce").to_numpy()
    weight = pd.to_numeric(df["Weight (%)"], errors="coerce").to_numpy()
    derivative = pd.to_numeric(df["Deriv. Weight (%/°C)"], errors="coerce").to_numpy()

    valid = np.isfinite(temperature) & np.isfinite(weight) & np.isfinite(derivative)
    temperature, weight, derivative = temperature[valid], weight[valid], derivative[valid]
    order = np.argsort(temperature)
    temperature, weight, derivative = temperature[order], weight[order], derivative[order]

    # Estimate the main mass-loss onset by intersecting the pre-event baseline
    # with the tangent at the steepest mass-loss point.
    smoothed = pd.Series(derivative).rolling( nine := 9, center=True, min_periods=1).mean().to_numpy()
    peak_index = int(np.argmin(smoothed))
    peak_temp = float(temperature[peak_index])

    baseline_mask = temperature <= peak_temp * 0.65
    if baseline_mask.sum() < 2:
        baseline_mask = temperature <= temperature[min(peak_index, max(1, len(temperature) // 5))]
    baseline_slope, baseline_intercept = np.polyfit(temperature[baseline_mask], weight[baseline_mask], 1)

    half_window = max(2, min(8, len(temperature) // 30))
    lo, hi = max(0, peak_index - half_window), min(len(temperature), peak_index + half_window + 1)
    tangent_slope, tangent_intercept = np.polyfit(temperature[lo:hi], weight[lo:hi], 1)

    denominator = tangent_slope - baseline_slope
    onset = (
        (baseline_intercept - tangent_intercept) / denominator
        if abs(denominator) > 1e-12
        else peak_temp
    )
    if not np.isfinite(onset) or onset < temperature[0] or onset > peak_temp:
        onset = peak_temp

    weight_data = [[float(x), float(y)] for x, y in zip(temperature, weight)]
    derivative_data = [[float(x), float(y)] for x, y in zip(temperature, derivative)]
    baseline_data = [
        [float(x), float(baseline_slope * x + baseline_intercept)]
        for x in temperature
        if onset <= x <= peak_temp
    ]
    tangent_data = [
        [float(x), float(tangent_slope * x + tangent_intercept)]
        for x in temperature
        if onset <= x <= peak_temp
    ]

    option = {
        "animation": False,
        "title": {
            "text": "Annealed TGA: Weight and Derivative",
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "tooltip": {"trigger": "axis", "axisPointer": {"type": "cross"}},
        "legend": {"top": 32, "data": ["Weight", "Derivative", "Onset baseline", "Peak tangent"]},
        "grid": {"left": 76, "right": 76, "top": 82, "bottom": 66},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 38,
            "min": float(np.min(temperature)),
            "max": float(np.max(temperature)),
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 48,
                "min": 0,
                "max": 105,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            },
            {
                "type": "value",
                "name": "Deriv. Weight (%/°C)",
                "nameLocation": "middle",
                "nameGap": 54,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            },
        ],
        "series": [
            {
                "name": "Weight",
                "type": "line",
                "showSymbol": False,
                "lineStyle": {"width": 2, "color": "#2878B5"},
                "data": weight_data,
            },
            {
                "name": "Derivative",
                "type": "line",
                "yAxisIndex": 1,
                "showSymbol": False,
                "lineStyle": {"width": 1.6, "color": "#D9534F"},
                "data": derivative_data,
            },
            {
                "name": "Onset baseline",
                "type": "line",
                "showSymbol": False,
                "yAxisIndex": 0,
                "lineStyle": {"width": 1.5, "type": "dashed", "color": "#2878B5"},
                "data": baseline_data,
            },
            {
                "name": "Peak tangent",
                "type": "line",
                "showSymbol": False,
                "yAxisIndex": 0,
                "lineStyle": {"width": 1.5, "type": "dashed", "color": "#E67E22"},
                "data": tangent_data,
                "markLine": {
                    "symbol": ["none", "none"],
                    "lineStyle": {"color": "#222222", "type": "dashed", "width": 1.5},
                    "label": {
                        "show": True,
                        "formatter": f"Onset: {onset:.1f} °C",
                        "position": "insideEndTop",
                    },
                    "data": [{"xAxis": float(onset)}],
                },
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 9,
                    "itemStyle": {"color": "#222222"},
                    "label": {
                        "show": True,
                        "formatter": f"{onset:.1f} °C",
                        "position": "top",
                    },
                    "data": [{"coord": [float(onset), float(baseline_slope * onset + baseline_intercept)]}],
                },
            },
        ],
    }

    result = {
        "figures": [
            {
                "name": "annealed_onset",
                "option": option,
                "width": 900,
                "height": 560,
                "dpi": 300,
                "formats": ["png"],
            }
        ]
    }
    print(json.dumps(result, ensure_ascii=False, allow_nan=False))

if __name__ == "__main__":
    main()
```
