import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/isothermal.txt"

df = pd.read_csv(DATA_FILE, sep=None, engine="python", comment="#")
df.columns = [str(c).strip() for c in df.columns]

# Preview indicates three columns: time (min), temperature (°C), and weight (%).
time_col, temp_col, weight_col = df.columns[:3]
for col in (time_col, temp_col, weight_col):
    df[col] = pd.to_numeric(df[col], errors="coerce")
df = df.dropna(subset=[time_col, temp_col, weight_col])

time = df[time_col].tolist()
temperature = df[temp_col].tolist()
weight = df[weight_col].tolist()

at_140 = df.loc[df[time_col].sub(140).abs().idxmin()]
weight_140 = float(at_140[weight_col])
time_140 = float(at_140[time_col])

option = {

    "tooltip": {"trigger": "axis"},

        "axisLabel": {"color": "#333"},
    },
    "yAxis": [
        {
            "type": "value",
            "name": "무게 (%)",
            "position": "left",
            "axisLine": {"show": True, "lineStyle": {"color": "#2563eb"}},
