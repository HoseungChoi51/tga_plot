import json

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"

# Read Annealed TGA data.
df = pd.read_excel(DATA_FILE, sheet_name="Annealed")
df.columns = [str(c).strip() for c in df.columns]

temperature_col = next(c for c in df.columns if "Temperature" in c)
weight_col = next(c for c in df.columns if "Weight" in c and "Deriv" not in c)
deriv_col = next(c for c in df.columns if "Deriv" in c)

data = df[[temperature_col, weight_col, deriv_col]].apply(pd.to_numeric, errors="coerce").dropna()
data = data.sort_values(temperature_col).drop_duplicates(subset=temperature_col)
temperature = data[temperature_col].to_numpy(dtype=float)
weight = data[weight_col].to_numpy(dtype=float)
derivative = data[deriv_col].to_numpy(dtype=float)

# Determine the principal mass-loss event from the most negative derivative.
peak_idx = int(np.argmin(derivative))
peak_temp = temperature[peak_idx]

# Fit a local tangent to the weight curve around the DTG minimum.
# Intersect the tangent with the initial-mass baseline to estimate onset.
local = (temperature >= peak_temp - 15) & (temperature <= peak_temp + 15)
if local.sum() < 2:
    lo = max(0, peak_idx - 5)
    hi = min(len(temperature), peak_idx + 6)
    local = np.zeros(len(temperature), dtype=bool)
    local[lo:hi] = True

slope, intercept = np.polyfit(temperature[local], weight[local], 1)
baseline = float(np.nanmedian(weight[temperature <= max(temperature[0] + 20, peak_temp - 60)]))
if not np.isfinite(baseline):
    baseline = float(weight[0])
onset_temp = (baseline - intercept) / slope if slope != 0 else float(temperature[peak_idx])

# Keep plotted onset marker within the observed temperature interval.
onset_temp = float(np.clip(onset_temp, temperature.min(), temperature.max()))
onset_weight = float(np.interp(onset_temp, temperature, weight))
peak_weight = float(np.interp(peak_temp, temperature, weight))
