import json
import math
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

sheets = ["PE", "PVC", "PET"]
colors = ["#2563eb", "#dc2626", "#16a34a"]
series = []

for sheet, color in zip(sheets, colors):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    x_col = next(c for c in df.columns if "Temperature" in str(c))
    y_col = next(c for c in df.columns if "Weight" in str(c) and "Deriv" not in str(c))

    data = []
    for x, y in zip(pd.to_numeric(df[x_col], errors="coerce"),
                    pd.to_numeric(df[y_col], errors="coerce")):
        if pd.notna(x) and pd.notna(y) and math.isfinite(float(x)) and math.isfinite(float(y)):
            data.append([float(x), float(y)])

    series.append({
        "name": sheet,
        "type": "line",
        "data": data,
        "showSymbol": False,
        "symbol": "none",
        "lineStyle": {"width": 2, "color": color},
        "itemStyle": {"color": color},
        "emphasis": {"focus": "series"},
        "animation": False
    })

option = {
    "animation": False,
    "title": {
        "text": "고분자 TGA 중량 변화",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16}
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "cross"}
    },
    "legend": {
        "data": sheets,
        "orient": "vertical",
        "right": 18,
        "top": "middle",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 12}
    },
    "grid": {"left": 75, "right": 125, "top": 65, "bottom": 65},
    "xAxis": {
        "type": "value",
        "name": "온도 (°C)",
        "nameLocation": "middle",
        "nameGap": 40,
        "min": 30,
        "max": 900,
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"}
    },
    "yAxis": {
        "type": "value",
        "name": "중량 (%)",
        "nameLocation": "middle",
        "nameGap": 50,
        "min": 0,
        "max": 105,
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        "splitLine": {"lineStyle": {"color": "#e5e7eb"}}
    },
    "series": series
}

print(json.dumps({
    "figures": [{
        "name": "polymer_tga_overlay",
        "option": option,
        "width": 800,
        "height": 520,
        "dpi": 300,
        "formats": ["png"]
    }]
}, ensure_ascii=False, allow_nan=False))
