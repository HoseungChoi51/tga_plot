import json
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

df = pd.read_excel(DATA_FILE, sheet_name="PVC")
df.columns = [str(c).strip() for c in df.columns]

temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce").to_numpy()
weight = pd.to_numeric(df["Weight (%)"], errors="coerce").to_numpy()

valid = np.isfinite(temperature) & np.isfinite(weight)
temperature = temperature[valid]
weight = weight[valid]

def weight_at(target):
    order = np.argsort(temperature)
    return float(np.interp(target, temperature[order], weight[order]))

start_temp, end_temp = 200.0, 400.0
start_weight = weight_at(start_temp)
end_weight = weight_at(end_temp)
mass_loss = start_weight - end_weight

# Keep the plotted data within the requested temperature interval and include
# interpolated boundary values so the highlighted interval is exact.
inside = (temperature > start_temp) & (temperature < end_temp)
plot_t = np.concatenate(([start_temp], temperature[inside], [end_temp]))
plot_w = np.concatenate(([start_weight], weight[inside], [end_weight]))
order = np.argsort(plot_t)
plot_t = plot_t[order]
plot_w = plot_w[order]

option = {

            "markArea": {
