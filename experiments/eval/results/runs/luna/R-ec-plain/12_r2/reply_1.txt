import json
import os
import re

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"


def clean_name(value):
    name = re.sub(r"[^\w가-힣-]+", "_", str(value).strip())
    return name.strip("_") or "plot"


def load_sheet(path, sheet_name):
    df = pd.read_excel(path, sheet_name=sheet_name)
    df.columns = [str(c).strip() for c in df.columns]

    temperature_col = next(
        (c for c in df.columns if "temperature" in c.lower() or "온도" in c),
        df.columns[1],
    )
    weight_col = next(
        (c for c in df.columns if "weight" in c.lower() or "중량" in c),
        df.columns[2],
    )
    derivative_col = next(
        (c for c in df.columns if "d.w" in c.lower() or "derivative" in c.lower()),
        df.columns[3],
    )

    temperature = pd.to_numeric(df[temperature_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    derivative = pd.to_numeric(df[derivative_col], errors="coerce")

    rows = []
    for x, y in zip(temperature, weight):
        if pd.notna(x) and pd.notna(y):
            rows.append([float(x), float(y)])

    derivative_rows = []
    for x, y in zip(temperature, derivative):
        if pd.notna(x) and pd.notna(y):
            derivative_rows.append([float(x), float(y)])

    return rows, derivative_rows, temperature_col, weight_col, derivative_col


figures = []
with pd.ExcelFile(DATA_FILE) as workbook:
