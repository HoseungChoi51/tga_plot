```python
import json
import os
import sys

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
RESIDUE_TEMPERATURE = 700.0


def clean_number(value):
    if pd.isna(value) or not np.isfinite(float(value)):
        return None
    return float(value)


def make_figure(sheet_name, df):
    # Select the temperature, weight, and derivative columns by their names.
    cols = list(df.columns)
    temperature_col = next((c for c in cols if "temperature" in str(c).lower()), cols[1])
    weight_col = next((c for c in cols if "weight" in str(c).lower()), cols[2])
    derivative_col = next(
        (c for c in cols if "d.w" in str(c).lower() or "derivative" in str(c).lower()),
        cols[3] if len(cols) > 3 else None,
    )

    temperature = pd.to_numeric(df[temperature_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    derivative = (
        pd.to_numeric(df[derivative_col], errors="coerce")
        if derivative_col is not None
        else pd.Series(np.nan, index=df.index)
    )

    valid = temperature.notna() & weight.notna()
    temperature = temperature[valid].to_numpy(dtype=float)
    weight = weight[valid].to_numpy(dtype=float)
    derivative = derivative[valid].to_numpy(dtype=float)

    # Interpolate residue at 700 °C, including data collected in descending order.
    order = np.argsort(temperature)
    temp_sorted = temperature[order]
    weight_sorted = weight[order]
    residue = (
        float(np.interp(RESIDUE_TEMPERATURE, temp_sorted, weight_sorted))
        if len(temp_sorted) and temp_sorted[0] <= RESIDUE_TEMPERATURE <= temp_sorted[-1]
        else None
    )

    series = [
        {
            "name": "Weight",
            "type": "line",
            "yAxisIndex": 0,
            "showSymbol": False,
            "lineStyle": {"width": 2, "color": "#2563A6"},
            "data": [
                [clean_number(x), clean_number(y)]
                for x, y in zip(temperature, weight)
                if clean_number(x) is not None and clean_number(y) is not None
            ],
        },
        {
            "name": "Derivative",
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "lineStyle": {"width": 1.5, "color": "#D97732"},
            "data": [
                [clean_number(x), clean_number(y)]
                for x, y in zip(temperature, derivative)
                if clean_number(x) is not None and clean_number(y) is not None
            ],
        },
    ]

    if residue is not None:
        series.append(
            {
                "name": "Residue at 700 °C",
                "type": "scatter",
                "yAxisIndex": 0,
                "symbolSize": 10,
                "itemStyle": {"color": "#B42318"},
                "data": [[RESIDUE_TEMPERATURE, round(residue, 2)]],
                "label": {
                    "show": True,
                    "position": "top",
                    "formatter": f"700 °C 잔류량: {residue:.2f}%",
                    "color": "#B42318",
                    "fontSize": 12,
                },
                "markLine": {
                    "silent": True,
                    "symbol": ["none", "none"],
                    "lineStyle": {"type": "dashed", "color": "#B42318", "width": 1},
                    "data": [{"xAxis": RESIDUE_TEMPERATURE}],
                },
            }
        )

    option = {
        "animation": False,
        "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        "title": {
            "text": str(sheet_name),
            "left": "center",
            "top": 8,
            "textStyle": {"fontSize": 16, "fontWeight": "normal"},
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"},
            "valueFormatter": "{value}",
        },
        "legend": {"top": 36, "data": ["Weight", "Derivative"]},
        "grid": {"left": 72, "right": 76, "top": 78, "bottom": 64},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 38,
            "axisLine": {"lineStyle": {"color": "#555"}},
            "splitLine": {"show": False},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True, "lineStyle": {"color": "#2563A6"}},
                "axisLabel": {"color": "#2563A6"},
                "splitLine": {"lineStyle": {"color": "#E5E7EB"}},
            },
            {
                "type": "value",
                "name": "D.W (%/°C)",
                "nameLocation": "middle",
                "nameGap": 52,
                "axisLine": {"show": True, "lineStyle": {"color": "#D97732"}},
                "axisLabel": {"color": "#D97732"},
                "splitLine": {"show": False},
            },
        ],
        "series": series,
    }

    return {
        "name": f"tga_{str(sheet_name).replace(' ', '_')}",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"],
    }


def main():
    path = DATA_FILE
    if not os.path.isfile(path):
        print(json.dumps({"figures": []}, ensure_ascii=False))
        return

    figures = []
    workbook = pd.ExcelFile(path)
    for sheet_name in workbook.sheet_names:
        df = pd.read_excel(path, sheet_name=sheet_name)
        if df.empty or len(df.columns) < 3:
            continue
        figures.append(make_figure(sheet_name, df))

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
