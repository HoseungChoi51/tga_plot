import json
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"

sheets = [
    ("Raw", "검정", "#222222"),
    ("Acid-washed", "Acid-washed", "#2878B5"),
    ("Annealed", "Annealed", "#D64541"),
]

series = []
for sheet, label, color in sheets:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    df.columns = [str(c).strip() for c in df.columns]

    x_col = next(c for c in df.columns if "Temperature" in c)
    weight_col = next(c for c in df.columns if "Weight" in c and "Deriv" not in c)
    derivative_col = next(c for c in df.columns if "Deriv" in c)

    x = pd.to_numeric(df[x_col], errors="coerce").to_numpy()
    weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy()
    derivative = pd.to_numeric(df[derivative_col], errors="coerce").to_numpy()

    weight_data = [
        [float(a), float(b)] for a, b in zip(x, weight)
        if np.isfinite(a) and np.isfinite(b)
    ]
    derivative_data = [
        [float(a), float(b)] for a, b in zip(x, derivative)
        if np.isfinite(a) and np.isfinite(b)
    ]

    series.extend([
        {
            "name": f"{label} Weight",
            "type": "line",
            "xAxisIndex": 0,
            "yAxisIndex": 0,
            "data": weight_data,
            "showSymbol": False,
            "lineStyle": {"width": 2, "color": color},
            "itemStyle": {"color": color},
            "emphasis": {"focus": "series"},
        },
        {
            "name": f"{label} Derivative",
            "type": "line",
            "xAxisIndex": 1,
            "yAxisIndex": 1,
            "data": derivative_data,
            "showSymbol": False,
            "lineStyle": {"width": 1.6, "color": color, "type": "dashed"},
            "itemStyle": {"color": color},
            "emphasis": {"focus": "series"},
        },
    ])

option = {
