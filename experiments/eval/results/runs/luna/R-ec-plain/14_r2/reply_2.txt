# Korean CSV files are often saved as CP949/EUC-KR rather than UTF-8.

    try:
        df = pd.read_csv(DATA_FILE, encoding=encoding)
        break
    except UnicodeDecodeError:
        continue
else:
    raise UnicodeError("CSV 파일의 문자 인코딩을 판별할 수 없습니다.")

x_col = "온도(°C)"

derivative_col = "미분무게(%/°C)"

for col in (x_col, weight_col, derivative_col):
    df[col] = pd.to_numeric(df[col], errors="coerce")

df = df.dropna(subset=[x_col, weight_col, derivative_col])

temperature = df[x_col].tolist()
weight = df[weight_col].tolist()
