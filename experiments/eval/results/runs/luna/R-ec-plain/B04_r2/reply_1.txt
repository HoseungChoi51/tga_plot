```python
import json
import os
import sys

import numpy as np
import pandas as pd
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/xrd.csv"


def main():
    df = pd.read_csv(DATA_FILE)
    if df.shape[1] < 2:
        raise ValueError("XRD 데이터에 2θ와 intensity 열이 모두 필요합니다.")

    x = pd.to_numeric(df.iloc[:, 0], errors="coerce").to_numpy()
    y = pd.to_numeric(df.iloc[:, 1], errors="coerce").to_numpy()
    valid = np.isfinite(x) & np.isfinite(y)
    x, y = x[valid], y[valid]

    order = np.argsort(x)
    x, y = x[order], y[order]

    peak_indices, _ = find_peaks(y)
    if peak_indices.size == 0:
        peak_indices = np.arange(len(y))

    strongest = peak_indices[np.argsort(y[peak_indices])[-5:]]
    strongest = strongest[np.argsort(x[strongest])]

    peak_data = [
        {
            "coord": [float(x[i]), float(y[i])],
            "name": f"{x[i]:.2f}°",
            "label": {
                "show": True,
                "position": "top",
                "formatter": f"{x[i]:.2f}°",
                "fontSize": 11,
                "color": "#c0392b",
                "distance": 8,
            },
            "itemStyle": {"color": "#c0392b"},
            "symbolSize": 8,
        }
        for i in strongest
    ]

    option = {
        "animation": False,
        "tooltip": {"trigger": "axis"},
        "grid": {"left": 72, "right": 24, "top": 68, "bottom": 66},
        "title": {
            "text": "XRD 패턴",
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "xAxis": {
            "type": "value",
            "name": "2θ (deg)",
            "nameLocation": "middle",
            "nameGap": 42,
            "min": float(np.min(x)),
            "max": float(np.max(x)),
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        },
        "yAxis": {
            "type": "value",
            "name": "Intensity (counts)",
            "nameLocation": "middle",
            "nameGap": 52,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
        },
        "series": [
            {
                "name": "Intensity",
                "type": "line",
                "data": [[float(xi), float(yi)] for xi, yi in zip(x, y)],
                "showSymbol": False,
                "lineStyle": {"width": 1.5, "color": "#2468a2"},
                "sampling": "lttb",
                "markPoint": {
                    "data": peak_data,
                    "symbol": "circle",
                    "label": {"fontFamily": "Noto Sans CJK KR"},
                },
            }
        ],
    }

    print(
        json.dumps(
            {
                "figures": [
                    {
                        "name": "xrd_pattern",
                        "option": option,
                        "width": 900,
                        "height": 600,
                        "dpi": 300,
                        "formats": ["png"],
                    }
                ]
            },
            ensure_ascii=False,
            allow_nan=False,
        )
    )


if __name__ == "__main__":
    main()
```
