```python
import json
import re
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

COLORS = ["#222222", "#d62728", "#1f77b4"]
WIDTH = 614
HEIGHT = 442
DPI = 300


def safe_name(value):
    name = re.sub(r"[^A-Za-z0-9가-힣_-]+", "_", str(value)).strip("_")
    return name or "plot"


def clean_number(value):
    if pd.isna(value):
        return None
    number = float(value)
    return number if np.isfinite(number) else None


def make_figure(sheet_name, frame):
    columns = list(frame.columns)

    def find_column(pattern, fallback_index):
        for column in columns:
            if re.search(pattern, str(column), re.IGNORECASE):
                return column
        return columns[fallback_index] if len(columns) > fallback_index else None

    x_col = find_column(r"temperature|temp", 1)
    y1_col = find_column(r"weight", 2)
    y2_col = find_column(r"d\.?\s*w|derivative", 3)

    if x_col is None or y1_col is None:
        return None

    x = pd.to_numeric(frame[x_col], errors="coerce")
    y1 = pd.to_numeric(frame[y1_col], errors="coerce")
    y2 = pd.to_numeric(frame[y2_col], errors="coerce") if y2_col is not None else None

    data1 = [
        [clean_number(xv), clean_number(yv)]
        for xv, yv in zip(x, y1)
        if pd.notna(xv) and pd.notna(yv)
    ]
    data2 = []
    if y2 is not None:
        data2 = [
            [clean_number(xv), clean_number(yv)]
            for xv, yv in zip(x, y2)
            if pd.notna(xv) and pd.notna(yv)
        ]

    if not data1:
        return None

    series = [
        {
            "name": str(y1_col),
            "type": "line",
            "data": data1,
            "showSymbol": False,
            "symbol": "none",
            "lineStyle": {"color": "#000000", "width": 2.5},
            "itemStyle": {"color": "#000000"},
            "yAxisIndex": 0,
            "smooth": False,
            "connectNulls": False,
        }
    ]

    if data2:
        series.append(
            {
                "name": str(y2_col),
                "type": "line",
                "data": data2,
                "showSymbol": False,
                "symbol": "none",
                "lineStyle": {"color": COLORS[1], "width": 1.8, "type": "dashed"},
                "itemStyle": {"color": COLORS[1]},
                "yAxisIndex": 1,
                "smooth": False,
                "connectNulls": False,
            }
        )

    option = {
        "animation": False,
        "backgroundColor": "#ffffff",
        "textStyle": {"fontFamily": "Noto Sans CJK KR, sans-serif", "color": "#333333"},
        "title": {
            "text": str(sheet_name),
            "left": "center",
            "top": 12,
            "textStyle": {"fontSize": 16, "fontWeight": "normal"},
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"},
        },
        "legend": {"top": 40, "left": "center"},
        "grid": {"left": 68, "right": 72, "top": 82, "bottom": 60},
        "xAxis": {
            "type": "value",
            "name": str(x_col),
            "nameLocation": "middle",
            "nameGap": 35,
            "scale": False,
            "axisLine": {"lineStyle": {"color": "#777777"}},
            "axisLabel": {"color": "#444444"},
            "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}},
        },
        "yAxis": [
            {
                "type": "value",
                "name": str(y1_col),
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True, "lineStyle": {"color": "#777777"}},
                "axisLabel": {"color": "#444444"},
                "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}},
            },
            {
                "type": "value",
                "name": str(y2_col) if y2_col is not None else "",
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True, "lineStyle": {"color": "#777777"}},
                "axisLabel": {"color": "#444444"},
                "splitLine": {"show": False},
            },
        ],
        "series": series,
    }

    return {
        "name": safe_name(sheet_name),
        "option": option,
        "width": WIDTH,
        "height": HEIGHT,
        "dpi": DPI,
        "formats": ["png"],
    }


def main():
    figures = []
    workbook = pd.ExcelFile(DATA_FILE)
    for sheet_name in workbook.sheet_names:
        frame = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        figure = make_figure(sheet_name, frame)
        if figure is not None:
            figures.append(figure)

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
