import json
import re
from pathlib import Path

import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/isothermal.txt"

def read_data(path):
    path = Path(path)
    for sep in (r"\s+", ",", "\t", ";"):
        try:
            df = pd.read_csv(path, sep=sep, engine="python", comment="#")
            if len(df.columns) >= 3:
                return df
        except Exception:
            pass
    raise ValueError(f"Could not read a three-column data table from {path}")

df = read_data(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

def find_column(pattern, fallback):
    for col in df.columns:
        if re.search(pattern, col, re.IGNORECASE):
            return col
    return df.columns[fallback]

time_col = find_column(r"time", 0)
temp_col = find_column(r"temperature|temp", 1)
weight_col = find_column(r"weight|mass", 2)

for col in (time_col, temp_col, weight_col):
    df[col] = pd.to_numeric(df[col], errors="coerce")

df = df[[time_col, temp_col, weight_col]].dropna().sort_values(time_col)
time = df[time_col].tolist()
temp = df[temp_col].tolist()
weight = df[weight_col].tolist()

# Mark the measurement at 140 minutes, using the nearest available observation.
idx = min(range(len(time)), key=lambda i: abs(time[i] - 140.0))
mark_x, mark_y = time[idx], weight[idx]
