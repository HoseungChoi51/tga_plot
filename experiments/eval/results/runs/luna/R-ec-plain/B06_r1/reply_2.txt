import csv
import json
import re
from pathlib import Path

import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/isothermal.txt"

def load_table(path):
    text = Path(path).read_text(encoding="utf-8-sig", errors="replace")
    lines = [line for line in text.splitlines() if line.strip()]
    if not lines:
        raise ValueError(f"No data found in {path}")

    # Detect common delimiters, including whitespace-separated instrument exports.
    sample = "\n".join(lines[:10])
    try:
        delimiter = csv.Sniffer().sniff(sample, delimiters=",\t;").delimiter
        rows = list(csv.reader(lines, delimiter=delimiter))
    except csv.Error:
        rows = [re.split(r"\s+", line.strip()) for line in lines]

    # Find the header row containing the time/temperature/weight field names.
    header_idx = None
    for i, row in enumerate(rows):
        joined = " ".join(str(v) for v in row)
        if re.search(r"time", joined, re.I) and re.search(r"weight", joined, re.I):
            header_idx = i
            break

    if header_idx is not None:
        headers = [str(v).strip() for v in rows[header_idx]]
        records = rows[header_idx + 1:]
        width = len(headers)
        records = [r[:width] + [None] * max(0, width - len(r)) for r in records]
        return pd.DataFrame(records, columns=headers)

    # Fall back to extracting numeric triples, ignoring textual preamble.
    numeric_rows = []
    for line in lines:
        vals = re.split(r"[\s,;\t]+", line.strip())
        if len(vals) < 3:
            continue
        try:
            numeric_rows.append([float(v) for v in vals[:3]])
        except ValueError:
            continue
    if not numeric_rows:
        raise ValueError(f"Could not find a data table in {path}")
    return pd.DataFrame(numeric_rows, columns=["Time (min)", "Temperature (°C)", "Weight (%)"])

df = load_table(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

def find_column(pattern, fallback):
    for col in df.columns:
        if re.search(pattern, col, re.IGNORECASE):
            return col
    return df.columns[fallback]

time_col = find_column(r"time", 0)
temp_col = find_column(r"temperature|temp", min(1, len(df.columns) - 1))
weight_col = find_column(r"weight|mass", min(2, len(df.columns) - 1))

for col in (time_col, temp_col, weight_col):
