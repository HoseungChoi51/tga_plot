```python
import json
import os
import re
import pandas as pd
import numpy as np

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

# 발표용 가로 10인치, 96 CSS px/in
WIDTH = 960
HEIGHT = 600
DPI = 300
FONT_SIZE = 16

df = pd.read_excel(DATA_FILE, sheet_name="PVC")
df.columns = [str(c).strip() for c in df.columns]

def find_column(pattern):
    matches = [c for c in df.columns if re.search(pattern, c, re.IGNORECASE)]
    if not matches:
        raise ValueError(f"필요한 열을 찾을 수 없습니다: {pattern}")
    return matches[0]

x_col = find_column(r"Temperature")
weight_col = find_column(r"Weight\s*\(%\)")
deriv_col = find_column(r"Deriv")

data = df[[x_col, weight_col, deriv_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[x_col].to_numpy(dtype=float)
weight = data[weight_col].to_numpy(dtype=float)
derivative = data[deriv_col].to_numpy(dtype=float)

# 값의 크기 차이를 읽기 쉽게 하기 위해 DTG를 양의 질량감소율로 표시
dtg = -derivative

option = {

    "backgroundColor": "#ffffff",

        "fontSize": FONT_SIZE,

    },

    },

        "axisPointer": {"type": "cross"},

    },
    "legend": {
        "data": ["Weight (%)", "DTG (%/°C, 질량감소율)"],
        "top": 56,
        "textStyle": {"fontSize": FONT_SIZE}
    },
    "grid": {
        "left": 95,
        "right": 100,
        "top": 115,
        "bottom": 85,
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 48,
        "min": 30,
        "max": 900,
        "axisLabel": {"fontSize": FONT_SIZE},
        "nameTextStyle": {"fontSize": FONT_SIZE},
        "axisLine": {"lineStyle": {"color": "#555555"}},
        "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}}
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 60,
            "min": 0,
            "max": 105,
            "axisLabel": {"fontSize": FONT_SIZE},
            "nameTextStyle": {"fontSize": FONT_SIZE},
            "axisLine": {"show": True, "lineStyle": {"color": "#2878b5"}},
            "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}}
        },
        {
            "type": "value",
            "name": "DTG (%/°C)",
            "nameLocation": "middle",
            "nameGap": 62,
            "min": 0,
            "max": max(0.1, float(np.nanpercentile(dtg, 99.5)) * 1.15),
            "axisLabel": {"fontSize": FONT_SIZE},
            "nameTextStyle": {"fontSize": FONT_SIZE},
            "axisLine": {"show": True, "lineStyle": {"color": "#d65f5f"}},
            "splitLine": {"show": False}

    "series": [
        {

            "type": "line",

            "showSymbol": False,
            "smooth": False,
            "lineStyle": {"width": 3, "color": "#2878b5"},
            "data": [[float(x), float(y)] for x, y in zip(temperature, weight)]
        },
        {

            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "smooth": False,
            "lineStyle": {"width": 3, "color": "#d65f5f"},
            "data": [[float(x), float(y)] for x, y in zip(temperature, dtg)]
        }
    ]
}

result = {
    "figures": [{
        "name": "pvc_tga",
        "option": option,
        "width": WIDTH,
        "height": HEIGHT,
        "dpi": DPI,
        "formats": ["png"]
    }]
}
print(json.dumps(result, ensure_ascii=False, allow_nan=False))
```
