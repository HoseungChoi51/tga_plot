```python
import json
import numpy as np
import pandas as pd
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/xrd.csv"

df = pd.read_csv(DATA_FILE)
x = pd.to_numeric(df.iloc[:, 0], errors="coerce").to_numpy()
y = pd.to_numeric(df.iloc[:, 1], errors="coerce").to_numpy()

valid = np.isfinite(x) & np.isfinite(y)
x, y = x[valid], y[valid]
order = np.argsort(x)
x, y = x[order], y[order]

# Find prominent local maxima, then select the five strongest.
peaks, _ = find_peaks(y, distance=max(1, len(y) // 200))
if len(peaks) < 5:
    peaks = np.arange(len(y))
top = peaks[np.argsort(y[peaks])[-5:]]
top = top[np.argsort(x[top])]

peak_points = [
    {
        "coord": [float(x[i]), float(y[i])],
        "value": f"{x[i]:.2f}°",
        "label": {
            "show": True,
            "formatter": f"{x[i]:.2f}°",
            "position": "top",
            "distance": 8,
            "fontSize": 11,
            "color": "#b42318",
            "backgroundColor": "rgba(255,255,255,0.85)",
            "padding": [2, 4],
        },
        "symbol": "circle",
        "symbolSize": 7,
        "itemStyle": {"color": "#b42318"},
    }
    for i in top
]

option = {
    "animation": False,
    "backgroundColor": "#ffffff",

    "title": {

    },

    },

        "type": "value",

        "max": float(np.max(x)),

    },
    "yAxis": {
        "type": "value",
        "name": "Intensity (counts)",

        "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}},
    },
    "series": [
        {
            "name": "Intensity",
            "type": "line",
            "data": [[float(a), float(b)] for a, b in zip(x, y)],
            "showSymbol": False,

            },
        }
    ],
}
```
