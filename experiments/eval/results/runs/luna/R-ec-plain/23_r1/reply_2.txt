import json

import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt"


def read_export(path):
    # Instrument text exports are often UTF-16 (BOM) or Windows-1252 encoded.
    text = None
    for encoding in ("utf-8-sig", "utf-16", "cp1252", "latin-1"):
        try:
            text = Path(path).read_text(encoding=encoding)
            break
        except UnicodeError:
            continue
    if text is None:
        raise ValueError(f"Could not decode input file: {path}")

    rows = []
    for line in text.splitlines():
        values = pd.to_numeric(
            pd.Series(line.replace(",", ".").replace("\t", " ").split()),
            errors="coerce",
        )
        # Keep only data lines with at least the expected x and mass values.
        if len(values) >= 3 and values.iloc[:3].notna().sum() >= 3:
            padded = list(values.iloc[:4])
            if len(padded) < 4:
                padded.append(float("nan"))
            rows.append(padded)

    if not rows:
        raise ValueError("No numeric TGA data rows found in the input file.")
    return pd.DataFrame(rows)
