```python
import json
import os

import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

COLORS = ["#5470C6", "#91CC75", "#FAC858"]


def main():
    excel = pd.ExcelFile(DATA_FILE)
    figures = []

    for index, sheet_name in enumerate(excel.sheet_names):
        df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        df.columns = [str(col).strip() for col in df.columns]

        required = ["Temperature", "Weight(%)", "D.W(%/C)"]
        if not all(col in df.columns for col in required):
            continue

        data = df[required].apply(pd.to_numeric, errors="coerce").dropna()
        if data.empty:
            continue

        x_min = float(data["Temperature"].min())
        x_max = float(data["Temperature"].max())

        option = {
            "animation": False,
            "title": {
                "text": sheet_name,
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
            },
            "tooltip": {"trigger": "axis"},
            "legend": {
                "data": ["Weight(%)", "D.W(%/C)"],
                "top": 34,
                "textStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
            "grid": {"left": 68, "right": 72, "top": 82, "bottom": 58},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 34,
                "min": x_min,
                "max": x_max,
                "interval": 100,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "min": 0,
                    "max": 105,
                    "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                },
                {
                    "type": "value",
                    "name": "D.W (%/°C)",
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "min": -0.05,
                    "max": 1.2,
                    "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                },
            ],
            "series": [
                {
                    "name": "Weight(%)",
                    "type": "line",
                    "showSymbol": False,
                    "yAxisIndex": 0,
                    "lineStyle": {"width": 2, "color": COLORS[index % len(COLORS)]},
                    "data": data[["Temperature", "Weight(%)"]].values.tolist(),
                },
                {
                    "name": "D.W(%/C)",
                    "type": "line",
                    "showSymbol": False,
                    "yAxisIndex": 1,
                    "lineStyle": {"width": 1.8, "color": "#EE6666"},
                    "data": data[["Temperature", "D.W(%/C)"]].values.tolist(),
                },
            ],
        }

        figures.append(
            {
                "name": f"tga_{index + 1}",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        )

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
