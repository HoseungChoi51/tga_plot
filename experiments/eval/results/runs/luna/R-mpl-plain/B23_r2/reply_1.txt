```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B23_r2/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

samples = [
    ("Raw", "black"),
    ("Acid-washed", "blue"),
    ("Annealed", "red"),
]

fig, ax = plt.subplots(figsize=(9, 6))

for sheet, color in samples:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    df.columns = [str(column).strip() for column in df.columns]

    x = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
    weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
    valid = x.notna() & weight.notna()

    ax.plot(x[valid], weight[valid], color=color, linewidth=1.8, label=sheet)

ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%)")
ax.set_title("TGA Curves")
ax.set_xlim(left=30)
ax.set_ylim(bottom=0)
ax.grid(True, linestyle="--", alpha=0.35)
ax.legend(frameon=False)
fig.tight_layout()

fig.savefig(OUT_DIR / "tga_weight_comparison.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
