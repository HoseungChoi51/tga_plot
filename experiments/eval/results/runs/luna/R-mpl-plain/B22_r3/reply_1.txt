from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B22_r3/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

# Read the Raw sheet and identify temperature and weight columns by their headers.
df = pd.read_excel(DATA_FILE, sheet_name="Raw")
temperature_col = next(c for c in df.columns if "Temperature" in str(c))
weight_col = next(c for c in df.columns if "Weight" in str(c) and "Deriv" not in str(c))

temperature = pd.to_numeric(df[temperature_col], errors="coerce")
weight = pd.to_numeric(df[weight_col], errors="coerce")
valid = temperature.notna() & weight.notna()
temperature = temperature[valid].to_numpy(dtype=float)
weight = weight[valid].to_numpy(dtype=float)

# Use the interpolated weight at 150 °C as the dry-basis reference (100%).
order = np.argsort(temperature)
temperature = temperature[order]
weight = weight[order]
reference_weight = float(np.interp(150.0, temperature, weight))
if not np.isfinite(reference_weight) or reference_weight == 0:
    raise ValueError("150 °C의 기준 무게를 계산할 수 없습니다.")

normalized_weight = weight / reference_weight * 100.0

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax = plt.subplots(figsize=(8, 5.5))
ax.plot(temperature, normalized_weight, color="#1f77b4", linewidth=1.8)
ax.axvline(150, color="0.45", linestyle="--", linewidth=1, alpha=0.8)
ax.axhline(100, color="0.65", linestyle=":", linewidth=1)

ax.set_xlabel("온도 (°C)")
ax.set_ylabel("무게 유지율 (%, 150 °C 기준)")
ax.set_title("Raw — 150 °C 건조 기준 무게")
ax.grid(True, linestyle=":", linewidth=0.7, alpha=0.6)
ax.margins(x=0.02)
fig.tight_layout()

fig.savefig(OUT_DIR / "raw_dry_basis_150C.png", dpi=300, bbox_inches="tight")
plt.close(fig)
