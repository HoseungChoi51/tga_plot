```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd


DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/10_r1/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

xls = pd.ExcelFile(DATA_FILE)
fig, ax = plt.subplots(figsize=(9, 6))

colors = ["#2563eb", "#dc2626", "#16a34a"]
residual_values = []

for i, sheet in enumerate(xls.sheet_names):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    df.columns = [str(col).strip() for col in df.columns]

    temp_col = next(col for col in df.columns if "temperature" in col.lower())
    weight_col = next(col for col in df.columns if "weight" in col.lower())

    data = df[[temp_col, weight_col]].apply(pd.to_numeric, errors="coerce").dropna()
    data = data.sort_values(temp_col)

    ax.plot(
        data[temp_col],
        data[weight_col],
        color=colors[i % len(colors)],
        linewidth=2,
        label=sheet,
    )

    # Interpolate the residual weight at 700 °C where the dataset spans that temperature.
    if data[temp_col].min() <= 700 <= data[temp_col].max():
        residual = float(
            pd.Series(data[weight_col].to_numpy(), index=data[temp_col].to_numpy())
            .groupby(level=0)
            .mean()
            .sort_index()
            .pipe(lambda s: __import__("numpy").interp(700, s.index.to_numpy(), s.to_numpy()))
        )
        residual_values.append((sheet, residual))

ax.set(
    title="TGA 곡선 및 700 °C 잔류량",
    xlabel="Temperature (°C)",
    ylabel="Weight (%)",
    xlim=(30, 900),
)
ax.grid(True, alpha=0.25)
ax.legend(loc="best", frameon=False)

# Mark 700 °C and annotate each sample's residual mass to two decimal places.
ax.axvline(700, color="0.35", linestyle="--", linewidth=1, alpha=0.8)
for i, (sheet, residual) in enumerate(residual_values):
    ax.scatter([700], [residual], color=colors[i % len(colors)], s=32, zorder=4)
    ax.annotate(
        f"{sheet}: {residual:.2f}%",
        xy=(700, residual),
        xytext=(8, 8 + 18 * i),
        textcoords="offset points",
        color=colors[i % len(colors)],
        fontsize=9,
        bbox={"boxstyle": "round,pad=0.25", "fc": "white", "ec": "none", "alpha": 0.8},
    )

fig.tight_layout()
fig.savefig(OUT_DIR / "tga_residual_700C.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
