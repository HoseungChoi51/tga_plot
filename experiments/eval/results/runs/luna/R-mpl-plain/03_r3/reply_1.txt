```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/03_r3/out"

# Korean font and consistent plot styling
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

df = pd.read_excel(DATA_FILE, sheet_name="1006R1-V acid")
temperature = pd.to_numeric(df["Temperature"], errors="coerce")
weight = pd.to_numeric(df["Weight(%)"], errors="coerce")
derivative = pd.to_numeric(df["D.W(%/C)"], errors="coerce")

data = pd.DataFrame({
    "Temperature": temperature,
    "Weight(%)": weight,
    "D.W(%/C)": derivative,
}).dropna()
data = data.sort_values("Temperature")
data = data[(data["Temperature"] >= 100) & (data["Temperature"] <= 800)]

x = data["Temperature"].to_numpy()
weight = data["Weight(%)"].to_numpy()
derivative = data["D.W(%/C)"].to_numpy()

# D.W(%/C) is negative during mass-loss events; mark the two strongest
# mass-loss peaks (the two most negative derivative values).
peak_indices, _ = find_peaks(-derivative, distance=max(1, len(derivative) // 20))
if len(peak_indices) < 2:
    peak_indices = np.argsort(derivative)[:min(2, len(derivative))]
else:
    peak_indices = peak_indices[np.argsort(derivative[peak_indices])[:2]]
peak_indices = np.sort(peak_indices)

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(
    x, weight, color="#2463A6", linewidth=2.2, label="Weight (%)"
)
derivative_line, = ax2.plot(
    x, derivative, color="#D9534F", linewidth=1.8, label="D.W. (%/°C)"
)

if len(peak_indices):
    peak_x = x[peak_indices]
    peak_y = derivative[peak_indices]
    ax2.scatter(
        peak_x, peak_y, color="#222222", edgecolor="white",
        linewidth=0.8, s=65, zorder=5, label="피크"
    )
    for i, (px, py) in enumerate(zip(peak_x, peak_y), start=1):
        ax2.annotate(
            f"피크 {i}\n{px:.1f} °C",
            xy=(px, py),
            xytext=(8, -22 if i % 2 else 12),
            textcoords="offset points",
            fontsize=9,
            color="#222222",
            arrowprops={"arrowstyle": "-", "color": "#555555", "lw": 0.8},
        )

ax1.set_xlim(100, 800)
ax1.set_xlabel("온도 (°C)")
ax1.set_ylabel("무게 (%)", color="#2463A6")
ax2.set_ylabel("D.W. (%/°C)", color="#D9534F")
ax1.tick_params(axis="y", labelcolor="#2463A6")
ax2.tick_params(axis="y", labelcolor="#D9534F")
ax1.set_title("1006R1-V acid TGA")
ax1.grid(True, linestyle="--", alpha=0.35)

handles = [weight_line, derivative_line]
labels = [line.get_label() for line in handles]
if len(peak_indices):
    handles.append(ax2.collections[0])
    labels.append("피크")
ax1.legend(handles, labels, loc="best", frameon=False)

fig.tight_layout()
fig.savefig(out_dir / "1006R1-V_acid_TGA.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
