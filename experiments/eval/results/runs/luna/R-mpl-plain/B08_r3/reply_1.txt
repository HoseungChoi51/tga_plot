from pathlib import Path

import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B08_r3/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

data = pd.read_excel(DATA_FILE, sheet_name="Raw")
temperature = pd.to_numeric(data["Temperature (°C)"], errors="coerce")
weight = pd.to_numeric(data["Weight (%)"], errors="coerce")
valid = temperature.notna() & weight.notna()

fig, ax = plt.subplots(figsize=(8, 5.5))
ax.plot(temperature[valid], weight[valid], color="black", linewidth=1.6)
ax.axhline(50, color="tab:red", linestyle="--", linewidth=1.2)

ax.text(
    0.98, 0.97, "공기, 10 °C/min",
    transform=ax.transAxes,
    ha="right", va="top",
    fontsize=11,
)

ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%)")
ax.set_title("Raw")
ax.set_xlim(left=temperature[valid].min(), right=temperature[valid].max())
ax.set_ylim(bottom=0, top=105)
ax.grid(True, alpha=0.25)

fig.tight_layout()
fig.savefig(OUT_DIR / "raw_weight.png", dpi=300, bbox_inches="tight")
plt.close(fig)
