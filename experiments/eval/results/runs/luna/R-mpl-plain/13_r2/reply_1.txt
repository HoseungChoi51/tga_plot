from pathlib import Path

import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/13_r2/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

# Read the CNT-01 sheet and use temperature as x, Weight (mg) as y.
df = pd.read_excel(DATA_FILE, sheet_name="CNT-01")
df.columns = [str(column).strip() for column in df.columns]

temperature_col = next(
    column for column in df.columns if "Temperature" in column
)
weight_col = next(
    column for column in df.columns if "Weight (mg)" in column
)

plot_data = df[[temperature_col, weight_col]].apply(
    pd.to_numeric, errors="coerce"
).dropna()

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax = plt.subplots(figsize=(8, 5))
ax.plot(
    plot_data[temperature_col],
    plot_data[weight_col],
    color="black",
    linewidth=1.8,
)

ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (mg)")
ax.set_title("CNT-01")
ax.grid(True, linestyle="--", linewidth=0.6, alpha=0.5)

fig.tight_layout()
fig.savefig(OUT_DIR / "weight_mg_vs_temperature.png", dpi=300, bbox_inches="tight")
plt.close(fig)
