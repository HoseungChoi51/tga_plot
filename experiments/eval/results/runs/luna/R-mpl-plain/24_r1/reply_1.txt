from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/24_r1/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

# Use the first worksheet and the columns identified in the data preview.
df = pd.read_excel(DATA_FILE, sheet_name=0)
df.columns = [str(column).strip() for column in df.columns]

temperature_col = "Temperature (°C)"
weight_col = "Weight (%)"
heat_flow_col = "Heat Flow (W/g)"

for column in (temperature_col, weight_col, heat_flow_col):
    if column not in df.columns:
        raise ValueError(f"Required column not found: {column}")

data = df[[temperature_col, weight_col, heat_flow_col]].apply(
    pd.to_numeric, errors="coerce"
).dropna()

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

line1, = ax1.plot(
    data[temperature_col],
    data[weight_col],
    color="tab:blue",
    linewidth=1.8,
    label="Weight (%)",
)
line2, = ax2.plot(
    data[temperature_col],
    data[heat_flow_col],
    color="tab:red",
    linewidth=1.6,
    label="Heat flow (W/g)",
)

ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="tab:blue")
ax2.set_ylabel("Heat flow (W/g)", color="tab:red")
ax1.tick_params(axis="y", labelcolor="tab:blue")
ax2.tick_params(axis="y", labelcolor="tab:red")
ax1.grid(True, linestyle="--", alpha=0.35)
ax1.legend(handles=[line1, line2], loc="best", frameon=False)

fig.tight_layout()
output_file = OUT_DIR / f"{DATA_FILE.stem}_tga_heat_flow.png"
fig.savefig(output_file, dpi=300, bbox_inches="tight")
plt.close(fig)
