```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/01_r1/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

fig, axes = plt.subplots(
    nrows=len(sheets),
    ncols=1,
    figsize=(9, 4.2 * len(sheets)),
    sharex=True,
    constrained_layout=True,
)
if len(sheets) == 1:
    axes = [axes]

for ax, (sheet_name, data) in zip(axes, sheets.items()):
    data.columns = [str(column).strip() for column in data.columns]
    temperature = pd.to_numeric(data["Temperature"], errors="coerce")
    weight = pd.to_numeric(data["Weight(%)"], errors="coerce")
    derivative = pd.to_numeric(data["D.W(%/C)"], errors="coerce")

    valid = temperature.notna()
    temperature, weight, derivative = (
        temperature[valid],
        weight[valid],
        derivative[valid],
    )

    ax.plot(temperature, weight, color="tab:blue", linewidth=1.8, label="Weight (%)")
    ax.set_title(sheet_name)
    ax.set_xlabel("Temperature (°C)")
    ax.set_ylabel("Weight (%)", color="tab:blue")
    ax.tick_params(axis="y", labelcolor="tab:blue")
    ax.grid(True, linestyle="--", alpha=0.35)

    ax2 = ax.twinx()
    ax2.plot(
        temperature,
        derivative,
        color="tab:red",
        linewidth=1.4,
        label="D.W (%/°C)",
    )
    ax2.set_ylabel("D.W (%/°C)", color="tab:red")
    ax2.tick_params(axis="y", labelcolor="tab:red")

    lines = ax.get_lines() + ax2.get_lines()
    ax.legend(lines, [line.get_label() for line in lines], loc="best", frameon=False)

fig.suptitle("TGA", fontsize=15)
output_path = OUT_DIR / "TGA.png"
fig.savefig(output_path, dpi=300, bbox_inches="tight")
plt.close(fig)
```
