```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.signal import find_peaks

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/xrd.csv")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B04_r2/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

# Read the CSV and use its first two columns as 2θ and intensity.
data = pd.read_csv(DATA_FILE)
if data.shape[1] < 2:
    raise ValueError("CSV 파일에 2θ와 intensity 데이터 열이 두 개 이상 있어야 합니다.")

x = pd.to_numeric(data.iloc[:, 0], errors="coerce").to_numpy()
y = pd.to_numeric(data.iloc[:, 1], errors="coerce").to_numpy()
valid = np.isfinite(x) & np.isfinite(y)
x, y = x[valid], y[valid]

if x.size == 0:
    raise ValueError("유효한 XRD 데이터가 없습니다.")

order = np.argsort(x)
x, y = x[order], y[order]

# Find local maxima, then select the five strongest distinct peaks.
peaks, _ = find_peaks(y, distance=max(1, len(y) // 150))
if peaks.size == 0:
    peaks = np.array([int(np.argmax(y))])

strongest = peaks[np.argsort(y[peaks])[::-1][:5]]
strongest = strongest[np.argsort(x[strongest])]

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(x, y, color="#1f77b4", linewidth=1.2)
ax.scatter(x[strongest], y[strongest], color="#d62728", s=35, zorder=3)

y_range = max(float(np.max(y) - np.min(y)), 1.0)
for idx in strongest:
    ax.annotate(
        f"{x[idx]:.2f}°",
        xy=(x[idx], y[idx]),
        xytext=(0, 8),
        textcoords="offset points",
        ha="center",
        va="bottom",
        fontsize=9,
        rotation=45,
    )

ax.set_xlabel(r"2$\theta$ (deg)")
ax.set_ylabel("Intensity (counts)")
ax.set_title("XRD Pattern")
ax.set_xlim(float(np.min(x)), float(np.max(x)))
ax.set_ylim(bottom=min(0, float(np.min(y)) - 0.03 * y_range))
ax.grid(True, alpha=0.25)
fig.tight_layout()

output_path = OUT_DIR / "xrd_pattern.png"
fig.savefig(output_path, dpi=300, bbox_inches="tight")
plt.close(fig)
```
