```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B17_r2/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

sheets = ["PE", "PVC", "PET"]
offsets = {"PE": 40, "PVC": 20, "PET": 0}

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax = plt.subplots(figsize=(9, 6))

for sheet in sheets:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
    weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
    offset = offsets[sheet]
    ax.plot(temperature, weight + offset, linewidth=1.8, label=f"{sheet} (+{offset}%)")

ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%) with vertical offset")
ax.set_title("Polymer TGA weight curves (20% vertical offset)")
ax.set_xlim(30, 900)
ax.set_ylim(0, 160)
ax.grid(True, alpha=0.3)
ax.legend(title="Sample", frameon=False)
fig.tight_layout()

fig.savefig(OUT_DIR / "polymer_weight_curves_offset.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
