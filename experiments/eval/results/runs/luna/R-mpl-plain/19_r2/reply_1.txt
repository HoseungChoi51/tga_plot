```python
from pathlib import Path

import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/19_r2/out"

GRAY = "#808080"
FONT_SIZE = 13

plt.rcParams.update({
    "font.family": "Noto Sans CJK KR",
    "font.size": FONT_SIZE,
    "text.color": GRAY,
    "axes.labelcolor": GRAY,
    "axes.titlecolor": GRAY,
    "xtick.color": GRAY,
    "ytick.color": GRAY,
    "legend.labelcolor": GRAY,
})

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df.columns = [str(col).strip() for col in df.columns]

    required = {"Temperature", "Weight(%)", "D.W(%/C)"}
    if not required.issubset(df.columns):
        continue

    data = df[["Temperature", "Weight(%)", "D.W(%/C)"]].apply(
        pd.to_numeric, errors="coerce"
    ).dropna()

    if data.empty:
        continue

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    line1, = ax1.plot(
        data["Temperature"],
        data["Weight(%)"],
        color="#1f77b4",
        linewidth=1.8,
        label="Weight (%)",
    )
    line2, = ax2.plot(
        data["Temperature"],
        data["D.W(%/C)"],
        color="#d95f02",
        linewidth=1.6,
        label="D.W (%/°C)",
    )

    ax1.set_title(sheet_name, color=GRAY, fontsize=FONT_SIZE)
    ax1.set_xlabel("Temperature (°C)", color=GRAY, fontsize=FONT_SIZE)
    ax1.set_ylabel("Weight (%)", color=GRAY, fontsize=FONT_SIZE)
    ax2.set_ylabel("D.W (%/°C)", color=GRAY, fontsize=FONT_SIZE)

    ax1.tick_params(axis="both", colors=GRAY, labelsize=FONT_SIZE)
    ax2.tick_params(axis="y", colors=GRAY, labelsize=FONT_SIZE)
    ax1.spines["bottom"].set_color(GRAY)
    ax1.spines["left"].set_color(GRAY)
    ax2.spines["right"].set_color(GRAY)
    ax1.grid(True, linestyle="--", alpha=0.3)

    legend = ax1.legend(
        [line1, line2],
        [line1.get_label(), line2.get_label()],
        loc="best",
        frameon=False,
        fontsize=FONT_SIZE,
    )
    for text in legend.get_texts():
        text.set_color(GRAY)

    fig.tight_layout()
    safe_name = "".join(
        char if char.isalnum() or char in ("-", "_") else "_" for char in sheet_name
    )
    fig.savefig(out_dir / f"{safe_name}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
