from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/17_r1/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheets = pd.read_excel(DATA_FILE, sheet_name=None)
colors = ["#1f77b4", "#d62728", "#2ca02c"]

fig, ax_weight = plt.subplots(figsize=(8, 5))
ax_dtg = ax_weight.twinx()

for (sheet_name, data), color in zip(sheets.items(), colors):
    data.columns = [str(column).strip() for column in data.columns]
    temperature = pd.to_numeric(data["Temperature"], errors="coerce")
    weight = pd.to_numeric(data["Weight(%)"], errors="coerce")
    dtg = pd.to_numeric(data["D.W(%/C)"], errors="coerce")

    valid_weight = temperature.notna() & weight.notna()
    valid_dtg = temperature.notna() & dtg.notna()

    ax_weight.plot(
        temperature[valid_weight],
        weight[valid_weight],
        color=color,
        linewidth=1.8,
        label=f"{sheet_name} 중량",
    )
    ax_dtg.plot(
        temperature[valid_dtg],
        dtg[valid_dtg],
        color=color,
        linewidth=1.5,
        linestyle="--",
        label=f"{sheet_name} 미분 중량",
    )

ax_weight.set_xlabel("온도 (°C)")
ax_weight.set_ylabel("중량 (%)")
ax_dtg.set_ylabel("미분 중량 (%/°C)")
ax_weight.set_title("TGA 및 미분 TGA 곡선")
ax_weight.grid(True, linestyle=":", alpha=0.5)

lines1, labels1 = ax_weight.get_legend_handles_labels()
lines2, labels2 = ax_dtg.get_legend_handles_labels()
ax_weight.legend(
    lines1 + lines2,
    labels1 + labels2,
    loc="best",
    fontsize=8,
    frameon=True,
)

fig.tight_layout()
fig.savefig(OUT_DIR / "TGA_comparison.png", dpi=600, bbox_inches="tight")
plt.close(fig)
