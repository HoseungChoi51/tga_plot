```python
from pathlib import Path

import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/11_r2/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df.columns = [str(column).strip() for column in df.columns]

    required = ["Temperature", "Weight(%)", "D.W(%/C)"]
    missing = [column for column in required if column not in df.columns]
    if missing:
        print(f"건너뜀: {sheet_name} — 열을 찾을 수 없음: {missing}")
        continue

    data = df[required].apply(pd.to_numeric, errors="coerce").dropna()
    if data.empty:
        print(f"건너뜀: {sheet_name} — 유효한 데이터가 없음")
        continue

    fig, ax_weight = plt.subplots(figsize=(9, 6))
    ax_derivative = ax_weight.twinx()

    ax_weight.plot(
        data["Temperature"],
        data["Weight(%)"],
        color="#2468a2",
        linewidth=2,
        label="Weight(%)",
    )
    ax_derivative.plot(
        data["Temperature"],
        data["D.W(%/C)"],
        color="#d46a28",
        linewidth=1.8,
        label="D.W(%/C)",
    )

    ax_weight.set_xlabel("Temperature (°C)")
    ax_weight.set_ylabel("Weight (%)", color="#2468a2")
    ax_derivative.set_ylabel("D.W (%/°C)", color="#d46a28")
    ax_weight.tick_params(axis="y", labelcolor="#2468a2")
    ax_derivative.tick_params(axis="y", labelcolor="#d46a28")
    ax_weight.set_title(sheet_name)
    ax_weight.grid(True, linestyle="--", alpha=0.35)

    # 피크 마커나 주석 없이 곡선만 표시
    lines = ax_weight.get_lines() + ax_derivative.get_lines()
    ax_weight.legend(lines, [line.get_label() for line in lines], loc="best", frameon=False)

    fig.tight_layout()
    output_name = "".join(
        character if character.isalnum() or character in "-_ " else "_"
        for character in sheet_name
    ).strip().replace(" ", "_")
    fig.savefig(OUT_DIR / f"{output_name}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
