from pathlib import Path

import pandas as pd
import matplotlib.pyplot as plt


DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/15_r2/out")


def find_column(columns, keyword):
    """열 이름에 keyword가 포함된 첫 번째 열을 찾습니다."""
    for column in columns:
        if keyword.lower() in str(column).lower():
            return column
    return None


def main():
    OUT_DIR.mkdir(parents=True, exist_ok=True)
    sheets = pd.read_excel(DATA_FILE, sheet_name=None)

    plt.rcParams["font.family"] = "Noto Sans CJK KR"
    plt.rcParams["axes.unicode_minus"] = False

    fig, ax = plt.subplots(figsize=(9, 6))
    colors = plt.get_cmap("tab10").colors

    plotted = 0
    for sheet_name, df in sheets.items():
        temperature_col = find_column(df.columns, "temperature")
        weight_col = find_column(df.columns, "weight")

        if temperature_col is None or weight_col is None:
            continue

        data = df[[temperature_col, weight_col]].apply(pd.to_numeric, errors="coerce").dropna()
        if data.empty:
            continue

        ax.plot(
            data[temperature_col],
            data[weight_col],
            label=sheet_name,
            color=colors[plotted % len(colors)],
            linewidth=1.8,
        )
        plotted += 1

    ax.set_xlabel("Temperature (°C)")
    ax.set_ylabel("Weight (%)")
    ax.set_title("TGA")
    ax.grid(True, linestyle="--", alpha=0.35)
    ax.legend(title="시트", frameon=True)
    fig.tight_layout()

    output_path = OUT_DIR / "TGA_all_sheets.png"
    fig.savefig(output_path, dpi=300, bbox_inches="tight")
    plt.close(fig)


if __name__ == "__main__":
    main()
