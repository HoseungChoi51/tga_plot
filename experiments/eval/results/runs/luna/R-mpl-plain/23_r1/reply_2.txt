```python
from pathlib import Path
import csv
import re

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/23_r1/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)


def split_fields(line):
    """Split tab/comma/semicolon/whitespace-delimited instrument export rows."""
    line = line.strip()
    if not line:
        return []
    if "\t" in line:
        return next(csv.reader([line], delimiter="\t", skipinitialspace=True))
    if "," in line:
        return next(csv.reader([line], delimiter=",", skipinitialspace=True))
    if ";" in line:
        return next(csv.reader([line], delimiter=";", skipinitialspace=True))
    return re.split(r"\s+", line)


def parse_number(value):
    value = value.strip().replace(",", ".")
    try:
        return float(value)
    except ValueError:
        return np.nan


lines = DATA_FILE.read_text(encoding="utf-8", errors="replace").splitlines()

# Locate the first row containing the four numeric data columns.
start = None
for i, line in enumerate(lines):
    fields = split_fields(line)
    if len(fields) >= 4 and sum(np.isfinite(parse_number(v)) for v in fields[:4]) >= 3:
        start = i
        break

if start is None:
    raise ValueError(f"No numeric data rows found in {DATA_FILE}")

rows = []
for line in lines[start:]:
    fields = split_fields(line)
    if len(fields) < 4:
        continue
    values = [parse_number(v) for v in fields[:4]]
    if np.isfinite(values).sum() >= 3:
        rows.append(values)

data = pd.DataFrame(rows, columns=["Time", "Temperature", "Mass", "Derivative"])
data = data.dropna(subset=["Temperature", "Mass"])
if data.empty:
    raise ValueError(f"No usable data rows found in {DATA_FILE}")

temperature = data["Temperature"].to_numpy()
mass = data["Mass"].to_numpy()

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax = plt.subplots(figsize=(9, 6))
ax.plot(temperature, mass, color="tab:blue", linewidth=1.8)
ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Mass (%)")
ax.set_xlim(np.nanmin(temperature), np.nanmax(temperature))
ax.grid(True, alpha=0.25)

residue = mass[-1]
ax.text(
    0.97,
    0.95,
    f"Residue: {residue:.2f}%",
    transform=ax.transAxes,
    ha="right",
    va="top",
    fontsize=12,
    bbox={"boxstyle": "round,pad=0.35", "facecolor": "white", "edgecolor": "0.7", "alpha": 0.9},
)

fig.tight_layout()
fig.savefig(OUT_DIR / "tga_plot.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
