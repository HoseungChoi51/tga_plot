from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B22_r1/out"

# Korean font and consistent minus-sign rendering
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

# Read the Raw worksheet and identify temperature and weight columns by name.
df = pd.read_excel(DATA_FILE, sheet_name="Raw")
temp_col = next(c for c in df.columns if "Temperature" in str(c))
weight_col = next(c for c in df.columns if "Weight" in str(c) and "Deriv" not in str(c))

data = df[[temp_col, weight_col]].apply(pd.to_numeric, errors="coerce").dropna()
data = data.sort_values(temp_col)

temperature = data[temp_col].to_numpy()
weight = data[weight_col].to_numpy()

# Use linearly interpolated weight at 150 °C as the 100% dry-basis reference.
reference_temp = 150.0
if reference_temp < temperature.min() or reference_temp > temperature.max():
    raise ValueError(f"{reference_temp:g} °C is outside the measured temperature range.")

reference_weight = np.interp(reference_temp, temperature, weight)
if not np.isfinite(reference_weight) or reference_weight == 0:
    raise ValueError("Could not determine a valid weight at 150 °C.")

dry_basis_weight = weight / reference_weight * 100.0

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

fig, ax = plt.subplots(figsize=(8, 5.5))
ax.plot(temperature, dry_basis_weight, color="#245a8d", linewidth=2)

ax.axvline(reference_temp, color="0.45", linestyle="--", linewidth=1)
ax.axhline(100, color="0.65", linestyle=":", linewidth=1)
ax.scatter([reference_temp], [100], color="#c44e52", zorder=3, s=35)

ax.set_xlabel("온도 (°C)")
ax.set_ylabel("무게 (건조 기준, %)")
ax.set_title("Raw 시트: 150 °C 무게를 100%로 정규화")
ax.grid(True, linestyle=":", linewidth=0.7, alpha=0.6)
ax.margins(x=0.02)
fig.tight_layout()

fig.savefig(out_dir / "raw_dry_basis_150C.png", dpi=300, bbox_inches="tight")
plt.close(fig)
