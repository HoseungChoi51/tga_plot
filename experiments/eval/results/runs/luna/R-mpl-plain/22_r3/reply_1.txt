```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/22_r3/out")

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False
OUT_DIR.mkdir(parents=True, exist_ok=True)

excel = pd.ExcelFile(DATA_FILE)

for sheet_name in excel.sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    df.columns = [str(col).strip() for col in df.columns]

    temperature_col = next(
        (col for col in df.columns if "temperature" in col.lower()),
        df.columns[1],
    )
    weight_col = next(
        (col for col in df.columns if "weight" in col.lower()),
        df.columns[2],
    )
    derivative_col = next(
        (col for col in df.columns if "d.w" in col.lower() or "dw" in col.lower()),
        df.columns[3],
    )

    temperature = pd.to_numeric(df[temperature_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    derivative = pd.to_numeric(df[derivative_col], errors="coerce")
    valid = temperature.notna()

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax1.plot(temperature[valid], weight[valid], color="tab:blue", linewidth=1.8)
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="tab:blue")
    ax1.tick_params(axis="y", labelcolor="tab:blue")
    ax1.grid(True, linestyle="--", alpha=0.35)

    ax2 = ax1.twinx()
    ax2.plot(
        temperature[valid],
        derivative[valid],
        color="green",
        linewidth=1.8,
    )
    ax2.set_ylabel(derivative_col, color="green")
    ax2.tick_params(axis="y", labelcolor="green")

    ax1.set_title(sheet_name)
    fig.tight_layout()

    output_path = OUT_DIR / f"{sheet_name}.png"
    fig.savefig(output_path, dpi=300, bbox_inches="tight")
    plt.close(fig)
```
