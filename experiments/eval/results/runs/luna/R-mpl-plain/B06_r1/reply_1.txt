from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/isothermal.txt")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B06_r1/out")

# 파일의 구분자를 자동으로 판별해 읽습니다.
df = pd.read_csv(DATA_FILE, sep=None, engine="python", comment="#")
df.columns = [str(col).strip() for col in df.columns]

time_col = "Time (min)"
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"

for col in (time_col, temp_col, weight_col):
    if col not in df.columns:
        raise ValueError(f"필요한 열을 찾을 수 없습니다: {col}")

for col in (time_col, temp_col, weight_col):
    df[col] = pd.to_numeric(df[col], errors="coerce")
df = df.dropna(subset=[time_col, temp_col, weight_col]).sort_values(time_col)

OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax_weight = plt.subplots(figsize=(9, 5.5))
ax_temp = ax_weight.twinx()

weight_line, = ax_weight.plot(
    df[time_col], df[weight_col],
    color="tab:blue", linewidth=2, label="무게"
)
temp_line, = ax_temp.plot(
    df[time_col], df[temp_col],
    color="tab:red", linewidth=2, label="온도"
)

# 140분 데이터가 있으면 해당 값을 사용하고, 없으면 선형 보간합니다.
target_time = 140.0
weight_at_140 = float(
    df.set_index(time_col)[weight_col].reindex(
        df[time_col].union(pd.Index([target_time])).sort_values()
    ).interpolate(method="index").loc[target_time]
)

ax_weight.scatter(
    [target_time], [weight_at_140],
    color="tab:blue", edgecolor="white", linewidth=0.8, s=65, zorder=5
)
ax_weight.annotate(
    f"{weight_at_140:.2f}%",
    xy=(target_time, weight_at_140),
    xytext=(-10, 12),
    textcoords="offset points",
    ha="right",
    color="tab:blue",
    fontsize=10,
    fontweight="bold",
)

ax_weight.set_xlabel("시간 (분)")
ax_weight.set_ylabel("무게 (%)", color="tab:blue")
ax_temp.set_ylabel("온도 (°C)", color="tab:red")
ax_weight.tick_params(axis="y", labelcolor="tab:blue")
ax_temp.tick_params(axis="y", labelcolor="tab:red")
ax_weight.grid(True, linestyle="--", alpha=0.35)
ax_weight.set_xlim(left=0)

ax_weight.legend(
    [weight_line, temp_line],
    ["무게", "온도"],
    loc="best",
    frameon=False,
)

fig.tight_layout()
fig.savefig(OUT_DIR / "isothermal_weight_temperature.png", dpi=300, bbox_inches="tight")
plt.close(fig)
