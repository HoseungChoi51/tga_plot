```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/23_r2/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)


def read_text_file(path):
    """Read instrument text exports with common Unicode and legacy encodings."""
    raw = path.read_bytes()
    for encoding in ("utf-8-sig", "utf-16", "cp1252", "latin1"):
        try:
            return raw.decode(encoding)
        except UnicodeDecodeError:
            continue
    return raw.decode("latin1", errors="replace")


# The export may contain metadata before the numeric data. Keep rows with at
# least three numeric fields, corresponding to temperature, mass, and derivative.
text = read_text_file(DATA_FILE)
rows = []
for line in text.splitlines():
    fields = line.replace(",", " ").replace(";", " ").split()
    values = []
    for field in fields:
        try:
            values.append(float(field))
        except ValueError:
            continue
    if len(values) >= 3:
        rows.append(values)

if not rows:
    raise ValueError("파일에서 유효한 숫자 데이터를 찾지 못했습니다.")

# Keep the first four numeric values per row, padding short rows as needed.
width = max(len(row) for row in rows)
table = pd.DataFrame([row + [np.nan] * (width - len(row)) for row in rows])
table = table.dropna(axis=1, how="all")

if table.shape[1] < 3:
    raise ValueError("온도와 질량 데이터 열을 찾지 못했습니다.")

temperature = pd.to_numeric(table.iloc[:, 1], errors="coerce").to_numpy()
mass = pd.to_numeric(table.iloc[:, 2], errors="coerce").to_numpy()
valid = np.isfinite(temperature) & np.isfinite(mass)
temperature, mass = temperature[valid], mass[valid]

if temperature.size == 0:
    raise ValueError("유효한 온도 및 질량 데이터가 없습니다.")

residue = float(mass[-1])

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax = plt.subplots(figsize=(8, 5.5))
ax.plot(temperature, mass, color="#1f77b4", linewidth=1.8)
ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Mass (%)")
ax.set_title("Thermogravimetric Analysis")
ax.grid(True, alpha=0.3)
ax.text(
    0.97,
    0.95,
    f"Residue: {residue:.2f}%",
    transform=ax.transAxes,
    ha="right",
    va="top",
    fontsize=12,
    bbox={
        "boxstyle": "round,pad=0.35",
        "facecolor": "white",
        "edgecolor": "0.7",
        "alpha": 0.9,
    },
)

fig.tight_layout()
fig.savefig(OUT_DIR / "tga_plot.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
