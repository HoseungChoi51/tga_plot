from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILES = [
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_A.csv",
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_B.txt",
]
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B05_r2/out"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def read_data(path):
    """Read CSV or whitespace/tab-delimited text, using the first row as headers."""
    suffix = Path(path).suffix.lower()
    if suffix == ".csv":
        return pd.read_csv(path)

    # Try common text delimiters; fall back to automatic separator detection.
    for sep in ("\t", r"\s+", ",", ";"):
        try:
            df = pd.read_csv(path, sep=sep, engine="python")
            if df.shape[1] >= 4:
                return df
        except Exception:
            pass
    return pd.read_csv(path, sep=None, engine="python")


def find_column(df, name):
    for col in df.columns:
        if str(col).strip().lower() == name.lower():
            return col
    raise ValueError(f"Required column not found: {name}. Available columns: {list(df.columns)}")


def main():
    out_dir = Path(OUT_DIR)
    out_dir.mkdir(parents=True, exist_ok=True)

    fig, ax = plt.subplots(figsize=(9, 6), constrained_layout=True)
    styles = [
        (DATA_FILES[0], "Lot A", "#1f77b4"),
        (DATA_FILES[1], "Lot B", "#d62728"),
    ]

    for file_path, label, color in styles:
        df = read_data(file_path)
        temperature = pd.to_numeric(
            df[find_column(df, "Temperature (°C)")], errors="coerce"
        )
        weight = pd.to_numeric(df[find_column(df, "Weight (%)")], errors="coerce")
        valid = temperature.notna() & weight.notna()

        ax.plot(
            temperature[valid],
            weight[valid],
            label=label,
            color=color,
            linewidth=1.8,
        )

    ax.set_xlabel("Temperature (°C)")
    ax.set_ylabel("Weight (%)")
    ax.set_title("TGA Comparison")
    ax.grid(True, linestyle="--", alpha=0.35)
    ax.legend(frameon=False)
    ax.margins(x=0.02)

    fig.savefig(out_dir / "tga_lot_comparison.png", dpi=300, bbox_inches="tight")
    plt.close(fig)


if __name__ == "__main__":
    main()
