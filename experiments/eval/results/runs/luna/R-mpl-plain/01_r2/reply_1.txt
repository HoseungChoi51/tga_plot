from pathlib import Path

import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/01_r2/out"

def main():
    out_dir = Path(OUT_DIR)
    out_dir.mkdir(parents=True, exist_ok=True)

    plt.rcParams["font.family"] = "Noto Sans CJK KR"
    plt.rcParams["axes.unicode_minus"] = False

    excel = pd.ExcelFile(DATA_FILE)
    for sheet_name in excel.sheet_names:
        df = pd.read_excel(excel, sheet_name=sheet_name)
        df.columns = [str(column).strip() for column in df.columns]

        temperature_col = next(
            (c for c in df.columns if "temperature" in c.lower()), df.columns[1]
        )
        weight_col = next(
            (c for c in df.columns if "weight" in c.lower()), df.columns[2]
        )
        derivative_col = next(
            (c for c in df.columns if "d.w" in c.lower() or "dw" in c.lower()),
            df.columns[3] if len(df.columns) > 3 else None,
        )

        temperature = pd.to_numeric(df[temperature_col], errors="coerce")
        weight = pd.to_numeric(df[weight_col], errors="coerce")
        valid = temperature.notna() & weight.notna()

        fig, ax_weight = plt.subplots(figsize=(9, 5.5))
        ax_weight.plot(
            temperature[valid], weight[valid], color="#1f77b4", linewidth=1.8,
            label="Weight (%)",
        )
        ax_weight.set_xlabel("Temperature (°C)")
        ax_weight.set_ylabel("Weight (%)", color="#1f77b4")
        ax_weight.tick_params(axis="y", labelcolor="#1f77b4")
        ax_weight.grid(True, linestyle="--", alpha=0.35)

        if derivative_col is not None:
            derivative = pd.to_numeric(df[derivative_col], errors="coerce")
            valid_derivative = temperature.notna() & derivative.notna()
            ax_derivative = ax_weight.twinx()
            ax_derivative.plot(
                temperature[valid_derivative],
                derivative[valid_derivative],
                color="#d62728",
                linewidth=1.5,
                label="D.W (%/°C)",
            )
            ax_derivative.set_ylabel("D.W (%/°C)", color="#d62728")
            ax_derivative.tick_params(axis="y", labelcolor="#d62728")

            lines = ax_weight.get_lines() + ax_derivative.get_lines()
            ax_weight.legend(lines, [line.get_label() for line in lines], loc="best")
        else:
            ax_weight.legend(loc="best")

        ax_weight.set_title(f"TGA: {sheet_name}")
        fig.tight_layout()

        safe_name = "".join(
            character if character.isalnum() or character in "-_." else "_"
            for character in sheet_name
        )
        fig.savefig(out_dir / f"{safe_name}_TGA.png", dpi=300, bbox_inches="tight")
        plt.close(fig)

if __name__ == "__main__":
    main()
