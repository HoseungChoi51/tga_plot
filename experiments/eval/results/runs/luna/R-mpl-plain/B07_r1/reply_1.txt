from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B07_r1/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

# 두 번째 시트(PVC)의 데이터를 읽습니다.
df = pd.read_excel(DATA_FILE, sheet_name=1)
temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
valid = temperature.notna() & weight.notna()
temperature = temperature[valid].to_numpy()
weight = weight[valid].to_numpy()

# 온도순으로 정렬한 뒤 요청 온도에서 무게를 선형 보간합니다.
order = np.argsort(temperature)
temperature = temperature[order]
weight = weight[order]
target_temperatures = [300, 600]
target_weights = np.interp(target_temperatures, temperature, weight)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax = plt.subplots(figsize=(9, 5.5))
ax.plot(temperature, weight, color="#2369a1", linewidth=2, label="PVC")
colors = ["#e07a26", "#b33b3b"]

for target_temp, target_weight, color in zip(
    target_temperatures, target_weights, colors
):
    ax.axvline(target_temp, color=color, linestyle="--", linewidth=1.2, alpha=0.8)
    ax.scatter(
        [target_temp],
        [target_weight],
        color=color,
        edgecolor="white",
        linewidth=0.8,
        s=65,
        zorder=3,
        label=f"{target_temp} °C: {target_weight:.2f}%",
    )
    ax.annotate(
        f"{target_temp} °C\n{target_weight:.2f}%",
        xy=(target_temp, target_weight),
        xytext=(8, 10),
        textcoords="offset points",
        color=color,
        fontsize=10,
        ha="left",
        va="bottom",
    )

ax.set_title("PVC 열중량 분석")
ax.set_xlabel("온도 (°C)")
ax.set_ylabel("무게 (%)")
ax.set_xlim(float(np.min(temperature)), float(np.max(temperature)))
ax.grid(True, alpha=0.25)
ax.legend(frameon=False)
fig.tight_layout()

fig.savefig(OUT_DIR / "pvc_weight_at_300_600C.png", dpi=300, bbox_inches="tight")
plt.close(fig)
