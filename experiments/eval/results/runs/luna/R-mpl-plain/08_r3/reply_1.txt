from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd


DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/08_r3/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx()

colors = plt.rcParams["axes.prop_cycle"].by_key()["color"]

for i, (sheet_name, df) in enumerate(sheets.items()):
    df.columns = [str(column).strip() for column in df.columns]
    required = ["Temperature", "Weight(%)", "D.W(%/C)"]
    if not all(column in df.columns for column in required):
        continue

    data = df[required].apply(pd.to_numeric, errors="coerce").dropna()
    color = colors[i % len(colors)]

    ax1.plot(
        data["Temperature"],
        data["Weight(%)"],
        color=color,
        linewidth=1.8,
        label=f"{sheet_name} 무게",
    )
    ax2.plot(
        data["Temperature"],
        data["D.W(%/C)"],
        color=color,
        linewidth=1.5,
        linestyle="--",
        label=f"{sheet_name} 미분 무게",
    )

ax1.set_xlabel("온도 (°C)")
ax1.set_ylabel("무게 (%)")
ax2.set_ylabel("미분 무게 (%/°C)")

ax1.grid(True, linestyle=":", alpha=0.5)
ax1.set_xlim(left=0)

lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc="best", fontsize=8, frameon=True)

fig.tight_layout()
fig.savefig(OUT_DIR / "TGA.png", dpi=300, bbox_inches="tight")
plt.close(fig)
