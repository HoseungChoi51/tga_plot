```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.signal import find_peaks

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/caox.csv")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B02_r2/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

df = pd.read_csv(DATA_FILE)
df.columns = [str(column).strip() for column in df.columns]


def find_column(fragment):
    matches = [column for column in df.columns if fragment.lower() in column.lower()]
    if not matches:
        raise ValueError(f"Cannot find column containing {fragment!r}: {list(df.columns)}")
    return matches[0]


temperature = pd.to_numeric(df[find_column("Temperature")], errors="coerce").to_numpy()
weight = pd.to_numeric(df[find_column("Weight (%)")], errors="coerce").to_numpy()
heat_flow = pd.to_numeric(df[find_column("Heat Flow")], errors="coerce").to_numpy()

valid = np.isfinite(temperature) & np.isfinite(weight) & np.isfinite(heat_flow)
temperature, weight, heat_flow = temperature[valid], weight[valid], heat_flow[valid]
order = np.argsort(temperature)
temperature, weight, heat_flow = temperature[order], weight[order], heat_flow[order]

# Smooth only for peak detection; plot the original Heat Flow measurements.
window = min(9, len(heat_flow) if len(heat_flow) % 2 else len(heat_flow) - 1)
if window >= 3:
    smooth_heat = (
        pd.Series(heat_flow)
        .rolling(window=window, center=True, min_periods=1)
        .mean()
        .to_numpy()
    )
else:
    smooth_heat = heat_flow.copy()

heat_range = float(np.nanmax(smooth_heat) - np.nanmin(smooth_heat))
prominence = max(heat_range * 0.025, 1e-8)
positive_peaks, _ = find_peaks(smooth_heat, prominence=prominence)
negative_peaks, _ = find_peaks(-smooth_heat, prominence=prominence)


def strongest(indices, count):
    baseline = float(np.nanmedian(smooth_heat))
    return sorted(
        list(indices),
        key=lambda index: abs(smooth_heat[index] - baseline),
        reverse=True,
    )[:count]


positive_candidates = strongest(positive_peaks, 2)
negative_candidates = strongest(negative_peaks, 2)

# Determine which direction contains the two stronger extrema. Heat-flow
# sign conventions vary, so classify that direction as endothermic.
if len(positive_candidates) >= 2 and len(negative_candidates) >= 1:
    positive_strength = sum(
        abs(smooth_heat[i] - np.nanmedian(smooth_heat)) for i in positive_candidates
    )
    negative_strength = sum(
        abs(smooth_heat[i] - np.nanmedian(smooth_heat)) for i in negative_candidates
    )
    if positive_strength >= negative_strength:
        endo_indices = strongest(positive_peaks, 2)
        exo_indices = strongest(negative_peaks, 1)
    else:
        endo_indices = strongest(negative_peaks, 2)
        exo_indices = strongest(positive_peaks, 1)
elif len(positive_peaks) >= 2:
    endo_indices = strongest(positive_peaks, 2)
    exo_indices = strongest(negative_peaks, 1)
else:
    endo_indices = strongest(negative_peaks, 2)
    exo_indices = strongest(positive_peaks, 1)

# Fall back to prominent trace extrema if the requested number of peaks is absent.
baseline = float(np.nanmedian(smooth_heat))
if len(endo_indices) < 2:
    fallback = sorted(
        range(len(smooth_heat)),
        key=lambda i: abs(smooth_heat[i] - baseline),
        reverse=True,
    )
    for index in fallback:
        if index not in exo_indices and index not in endo_indices:
            endo_indices.append(index)
        if len(endo_indices) >= 2:
            break

if not exo_indices:
    fallback = sorted(
        range(len(smooth_heat)),
        key=lambda i: abs(smooth_heat[i] - baseline),
        reverse=True,
    )
    exo_indices = [i for i in fallback if i not in endo_indices][:1]

endo_indices = sorted(endo_indices[:2], key=lambda i: temperature[i])
exo_indices = exo_indices[:1]

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(
    temperature, weight, color="#2468a2", linewidth=1.8, label="Weight"
)
heat_line, = ax2.plot(
    temperature, heat_flow, color="#c74440", linewidth=1.6, label="Heat Flow"
)

ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#2468a2")
ax2.set_ylabel("Heat Flow (W/g)", color="#c74440")
ax1.tick_params(axis="y", labelcolor="#2468a2")
ax2.tick_params(axis="y", labelcolor="#c74440")
ax1.grid(True, linestyle="--", alpha=0.35)

for number, index in enumerate(endo_indices, start=1):
    ax2.scatter(temperature[index], heat_flow[index], color="#7b2cbf", s=48, zorder=5)
    ax2.annotate(
        f"흡열 {number}: {temperature[index]:.1f} °C",
        xy=(temperature[index], heat_flow[index]),
        xytext=(10, 16 if number == 1 else -28),
        textcoords="offset points",
        fontsize=9,
        color="#64209a",
        arrowprops=dict(arrowstyle="->", color="#64209a", lw=0.9),
        bbox=dict(boxstyle="round,pad=0.25", fc="white", ec="#b99bd4", alpha=0.9),
    )

for index in exo_indices:
    ax2.scatter(temperature[index], heat_flow[index], color="#e58b18", s=52, zorder=5)
    ax2.annotate(
        f"발열: {temperature[index]:.1f} °C",
        xy=(temperature[index], heat_flow[index]),
        xytext=(10, 18),
        textcoords="offset points",
        fontsize=9,
        color="#a45b00",
        arrowprops=dict(arrowstyle="->", color="#a45b00", lw=0.9),
        bbox=dict(boxstyle="round,pad=0.25", fc="white", ec="#e8bd7a", alpha=0.9),
    )

ax1.set_title("TGA 및 Heat Flow")
ax1.legend([weight_line, heat_line], ["Weight", "Heat Flow"], loc="best")
fig.tight_layout()
fig.savefig(OUT_DIR / "caox_tga_heat_flow_peaks.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
