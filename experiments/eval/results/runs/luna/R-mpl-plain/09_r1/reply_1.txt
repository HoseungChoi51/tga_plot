```python
from pathlib import Path

import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/09_r1/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df.columns = [str(col).strip() for col in df.columns]

    temperature = pd.to_numeric(df["Temperature"], errors="coerce")
    weight = pd.to_numeric(df["Weight(%)"], errors="coerce")
    derivative = pd.to_numeric(df["D.W(%/C)"], errors="coerce")
    valid = temperature.notna()

    fig, ax_weight = plt.subplots(figsize=(8, 5))
    ax_weight.plot(
        temperature[valid],
        weight[valid],
        color="tab:blue",
        linewidth=1.8,
        label="Weight (%)",
    )
    ax_weight.set_xlabel("Temperature (°C)")
    ax_weight.set_ylabel("Weight (%)", color="tab:blue")
    ax_weight.tick_params(axis="y", labelcolor="tab:blue", direction="out", labelsize=9)
    ax_weight.tick_params(axis="x", direction="out", labelsize=9)

    ax_derivative = ax_weight.twinx()
    ax_derivative.plot(
        temperature[valid],
        derivative[valid],
        color="tab:red",
        linewidth=1.5,
        label="D.W (%/°C)",
    )
    ax_derivative.set_ylabel("D.W (%/°C)", color="tab:red")
    ax_derivative.tick_params(
        axis="y", labelcolor="tab:red", direction="out", labelsize=9
    )

    ax_weight.set_title(sheet_name)
    ax_weight.grid(True, linestyle="--", alpha=0.35)
    fig.tight_layout()

    output_name = "".join(
        char if char.isalnum() or char in "-_ " else "_" for char in sheet_name
    ).strip().replace(" ", "_")
    fig.savefig(OUT_DIR / f"{output_name}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
