```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/06_r2/out")

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False
OUT_DIR.mkdir(parents=True, exist_ok=True)


def find_column(df, candidates):
    lowered = {str(col).strip().lower(): col for col in df.columns}
    for candidate in candidates:
        for key, original in lowered.items():
            if candidate in key:
                return original
    return None


def safe_filename(name):
    return "".join(c if c.isalnum() or c in "-_ ." else "_" for c in str(name)).strip()


sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    x_col = find_column(df, ["temperature", "temp"])
    weight_col = find_column(df, ["weight"])
    dtg_col = find_column(df, ["d.w", "dtg", "derivative"])

    if x_col is None or weight_col is None or dtg_col is None:
        numeric_cols = df.select_dtypes(include="number").columns.tolist()
        if len(numeric_cols) < 3:
            continue
        x_col, weight_col, dtg_col = numeric_cols[:3]

    data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
    if data.empty:
        continue

    x = data[x_col].to_numpy()
    weight = data[weight_col].to_numpy()
    dtg = data[dtg_col].to_numpy()

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    ax1.plot(x, weight, color="#1f4e9c", linewidth=1.8)
    ax2.plot(x, -dtg, color="#c0392b", linewidth=1.5)

    ax1.set_title(str(sheet_name))
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="#1f4e9c")
    ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
    ax1.tick_params(axis="y", labelcolor="#1f4e9c")
    ax2.tick_params(axis="y", labelcolor="#c0392b")

    # 요청에 따라 격자는 켜고 범례는 표시하지 않습니다.
    ax1.grid(True, which="major", linestyle="--", linewidth=0.6, alpha=0.6)

    residue_idx = int(np.argmin(np.abs(x - 800)))
    ax1.plot(x[residue_idx], weight[residue_idx], marker="o", color="#1f4e9c")
    ax1.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        (x[residue_idx], weight[residue_idx]),
        xytext=(8, 8),
        textcoords="offset points",
        color="#1f4e9c",
    )

    peak_candidates = np.flatnonzero(x > 150)
    if peak_candidates.size:
        peak_idx = peak_candidates[np.argmax((-dtg)[peak_candidates])]
        ax2.plot(x[peak_idx], -dtg[peak_idx], marker="o", color="#c0392b")
        ax2.annotate(
            f"{x[peak_idx]:.0f} °C",
            (x[peak_idx], -dtg[peak_idx]),
            xytext=(8, 8),
            textcoords="offset points",
            color="#c0392b",
        )

    fig.tight_layout()
    fig.savefig(OUT_DIR / f"{safe_filename(sheet_name)}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
