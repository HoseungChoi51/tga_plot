from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt


DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/01_r2/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def find_column(columns, candidates):
    """Return the first column whose normalized name contains a candidate."""
    normalized = {col: str(col).strip().lower().replace(" ", "") for col in columns}
    for candidate in candidates:
        key = candidate.lower().replace(" ", "")
        for col, name in normalized.items():
            if key in name:
                return col
    return None


def safe_filename(name):
    cleaned = "".join("_" if char in '<>:"/\\|?*' else char for char in str(name))
    return cleaned.strip().replace(" ", "_") or "sheet"


sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, data in sheets.items():
    x_col = find_column(data.columns, ["temperature", "temp", "온도"])
    weight_col = find_column(data.columns, ["weight(%)", "weight", "무게", "중량"])
    dtg_col = find_column(data.columns, ["d.w", "dtg", "derivative", "미분"])

    if x_col is None or weight_col is None or dtg_col is None:
        if data.shape[1] < 4:
            continue
        x_col, weight_col, dtg_col = data.columns[1], data.columns[2], data.columns[3]

    x = pd.to_numeric(data[x_col], errors="coerce").to_numpy(dtype=float)
    weight = pd.to_numeric(data[weight_col], errors="coerce").to_numpy(dtype=float)
    dtg = pd.to_numeric(data[dtg_col], errors="coerce").to_numpy(dtype=float)
    valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
    x, weight, dtg = x[valid], weight[valid], dtg[valid]

    if x.size == 0:
        continue

    order = np.argsort(x)
    x, weight, dtg = x[order], weight[order], dtg[order]

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    line1, = ax1.plot(x, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
    line2, = ax2.plot(x, -dtg, color="#c0392b", linewidth=1.6, label="-DTG (%/°C)")

    # Residue: measured weight at the point nearest 800 °C.
    residue_idx = int(np.argmin(np.abs(x - 800)))
    ax1.scatter(x[residue_idx], weight[residue_idx], color="#1f4e9c", s=35, zorder=5)
    ax1.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        xy=(x[residue_idx], weight[residue_idx]),
        xytext=(8, 10),
        textcoords="offset points",
        color="#1f4e9c",
        fontsize=9,
    )

    # Peak: highest positive -DTG signal above 150 °C.
    peak_candidates = np.flatnonzero((x > 150) & np.isfinite(dtg))
    if peak_candidates.size:
        peak_idx = peak_candidates[np.argmax(-dtg[peak_candidates])]
        ax2.scatter(x[peak_idx], -dtg[peak_idx], color="#c0392b", s=35, zorder=5)
        ax2.annotate(
            f"{x[peak_idx]:.0f} °C",
            xy=(x[peak_idx], -dtg[peak_idx]),
            xytext=(8, 10),
            textcoords="offset points",
            color="#c0392b",
            fontsize=9,
        )

    ax1.set_title(str(sheet_name))
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="#1f4e9c")
    ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
    ax1.tick_params(axis="y", labelcolor="#1f4e9c")
    ax2.tick_params(axis="y", labelcolor="#c0392b")
    ax1.grid(True, linestyle="--", alpha=0.35)
    ax1.legend([line1, line2], [line1.get_label(), line2.get_label()], loc="center left")
    fig.tight_layout()

    output_path = OUT_DIR / f"{safe_filename(sheet_name)}.png"
    fig.savefig(output_path, dpi=300, bbox_inches="tight")
    plt.close(fig)
