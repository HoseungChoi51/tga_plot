```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B16_r2/out"

# Korean-capable font
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

# Read the CSV, allowing common encodings.
try:
    df = pd.read_csv(DATA_FILE)
except UnicodeDecodeError:
    df = pd.read_csv(DATA_FILE, encoding="cp949")

# Identify columns from their names, with positional fallbacks.
def find_column(keywords, fallback_index):
    for col in df.columns:
        name = str(col).strip().lower()
        if any(keyword in name for keyword in keywords):
            return col
    return df.columns[fallback_index]

x_col = find_column(["temperature", "temp"], 1)
weight_col = find_column(["weight"], 2)
dtg_col = find_column(["deriv", "dtg"], 3)

data = df[[x_col, weight_col, dtg_col]].copy()
for col in data.columns:
    data[col] = pd.to_numeric(data[col], errors="coerce")
data = data.dropna().sort_values(x_col)

temperature = data[x_col].to_numpy()
weight = data[weight_col].to_numpy()
dtg = data[dtg_col].to_numpy()

# Separate, vertically stacked panels sharing the temperature axis.
fig, (ax_weight, ax_dtg) = plt.subplots(
    2, 1, figsize=(9, 7), sharex=True,
    gridspec_kw={"height_ratios": [1, 1], "hspace": 0.08}
)

ax_weight.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight")
ax_weight.set_ylabel("Weight (%)")
ax_weight.set_title(Path(DATA_FILE).stem)
ax_weight.grid(True, alpha=0.25)
ax_weight.legend(loc="best")

# Plot DTG inverted so mass-loss peaks point upward, as in the standard TGA plot.
ax_dtg.plot(temperature, -dtg, color="#c0392b", linewidth=1.6, label="−DTG")
ax_dtg.set_xlabel("Temperature (°C)")
ax_dtg.set_ylabel("−DTG (%/°C)")
ax_dtg.grid(True, alpha=0.25)
ax_dtg.legend(loc="best")

# Residue label and marker at the measured point nearest 800 °C.
residue_idx = int(np.argmin(np.abs(temperature - 800)))
ax_weight.plot(temperature[residue_idx], weight[residue_idx], "o", color="#1f4e9c")
ax_weight.annotate(
    f"Residue {weight[residue_idx]:.1f}%",
    xy=(temperature[residue_idx], weight[residue_idx]),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
    fontsize=9,
)

# Peak label and marker for the largest DTG mass-loss peak above 150 °C.
peak_candidates = np.flatnonzero(temperature > 150)
if peak_candidates.size:
    peak_idx = peak_candidates[np.argmax(-dtg[peak_candidates])]
    ax_dtg.plot(temperature[peak_idx], -dtg[peak_idx], "o", color="#c0392b")
    ax_dtg.annotate(
        f"{temperature[peak_idx]:.0f} °C",
        xy=(temperature[peak_idx], -dtg[peak_idx]),
        xytext=(8, 10),
        textcoords="offset points",
        color="#c0392b",
        fontsize=9,
    )

fig.tight_layout()
fig.savefig(out_dir / "caox_separate_weight_dtg.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
