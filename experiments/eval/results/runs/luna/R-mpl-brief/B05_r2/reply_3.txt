```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILES = [
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_A.csv",
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_B.txt",
]
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B05_r2/out"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def read_data(path):
    """Read CSV or text data, accommodating whitespace and unusual text delimiters."""
    if Path(path).suffix.lower() == ".csv":
        return pd.read_csv(path)

    # sep=None can infer delimiters from text files; try explicit formats too.
    attempts = [
        {"sep": None, "engine": "python"},
        {"sep": r"\s+", "engine": "python"},
        {"sep": "\t", "engine": "python"},
        {"sep": ",", "engine": "python"},
        {"sep": ";", "engine": "python"},
    ]
    for kwargs in attempts:
        try:
            frame = pd.read_csv(path, **kwargs)
            if frame.shape[1] >= 4:
                return frame
        except Exception:
            continue

    # Last resort: skip non-tabular leading lines and infer whitespace columns.
    return pd.read_csv(path, sep=r"\s+", engine="python", comment="#")


def normalize_name(value):
    return "".join(ch.lower() for ch in str(value) if ch.isalnum())


def find_column(frame, terms, fallback_index):
    normalized_terms = [normalize_name(term) for term in terms]
    for col in frame.columns:
        name = normalize_name(col)
        if any(term in name for term in normalized_terms):
            return col

    # The supplied files have the standard TGA layout:
    # Time, Temperature, Weight, Derivative Weight.
    if frame.shape[1] >= 4:
        return frame.columns[fallback_index]
    raise KeyError(f"Required column not found; columns are {list(frame.columns)}")


def numeric_series(series):
    # Handles numeric text with commas, spaces, and unit annotations.
    text = series.astype(str).str.strip().str.replace(",", "", regex=False)
    return pd.to_numeric(text, errors="coerce")


fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()
colors = ("#1f4e9c", "#c0392b")
styles = ("-", "--")

for path, label, color, style in zip(DATA_FILES, ("Lot A", "Lot B"), colors, styles):
    data = read_data(path)

    x_col = find_column(data, ("temperature", "temp"), 1)
    weight_col = find_column(data, ("weight", "mass"), 2)
    dtg_col = find_column(data, ("deriv", "dtg"), 3)

    x = numeric_series(data[x_col]).to_numpy(dtype=float)
    weight = numeric_series(data[weight_col]).to_numpy(dtype=float)
    dtg = numeric_series(data[dtg_col]).to_numpy(dtype=float)

    valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
    x, weight, dtg = x[valid], weight[valid], dtg[valid]

    # Some text files may have been parsed as one column. Re-read as whitespace
    # separated data and use the standard four-column TGA arrangement.
    if x.size == 0 and Path(path).suffix.lower() != ".csv":
        data = pd.read_csv(path, sep=r"\s+", engine="python", header=0)
        if data.shape[1] >= 4:
            x = numeric_series(data.iloc[:, 1]).to_numpy(dtype=float)
            weight = numeric_series(data.iloc[:, 2]).to_numpy(dtype=float)
            dtg = numeric_series(data.iloc[:, 3]).to_numpy(dtype=float)
            valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
            x, weight, dtg = x[valid], weight[valid], dtg[valid]

    if x.size == 0:
        raise ValueError(f"No numeric TGA rows found in {path}")

    order = np.argsort(x)
    x, weight, dtg = x[order], weight[order], dtg[order]

    ax1.plot(x, weight, color=color, linestyle=style, linewidth=1.8, label=label)
    ax2.plot(x, -dtg, color=color, linestyle=style, linewidth=1.5, alpha=0.7,
             label=f"{label} DTG")

    residue_idx = int(np.argmin(np.abs(x - 800)))
    ax1.plot(x[residue_idx], weight[residue_idx], marker="o", color=color,
             markersize=5, linestyle="None")
    ax1.annotate(
        f"{label} residue {weight[residue_idx]:.1f}%",
        (x[residue_idx], weight[residue_idx]),
        xytext=(6, 7), textcoords="offset points", color=color, fontsize=9,
    )

    peak_mask = x > 150
    if np.any(peak_mask):
        peak_indices = np.flatnonzero(peak_mask)
        peak_idx = peak_indices[int(np.argmax((-dtg)[peak_mask]))]
        ax2.plot(x[peak_idx], -dtg[peak_idx], marker="^", color=color,
                 markersize=6, linestyle="None")
        ax2.annotate(
            f"{x[peak_idx]:.0f} °C",
            (x[peak_idx], -dtg[peak_idx]),
            xytext=(6, -13), textcoords="offset points", color=color, fontsize=9,
        )

ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("DTG (-dWeight/dT, %/°C)")
ax1.set_title("Lot A and Lot B")
ax1.grid(True, alpha=0.25)

handles1, labels1 = ax1.get_legend_handles_labels()
handles2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(handles1 + handles2, labels1 + labels2, loc="center left", frameon=True)

fig.tight_layout()
out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)
fig.savefig(out_dir / "cnt_lot_A_vs_B.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
