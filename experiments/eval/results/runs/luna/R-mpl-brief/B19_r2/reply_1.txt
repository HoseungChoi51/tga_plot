```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B19_r2/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

# PVC 시트 데이터 불러오기
df = pd.read_excel(DATA_FILE, sheet_name="PVC")
df.columns = [str(column).strip() for column in df.columns]

def find_column(names, keywords):
    for column in names:
        if any(keyword in column.lower() for keyword in keywords):
            return column
    raise ValueError(f"필요한 열을 찾지 못했습니다: {keywords}")

temperature_col = find_column(df.columns, ["temperature", "온도"])
weight_col = find_column(df.columns, ["weight", "무게", "질량"])

data = df[[temperature_col, weight_col]].apply(pd.to_numeric, errors="coerce").dropna()
data = data.sort_values(temperature_col)

# 200 °C와 400 °C에서의 질량은 실제 측정점으로부터 선형 보간
temperature = data[temperature_col].to_numpy()
weight = data[weight_col].to_numpy()

if temperature.min() > 200 or temperature.max() < 400:
    raise ValueError("데이터 온도 범위에 200–400 °C가 모두 포함되어 있지 않습니다.")

weight_200 = np.interp(200, temperature, weight)
weight_400 = np.interp(400, temperature, weight)
mass_loss = weight_200 - weight_400

fig, ax = plt.subplots(figsize=(9, 5.5))
ax.plot(temperature, weight, color="#1f4e9c", linewidth=2, label="질량")

ax.axvline(200, color="gray", linestyle="--", linewidth=1)
ax.axvline(400, color="gray", linestyle="--", linewidth=1)
ax.scatter([200, 400], [weight_200, weight_400], color="#1f4e9c", zorder=3)

# 감소 구간을 음영 처리하고 질량 감소량을 그래프에 표시
mask = (temperature >= 200) & (temperature <= 400)
ax.fill_between(temperature[mask], weight[mask], weight_200, color="#c0392b", alpha=0.18)

annotation_y = (weight_200 + weight_400) / 2
ax.annotate(
    f"200–400 °C 질량 감소량: {mass_loss:.2f}%",
    xy=(300, annotation_y),
    xytext=(0.53, 0.18),
    textcoords="axes fraction",
    ha="center",
    va="center",
    bbox=dict(boxstyle="round,pad=0.4", facecolor="white", edgecolor="#c0392b", alpha=0.95),
    arrowprops=dict(arrowstyle="->", color="#c0392b", linewidth=1.2),
)

ax.set_title("PVC")
ax.set_xlabel("온도 (°C)")
ax.set_ylabel("질량 (%)")
ax.set_xlim(left=temperature.min(), right=temperature.max())
ax.grid(True, linestyle=":", alpha=0.5)
ax.legend(loc="best")
fig.tight_layout()

fig.savefig(OUT_DIR / "PVC_mass_loss_200_400C.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
