```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/07_r1/out")

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

OUT_DIR.mkdir(parents=True, exist_ok=True)


def find_column(columns, keywords):
    names = [str(col) for col in columns]
    lowered = [name.lower() for name in names]
    for keyword in keywords:
        for index, name in enumerate(lowered):
            if keyword in name:
                return names[index]
    return None


def safe_filename(name):
    cleaned = "".join(ch if ch.isalnum() or ch in "-_ ." else "_" for ch in str(name))
    return cleaned.strip().replace(" ", "_") or "sheet"


sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df = df.dropna(how="all")
    if df.empty:
        continue

    x_col = find_column(df.columns, ["temperature", "temp"])
    weight_col = find_column(df.columns, ["weight", "mass"])
    dtg_col = find_column(df.columns, ["d.w", "dtg", "derivative", "dw"])

    if x_col is None or weight_col is None or dtg_col is None:
        continue

    data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
    if data.empty:
        continue

    temperature = data[x_col].to_numpy()
    weight = data[weight_col].to_numpy()
    dtg = data[dtg_col].to_numpy()

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    weight_line, = ax1.plot(
        temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)"
    )
    dtg_line, = ax2.plot(
        temperature, -dtg, color="#c0392b", linewidth=1.5, label="-DTG"
    )

    ax1.set_title(str(sheet_name))
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="#1f4e9c")
    ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
    ax1.tick_params(axis="y", labelcolor="#1f4e9c")
    ax2.tick_params(axis="y", labelcolor="#c0392b")
    ax1.grid(True, alpha=0.25)

    # Residue at the measured point nearest 800 °C.
    residue_idx = int(np.argmin(np.abs(temperature - 800)))
    residue_temp = temperature[residue_idx]
    residue_value = weight[residue_idx]
    ax1.scatter([residue_temp], [residue_value], color="#1f4e9c", s=28, zorder=4)
    ax1.annotate(
        f"Residue {residue_value:.1f}%",
        (residue_temp, residue_value),
        xytext=(8, 10),
        textcoords="offset points",
        color="#1f4e9c",
        fontsize=9,
    )

    # Highest positive displayed DTG peak above 150 °C.
    peak_candidates = np.flatnonzero((temperature > 150) & (-dtg > 0))
    if peak_candidates.size:
        peak_idx = peak_candidates[np.argmax((-dtg)[peak_candidates])]
        peak_temp = temperature[peak_idx]
        ax2.scatter([peak_temp], [(-dtg)[peak_idx]], color="#c0392b", s=28, zorder=4)
        ax2.annotate(
            f"{peak_temp:.0f} °C",
            (peak_temp, (-dtg)[peak_idx]),
            xytext=(8, 10),
            textcoords="offset points",
            color="#c0392b",
            fontsize=9,
        )

    ax1.legend(
        [weight_line, dtg_line],
        [weight_line.get_label(), dtg_line.get_label()],
        loc="upper right",
        fontsize=9,
    )

    fig.tight_layout()
    fig.savefig(OUT_DIR / f"{safe_filename(sheet_name)}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
