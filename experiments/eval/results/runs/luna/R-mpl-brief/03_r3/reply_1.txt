```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/03_r3/out")
SHEET_NAME = "1006R1-V acid"

OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
df.columns = [str(column).strip() for column in df.columns]

temperature_col = next(col for col in df.columns if "temperature" in col.lower())
weight_col = next(col for col in df.columns if "weight" in col.lower() and "d.w" not in col.lower())
dtg_col = next(col for col in df.columns if "d.w" in col.lower() or "dtg" in col.lower())

temperature = pd.to_numeric(df[temperature_col], errors="coerce")
weight = pd.to_numeric(df[weight_col], errors="coerce")
dtg = pd.to_numeric(df[dtg_col], errors="coerce")
valid = temperature.notna() & weight.notna() & dtg.notna()

temperature = temperature[valid].to_numpy()
weight = weight[valid].to_numpy()
dtg = dtg[valid].to_numpy()

# Find the two largest DTG mass-loss peaks above 150 °C, separated to avoid
# labeling adjacent points from the same peak.
from scipy.signal import find_peaks

eligible = np.flatnonzero(temperature > 150)
local_peaks, _ = find_peaks(-dtg[eligible])
peak_indices = eligible[local_peaks]

if len(peak_indices) < 2:
    peak_indices = eligible[np.argsort(dtg[eligible])[:2]]
else:
    peak_indices = peak_indices[np.argsort(dtg[peak_indices])[:2]]

peak_indices = peak_indices[np.argsort(temperature[peak_indices])]

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

ax1.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
ax2.plot(temperature, -dtg, color="#c0392b", linewidth=1.5, label="DTG")

ax1.set_xlim(100, 800)
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.set_title(SHEET_NAME)

# Preserve the standard residue annotation at the measured point nearest 800 °C.
residue_idx = int(np.argmin(np.abs(temperature - 800)))
ax1.scatter(temperature[residue_idx], weight[residue_idx], color="#1f4e9c", zorder=5)
ax1.annotate(
    f"Residue {weight[residue_idx]:.1f}%",
    (temperature[residue_idx], weight[residue_idx]),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
)

# Mark and label the two highest DTG peaks above 150 °C.
for rank, idx in enumerate(peak_indices, start=1):
    peak_height = -dtg[idx]
    ax2.scatter(temperature[idx], peak_height, color="#c0392b", zorder=6)
    ax2.annotate(
        f"{temperature[idx]:.0f} °C",
        (temperature[idx], peak_height),
        xytext=(7, 8 if rank == 1 else -16),
        textcoords="offset points",
        color="#c0392b",
    )

lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc="center left")
ax1.grid(True, alpha=0.25)
fig.tight_layout()

fig.savefig(OUT_DIR / "1006R1-V_acid.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
