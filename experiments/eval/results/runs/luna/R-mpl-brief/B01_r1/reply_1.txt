```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/caox.csv")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B01_r1/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

# CSV data: temperature, weight, and derivative weight
df = pd.read_csv(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

temperature_col = next(c for c in df.columns if "temperature" in c.lower())
weight_col = next(c for c in df.columns if "weight (%)" in c.lower())
dtg_col = next(c for c in df.columns if "deriv" in c.lower() and "weight" in c.lower())

temperature = pd.to_numeric(df[temperature_col], errors="coerce")
weight = pd.to_numeric(df[weight_col], errors="coerce")
dtg = pd.to_numeric(df[dtg_col], errors="coerce")
valid = temperature.notna() & weight.notna() & dtg.notna()
temperature, weight, dtg = temperature[valid], weight[valid], dtg[valid]

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

ax1.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
ax2.plot(temperature, -dtg, color="#c0392b", linewidth=1.5, label="DTG")

# Highlight mass-loss intervals with distinct, translucent colours and labels.
regions = [
    (120, 250, "H₂O", "#72b7b2"),
    (430, 580, "CO", "#f2cf5b"),
    (640, 830, "CO₂", "#b279a2"),
]
for start, end, label, color in regions:
    ax1.axvspan(start, end, color=color, alpha=0.22, linewidth=0, zorder=0)
    ax1.text(
        (start + end) / 2,
        1.015,
        label,
        transform=ax1.get_xaxis_transform(),
        ha="center",
        va="bottom",
        fontsize=11,
        color=color,
        fontweight="bold",
        clip_on=False,
    )

# Residue: measured weight at the point nearest 800 °C.
residue_idx = (temperature - 800).abs().idxmin()
residue_x = temperature.loc[residue_idx]
residue_y = weight.loc[residue_idx]
ax1.scatter([residue_x], [residue_y], color="#1f4e9c", zorder=5)
ax1.annotate(
    f"Residue {residue_y:.1f}%",
    xy=(residue_x, residue_y),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
    fontsize=9,
)

# Highest upward-pointing DTG peak above 150 °C.
peak_candidates = dtg[temperature > 150]
if not peak_candidates.empty:
    peak_idx = peak_candidates.idxmin()
    peak_x = temperature.loc[peak_idx]
    peak_y = -dtg.loc[peak_idx]
    ax2.scatter([peak_x], [peak_y], color="#c0392b", zorder=5)
    ax2.annotate(
        f"{peak_x:.0f} °C",
        xy=(peak_x, peak_y),
        xytext=(8, 10),
        textcoords="offset points",
        color="#c0392b",
        fontsize=9,
    )

ax1.set_title("caox")
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("DTG (−%/°C)", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.grid(True, alpha=0.25)

lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc="center left")

fig.tight_layout()
fig.savefig(OUT_DIR / "caox.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
