```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/23_r3/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def read_instrument_text(path):
    """Find and load the numeric data table, tolerating instrument metadata."""
    text = None
    for encoding in ("utf-8-sig", "cp949", "latin-1"):
        try:
            text = path.read_text(encoding=encoding)
            break
        except UnicodeDecodeError:
            continue
    if text is None:
        text = path.read_text(encoding="utf-8", errors="replace")

    rows = []
    for line in text.splitlines():
        # Handle whitespace, comma, semicolon, and tab-delimited exports.
        parts = line.replace(",", " ").replace(";", " ").split()
        if len(parts) < 4:
            continue

        values = []
        try:
            for token in parts[:4]:
                values.append(float(token.strip().strip('"')))
        except ValueError:
            continue
        rows.append(values)

    if not rows:
        # Fallback for consistently delimited files, with or without a preamble.
        for separator in (r"\s+", ",", ";", "\t"):
            try:
                candidate = pd.read_csv(
                    path, sep=separator, header=None, comment="#",
                    engine="python", encoding="utf-8", skiprows=10
                )
                candidate = candidate.apply(pd.to_numeric, errors="coerce")
                candidate = candidate.dropna(how="all").dropna(axis=1, how="all")
                if candidate.shape[1] >= 4 and len(candidate):
                    return candidate.iloc[:, :4].reset_index(drop=True)
            except Exception:
                pass
        raise ValueError(f"No numeric four-column data found in {path}")

    return pd.DataFrame(rows, columns=["time", "temperature", "weight", "dtg"])


data = read_instrument_text(DATA_FILE)
temperature = data.iloc[:, 1].to_numpy(dtype=float)
weight = data.iloc[:, 2].to_numpy(dtype=float)
dtg = data.iloc[:, 3].to_numpy(dtype=float)

valid = np.isfinite(temperature) & np.isfinite(weight) & np.isfinite(dtg)
temperature, weight, dtg = temperature[valid], weight[valid], dtg[valid]
if temperature.size == 0:
    raise ValueError("No valid data points to plot.")

residue_idx = int(np.argmin(np.abs(temperature - 800)))
residue_temp = temperature[residue_idx]
residue_weight = weight[residue_idx]

peak_mask = temperature > 150
if peak_mask.any():
    peak_indices = np.flatnonzero(peak_mask)
    peak_idx = int(peak_indices[np.argmax(-dtg[peak_mask])])
else:
    peak_idx = int(np.argmax(-dtg))
peak_temp = temperature[peak_idx]

fig, ax1 = plt.subplots(figsize=(8, 5))
ax2 = ax1.twinx()

weight_line, = ax1.plot(
    temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight"
)
dtg_line, = ax2.plot(
    temperature, -dtg, color="#c0392b", linewidth=1.5, label="DTG"
)

ax1.scatter(
    [residue_temp], [residue_weight], color="#1f4e9c",
    edgecolor="white", linewidth=0.6, zorder=5
)
ax1.annotate(
    f"Residue: {residue_weight:.1f}%",
    xy=(residue_temp, residue_weight),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
    fontsize=9,
)

ax2.scatter(
    [peak_temp], [-dtg[peak_idx]], color="#c0392b",
    edgecolor="white", linewidth=0.6, zorder=5
)
ax2.annotate(
    f"{peak_temp:.0f} °C",
    xy=(peak_temp, -dtg[peak_idx]),
    xytext=(8, 10),
    textcoords="offset points",
    color="#c0392b",
    fontsize=9,
)

ax1.set_title("instrument_export")
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("DTG", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.grid(True, alpha=0.25)
ax1.legend([weight_line, dtg_line], ["Weight", "DTG"], loc="center left")

fig.tight_layout()
fig.savefig(OUT_DIR / "instrument_export.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
