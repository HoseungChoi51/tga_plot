```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd


DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/caox.csv")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B12_r3/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

# Read the CSV and identify temperature and weight columns from their headers.
df = pd.read_csv(DATA_FILE)
columns = {str(col).strip().lower(): col for col in df.columns}

temperature_col = next(
    (col for name, col in columns.items() if "temperature" in name or "temp" in name),
    df.columns[1],
)
weight_col = next(
    (col for name, col in columns.items() if "weight" in name and "deriv" not in name),
    df.columns[2],
)

temperature = pd.to_numeric(df[temperature_col], errors="coerce")
weight = pd.to_numeric(df[weight_col], errors="coerce")
valid = temperature.notna() & weight.notna()

temperature = temperature[valid]
weight = weight[valid]

# Label the measured point nearest to 900 °C.
residue_idx = (temperature - 900).abs().idxmin()
residue_temp = float(temperature.loc[residue_idx])
residue_weight = float(weight.loc[residue_idx])

plt.rcParams["font.family"] = "Noto Sans CJK KR"
fig, ax = plt.subplots(figsize=(8, 5))

ax.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight")
ax.set_xlim(100, 900)
ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%)")
ax.set_title("caox")
ax.grid(True, alpha=0.3)

ax.scatter(
    [residue_temp],
    [residue_weight],
    color="#1f4e9c",
    edgecolor="white",
    linewidth=0.8,
    zorder=3,
)
ax.annotate(
    f"Residue {residue_weight:.1f}%",
    xy=(residue_temp, residue_weight),
    xytext=(-8, 12),
    textcoords="offset points",
    ha="right",
    va="bottom",
    color="#1f4e9c",
)

fig.tight_layout()
fig.savefig(OUT_DIR / "caox_weight.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
