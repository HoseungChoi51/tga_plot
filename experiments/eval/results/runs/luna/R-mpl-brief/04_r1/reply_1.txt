from pathlib import Path

import pandas as pd
import matplotlib.pyplot as plt
from matplotlib import font_manager

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/04_r1/out")
SHEET_NAME = "1006R1-V normal"

OUT_DIR.mkdir(parents=True, exist_ok=True)

# Korean font
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
df.columns = [str(column).strip() for column in df.columns]

def find_column(keywords):
    for column in df.columns:
        if any(word.lower() in column.lower() for word in keywords):
            return column
    raise ValueError(f"Could not find a column matching {keywords}")

x_col = find_column(["Temperature"])
weight_col = find_column(["Weight"])
dtg_col = find_column(["D.W", "DTG", "Derivative"])

data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[x_col]
weight = data[weight_col]
dtg_up = -data[dtg_col]

fig, ax1 = plt.subplots(figsize=(8, 6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(temperature, weight, color="#1f4e9c", label="Weight")
dtg_line, = ax2.plot(temperature, dtg_up, color="#c0392b", label="DTG")

# Residue at the measured point nearest 800 °C
residue_idx = (temperature - 800).abs().idxmin()
residue_temp = temperature.loc[residue_idx]
residue_weight = weight.loc[residue_idx]
ax1.plot(residue_temp, residue_weight, marker="o", color="#1f4e9c")
ax1.annotate(
    f"Residue {residue_weight:.1f}%",
    (residue_temp, residue_weight),
    xytext=(8, 8),
    textcoords="offset points",
    color="#1f4e9c",
)

# Highest upward DTG peak above 150 °C
peak_mask = temperature > 150
peak_idx = dtg_up.loc[peak_mask].idxmax()
peak_temp = temperature.loc[peak_idx]
peak_value = dtg_up.loc[peak_idx]
ax2.plot(peak_temp, peak_value, marker="o", color="#c0392b")
ax2.annotate(
    f"{peak_temp:.0f} °C",
    (peak_temp, peak_value),
    xytext=(8, 8),
    textcoords="offset points",
    color="#c0392b",
)

ax1.set_title("CNT 산화 거동", fontsize=18, color="navy")
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("DTG (-%/°C)")
ax1.legend([weight_line, dtg_line], ["Weight", "DTG"], loc="center left")
ax1.grid(True, alpha=0.3)
fig.tight_layout()

fig.savefig(OUT_DIR / "1006R1-V normal.png", dpi=300, bbox_inches="tight")
plt.close(fig)
