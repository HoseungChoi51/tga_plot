```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B17_r1/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheets = pd.read_excel(DATA_FILE, sheet_name=None)
fig, ax = plt.subplots(figsize=(10, 6.5))

colors = ["#1f4e9c", "#c0392b", "#2e8b57"]
offset_step = 20.0

for index, (sheet_name, df) in enumerate(sheets.items()):
    # Use temperature and weight columns identified from their headers.
    columns = {str(col).strip().lower(): col for col in df.columns}
    x_col = next((col for key, col in columns.items() if "temperature" in key), df.columns[1])
    weight_col = next((col for key, col in columns.items() if "weight" in key and "deriv" not in key), df.columns[2])

    x = pd.to_numeric(df[x_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    valid = x.notna() & weight.notna()

    offset = index * offset_step
    ax.plot(x[valid], weight[valid] + offset, color=colors[index % len(colors)],
            linewidth=1.8, label=f"{sheet_name} (+{offset:g}%)")

ax.set_title("세 시트 무게 곡선 (20% offset)")
ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%) + offset")
ax.legend(loc="best", frameon=False)
ax.grid(True, linestyle="--", alpha=0.35)
fig.tight_layout()

fig.savefig(OUT_DIR / "polymer_weight_offset.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
