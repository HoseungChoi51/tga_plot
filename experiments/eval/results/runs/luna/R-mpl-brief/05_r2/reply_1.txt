from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/05_r2/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

def find_column(columns, terms):
    for col in columns:
        name = str(col).strip().lower()
        if any(term in name for term in terms):
            return col
    return None

def safe_filename(name):
    return "".join(c if c.isalnum() or c in "-_." else "_" for c in str(name))

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    x_col = find_column(df.columns, ["temperature", "temp"])
    weight_col = find_column(df.columns, ["weight"])
    dtg_col = find_column(df.columns, ["d.w", "dtg", "derivative"])

    if x_col is None or weight_col is None or dtg_col is None:
        raise ValueError(f"'{sheet_name}' 시트에서 온도, 중량, DTG 열을 찾을 수 없습니다.")

    data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
    if data.empty:
        continue

    temperature = data[x_col].to_numpy()
    weight = data[weight_col].to_numpy()
    dtg_up = -data[dtg_col].to_numpy()

    residue_idx = int(np.argmin(np.abs(temperature - 800)))
    peak_candidates = np.flatnonzero(temperature > 150)
    if peak_candidates.size:
        peak_idx = int(peak_candidates[np.argmax(dtg_up[peak_candidates])])
    else:
        peak_idx = int(np.argmax(dtg_up))

    fig, ax1 = plt.subplots(figsize=(10, 6))
    ax2 = ax1.twinx()

    ax1.plot(temperature, weight, color="black", linewidth=2.5, label="Weight")
    ax2.plot(temperature, dtg_up, color="#c0392b", linestyle="--", label="DTG")

    ax1.scatter(temperature[residue_idx], weight[residue_idx], color="black", zorder=4)
    ax1.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        (temperature[residue_idx], weight[residue_idx]),
        xytext=(8, 10),
        textcoords="offset points",
        color="black",
    )

    ax2.scatter(temperature[peak_idx], dtg_up[peak_idx], color="#c0392b", zorder=4)
    ax2.annotate(
        f"{temperature[peak_idx]:.0f} °C",
        (temperature[peak_idx], dtg_up[peak_idx]),
        xytext=(8, 10),
        textcoords="offset points",
        color="#c0392b",
    )

    ax1.set_title(str(sheet_name))
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)")
    ax2.set_ylabel("DTG (%/°C)")
    ax1.grid(True, alpha=0.3)

    handles1, labels1 = ax1.get_legend_handles_labels()
    handles2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(handles1 + handles2, labels1 + labels2, loc="center left")

    fig.tight_layout()
    fig.savefig(OUT_DIR / f"{safe_filename(sheet_name)}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
