```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt


DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/korean_excel.csv")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/14_r1/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def read_csv_with_fallback(path):
    """Read CSV files whose encoding may be Korean Windows/Excel encoding."""
    encodings = ("utf-8-sig", "cp949", "euc-kr", "utf-16", "latin1")
    last_error = None
    for encoding in encodings:
        try:
            return pd.read_csv(path, encoding=encoding)
        except UnicodeDecodeError as error:
            last_error = error
    raise last_error


df = read_csv_with_fallback(DATA_FILE)
df.columns = [str(column).strip() for column in df.columns]

temperature = pd.to_numeric(df["온도(°C)"], errors="coerce")
weight = pd.to_numeric(df["무게(%)"], errors="coerce")
dtg = pd.to_numeric(df["미분무게(%/°C)"], errors="coerce")

valid = temperature.notna() & weight.notna() & dtg.notna()
temperature = temperature[valid].to_numpy()
weight = weight[valid].to_numpy()
dtg = dtg[valid].to_numpy()

order = np.argsort(temperature)
temperature = temperature[order]
weight = weight[order]
dtg = dtg[order]

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

ax1.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="무게")
ax2.plot(temperature, -dtg, color="#c0392b", linewidth=1.5, label="DTG")

ax1.set_title(DATA_FILE.stem)
ax1.set_xlabel("온도 (°C)")
ax1.set_ylabel("무게 (%)", color="#1f4e9c")
ax2.set_ylabel("DTG (-미분무게, %/°C)", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.grid(True, alpha=0.25)

# Residue at the measured point nearest 800 °C.
residue_idx = int(np.argmin(np.abs(temperature - 800)))
residue_temp = temperature[residue_idx]
residue_weight = weight[residue_idx]
ax1.scatter([residue_temp], [residue_weight], color="#1f4e9c", zorder=5)
ax1.annotate(
    f"Residue {residue_weight:.1f}%",
    xy=(residue_temp, residue_weight),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
)

# Highest upward-pointing DTG peak above 150 °C.
peak_candidates = np.flatnonzero(temperature > 150)
if peak_candidates.size:
    peak_idx = peak_candidates[np.argmax(-dtg[peak_candidates])]
    peak_temp = temperature[peak_idx]
    peak_value = -dtg[peak_idx]
    ax2.scatter([peak_temp], [peak_value], color="#c0392b", zorder=5)
    ax2.annotate(
        f"{peak_temp:.0f} °C",
        xy=(peak_temp, peak_value),
        xytext=(8, 10),
        textcoords="offset points",
        color="#c0392b",
    )

lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc="center left")

fig.tight_layout()
fig.savefig(OUT_DIR / f"{DATA_FILE.stem}.png", dpi=300, bbox_inches="tight")
fig.savefig(OUT_DIR / f"{DATA_FILE.stem}.pdf", bbox_inches="tight")
plt.close(fig)
```
