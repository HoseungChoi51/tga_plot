```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B09_r2/out"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

df = pd.read_excel(DATA_FILE, sheet_name="PE")
df.columns = [str(col).strip() for col in df.columns]

temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
dtg = pd.to_numeric(df["Deriv. Weight (%/°C)"], errors="coerce")

valid = temperature.notna() & weight.notna()
temperature = temperature[valid].to_numpy()
weight = weight[valid].to_numpy()
dtg = dtg[valid].to_numpy()

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(
    temperature, weight, color="red", linewidth=3, label="Weight (%)"
)
ax2.plot(
    temperature, -dtg, color="#c0392b", linewidth=1.5, label="DTG"
)

# 잔류량: 800 °C에 가장 가까운 측정점
residue_idx = int(np.nanargmin(np.abs(temperature - 800)))
residue_temp = temperature[residue_idx]
residue_weight = weight[residue_idx]
ax1.plot(residue_temp, residue_weight, "o", color="red", markersize=5)
ax1.annotate(
    f"Residue {residue_weight:.1f}%",
    (residue_temp, residue_weight),
    xytext=(8, 8),
    textcoords="offset points",
    color="red",
)

# DTG 피크: 150 °C 초과 영역에서 -DTG가 가장 높은 지점
peak_candidates = np.flatnonzero((temperature > 150) & np.isfinite(dtg))
if peak_candidates.size:
    peak_idx = peak_candidates[np.argmax(-dtg[peak_candidates])]
    ax2.plot(temperature[peak_idx], -dtg[peak_idx], "o", color="#c0392b", markersize=5)
    ax2.annotate(
        f"{temperature[peak_idx]:.0f} °C",
        (temperature[peak_idx], -dtg[peak_idx]),
        xytext=(8, 8),
        textcoords="offset points",
        color="#c0392b",
    )

ax1.set_title("PE")
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("-DTG (%/°C)")
ax1.set_xlim(100, 700)
ax1.set_xticks(np.arange(100, 701, 100))
ax1.grid(True, alpha=0.3)

ax1.legend([weight_line], ["Weight (%)"], loc="center left")
fig.tight_layout()
fig.savefig(out_dir / "PE.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
