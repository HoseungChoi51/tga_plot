```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B10_r1/out")
SHEET_NAME = "Annealed"

OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams.update({
    "font.family": "Noto Sans CJK KR",
    "font.size": 8,
    "axes.titlesize": 8,
    "axes.labelsize": 8,
    "xtick.labelsize": 8,
    "ytick.labelsize": 8,
    "legend.fontsize": 8,
    "axes.unicode_minus": False,
})

data = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
data.columns = [str(col).strip() for col in data.columns]

temperature_col = next(
    col for col in data.columns
    if "temperature" in col.lower() or "온도" in col
)
weight_col = next(
    col for col in data.columns
    if "weight" in col.lower() and "%" in col
)
dtg_col = next(
    col for col in data.columns
    if "deriv" in col.lower() or "dtg" in col.lower()
)

temperature = pd.to_numeric(data[temperature_col], errors="coerce")
weight = pd.to_numeric(data[weight_col], errors="coerce")
dtg = pd.to_numeric(data[dtg_col], errors="coerce")

valid = temperature.notna() & weight.notna() & dtg.notna()
temperature = temperature[valid].to_numpy()
weight = weight[valid].to_numpy()
dtg = dtg[valid].to_numpy()

fig, ax1 = plt.subplots(figsize=(3.5, 2.6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(
    temperature, weight,
    color="black", linestyle="-", linewidth=1.0,
    label="Weight (%)",
)
dtg_line, = ax2.plot(
    temperature, -dtg,
    color="black", linestyle="--", linewidth=1.0,
    label="DTG (%/°C)",
)

# Residue: measured weight at the point nearest 800 °C.
residue_idx = int(np.argmin(np.abs(temperature - 800)))
residue_temp = temperature[residue_idx]
residue_value = weight[residue_idx]
ax1.plot(
    residue_temp, residue_value, marker="o", markersize=3.5,
    markerfacecolor="white", markeredgecolor="black",
    linestyle="none", zorder=5,
)
ax1.annotate(
    f"Residue {residue_value:.1f}%",
    xy=(residue_temp, residue_value),
    xytext=(5, 7), textcoords="offset points",
    fontsize=8, color="black",
)

# DTG peak: highest upward peak above 150 °C.
peak_candidates = np.flatnonzero(temperature > 150)
peak_idx = peak_candidates[np.argmax((-dtg)[peak_candidates])]
peak_temp = temperature[peak_idx]
peak_value = -dtg[peak_idx]
ax2.plot(
    peak_temp, peak_value, marker="s", markersize=3.5,
    markerfacecolor="white", markeredgecolor="black",
    linestyle="none", zorder=5,
)
ax2.annotate(
    f"{peak_temp:.0f} °C",
    xy=(peak_temp, peak_value),
    xytext=(5, 6), textcoords="offset points",
    fontsize=8, color="black",
)

ax1.set_title(SHEET_NAME)
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("DTG (%/°C)")

ax1.legend(
    [weight_line, dtg_line],
    [weight_line.get_label(), dtg_line.get_label()],
    loc="center left",
    frameon=False,
)

fig.tight_layout()
fig.savefig(OUT_DIR / "Annealed.png", dpi=600, bbox_inches="tight")
plt.close(fig)
```
