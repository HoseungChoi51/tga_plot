```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib.ticker import AutoMinorLocator

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B20_r2/out")
SHEET_NAME = "Acid-washed"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def find_column(columns, keywords):
    """Return the first column whose normalized name contains one of the keywords."""
    normalized = {col: str(col).strip().lower() for col in columns}
    for keyword in keywords:
        for col, name in normalized.items():
            if keyword in name:
                return col
    return None


OUT_DIR.mkdir(parents=True, exist_ok=True)

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
df.columns = [str(col).strip() for col in df.columns]

x_col = find_column(df.columns, ["temperature", "temp", "온도"])
weight_col = find_column(df.columns, ["weight", "mass", "무게", "중량"])
dtg_col = find_column(df.columns, ["deriv", "dtg", "미분"])

if x_col is None or weight_col is None or dtg_col is None:
    raise ValueError(f"필요한 온도/무게/DTG 열을 찾을 수 없습니다: {list(df.columns)}")

data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[x_col].to_numpy()
weight = data[weight_col].to_numpy()
dtg = data[dtg_col].to_numpy()

fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(
    temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)"
)
dtg_line, = ax2.plot(
    temperature, -dtg, color="#c0392b", linewidth=1.6, label="-DTG (%/°C)"
)

# Residue: measured weight nearest 800 °C
residue_idx = int(np.argmin(np.abs(temperature - 800)))
ax1.plot(
    temperature[residue_idx], weight[residue_idx],
    marker="o", color="#1f4e9c", markersize=6
)
ax1.annotate(
    f"Residue {weight[residue_idx]:.1f}%",
    (temperature[residue_idx], weight[residue_idx]),
    xytext=(8, 9), textcoords="offset points", color="#1f4e9c"
)

# Highest positive DTG peak above 150 °C
peak_candidates = np.flatnonzero(temperature > 150)
if peak_candidates.size:
    peak_idx = peak_candidates[np.argmax(-dtg[peak_candidates])]
    ax2.plot(
        temperature[peak_idx], -dtg[peak_idx],
        marker="o", color="#c0392b", markersize=6
    )
    ax2.annotate(
        f"{temperature[peak_idx]:.0f} °C",
        (temperature[peak_idx], -dtg[peak_idx]),
        xytext=(8, 9), textcoords="offset points", color="#c0392b"
    )

ax1.set_title(SHEET_NAME)
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
ax1.tick_params(axis="y", colors="#1f4e9c")
ax2.tick_params(axis="y", colors="#c0392b")

# Add minor ticks on the x-axis.
ax1.xaxis.set_minor_locator(AutoMinorLocator())
ax1.tick_params(axis="x", which="minor", length=3)

ax1.legend([weight_line, dtg_line], [weight_line.get_label(), dtg_line.get_label()],
           loc="center left")
fig.tight_layout()

output_path = OUT_DIR / "Acid-washed.png"
fig.savefig(output_path, dpi=300, bbox_inches="tight")
plt.close(fig)
```
