from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B08_r1/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def find_column(columns, *keywords):
    """Find a column whose name contains any of the supplied keywords."""
    for col in columns:
        name = str(col).strip().lower()
        if any(keyword.lower() in name for keyword in keywords):
            return col
    return None


df = pd.read_excel(DATA_FILE, sheet_name="Raw")
df.columns = [str(col).strip() for col in df.columns]

x_col = find_column(df.columns, "temperature", "temp")
weight_col = find_column(df.columns, "weight")
dtg_col = find_column(df.columns, "deriv", "dtg")

if x_col is None or weight_col is None or dtg_col is None:
    raise ValueError("Raw 시트에서 온도, 무게, DTG 열을 찾을 수 없습니다.")

data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[x_col].to_numpy()
weight = data[weight_col].to_numpy()
dtg = data[dtg_col].to_numpy()

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(
    temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)"
)
dtg_line, = ax2.plot(
    temperature, -dtg, color="#c0392b", linewidth=1.5, label="-DTG (%/°C)"
)

# Residue at the measured point nearest 800 °C
residue_idx = int(np.argmin(np.abs(temperature - 800)))
residue_temp = temperature[residue_idx]
residue_value = weight[residue_idx]
ax1.scatter(
    residue_temp, residue_value, color="#1f4e9c", edgecolor="white",
    linewidth=0.7, zorder=5
)
ax1.annotate(
    f"Residue {residue_value:.1f}%",
    xy=(residue_temp, residue_value),
    xytext=(7, 9),
    textcoords="offset points",
    color="#1f4e9c",
    fontsize=9,
)

# Highest upward-pointing DTG peak above 150 °C
peak_candidates = np.flatnonzero(temperature > 150)
if peak_candidates.size:
    peak_idx = peak_candidates[np.argmax((-dtg)[peak_candidates])]
    peak_temp = temperature[peak_idx]
    peak_value = (-dtg)[peak_idx]
    ax2.scatter(
        peak_temp, peak_value, color="#c0392b", edgecolor="white",
        linewidth=0.7, zorder=5
    )
    ax2.annotate(
        f"{peak_temp:.0f} °C",
        xy=(peak_temp, peak_value),
        xytext=(7, 9),
        textcoords="offset points",
        color="#c0392b",
        fontsize=9,
    )

# Horizontal reference line at 50% weight
ax1.axhline(50, color="#555555", linestyle="--", linewidth=1.0, alpha=0.8)

ax1.set_title("Raw")
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.grid(True, which="major", linestyle=":", linewidth=0.7, alpha=0.55)

ax1.legend(
    handles=[weight_line, dtg_line],
    loc="center left",
    frameon=True,
)

ax1.text(
    0.98, 0.97, "공기, 10 °C/min",
    transform=ax1.transAxes,
    ha="right", va="top",
    fontsize=10,
    bbox={"facecolor": "white", "edgecolor": "none", "alpha": 0.8, "pad": 3},
)

fig.tight_layout()
fig.savefig(OUT_DIR / "Raw.png", dpi=300, bbox_inches="tight")
plt.close(fig)
