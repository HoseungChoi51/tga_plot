from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B09_r3/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheet_name = "PE"
df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)

# 열 이름을 기준으로 온도, 무게, DTG 열을 찾습니다.
def find_column(keywords):
    for column in df.columns:
        if any(word.lower() in str(column).lower() for word in keywords):
            return column
    raise ValueError(f"열을 찾을 수 없습니다: {keywords}")

temperature_col = find_column(["temperature", "온도"])
weight_col = find_column(["weight", "무게"])
dtg_col = find_column(["deriv", "dtg", "미분"])

temperature = pd.to_numeric(df[temperature_col], errors="coerce")
weight = pd.to_numeric(df[weight_col], errors="coerce")
dtg = pd.to_numeric(df[dtg_col], errors="coerce")

valid = temperature.notna() & weight.notna() & dtg.notna()
temperature = temperature[valid].to_numpy()
weight = weight[valid].to_numpy()
dtg = dtg[valid].to_numpy()

fig, ax1 = plt.subplots(figsize=(8, 6))

# y1: 무게
line1, = ax1.plot(
    temperature, weight,
    color="red",
    linewidth=3,
    label="Weight (%)",
)
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax1.set_xlim(100, 700)
ax1.set_xticks(np.arange(100, 701, 100))
ax1.set_title(sheet_name)

# y2: 부호를 반전해 위쪽으로 향하는 DTG 피크 표시
ax2 = ax1.twinx()
line2, = ax2.plot(
    temperature, -dtg,
    color="#c0392b",
    linewidth=1.8,
    label="DTG (%/°C)",
)
ax2.set_ylabel("DTG (%/°C)")

# 잔류량: 800 °C에 가장 가까운 측정값
residue_index = int(np.argmin(np.abs(temperature - 800)))
residue_x = temperature[residue_index]
residue_y = weight[residue_index]
ax1.plot(residue_x, residue_y, "o", color="red", markersize=5)
ax1.annotate(
    f"Residue {residue_y:.1f}%",
    xy=(residue_x, residue_y),
    xytext=(8, 8),
    textcoords="offset points",
    color="red",
)

# 피크: 150 °C 초과 영역에서 가장 높은 부호 반전 DTG 값
peak_mask = temperature > 150
if peak_mask.any():
    peak_indices = np.flatnonzero(peak_mask)
    peak_index = peak_indices[np.argmax(-dtg[peak_mask])]
    peak_x = temperature[peak_index]
    peak_y = -dtg[peak_index]
    ax2.plot(peak_x, peak_y, "o", color="#c0392b", markersize=5)
    ax2.annotate(
        f"{peak_x:.0f} °C",
        xy=(peak_x, peak_y),
        xytext=(8, 8),
        textcoords="offset points",
        color="#c0392b",
    )

ax1.legend([line1, line2], [line1.get_label(), line2.get_label()], loc="center left")
fig.tight_layout()
fig.savefig(OUT_DIR / "PE.png", dpi=300, bbox_inches="tight")
plt.close(fig)
