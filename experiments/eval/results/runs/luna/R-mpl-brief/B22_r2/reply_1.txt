from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B22_r2/out")

SHEET_NAME = "Raw"
REFERENCE_TEMP = 150.0
RESIDUE_TEMP = 800.0
PEAK_MIN_TEMP = 150.0

OUT_DIR.mkdir(parents=True, exist_ok=True)

# Korean text font
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
df.columns = [str(column).strip() for column in df.columns]

def find_column(name_part):
    matches = [column for column in df.columns if name_part.lower() in column.lower()]
    if not matches:
        raise ValueError(f"'{name_part}' 열을 찾을 수 없습니다. 열: {list(df.columns)}")
    return matches[0]

temperature_col = find_column("Temperature")
weight_col = find_column("Weight")
dtg_col = find_column("Deriv")

data = df[[temperature_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[temperature_col].to_numpy(dtype=float)
weight = data[weight_col].to_numpy(dtype=float)
dtg = data[dtg_col].to_numpy(dtype=float)

if len(temperature) == 0:
    raise ValueError("유효한 데이터가 없습니다.")

# Normalize weight and DTG to 100% at the measured point nearest 150 °C.
reference_index = int(np.nanargmin(np.abs(temperature - REFERENCE_TEMP)))
reference_weight = weight[reference_index]
if not np.isfinite(reference_weight) or reference_weight == 0:
    raise ValueError("150 °C 기준 무게가 유효하지 않아 건조 기준 환산을 할 수 없습니다.")

scale = 100.0 / reference_weight
weight_dry_basis = weight * scale
dtg_dry_basis = dtg * scale

# Residue at the measured point nearest 800 °C.
residue_index = int(np.nanargmin(np.abs(temperature - RESIDUE_TEMP)))
residue_temp = temperature[residue_index]
residue_value = weight_dry_basis[residue_index]

# Highest positive-pointing DTG peak above 150 °C (original DTG is negative during mass loss).
peak_candidates = np.flatnonzero((temperature > PEAK_MIN_TEMP) & np.isfinite(dtg_dry_basis))
if len(peak_candidates):
    peak_index = peak_candidates[np.argmax(-dtg_dry_basis[peak_candidates])]
    peak_temp = temperature[peak_index]
    peak_value = -dtg_dry_basis[peak_index]
else:
    peak_index = None

fig, ax1 = plt.subplots(figsize=(8, 6))
ax2 = ax1.twinx()

line1, = ax1.plot(
    temperature, weight_dry_basis,
    color="#1f4e9c", linewidth=1.8, label="Weight (dry basis)"
)
line2, = ax2.plot(
    temperature, -dtg_dry_basis,
    color="#c0392b", linewidth=1.5, label="DTG (dry basis)"
)

ax1.set_title(SHEET_NAME)
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("DTG (%/°C)")
ax1.grid(True, alpha=0.25)

ax1.scatter(
    temperature[residue_index], residue_value,
    color="#1f4e9c", edgecolor="white", linewidth=0.7, zorder=5
)
ax1.annotate(
    f"Residue {residue_value:.1f}%",
    (temperature[residue_index], residue_value),
    xytext=(8, 10), textcoords="offset points",
    color="#1f4e9c", fontsize=9
)

if peak_index is not None:
    ax2.scatter(
        peak_temp, peak_value,
        color="#c0392b", edgecolor="white", linewidth=0.7, zorder=5
    )
    ax2.annotate(
        f"{peak_temp:.0f} °C",
        (peak_temp, peak_value),
        xytext=(8, 10), textcoords="offset points",
        color="#c0392b", fontsize=9
    )

ax1.legend([line1, line2], [line1.get_label(), line2.get_label()], loc="center left")
fig.tight_layout()
fig.savefig(OUT_DIR / "Raw_dry_basis.png", dpi=300, bbox_inches="tight")
plt.close(fig)
