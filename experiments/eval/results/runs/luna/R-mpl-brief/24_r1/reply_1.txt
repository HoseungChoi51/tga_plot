from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd


DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/24_r1/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def find_column(columns, keyword):
    for column in columns:
        if keyword.lower() in str(column).lower():
            return column
    raise ValueError(f"열을 찾을 수 없습니다: {keyword}")


sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, data in sheets.items():
    temperature_col = find_column(data.columns, "Temperature")
    weight_col = find_column(data.columns, "Weight (%)")
    heat_flow_col = find_column(data.columns, "Heat Flow")

    temperature = pd.to_numeric(data[temperature_col], errors="coerce")
    weight = pd.to_numeric(data[weight_col], errors="coerce")
    heat_flow = pd.to_numeric(data[heat_flow_col], errors="coerce")

    valid = temperature.notna() & weight.notna() & heat_flow.notna()
    temperature = temperature[valid]
    weight = weight[valid]
    heat_flow = heat_flow[valid]

    if temperature.empty:
        continue

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    ax1.plot(temperature, weight, color="#1f4e9c", label="Weight (%)")
    ax2.plot(temperature, heat_flow, color="#c0392b", label="Heat Flow (W/g)")

    residue_index = (temperature - 800).abs().idxmin()
    residue_temp = temperature.loc[residue_index]
    residue = weight.loc[residue_index]
    ax1.scatter(
        residue_temp, residue, color="#1f4e9c", edgecolor="white",
        linewidth=0.8, zorder=5
    )
    ax1.annotate(
        f"Residue {residue:.1f}%",
        xy=(residue_temp, residue),
        xytext=(8, 10),
        textcoords="offset points",
        color="#1f4e9c",
    )

    ax1.set_title(str(sheet_name))
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="#1f4e9c")
    ax2.set_ylabel("Heat flow (W/g)", color="#c0392b")
    ax1.tick_params(axis="y", labelcolor="#1f4e9c")
    ax2.tick_params(axis="y", labelcolor="#c0392b")
    ax1.grid(True, alpha=0.3)

    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(lines1 + lines2, labels1 + labels2, loc="center left")

    fig.tight_layout()
    output_path = OUT_DIR / f"{sheet_name}.png"
    fig.savefig(output_path, dpi=300, bbox_inches="tight")
    plt.close(fig)
