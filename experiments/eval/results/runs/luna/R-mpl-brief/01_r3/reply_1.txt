```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/01_r3/out")

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

OUT_DIR.mkdir(parents=True, exist_ok=True)


def find_column(df, candidates):
    """Find a column by case-insensitive partial match."""
    normalized = {str(col).strip().lower(): col for col in df.columns}
    for candidate in candidates:
        for name, original in normalized.items():
            if candidate in name:
                return original
    return None


def safe_filename(name):
    cleaned = "".join(ch if ch.isalnum() or ch in "-_ ." else "_" for ch in str(name))
    return cleaned.strip().replace(" ", "_") or "sheet"


excel = pd.ExcelFile(DATA_FILE)
for sheet_name in excel.sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    df = df.dropna(how="all")

    x_col = find_column(df, ["temperature", "temp"])
    weight_col = find_column(df, ["weight", "mass"])
    dtg_col = find_column(df, ["d.w", "dtg", "derivative"])

    if x_col is None or weight_col is None or dtg_col is None:
        # For conventional TGA sheets, use the second, third, and fourth columns.
        if df.shape[1] < 4:
            continue
        x_col, weight_col, dtg_col = df.columns[1], df.columns[2], df.columns[3]

    x = pd.to_numeric(df[x_col], errors="coerce").to_numpy()
    weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy()
    dtg = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy()

    valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
    x, weight, dtg = x[valid], weight[valid], dtg[valid]
    if x.size == 0:
        continue

    order = np.argsort(x)
    x, weight, dtg = x[order], weight[order], dtg[order]

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    weight_line, = ax1.plot(x, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
    dtg_line, = ax2.plot(x, -dtg, color="#c0392b", linewidth=1.6, label="-DTG (%/°C)")

    # Residue: measured weight at the point nearest 800 °C.
    residue_idx = int(np.argmin(np.abs(x - 800)))
    residue_x, residue_y = x[residue_idx], weight[residue_idx]
    ax1.plot(residue_x, residue_y, marker="o", color="#1f4e9c", markersize=5)
    ax1.annotate(
        f"Residue {residue_y:.1f}%",
        xy=(residue_x, residue_y),
        xytext=(8, 10),
        textcoords="offset points",
        color="#1f4e9c",
        fontsize=9,
    )

    # Peak: highest upward-pointing DTG peak above 150 °C.
    peak_mask = x > 150
    peak_x = peak_y = None
    if np.any(peak_mask):
        peak_indices = np.flatnonzero(peak_mask)
        peak_idx = int(peak_indices[np.argmax((-dtg)[peak_mask])])
        peak_x, peak_y = x[peak_idx], -dtg[peak_idx]
        ax2.plot(peak_x, peak_y, marker="o", color="#c0392b", markersize=5)
        ax2.annotate(
            f"{peak_x:.0f} °C",
            xy=(peak_x, peak_y),
            xytext=(8, 10),
            textcoords="offset points",
            color="#c0392b",
            fontsize=9,
        )

    ax1.set_title(str(sheet_name))
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="#1f4e9c")
    ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
    ax1.tick_params(axis="y", labelcolor="#1f4e9c")
    ax2.tick_params(axis="y", labelcolor="#c0392b")
    ax1.grid(True, linestyle="--", alpha=0.35)
    ax1.legend([weight_line, dtg_line], ["Weight (%)", "-DTG (%/°C)"], loc="center left")

    fig.tight_layout()
    fig.savefig(OUT_DIR / f"{safe_filename(sheet_name)}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
