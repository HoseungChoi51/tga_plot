from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/rheology.csv")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B03_r2/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

# Read the CSV and use the first three columns: frequency, G', and G''.
df = pd.read_csv(DATA_FILE)
if df.shape[1] < 3:
    raise ValueError("CSV 파일에 주파수, G', G'' 세 개의 열이 필요합니다.")

x = pd.to_numeric(df.iloc[:, 0], errors="coerce")
g_storage = pd.to_numeric(df.iloc[:, 1], errors="coerce")
g_loss = pd.to_numeric(df.iloc[:, 2], errors="coerce")
data = pd.DataFrame({"x": x, "g_storage": g_storage, "g_loss": g_loss}).dropna()

# Log scales require positive values.
data = data[(data["x"] > 0) & (data["g_storage"] > 0) & (data["g_loss"] > 0)]
if data.empty:
    raise ValueError("로그 스케일에 사용할 수 있는 양수 데이터가 없습니다.")

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax1 = plt.subplots(figsize=(8, 6))
ax2 = ax1.twinx()

line1, = ax1.plot(
    data["x"], data["g_storage"],
    color="#1f4e9c", marker="o", linewidth=2, markersize=4,
    label="Storage modulus G′",
)
line2, = ax2.plot(
    data["x"], data["g_loss"],
    color="#c0392b", marker="s", linewidth=2, markersize=4,
    label="Loss modulus G″",
)

ax1.set_xscale("log")
ax1.set_yscale("log")
ax2.set_yscale("log")

ax1.set_xlabel("Angular frequency (rad/s)")
ax1.set_ylabel("Storage modulus G′ (Pa)", color="#1f4e9c")
ax2.set_ylabel("Loss modulus G″ (Pa)", color="#c0392b")
ax1.tick_params(axis="y", colors="#1f4e9c")
ax2.tick_params(axis="y", colors="#c0392b")
ax1.grid(True, which="both", linestyle="--", alpha=0.35)

ax1.legend([line1, line2], [line1.get_label(), line2.get_label()], loc="best")
fig.tight_layout()

output_path = OUT_DIR / "rheology_frequency_moduli.png"
fig.savefig(output_path, dpi=300, bbox_inches="tight")
plt.close(fig)
