```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/16_r1/out"
SHEET_NAME = "1006R1-V normal"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
df.columns = [str(col).strip() for col in df.columns]

def find_column(names, candidates):
    for col in names:
        normalized = col.lower().replace(" ", "")
        if any(candidate in normalized for candidate in candidates):
            return col
    raise ValueError(f"필요한 열을 찾을 수 없습니다: {candidates}")

temperature_col = find_column(df.columns, ("temperature", "temp"))
weight_col = find_column(df.columns, ("weight", "mass"))

temperature = pd.to_numeric(df[temperature_col], errors="coerce")
weight = pd.to_numeric(df[weight_col], errors="coerce")
valid = temperature.notna() & weight.notna()
temperature = temperature[valid].to_numpy()
weight = weight[valid].to_numpy()

if temperature.size == 0:
    raise ValueError("그릴 수 있는 온도/무게 데이터가 없습니다.")

fig, ax = plt.subplots(figsize=(8, 5))
ax.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight")

# Residue: weight at the measured point nearest 800 °C.
residue_idx = int(np.argmin(np.abs(temperature - 800)))
residue_temp = temperature[residue_idx]
residue_weight = weight[residue_idx]
ax.scatter([residue_temp], [residue_weight], color="#1f4e9c", zorder=3)
ax.annotate(
    f"Residue {residue_weight:.1f}%",
    (residue_temp, residue_weight),
    xytext=(8, 10),
    textcoords="offset points",
    fontsize=9,
)

ax.set_title(SHEET_NAME)
ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%)")
ax.grid(True, alpha=0.3)
ax.legend(loc="center left")
fig.tight_layout()
fig.savefig(out_dir / "1006R1-V_normal.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
