```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/19_r3/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

# 전체 텍스트: 회색, 13pt
TEXT_COLOR = "gray"
FONT_SIZE = 13
plt.rcParams.update({
    "font.family": "Noto Sans CJK KR",
    "font.size": FONT_SIZE,
    "text.color": TEXT_COLOR,
    "axes.labelcolor": TEXT_COLOR,
    "axes.titlecolor": TEXT_COLOR,
    "xtick.color": TEXT_COLOR,
    "ytick.color": TEXT_COLOR,
    "legend.fontsize": FONT_SIZE,
})

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df.columns = [str(c).strip() for c in df.columns]

    def find_column(tokens):
        for col in df.columns:
            if any(token in col.lower() for token in tokens):
                return col
        return None

    x_col = find_column(["temperature", "temp"])
    weight_col = find_column(["weight"])
    dtg_col = find_column(["d.w", "dtg", "derivative"])

    if x_col is None or weight_col is None or dtg_col is None:
        # 알려진 TGA 형식의 첫 네 열을 사용
        if len(df.columns) < 4:
            continue
        x_col, weight_col, dtg_col = df.columns[1], df.columns[2], df.columns[3]

    data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
    if data.empty:
        continue

    temperature = data[x_col].to_numpy()
    weight = data[weight_col].to_numpy()
    dtg = data[dtg_col].to_numpy()

    fig, ax1 = plt.subplots(figsize=(10, 6))
    ax2 = ax1.twinx()

    weight_line, = ax1.plot(
        temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)"
    )
    dtg_line, = ax2.plot(
        temperature, -dtg, color="#c0392b", linewidth=1.5, label="-DTG"
    )

    # 800 °C에 가장 가까운 측정점의 잔류량
    residue_idx = int(np.argmin(np.abs(temperature - 800)))
    ax1.scatter(
        temperature[residue_idx], weight[residue_idx],
        color="#1f4e9c", edgecolor="white", linewidth=0.7, zorder=5
    )
    ax1.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        (temperature[residue_idx], weight[residue_idx]),
        xytext=(8, 10), textcoords="offset points",
        color=TEXT_COLOR, fontsize=FONT_SIZE
    )

    # 150 °C 초과 구간에서 -DTG가 가장 큰 피크
    eligible = np.flatnonzero(temperature > 150)
    if eligible.size:
        peak_idx = int(eligible[np.argmax(-dtg[eligible])])
        ax2.scatter(
            temperature[peak_idx], -dtg[peak_idx],
            color="#c0392b", edgecolor="white", linewidth=0.7, zorder=5
        )
        ax2.annotate(
            f"{temperature[peak_idx]:.0f} °C",
            (temperature[peak_idx], -dtg[peak_idx]),
            xytext=(8, 10), textcoords="offset points",
            color=TEXT_COLOR, fontsize=FONT_SIZE
        )

    ax1.set_title(sheet_name, color=TEXT_COLOR, fontsize=FONT_SIZE)
    ax1.set_xlabel("Temperature (°C)", color=TEXT_COLOR, fontsize=FONT_SIZE)
    ax1.set_ylabel("Weight (%)", color=TEXT_COLOR, fontsize=FONT_SIZE)
    ax2.set_ylabel("-DTG (%/°C)", color=TEXT_COLOR, fontsize=FONT_SIZE)

    ax1.tick_params(axis="both", colors=TEXT_COLOR, labelsize=FONT_SIZE)
    ax2.tick_params(axis="y", colors=TEXT_COLOR, labelsize=FONT_SIZE)
    ax1.spines["left"].set_color(TEXT_COLOR)
    ax1.spines["bottom"].set_color(TEXT_COLOR)
    ax1.spines["top"].set_color(TEXT_COLOR)
    ax2.spines["right"].set_color(TEXT_COLOR)

    legend = ax1.legend(
        [weight_line, dtg_line], ["Weight (%)", "-DTG"],
        loc="center left", frameon=False
    )
    for text in legend.get_texts():
        text.set_color(TEXT_COLOR)
        text.set_fontsize(FONT_SIZE)

    fig.tight_layout()
    safe_name = "".join(c if c.isalnum() or c in "-_ ." else "_" for c in sheet_name).strip()
    fig.savefig(OUT_DIR / f"{safe_name or 'TGA'}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
