```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from matplotlib import font_manager

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/22_r3/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

# Configure Korean font when available.
available_fonts = {font.name for font in font_manager.fontManager.ttflist}
if "Noto Sans CJK KR" in available_fonts:
    plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def safe_filename(name):
    return "".join(c if c.isalnum() or c in "-_." else "_" for c in str(name))


sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df.columns = [str(column).strip() for column in df.columns]

    def find_column(terms):
        for column in df.columns:
            normalized = column.lower().replace(" ", "")
            if any(term in normalized for term in terms):
                return column
        return None

    x_col = find_column(["temperature", "temp", "온도"])
    weight_col = find_column(["weight(%)", "weight%", "weight", "무게", "중량"])
    dtg_col = find_column(["d.w", "dtg", "derivative", "미분"])

    # Fall back to the expected TGA column order: time, temperature, weight, DTG.
    if x_col is None and len(df.columns) >= 2:
        x_col = df.columns[1]
    if weight_col is None and len(df.columns) >= 3:
        weight_col = df.columns[2]
    if dtg_col is None and len(df.columns) >= 4:
        dtg_col = df.columns[3]

    if x_col is None or weight_col is None or dtg_col is None:
        continue

    data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
    if data.empty:
        continue

    temperature = data[x_col].to_numpy()
    weight = data[weight_col].to_numpy()
    dtg = data[dtg_col].to_numpy()

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    line1, = ax1.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
    line2, = ax2.plot(temperature, -dtg, color="green", linewidth=1.6, label="DTG")

    ax1.set_title(str(sheet_name))
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)")
    ax2.set_ylabel("DTG (%/°C)", color="green")
    ax2.tick_params(axis="y", colors="green")
    ax2.spines["right"].set_color("green")

    # Residue at the measured point closest to 800 °C.
    residue_idx = int(np.argmin(np.abs(temperature - 800)))
    ax1.plot(temperature[residue_idx], weight[residue_idx], "o", color="#1f4e9c", markersize=5)
    ax1.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        (temperature[residue_idx], weight[residue_idx]),
        xytext=(8, 10),
        textcoords="offset points",
        color="#1f4e9c",
        fontsize=9,
    )

    # Highest upward DTG peak above 150 °C.
    peak_candidates = np.flatnonzero(temperature > 150)
    if peak_candidates.size:
        peak_idx = peak_candidates[np.argmax((-dtg)[peak_candidates])]
        ax2.plot(temperature[peak_idx], -dtg[peak_idx], "o", color="green", markersize=5)
        ax2.annotate(
            f"{temperature[peak_idx]:.0f} °C",
            (temperature[peak_idx], -dtg[peak_idx]),
            xytext=(8, 10),
            textcoords="offset points",
            color="green",
            fontsize=9,
        )

    ax1.legend([line1, line2], [line1.get_label(), line2.get_label()], loc="center left")
    ax1.grid(True, alpha=0.25)
    fig.tight_layout()
    fig.savefig(OUT_DIR / f"{safe_filename(sheet_name)}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
