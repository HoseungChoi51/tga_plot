```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/03_r2/out"
SHEET_NAME = "1006R1-V acid"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
df.columns = [str(c).strip() for c in df.columns]

def find_column(tokens):
    for column in df.columns:
        name = column.lower().replace(" ", "")
        if any(token in name for token in tokens):
            return column
    return None

x_col = find_column(["temperature", "temp", "온도"])
weight_col = find_column(["weight", "mass", "중량", "무게"])
dtg_col = find_column(["d.w", "dtg", "derivative", "미분"])

# Fall back to the preview's known column order if names are not recognizable.
if x_col is None:
    x_col = df.columns[1]
if weight_col is None:
    weight_col = df.columns[2]
if dtg_col is None:
    dtg_col = df.columns[3]

data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[x_col].to_numpy()
weight = data[weight_col].to_numpy()
dtg = data[dtg_col].to_numpy()

# The plotted DTG is negated so mass-loss peaks point upward.
dtg_plot = -dtg
eligible = temperature > 150
peak_indices, _ = find_peaks(dtg_plot, prominence=0)
peak_indices = peak_indices[eligible[peak_indices]]

if len(peak_indices) >= 2:
    peak_indices = peak_indices[np.argsort(dtg_plot[peak_indices])[-2:]]
else:
    candidates = np.flatnonzero(eligible)
    peak_indices = candidates[np.argsort(dtg_plot[candidates])[-min(2, len(candidates)):]]

peak_indices = peak_indices[np.argsort(temperature[peak_indices])]

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

ax1.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
ax2.plot(temperature, dtg_plot, color="#c0392b", linewidth=1.5, label="DTG (-D.W)")

ax1.set_xlim(100, 800)
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("DTG (-D.W, %/°C)", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.set_title(SHEET_NAME)
ax1.grid(True, alpha=0.25)

# Residue: measurement nearest 800 °C.
residue_idx = int(np.argmin(np.abs(temperature - 800)))
ax1.scatter(temperature[residue_idx], weight[residue_idx], color="#1f4e9c",
            edgecolor="white", zorder=5)
ax1.annotate(
    f"Residue {weight[residue_idx]:.1f}%",
    (temperature[residue_idx], weight[residue_idx]),
    xytext=(-8, 10), textcoords="offset points", ha="right",
    color="#1f4e9c", fontsize=9
)

# Mark the two highest DTG peaks above 150 °C.
for idx in peak_indices:
    ax2.scatter(temperature[idx], dtg_plot[idx], color="#c0392b",
                edgecolor="white", zorder=5)
    ax2.annotate(
        f"{temperature[idx]:.0f} °C",
        (temperature[idx], dtg_plot[idx]),
        xytext=(5, 8), textcoords="offset points",
        color="#c0392b", fontsize=9
    )

lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc="center left")

fig.tight_layout()
fig.savefig(out_dir / "1006R1-V_acid.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
