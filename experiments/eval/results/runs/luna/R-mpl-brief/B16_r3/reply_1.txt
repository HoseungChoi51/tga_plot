```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B16_r3/out"

# Korean-capable font
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)


def read_data(path):
    suffix = Path(path).suffix.lower()
    if suffix in (".xlsx", ".xls"):
        sheets = pd.read_excel(path, sheet_name=None)
    else:
        df = pd.read_csv(path, sep=None, engine="python")
        sheets = {Path(path).stem: df}
    return sheets


def find_column(df, candidates, fallback_index):
    normalized = {str(col).strip().lower(): col for col in df.columns}
    for candidate in candidates:
        for name, original in normalized.items():
            if candidate in name:
                return original
    return df.columns[fallback_index]


for sheet_name, df in read_data(DATA_FILE).items():
    if df.empty or len(df.columns) < 4:
        continue

    x_col = find_column(df, ["temperature"], 1)
    weight_col = find_column(df, ["weight"], 2)
    dtg_col = find_column(df, ["deriv", "dtg"], 3)

    data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
    if data.empty:
        continue

    x = data[x_col].to_numpy()
    weight = data[weight_col].to_numpy()
    dtg = data[dtg_col].to_numpy()

    fig, (ax_weight, ax_dtg) = plt.subplots(
        2, 1, figsize=(9, 8), sharex=True,
        gridspec_kw={"height_ratios": [1, 1], "hspace": 0.08}
    )

    ax_weight.plot(x, weight, color="#1f4e9c", linewidth=1.6, label="Weight")
    ax_weight.set_ylabel("Weight (%)")
    ax_weight.grid(True, alpha=0.25)
    ax_weight.legend(loc="best")

    # Plot DTG as a positive-up curve, as is standard for mass-loss peaks.
    dtg_up = -dtg
    ax_dtg.plot(x, dtg_up, color="#c0392b", linewidth=1.4, label="DTG")
    ax_dtg.set_xlabel("Temperature (°C)")
    ax_dtg.set_ylabel("DTG (-%/°C)")
    ax_dtg.grid(True, alpha=0.25)
    ax_dtg.legend(loc="best")

    # Mark residue at the measured point nearest 800 °C.
    residue_idx = int(np.argmin(np.abs(x - 800)))
    ax_weight.scatter(x[residue_idx], weight[residue_idx], color="#1f4e9c", zorder=3)
    ax_weight.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        (x[residue_idx], weight[residue_idx]),
        xytext=(8, 8), textcoords="offset points",
        color="#1f4e9c", fontsize=9
    )

    # Mark the highest positive DTG peak above 150 °C.
    eligible = np.flatnonzero(x > 150)
    if eligible.size:
        peak_idx = int(eligible[np.argmax(dtg_up[eligible])])
        ax_dtg.scatter(x[peak_idx], dtg_up[peak_idx], color="#c0392b", zorder=3)
        ax_dtg.annotate(
            f"{x[peak_idx]:.0f} °C",
            (x[peak_idx], dtg_up[peak_idx]),
            xytext=(8, 8), textcoords="offset points",
            color="#c0392b", fontsize=9
        )

    ax_weight.set_title(str(sheet_name))
    fig.align_ylabels()
    fig.tight_layout()

    safe_name = "".join(c if c.isalnum() or c in "-_." else "_" for c in str(sheet_name))
    fig.savefig(out_dir / f"{safe_name}_separate_weight_dtg.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
