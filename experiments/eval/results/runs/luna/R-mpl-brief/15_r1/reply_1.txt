```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from matplotlib.lines import Line2D

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/15_r1/out"

# Korean-capable font and consistent styling
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

Path(OUT_DIR).mkdir(parents=True, exist_ok=True)

sheets = pd.read_excel(DATA_FILE, sheet_name=None)
colors = ["#1f4e9c", "#c0392b", "#2e8b57", "#8e44ad", "#e67e22", "#16a085"]

fig, ax_weight = plt.subplots(figsize=(11, 7))
ax_dtg = ax_weight.twinx()

for idx, (sheet_name, df) in enumerate(sheets.items()):
    color = colors[idx % len(colors)]

    # Identify the standard TGA columns by their header names.
    normalized = {str(col).strip().lower(): col for col in df.columns}
    temperature_col = next(
        (col for key, col in normalized.items() if "temperature" in key or "temp" in key),
        df.columns[1] if len(df.columns) > 1 else None,
    )
    weight_col = next(
        (col for key, col in normalized.items() if "weight" in key),
        df.columns[2] if len(df.columns) > 2 else None,
    )
    dtg_col = next(
        (
            col for key, col in normalized.items()
            if "dtg" in key or "d.w" in key or "derivative" in key
        ),
        df.columns[3] if len(df.columns) > 3 else None,
    )

    if temperature_col is None or weight_col is None or dtg_col is None:
        continue

    x = pd.to_numeric(df[temperature_col], errors="coerce").to_numpy(dtype=float)
    weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy(dtype=float)
    dtg = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy(dtype=float)

    valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
    x, weight, dtg = x[valid], weight[valid], dtg[valid]
    if x.size == 0:
        continue

    order = np.argsort(x)
    x, weight, dtg = x[order], weight[order], dtg[order]

    ax_weight.plot(x, weight, color=color, linewidth=1.8, label=sheet_name)
    # DTG values are negative during mass loss; multiply by -1 so peaks point up.
    ax_dtg.plot(x, -dtg, color=color, linewidth=1.5, linestyle="--", alpha=0.9)

ax_weight.set_xlabel("Temperature (°C)")
ax_weight.set_ylabel("Weight (%)", color="#1f4e9c")
ax_dtg.set_ylabel("-DTG (%/°C)", color="#c0392b")
ax_weight.tick_params(axis="y", colors="#1f4e9c")
ax_dtg.tick_params(axis="y", colors="#c0392b")
ax_weight.set_title("TGA — All Sheets")
ax_weight.grid(True, linestyle=":", alpha=0.45)

# Separate legend entries identify each sheet's paired weight and DTG curves.
sheet_handles = [
    Line2D([0], [0], color=colors[i % len(colors)], linewidth=1.8, label=name)
    for i, name in enumerate(sheets.keys())
]
style_handles = [
    Line2D([0], [0], color="black", linewidth=1.8, label="Weight"),
    Line2D([0], [0], color="black", linewidth=1.5, linestyle="--", label="-DTG"),
]
ax_weight.legend(
    handles=sheet_handles + style_handles,
    loc="center left",
    bbox_to_anchor=(1.08, 0.5),
    frameon=True,
)

fig.tight_layout()
output_path = Path(OUT_DIR) / "TGA_all_sheets_overlay.png"
fig.savefig(output_path, dpi=300, bbox_inches="tight")
plt.close(fig)
```
