```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/korean_excel.csv")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/14_r3/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

df = pd.read_csv(DATA_FILE, encoding="utf-8-sig")

def find_column(frame, candidates, fallback_index):
    for column in frame.columns:
        name = str(column).strip().lower()
        if any(candidate in name for candidate in candidates):
            return column
    return frame.columns[fallback_index]

x_col = find_column(df, ["온도", "temperature", "temp"], 1)
weight_col = find_column(df, ["무게", "weight", "mass"], 2)
dtg_col = find_column(df, ["미분무게", "dtg", "derivative"], 3)

x = pd.to_numeric(df[x_col], errors="coerce")
weight = pd.to_numeric(df[weight_col], errors="coerce")
dtg = pd.to_numeric(df[dtg_col], errors="coerce")
valid = x.notna() & weight.notna() & dtg.notna()
x, weight, dtg = x[valid], weight[valid], dtg[valid]

fig, ax1 = plt.subplots(figsize=(10, 6))
ax1.plot(x, weight, color="#1f4e9c", linewidth=1.8, label=str(weight_col))
ax1.set_xlabel(str(x_col))
ax1.set_ylabel(str(weight_col), color="#1f4e9c")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")

ax2 = ax1.twinx()
dtg_up = -dtg
ax2.plot(x, dtg_up, color="#c0392b", linewidth=1.5, label=f"-{dtg_col}")
ax2.set_ylabel(f"-{dtg_col}", color="#c0392b")
ax2.tick_params(axis="y", labelcolor="#c0392b")

# 잔류량: 측정 온도 중 800 °C에 가장 가까운 지점
residue_idx = (x - 800).abs().idxmin()
residue_x = float(x.loc[residue_idx])
residue_y = float(weight.loc[residue_idx])
ax1.scatter([residue_x], [residue_y], color="#1f4e9c", zorder=4)
ax1.annotate(
    f"Residue {residue_y:.1f}%",
    (residue_x, residue_y),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
)

# 피크: 150 °C 초과 구간에서 가장 높은 -DTG 값
peak_mask = x > 150
if peak_mask.any():
    peak_idx = dtg_up[peak_mask].idxmax()
    peak_x = float(x.loc[peak_idx])
    peak_y = float(dtg_up.loc[peak_idx])
    ax2.scatter([peak_x], [peak_y], color="#c0392b", zorder=4)
    ax2.annotate(
        f"{peak_x:.0f} °C",
        (peak_x, peak_y),
        xytext=(8, 10),
        textcoords="offset points",
        color="#c0392b",
    )

ax1.set_title(DATA_FILE.stem)
ax1.grid(True, alpha=0.3)
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc="center left")
fig.tight_layout()

fig.savefig(OUT_DIR / f"{DATA_FILE.stem}.png", dpi=300, bbox_inches="tight")
fig.savefig(OUT_DIR / f"{DATA_FILE.stem}.pdf", bbox_inches="tight")
plt.close(fig)
```
