```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/xrd.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B04_r3/out"

# Korean-capable font; this also works when labels contain Korean.
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

data = pd.read_csv(DATA_FILE)
if data.shape[1] < 2:
    raise ValueError("XRD 데이터에는 x와 intensity 열이 최소 두 개 필요합니다.")

# Use the first column as 2θ and the second as intensity.
x = pd.to_numeric(data.iloc[:, 0], errors="coerce").to_numpy()
y = pd.to_numeric(data.iloc[:, 1], errors="coerce").to_numpy()
valid = np.isfinite(x) & np.isfinite(y)
x, y = x[valid], y[valid]

if len(x) == 0:
    raise ValueError("유효한 XRD 데이터가 없습니다.")

order = np.argsort(x)
x, y = x[order], y[order]

# Locate local maxima, then select the five strongest distinct peaks.
peaks, _ = find_peaks(y, distance=max(1, len(y) // 300))
if len(peaks) == 0:
    peaks = np.array([int(np.argmax(y))])

strongest = peaks[np.argsort(y[peaks])[-5:]]
strongest = strongest[np.argsort(x[strongest])]

fig, ax = plt.subplots(figsize=(9, 5.5))
ax.plot(x, y, color="#1f4e9c", linewidth=1.2, label="Intensity")
ax.set_xlabel("2θ (deg)")
ax.set_ylabel("Intensity (counts)")
ax.set_title("XRD Pattern")
ax.grid(True, alpha=0.25)

y_span = max(float(np.nanmax(y) - np.nanmin(y)), 1.0)
for i, idx in enumerate(strongest):
    peak_x, peak_y = x[idx], y[idx]
    ax.plot(peak_x, peak_y, marker="o", markersize=4, color="#c0392b")
    # Alternate label heights slightly to reduce overlap.
    offset = 9 + (i % 2) * 13
    ax.annotate(
        f"{peak_x:.2f}°",
        xy=(peak_x, peak_y),
        xytext=(0, offset),
        textcoords="offset points",
        ha="center",
        va="bottom",
        fontsize=9,
        color="#8e2a20",
        arrowprops=dict(arrowstyle="-", color="#8e2a20", lw=0.7),
    )

ax.margins(x=0.02, y=0.18)
fig.tight_layout()
fig.savefig(out_dir / "xrd_pattern.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
