```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/12_r2/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df.columns = [str(column).strip() for column in df.columns]

    def find_column(keywords):
        for column in df.columns:
            if any(keyword.lower() in column.lower() for keyword in keywords):
                return column
        return None

    temperature_col = find_column(["temperature", "temp"])
    weight_col = find_column(["weight"])
    dtg_col = find_column(["d.w", "dtg", "derivative"])

    # TGA 열 이름을 찾지 못하는 경우 미리보기 구조의 B/C/D 열을 사용합니다.
    if temperature_col is None:
        temperature_col = df.columns[1]
    if weight_col is None:
        weight_col = df.columns[2]
    if dtg_col is None:
        dtg_col = df.columns[3]

    temperature = pd.to_numeric(df[temperature_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    dtg = pd.to_numeric(df[dtg_col], errors="coerce")
    valid = temperature.notna() & weight.notna() & dtg.notna()
    temperature = temperature[valid]
    weight = weight[valid]
    dtg = dtg[valid]

    if temperature.empty:
        continue

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    ax1.plot(temperature, weight, color="#1f4e9c", label="Weight (%)")
    ax2.plot(temperature, -dtg, color="#c0392b", label="DTG")

    ax1.set_title(sheet_name)
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="#1f4e9c")
    ax2.set_ylabel("DTG (-%/°C)", color="#c0392b")
    ax1.tick_params(axis="y", labelcolor="#1f4e9c")
    ax2.tick_params(axis="y", labelcolor="#c0392b")
    ax1.grid(True, alpha=0.25)

    # 450 °C 산화 시작 기준선과 라벨
    ax1.axvline(450, color="#555555", linestyle="--", linewidth=1.4)
    ax1.text(
        450, 0.98, "산화 시작",
        transform=ax1.get_xaxis_transform(),
        ha="left", va="top",
        color="#333333",
        bbox=dict(facecolor="white", edgecolor="none", alpha=0.8, pad=2),
    )

    # 800 °C에 가장 가까운 측정점의 잔류량 표시
    residue_idx = (temperature - 800).abs().idxmin()
    residue_temp = temperature.loc[residue_idx]
    residue = weight.loc[residue_idx]
    ax1.plot(residue_temp, residue, marker="o", color="#1f4e9c")
    ax1.annotate(
        f"Residue {residue:.1f}%",
        (residue_temp, residue),
        xytext=(8, 8), textcoords="offset points",
        color="#1f4e9c",
    )

    # 150 °C 초과 구간에서 가장 높은 DTG 피크 표시
    peak_region = temperature > 150
    if peak_region.any():
        peak_idx = (-dtg[peak_region]).idxmax()
        peak_temp = temperature.loc[peak_idx]
        peak_value = -dtg.loc[peak_idx]
        ax2.plot(peak_temp, peak_value, marker="o", color="#c0392b")
        ax2.annotate(
            f"{peak_temp:.0f} °C",
            (peak_temp, peak_value),
            xytext=(8, 8), textcoords="offset points",
            color="#c0392b",
        )

    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(lines1 + lines2, labels1 + labels2, loc="center left")

    fig.tight_layout()
    output_name = "".join(
        character if character.isalnum() or character in " -_." else "_"
        for character in sheet_name
    ).strip().replace(" ", "_")
    fig.savefig(OUT_DIR / f"{output_name}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
