```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B13_r2/out"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

# Load the requested sheet and identify its temperature and weight columns.
df = pd.read_excel(DATA_FILE, sheet_name="PVC")
temperature_col = next(c for c in df.columns if "temperature" in str(c).lower())
weight_col = next(c for c in df.columns if "weight" in str(c).lower() and "deriv" not in str(c).lower())

data = df[[temperature_col, weight_col]].apply(pd.to_numeric, errors="coerce").dropna()
data = data.sort_values(temperature_col).reset_index(drop=True)
temperature = data[temperature_col].to_numpy(dtype=float)
weight = data[weight_col].to_numpy(dtype=float)

# Find two pronounced mass-loss regions as the two strongest negative
# weight slopes, separated sufficiently to represent distinct events.
slope = np.gradient(weight, temperature)
candidate_indices = np.where((temperature >= 150) & (slope < 0))[0]
if candidate_indices.size == 0:
    candidate_indices = np.where(slope < 0)[0]

ranked = candidate_indices[np.argsort(slope[candidate_indices])]
selected = []
min_separation_c = 50.0
for idx in ranked:
    if all(abs(temperature[idx] - temperature[chosen]) >= min_separation_c for chosen in selected):
        selected.append(int(idx))
        if len(selected) == 2:
            break

# If slope extrema are not sufficiently separated, use the strongest
# local mass drops in non-overlapping temperature regions as a fallback.
if len(selected) < 2:
    split = np.nanmedian(temperature)
    for region in (temperature <= split, temperature > split):
        region_idx = np.where(region & (slope < 0))[0]
        if region_idx.size:
            idx = int(region_idx[np.argmin(slope[region_idx])])
            if all(abs(temperature[idx] - temperature[chosen]) >= 10 for chosen in selected):
                selected.append(idx)
        if len(selected) == 2:
            break

selected = sorted(selected, key=lambda i: temperature[i])

fig, ax = plt.subplots(figsize=(9, 6))
ax.plot(temperature, weight, color="#1f4e9c", linewidth=2, label="Weight (%)")
ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%)")
ax.set_title("PVC")
ax.grid(True, alpha=0.25)

for number, idx in enumerate(selected[:2], start=1):
    x, y = temperature[idx], weight[idx]
    ax.scatter(x, y, color="#c0392b", edgecolor="white", linewidth=0.8, s=65, zorder=3)
    ax.annotate(
        f"{x:.0f} °C",
        xy=(x, y),
        xytext=(8, 14 if number == 1 else -22),
        textcoords="offset points",
        color="#c0392b",
        fontsize=11,
        fontweight="bold",
        arrowprops=dict(arrowstyle="-", color="#c0392b", lw=1),
    )

fig.tight_layout()
fig.savefig(out_dir / "PVC_mass_drop_temperatures.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
