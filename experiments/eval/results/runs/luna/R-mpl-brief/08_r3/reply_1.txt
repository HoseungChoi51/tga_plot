from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/08_r3/out")

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

OUT_DIR.mkdir(parents=True, exist_ok=True)


def find_column(columns, keywords):
    """Find a column by matching any keyword, case-insensitively."""
    for column in columns:
        name = str(column).strip().lower()
        if any(keyword in name for keyword in keywords):
            return column
    return None


def safe_filename(name):
    return "".join(
        "_" if character in '<>:"/\\|?*' else character
        for character in str(name)
    ).strip() or "sheet"


excel = pd.ExcelFile(DATA_FILE)

for sheet_name in excel.sheet_names:
    data = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    data.columns = [str(column).strip() for column in data.columns]

    x_col = find_column(data.columns, ("temperature", "temp", "온도"))
    weight_col = find_column(data.columns, ("weight", "무게"))
    dtg_col = find_column(data.columns, ("d.w", "dtg", "derivative", "미분"))

    if x_col is None or weight_col is None or dtg_col is None:
        numeric_columns = [
            column for column in data.columns
            if pd.to_numeric(data[column], errors="coerce").notna().any()
        ]
        if len(numeric_columns) < 4:
            raise ValueError(f"시트 '{sheet_name}'에서 TGA 데이터 열을 찾을 수 없습니다.")
        x_col = x_col or numeric_columns[1]
        weight_col = weight_col or numeric_columns[2]
        dtg_col = dtg_col or numeric_columns[3]

    plot_data = data[[x_col, weight_col, dtg_col]].copy()
    for column in plot_data.columns:
        plot_data[column] = pd.to_numeric(plot_data[column], errors="coerce")
    plot_data = plot_data.dropna().sort_values(x_col)

    if plot_data.empty:
        continue

    temperature = plot_data[x_col].to_numpy()
    weight = plot_data[weight_col].to_numpy()
    dtg_up = -plot_data[dtg_col].to_numpy()

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    ax1.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight")
    ax2.plot(temperature, dtg_up, color="#c0392b", linewidth=1.5, label="DTG")

    ax1.set_title(str(sheet_name))
    ax1.set_xlabel("온도 (°C)")
    ax1.set_ylabel("무게 (%)", color="#1f4e9c")
    ax2.set_ylabel("미분 무게 (%/°C)", color="#c0392b")
    ax1.tick_params(axis="y", labelcolor="#1f4e9c")
    ax2.tick_params(axis="y", labelcolor="#c0392b")
    ax1.grid(True, alpha=0.25)

    # Residue: measured weight at the point nearest 800 °C.
    residue_index = int(np.argmin(np.abs(temperature - 800)))
    residue_temp = temperature[residue_index]
    residue_weight = weight[residue_index]
    ax1.plot(
        residue_temp, residue_weight, marker="o", color="#1f4e9c",
        markersize=5, linestyle="none",
    )
    ax1.annotate(
        f"Residue {residue_weight:.1f}%",
        (residue_temp, residue_weight),
        xytext=(8, 10),
        textcoords="offset points",
        color="#1f4e9c",
        fontsize=9,
    )

    # Peak: highest upward-pointing DTG value above 150 °C.
    peak_candidates = np.flatnonzero(temperature > 150)
    if peak_candidates.size:
        peak_index = peak_candidates[np.argmax(dtg_up[peak_candidates])]
        peak_temp = temperature[peak_index]
        peak_value = dtg_up[peak_index]
        ax2.plot(
            peak_temp, peak_value, marker="o", color="#c0392b",
            markersize=5, linestyle="none",
        )
        ax2.annotate(
            f"{peak_temp:.0f} °C",
            (peak_temp, peak_value),
            xytext=(8, 10),
            textcoords="offset points",
            color="#c0392b",
            fontsize=9,
        )

    # Combine both axes' curves in a center-left legend.
    handles1, labels1 = ax1.get_legend_handles_labels()
    handles2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(handles1 + handles2, labels1 + labels2, loc="center left")

    fig.tight_layout()
    output_path = OUT_DIR / f"{safe_filename(sheet_name)}.png"
    fig.savefig(output_path, dpi=300, bbox_inches="tight")
    plt.close(fig)
