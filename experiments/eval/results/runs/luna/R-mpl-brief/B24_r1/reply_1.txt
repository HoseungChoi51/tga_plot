```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B24_r1/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

df = pd.read_excel(DATA_FILE, sheet_name="PE")

# Use the columns identified from the worksheet preview.
temperature_c = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
temperature_k = temperature_c + 273.15
weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
dtg = pd.to_numeric(df["Deriv. Weight (%/°C)"], errors="coerce")

valid = temperature_k.notna() & weight.notna() & dtg.notna()
temperature_k = temperature_k[valid].to_numpy()
weight = weight[valid].to_numpy()
dtg = dtg[valid].to_numpy()

fig, ax1 = plt.subplots(figsize=(8, 5))
ax2 = ax1.twinx()

ax1.plot(temperature_k, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
ax2.plot(temperature_k, -dtg, color="#c0392b", linewidth=1.5, label="DTG (−%/°C)")

# Residue at the measured point nearest 800 °C (1073.15 K).
residue_target_k = 800.0 + 273.15
residue_idx = int(np.argmin(np.abs(temperature_k - residue_target_k)))
residue_x = temperature_k[residue_idx]
residue_y = weight[residue_idx]
ax1.scatter([residue_x], [residue_y], color="#1f4e9c", edgecolor="white", zorder=5)
ax1.annotate(
    f"Residue {residue_y:.1f}%",
    xy=(residue_x, residue_y),
    xytext=(8, 12),
    textcoords="offset points",
    color="#1f4e9c",
    fontsize=9,
)

# Highest upward DTG peak above 150 °C (423.15 K).
peak_candidates = np.flatnonzero(temperature_k > 150.0 + 273.15)
if peak_candidates.size:
    peak_idx = peak_candidates[np.argmax((-dtg)[peak_candidates])]
    peak_x = temperature_k[peak_idx]
    peak_y = -dtg[peak_idx]
    ax2.scatter([peak_x], [peak_y], color="#c0392b", edgecolor="white", zorder=5)
    ax2.annotate(
        f"{peak_x:.0f} K",
        xy=(peak_x, peak_y),
        xytext=(8, 10),
        textcoords="offset points",
        color="#c0392b",
        fontsize=9,
    )

ax1.set_title("PE")
ax1.set_xlabel("Temperature (K)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("DTG (−%/°C)", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.grid(True, linestyle="--", alpha=0.35)

lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc="center left")

fig.tight_layout()
fig.savefig(OUT_DIR / "PE_temperature_K.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
