```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os
from matplotlib import font_manager

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'

DATA_FILE = "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/CNT/Plot/eval/results/runs/sx/R-mpl-brief/07_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

xls = pd.ExcelFile(DATA_FILE)

for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name, header=0)
    
    # Identify columns based on the preview data structure
    # Assuming standard TGA format from preview: Time, Temperature, Weight(%), D.W(%/C)
    cols = df.columns.tolist()
    
    # Find temperature column (usually second or named containing 'Temp')
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for c in cols:
        if 'Temperature' in str(c):
            temp_col = c
        elif 'Weight' in str(c) and '%' in str(c):
            weight_col = c
        elif 'D.W' in str(c) or 'DTG' in str(c) or 'derivative' in str(c).lower():
            dtg_col = c
            
    # Fallback to positional if names are ambiguous but structure matches preview
    if temp_col is None and len(cols) >= 2: temp_col = cols[1]
    if weight_col is None and len(cols) >= 3: weight_col = cols[2]
    if dtg_col is None and len(cols) >= 4: dtg_col = cols[3]

    x = df[temp_col].values
    y1 = df[weight_col].values
    y2 = df[dtg_col].values * -1  # Invert DTG so peaks point up

    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot Weight (%) on left axis
    line1 = ax1.plot(x, y1, color='#1f4e9c', label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color='#1f4e9c', fontsize=12)
    ax1.tick_params(axis='y', labelcolor='#1f4e9c')
    
    # Create right axis for DTG
    ax2 = ax1.twinx()
    line2 = ax2.plot(x, y2, color='#c0392b', label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color='#c0392b', fontsize=12)
    ax2.tick_params(axis='y', labelcolor='#c0392b')
    
    # Title
    ax1.set_title(sheet_name, fontsize=14)
    
    # Calculate Residue at ~800°C
    mask_800 = (x >= 790) & (x <= 810)
    if not np.any(mask_800):
        # If no points near 800, take the last point or closest to 800
        idx_800 = np.argmin(np.abs(x - 800))
    else:
        idx_800 = np.where(mask_800)[0][0] # Take first one in range
        
    residue_val = y1[idx_800]
    residue_temp = x[idx_800]
    
    # Add residue marker and text
    ax1.plot(residue_temp, residue_val, 'o', color='#1f4e9c', markersize=5)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(residue_temp, residue_val), 
                 xytext=(10, 10), textcoords='offset points',
                 fontsize=9, color='#1f4e9c',
                 arrowprops=dict(arrowstyle='->', color='#1f4e9c'))
                 
    # Calculate Peak temperature above 150°C
    mask_peak = x > 150
    if np.any(mask_peak):
        y2_masked = y2[mask_peak]
        x_masked = x[mask_peak]
        
        # Find index of max value in masked array
        max_idx_in_masked = np.argmax(y2_masked)
        peak_temp = x_masked[max_idx_in_masked]
        peak_val = y2_masked[max_idx_in_masked]
        
        # Add peak marker and text on ax2
        ax2.plot(peak_temp, peak_val, 'v', color='#c0392b', markersize=5)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_val), 
                     xytext=(-10, 10), textcoords='offset points',
                     fontsize=9, color='#c0392b', ha='right',
                     arrowprops=dict(arrowstyle='->', color='#c0392b'))

    # Legend settings: Move to upper right, font size 9
    lines = line1 + line2
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='upper right', fontsize=9)
    
    # Grid
    ax1.grid(True, linestyle='--', alpha=0.5)
    
    # Save
    safe_sheet_name = sheet_name.replace('/', '_').replace('\\', '_')
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    plt.tight_layout()
    plt.savefig(out_path, dpi=300)
    plt.close(fig)

print("Plots saved.")
```
