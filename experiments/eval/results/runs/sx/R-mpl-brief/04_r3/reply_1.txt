```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/04_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read the specific sheet
sheet_name = "1006R1-V normal"
df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)

# Identify columns based on preview
x_col = "Temperature"
y1_col = "Weight(%)"
y2_col = "D.W(%/C)"

x = df[x_col].values
y1 = df[y1_col].values
y2 = df[y2_col].values

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight (%) on left y-axis
color1 = '#1f4e9c'
ax1.plot(x, y1, color=color1, label='Weight (%)')
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
ax1.tick_params(axis='y', labelcolor=color1)
ax1.grid(True, linestyle='--', alpha=0.5)

# Create second y-axis for DTG
ax2 = ax1.twinx()
color2 = '#c0392b'
# Multiply DTG by -1 so peaks point up
dtg_inverted = y2 * -1
ax2.plot(x, dtg_inverted, color=color2, label='DTG (%/°C)')
ax2.set_ylabel('DTG (%/°C)', color=color2, fontsize=12)
ax2.tick_params(axis='y', labelcolor=color2)

# Find Residue at nearest 800°C
idx_residue = np.argmin(np.abs(x - 800))
residue_val = y1[idx_residue]
residue_temp = x[idx_residue]

# Find Peak temperature of highest DTG peak above 150°C
mask_peak = x > 150
if np.any(mask_peak):
    # We want the minimum of original y2 (most negative) or maximum of inverted y2
    dtg_masked = dtg_inverted[mask_peak]
    idx_max_dtg_local = np.argmax(dtg_masked)
    # Map back to original index
    indices_above_150 = np.where(mask_peak)[0]
    idx_peak = indices_above_150[idx_max_dtg_local]
    peak_temp = x[idx_peak]
else:
    peak_temp = None
    idx_peak = None

# Add markers and labels for Residue
ax1.plot(residue_temp, residue_val, 'o', color=color1, markersize=8)
ax1.annotate(f'Residue {residue_val:.1f}%', 
             xy=(residue_temp, residue_val), 
             xytext=(residue_temp + 20, residue_val + 5),
             arrowprops=dict(facecolor='black', shrink=0.05),
             fontsize=10, color='black')

# Add markers and labels for Peak
if peak_temp is not None:
    ax2.plot(peak_temp, dtg_inverted[idx_peak], 'o', color=color2, markersize=8)
    ax2.annotate(f'{peak_temp:.0f} °C', 
                 xy=(peak_temp, dtg_inverted[idx_peak]), 
                 xytext=(peak_temp + 20, dtg_inverted[idx_peak] + 0.1),
                 arrowprops=dict(facecolor='black', shrink=0.05),
                 fontsize=10, color='black')

# Title settings
title_text = 'CNT 산화 거동'
ax1.set_title(title_text, fontsize=18, color='navy')

# Legend
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left')

plt.tight_layout()

# Save figure
output_filename = f"{sheet_name.replace(' ', '_')}.png"
output_path = os.path.join(OUT_DIR, output_filename)
plt.savefig(output_path, dpi=300)
plt.close()
```
