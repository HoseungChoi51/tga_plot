```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os
from pathlib import Path

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/07_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

for sheet_name in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)
    
    # Identify columns based on the preview data structure
    # Assuming standard TGA layout: Time, Temperature, Weight(%), D.W(%/C)
    cols = df.columns.tolist()
    
    # Find temperature column (usually 2nd or named 'Temperature')
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for c in cols:
        if 'temperature' in str(c).lower():
            temp_col = c
        elif 'weight' in str(c).lower() and '%' in str(c):
            weight_col = c
        elif 'd.w' in str(c).lower() or 'dtg' in str(c).lower():
            dtg_col = c
            
    # Fallback to positional if names aren't clear enough, but preview suggests names are good
    if temp_col is None:
        temp_col = cols[1] # B column usually
    if weight_col is None:
        weight_col = cols[2] # C column usually
    if dtg_col is None:
        dtg_col = cols[3] # D column usually
        
    x = df[temp_col].values
    y1 = df[weight_col].values
    y2 = df[dtg_col].values
    
    # Plotting
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    color1 = '#1f4e9c'
    color2 = '#c0392b'
    
    # Left axis: Weight %
    line1 = ax1.plot(x, y1, color=color1, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color1)
    
    # Right axis: DTG (%/°C) * -1 so peaks point up
    ax2 = ax1.twinx()
    # Multiply by -1 to make peaks positive/upward
    y2_plot = -y2 
    line2 = ax2.plot(x, y2_plot, color=color2, linestyle='--', label='DTG (-%/°C)')
    ax2.set_ylabel('DTG (-%/°C)', color=color2, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color2)
    
    # Title
    ax1.set_title(sheet_name, fontsize=14)
    
    # Calculate Residue at nearest 800°C
    # If max temp < 800, use last point? Or nearest available.
    # The prompt says "nearest 800 °C". 
    mask_800 = np.argmin(np.abs(x - 800))
    residue_val = y1[mask_800]
    residue_temp = x[mask_800]
    
    # Mark residue on curve
    ax1.scatter([residue_temp], [residue_val], color=color1, zorder=5)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(residue_temp, residue_val), 
                 xytext=(residue_temp + 20, residue_val - 5),
                 arrowprops=dict(facecolor=color1, shrink=0.05),
                 fontsize=10, color=color1)
                 
    # Calculate Peak: Highest DTG peak above 150°C
    # We look for the minimum of original DTG (which becomes maximum of -DTG)
    # Condition: Temp > 150
    valid_mask = x > 150
    if np.any(valid_mask):
        # In original data, loss is negative. So we look for min(y2) where x>150
        y2_valid = y2[valid_mask]
        x_valid = x[valid_mask]
        
        # Find index of minimum value (most negative = biggest mass loss rate)
        min_idx_local = np.argmin(y2_valid)
        peak_temp = x_valid[min_idx_local]
        peak_dtg_val = y2_valid[min_idx_local]
        
        # Mark peak on DTG curve (using the plotted values which are inverted)
        # On ax2, the value is -peak_dtg_val
        plot_peak_y = -peak_dtg_val
        
        ax2.scatter([peak_temp], [plot_peak_y], color=color2, zorder=5)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, plot_peak_y), 
                     xytext=(peak_temp + 20, plot_peak_y + 0.1),
                     arrowprops=dict(facecolor=color2, shrink=0.05),
                     fontsize=10, color=color2)
                     
    # Legend configuration
    # User request: Move legend to top right, font size 9
    lines = line1 + line2
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='upper right', fontsize=9)
    
    # Grid
    ax1.grid(True, linestyle=':', alpha=0.6)
    
    # Save
    safe_sheet_name = "".join([c if c.isalnum() else '_' for c in sheet_name])
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    plt.tight_layout()
    plt.savefig(out_path, dpi=300)
    plt.close(fig)

print("Plots saved.")
```
