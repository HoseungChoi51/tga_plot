```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import find_peaks
import os

# Set font for Korean text if needed, though labels are mostly English in TGA standard here.
# Using 'Noto Sans CJK KR' as requested by environment description.
plt.rcParams['font.family'] = 'sans-serif'
plt.rcParams['font.sans-serif'] = ['Noto Sans CJK KR', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B10_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

def plot_tga_blackwhite(file_path, sheet_name, out_dir):
    # Read specific sheet
    df = pd.read_excel(file_path, sheet_name=sheet_name)
    
    # Identify columns based on standard names from preview
    # B: Temperature (°C), C: Weight (%), D: Deriv. Weight (%/°C)
    temp_col = 'Temperature (°C)'
    weight_col = 'Weight (%)'
    dtg_col = 'Deriv. Weight (%/°C)'
    
    # Ensure numeric types
    temp = pd.to_numeric(df[temp_col], errors='coerce')
    weight = pd.to_numeric(df[weight_col], errors='coerce')
    dtg = pd.to_numeric(df[dtg_col], errors='coerce')
    
    # Drop rows with NaN in critical columns
    mask = ~(temp.isna() | weight.isna() | dtg.isna())
    temp = temp[mask]
    weight = weight[mask]
    dtg = dtg[mask]
    
    # Sort by temperature just in case
    sort_idx = np.argsort(temp.values)
    temp = temp.iloc[sort_idx].values
    weight = weight.iloc[sort_idx].values
    dtg = dtg.iloc[sort_idx].values
    
    # Create figure
    fig, ax1 = plt.subplots(figsize=(3.5, 2.6))
    
    # Plot Weight (%) on left y-axis (y1)
    line1, = ax1.plot(temp, weight, color='black', linestyle='-', linewidth=1.2, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=8)
    ax1.set_ylabel('Weight (%)', fontsize=8, color='black')
    ax1.tick_params(axis='both', which='major', labelsize=8)
    ax1.tick_params(axis='y', colors='black')
    
    # Plot DTG on right y-axis (y2)
    ax2 = ax1.twinx()
    # Multiply DTG by -1 so peaks point up
    dtg_inverted = -dtg
    line2, = ax2.plot(temp, dtg_inverted, color='gray', linestyle='--', linewidth=1.0, label='DTG (-dW/dT)')
    ax2.set_ylabel('DTG (%/°C)', fontsize=8, color='gray')
    ax2.tick_params(axis='y', colors='gray', labelsize=8)
    
    # Title is the sheet name
    ax1.set_title(sheet_name, fontsize=9, pad=10)
    
    # Find Residue at ~800°C
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = weight[idx_800]
    residue_temp = temp[idx_800]
    
    # Add marker and label for residue
    ax1.plot(residue_temp, residue_val, 'o', color='black', markersize=4)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(residue_temp, residue_val), 
                 xytext=(residue_temp - 100, residue_val + 5),
                 fontsize=7,
                 arrowprops=dict(facecolor='black', shrink=0.05, width=0.5, headwidth=3),
                 color='black')
                 
    # Find Peak: Highest DTG peak above 150°C
    # We look at the inverted DTG (which is positive for mass loss) to find the maximum
    # But strictly speaking, we want the peak of the original DTG (most negative value) or highest magnitude?
    # Standard practice: "Highest DTG peak" usually means the local maximum of the -DTG curve (upward peak).
    
    valid_mask = temp > 150
    temp_valid = temp[valid_mask]
    dtg_inv_valid = dtg_inverted[valid_mask]
    
    if len(dtg_inv_valid) > 0:
        # Use scipy to find peaks, then pick the one with the highest prominence/value
        peaks, properties = find_peaks(dtg_inv_valid, height=0.01, distance=10)
        
        if len(peaks) > 0:
            # Get indices of peaks in the valid subset
            peak_indices = peaks
            
            # Find the peak with the maximum height among detected peaks
            max_peak_idx_local = np.argmax(dtg_inv_valid[peak_indices])
            best_peak_local_idx = peak_indices[max_peak_idx_local]
            
            peak_temp = temp_valid[best_peak_local_idx]
            peak_dtg_inv_val = dtg_inv_valid[best_peak_local_idx]
            
            # Marker on the DTG curve (ax2)
            ax2.plot(peak_temp, peak_dtg_inv_val, 'v', color='gray', markersize=4)
            
            # Annotate temperature
            # Place annotation slightly above the peak
            ax2.annotate(f'{peak_temp:.0f} °C', 
                         xy=(peak_temp, peak_dtg_inv_val), 
                         xytext=(peak_temp, peak_dtg_inv_val + 0.1),
                         ha='center', va='bottom',
                         fontsize=7,
                         color='gray')
        else:
            # Fallback if no clear peak found via scipy, just take global max in range
            peak_idx_global_valid = np.argmax(dtg_inv_valid)
            peak_temp = temp_valid[peak_idx_global_valid]
            peak_dtg_inv_val = dtg_inv_valid[peak_idx_global_valid]
            
            ax2.plot(peak_temp, peak_dtg_inv_val, 'v', color='gray', markersize=4)
            ax2.annotate(f'{peak_temp:.0f} °C', 
                         xy=(peak_temp, peak_dtg_inv_val), 
                         xytext=(peak_temp, peak_dtg_inv_val + 0.1),
                         ha='center', va='bottom',
                         fontsize=7,
                         color='gray')

    # Legend
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='center left', fontsize=7, frameon=False)
    
    # Grid
    ax1.grid(True, linestyle=':', alpha=0.5, color='gray')
    
    # Layout adjustment
    plt.tight_layout()
    
    # Save
    out_file = os.path.join(out_dir, f"{sheet_name.replace(' ', '_')}.png")
    plt.savefig(out_file, dpi=600, bbox_inches='tight')
    plt.close(fig)

# Process only Annealed sheet as requested
plot_tga_blackwhite(DATA_FILE, 'Annealed', OUT_DIR)
```
