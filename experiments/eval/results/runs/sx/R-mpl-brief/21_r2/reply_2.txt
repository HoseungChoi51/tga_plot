```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/blank_rows_on_top.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/21_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read the Excel file with header at row 8 (0-indexed)
df = pd.read_excel(DATA_FILE, sheet_name='Sheet1', header=8)

# The user requested: C column -> X axis, D column -> Y1 axis, E column -> Y2 axis
# In pandas DataFrame, columns are accessed by name or position.
# Based on the preview:
# Col B is "Time", Col C is "Temperature", Col D is "Weight(%)", Col E is "D.W(%/C)"
# Since we read with header=8, the columns in df will correspond to these names if they were in row 8.
# However, sometimes headers might be unnamed or have spaces. Let's inspect available columns and match them.
# The error suggests exact string matching failed. Let's try accessing by index if names don't match exactly, 
# or clean the names first.

# Print columns to debug (optional, but good for robustness logic)
# print(df.columns.tolist())

# Strategy: Try to find columns by partial match or use positional indexing if names are unreliable.
# User specified C, D, E columns from the original Excel.
# When reading with header=8, the data starts at row 9.
# Column indices in the dataframe:
# If the header row had values, they become column names.
# Let's assume the standard TGA structure found in the preview.

cols = df.columns.tolist()

# Helper to find column index/name by looking for keywords
def find_col(cols, keyword):
    for c in cols:
        if keyword.lower() in str(c).lower():
            return c
    return None

# Find Temperature (X), Weight (Y1), DTG/D.W (Y2)
col_x = find_col(cols, 'temperature')
if col_x is None:
    # Fallback to positional if name search fails? 
    # The preview says C is Temperature. If header parsing shifted things, maybe we need positional.
    # But let's try strict names from preview first, stripping whitespace just in case.
    col_x = find_col(cols, 'temp')

col_y1 = find_col(cols, 'weight')
col_y2 = find_col(cols, 'd.w') or find_col(cols, 'dtg') or find_col(cols, 'derivative')

# If specific names aren't found, fall back to positional based on the preview description:
# Preview: B=Time, C=Temp, D=Weight, E=D.W
# In df (read with header=8), the columns should map to these.
# If col_x etc are still None, use positions. Note: Time is usually col 0 if B is first data col.
# So Temp is likely index 1, Weight index 2, DW index 3.

if col_x is None:
    # Assume order: Time, Temp, Weight, DW
    if len(df.columns) >= 4:
        col_x = df.columns[1]
    else:
        raise ValueError("Could not identify Temperature column")

if col_y1 is None:
    if len(df.columns) >= 3:
        col_y1 = df.columns[2]
    else:
        raise ValueError("Could not identify Weight column")

if col_y2 is None:
    if len(df.columns) >= 4:
        col_y2 = df.columns[3]
    else:
        raise ValueError("Could not identify DTG column")

# Extract data
df_clean = df[[col_x, col_y1, col_y2]].dropna()

x = df_clean[col_x].values
y1 = df_clean[col_y1].values
y2 = df_clean[col_y2].values

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(10, 6))

# Plot Weight (%) on left Y-axis
color1 = '#1f4e9c'
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
line1, = ax1.plot(x, y1, color=color1, label='Weight (%)')
ax1.tick_params(axis='y', labelcolor=color1)

# Create secondary axis for DTG
ax2 = ax1.twinx()
color2 = '#c0392b'
ax2.set_ylabel('DTG (%/°C)', color=color2, fontsize=12)
# Multiply by -1 so peaks point up
line2, = ax2.plot(x, -y2, color=color2, label='DTG (-1 * %/°C)')
ax2.tick_params(axis='y', labelcolor=color2)

# Find Residue at nearest 800°C
idx_800 = np.argmin(np.abs(x - 800))
residue_val = y1[idx_800]
temp_residue = x[idx_800]

# Add residue marker and label
ax1.scatter(temp_residue, residue_val, color=color1, zorder=5)
ax1.annotate(f'Residue {residue_val:.1f}%', 
             xy=(temp_residue, residue_val), 
             xytext=(10, -15), textcoords='offset points',
             color=color1, fontsize=10, fontweight='bold')

# Find Peak temperature above 150°C (highest DTG peak means most negative original value, or highest positive after -1 multiplication)
# We look for max of -y2 (which is the plotted value) for x > 150
mask_peak = x > 150
if np.any(mask_peak):
    # Find index of maximum value in -y2 within the mask
    neg_y2 = -y2
    idx_max_neg_y2 = np.argmax(neg_y2[mask_peak])
    # Map back to original index
    indices_above_150 = np.where(mask_peak)[0]
    peak_idx = indices_above_150[idx_max_neg_y2]
    
    peak_temp = x[peak_idx]
    peak_val_plotted = -y2[peak_idx]
    
    # Add peak marker on ax2
    ax2.scatter(peak_temp, peak_val_plotted, color=color2, zorder=5)
    ax2.annotate(f'{peak_temp:.0f} °C', 
                 xy=(peak_temp, peak_val_plotted), 
                 xytext=(10, 10), textcoords='offset points',
                 color=color2, fontsize=10, fontweight='bold')

# Title
sheet_name = "Sheet1"
plt.title(sheet_name, fontsize=14)

# Legend
lines = [line1, line2]
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='center left')

# Grid
ax1.grid(True, linestyle='--', alpha=0.5)

# Save plot
out_path = os.path.join(OUT_DIR, f"{sheet_name}.png")
plt.savefig(out_path, dpi=300, bbox_inches='tight')
plt.close()
```
