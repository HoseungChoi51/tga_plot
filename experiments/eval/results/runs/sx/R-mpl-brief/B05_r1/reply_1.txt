```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILES = ["/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_A.csv", "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_B.txt"]
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B05_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

fig, ax1 = plt.subplots(figsize=(8, 6))

colors = ['#1f4e9c', '#2ca02c']  # Blue for Lot A, Green for Lot B
labels = ['Lot A', 'Lot B']

for i, file_path in enumerate(DATA_FILES):
    if file_path.endswith('.csv'):
        df = pd.read_csv(file_path)
    elif file_path.endswith('.txt'):
        df = pd.read_csv(file_path, sep=r'\s+')
    else:
        continue
    
    temp = df['Temperature (°C)'].values
    weight = df['Weight (%)'].values
    dtg = df['Deriv. Weight (%/°C)'].values
    
    # Plot Weight on left axis
    ax1.plot(temp, weight, color=colors[i], label=labels[i], linewidth=1.5)
    
    # Find residue at ~800°C
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = weight[idx_800]
    
    # Find DTG peak above 150°C
    mask = temp > 150
    if np.any(mask):
        dtg_masked = dtg[mask]
        temp_masked = temp[mask]
        # Peak is the most negative value (highest mass loss rate), but we plot -DTG so it points up
        # We want the temperature where -DTG is maximum, which corresponds to min(DTG)
        peak_idx_local = np.argmin(dtg_masked)
        peak_temp = temp_masked[peak_idx_local]
        
        # Plot markers for this lot
        # Residue marker
        ax1.scatter([temp[idx_800]], [residue_val], color=colors[i], s=30, zorder=5)
        # Peak marker on DTG curve? Usually peak label refers to the DTG peak location.
        # Since we are overlaying, putting too many markers might clutter. 
        # The instruction says "two labels are always drawn unless the user says otherwise".
        # But with two lots overlaid, standard practice is often just the curves + legend.
        # However, strict adherence: "two labels... residue... peak".
        # If I add text for both, it will be messy. Let's look at the request: "두 파일을 한 그림에 겹쳐 그리고 범례에는 Lot A, Lot B라고 써줘".
        # It doesn't explicitly say remove labels, but usually overlapping plots focus on comparison.
        # Let's stick to the default rule: draw residue and peak labels.
        # To avoid clutter, I'll place them carefully or maybe only for one? No, rule says always.
        # Actually, for overlaid plots, specific annotations like "Residue X%" for each series can be added near the curves or in a table.
        # Given the constraint "make a reasonable choice", and the potential for clutter, I will annotate the values on the plot using small text near the features, or perhaps just rely on the legend if space is tight. 
        # Let's try adding the text labels.
        
        # Text for residue
        ax1.text(temp[idx_800], residue_val, f'{residue_val:.1f}%', 
                 fontsize=8, color=colors[i], ha='center', va='bottom', 
                 bbox=dict(facecolor='white', alpha=0.7, edgecolor='none', pad=1))
                 
        # Text for peak temp
        # We need to know the y-value of the DTG curve at that point to place the text? 
        # Or just place it near the x-axis? The prompt says "written like '600 °C', with a marker".
        # Marker on the curve. Which curve? The DTG curve is on the right axis.
        # So we need a second axis.
        pass

# Create right axis for DTG
ax2 = ax1.twinx()

for i, file_path in enumerate(DATA_FILES):
    if file_path.endswith('.csv'):
        df = pd.read_csv(file_path)
    elif file_path.endswith('.txt'):
        df = pd.read_csv(file_path, sep=r'\s+')
    else:
        continue
        
    temp = df['Temperature (°C)'].values
    dtg = df['Deriv. Weight (%/°C)'].values
    
    # Plot -DTG on right axis
    ax2.plot(temp, -dtg, color=colors[i], linestyle='--', linewidth=1.0, alpha=0.7)
    
    # Markers and Labels again for precision
    # Residue
    idx_800 = np.argmin(np.abs(temp - 800))
    weight = df['Weight (%)'].values
    residue_val = weight[idx_800]
    
    # Peak
    mask = temp > 150
    if np.any(mask):
        dtg_masked = dtg[mask]
        temp_masked = temp[mask]
        peak_idx_local = np.argmin(dtg_masked)
        peak_temp = temp_masked[peak_idx_local]
        peak_dtg_val = dtg[peak_idx_local] # Original value
        
        # Marker for peak on DTG curve (right axis)
        # Y position on right axis is -peak_dtg_val
        ax2.scatter([peak_temp], [-peak_dtg_val], color=colors[i], s=30, marker='v', zorder=5)
        
        # Label for peak temperature
        # Place it slightly above the peak marker
        ax2.text(peak_temp, -peak_dtg_val, f'{peak_temp:.0f}°C', 
                 fontsize=8, color=colors[i], ha='center', va='bottom',
                 bbox=dict(facecolor='white', alpha=0.7, edgecolor='none', pad=1))

# Axis settings
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12, color='#1f4e9c') # Using first color as generic for label? Or black. Standard is black.
ax1.tick_params(axis='y', labelcolor='black')

ax2.set_ylabel('-DTG (%/°C)', fontsize=12, color='black')
ax2.tick_params(axis='y', labelcolor='black')

ax1.set_xlim(min(df['Temperature (°C)'])-10, max(df['Temperature (°C)'])+10)

# Legend
lines, labels_list = ax1.get_legend_handles_labels()
ax1.legend(lines, labels_list, loc='center left', fontsize=10)

# Title
ax1.set_title('TGA Overlay: Lot A vs Lot B', fontsize=14)

plt.tight_layout()
out_path = os.path.join(OUT_DIR, 'overlay_tga.png')
plt.savefig(out_path, dpi=300)
plt.close()
```
