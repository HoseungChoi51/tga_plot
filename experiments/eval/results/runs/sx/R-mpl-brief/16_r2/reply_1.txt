```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/16_r2/out"
SHEET_NAME = "1006R1-V normal"

# Read data
df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)

# Extract columns based on preview info
x_col = 'Temperature'
y1_col = 'Weight(%)'

# Prepare plot
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight (%)
ax1.plot(df[x_col], df[y1_col], color='#1f4e9c', label='Weight (%)')
ax1.set_xlabel('Temperature (°C)')
ax1.set_ylabel('Weight (%)')
ax1.set_title(SHEET_NAME)

# Add residue marker at nearest point to 800°C
# Find index of temperature closest to 800
idx_800 = (df[x_col] - 800).abs().idxmin()
temp_800 = df.loc[idx_800, x_col]
weight_800 = df.loc[idx_800, y1_col]

# Draw marker and text for residue
ax1.plot(temp_800, weight_800, 'o', color='#1f4e9c', markersize=5)
ax1.annotate(f'Residue {weight_800:.1f}%', 
             xy=(temp_800, weight_800), 
             xytext=(10, 10), 
             textcoords='offset points',
             fontsize=10, 
             color='#1f4e9c',
             arrowprops=dict(arrowstyle='->', color='#1f4e9c'))

# Legend
ax1.legend(loc='center left')

# Ensure output directory exists
os.makedirs(OUT_DIR, exist_ok=True)

# Save figure
output_path = os.path.join(OUT_DIR, f"{SHEET_NAME.replace(' ', '_').replace('/', '_')}.png")
plt.tight_layout()
plt.savefig(output_path, dpi=300)
plt.close()
```
