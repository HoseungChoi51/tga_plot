```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.font_manager as fm

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B14_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read only the PVC sheet
df = pd.read_excel(DATA_FILE, sheet_name="PVC")

# Identify columns based on standard TGA structure from preview
# Col B: Temperature (°C), Col C: Weight (%), Col D: Deriv. Weight (%/°C)
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

# Extract data
temp = df[temp_col].values
weight = df[weight_col].values
dtg = df[dtg_col].values

# Prepare DTG for plotting (invert sign so peaks point up)
dtg_plot = -dtg

# Find Residue at 800°C
idx_800 = np.argmin(np.abs(temp - 800))
residue_temp = temp[idx_800]
residue_weight = weight[idx_800]

# Find Peak temperature (highest DTG peak above 150°C)
mask_peak = temp > 150
if np.any(mask_peak):
    # We want the highest peak in dtg_plot (which is -dtg). 
    # Since dtg is negative during mass loss, -dtg is positive.
    # So we look for max value in dtg_plot where temp > 150
    dtg_masked = dtg_plot[mask_peak]
    temp_masked = temp[mask_peak]
    
    if len(dtg_masked) > 0:
        idx_max_dtg = np.argmax(dtg_masked)
        peak_temp = temp_masked[idx_max_dtg]
        peak_val = dtg_masked[idx_max_dtg]
    else:
        peak_temp = None
        peak_val = None
else:
    peak_temp = None
    peak_val = None

# Plotting
fig, ax1 = plt.subplots(figsize=(10, 6))

# Y1 Axis: Weight (%)
color_y1 = '#1f4e9c'
ax1.plot(temp, weight, color=color_y1, label='Weight')
ax1.set_xlabel('Temperature (°C)', fontsize=16)
ax1.set_ylabel('Weight (%)', color=color_y1, fontsize=16)
ax1.tick_params(axis='y', labelcolor=color_y1, labelsize=16)
ax1.tick_params(axis='x', labelsize=16)

# Y2 Axis: DTG
ax2 = ax1.twinx()
color_y2 = '#c0392b'
ax2.plot(temp, dtg_plot, color=color_y2, label='DTG')
ax2.set_ylabel('DTG (%/°C)', color=color_y2, fontsize=16)
ax2.tick_params(axis='y', labelcolor=color_y2, labelsize=16)

# Title
sheet_name = "PVC"
ax1.set_title(sheet_name, fontsize=16)

# Legend
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left', fontsize=16)

# Annotations
# Residue marker and label
ax1.plot(residue_temp, residue_weight, 'o', color=color_y1, markersize=8)
ax1.annotate(f'Residue {residue_weight:.1f}%', 
             xy=(residue_temp, residue_weight), 
             xytext=(10, -20), textcoords='offset points',
             fontsize=16, color=color_y1,
             arrowprops=dict(facecolor=color_y1, shrink=0.05, width=1, headwidth=5))

# Peak marker and label
if peak_temp is not None:
    # Find the corresponding y-value on the DTG curve for annotation position
    # Note: The annotation should probably point to the peak on the DTG curve visually
    # But usually "Peak" refers to the temperature. Let's mark it on the DTG curve.
    # We need the index in the original array for correct plotting
    full_idx_peak = np.where((temp == peak_temp))[0][0]
    
    ax2.plot(peak_temp, dtg_plot[full_idx_peak], 'o', color=color_y2, markersize=8)
    ax2.annotate(f'{peak_temp:.0f} °C', 
                 xy=(peak_temp, dtg_plot[full_idx_peak]), 
                 xytext=(10, 10), textcoords='offset points',
                 fontsize=16, color=color_y2,
                 arrowprops=dict(facecolor=color_y2, shrink=0.05, width=1, headwidth=5))

plt.tight_layout()

out_path = os.path.join(OUT_DIR, f"{sheet_name}.png")
plt.savefig(out_path, dpi=300)
plt.close()
```
