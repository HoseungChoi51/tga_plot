```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/05_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

xls = pd.ExcelFile(DATA_FILE)

for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name, header=1)
    
    # Identify columns based on standard TGA structure from preview
    x_col = None
    y1_col = None
    y2_col = None
    
    cols = df.columns.tolist()
    
    for c in cols:
        if 'Temperature' in str(c):
            x_col = c
        elif 'Weight' in str(c) and '%' in str(c):
            y1_col = c
        elif 'D.W' in str(c) or 'DTG' in str(c):
            y2_col = c
            
    # Fallback to indices if names don't match strictly
    if x_col is None: x_col = cols[1]
    if y1_col is None: y1_col = cols[2]
    if y2_col is None: y2_col = cols[3]

    temp = df[x_col].values
    weight = df[y1_col].values
    dtg = df[y2_col].values
    
    # Plotting
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Y1 (Left): Weight (%) - Black, linewidth 2.5
    color_y1 = 'black'
    ax1.plot(temp, weight, color=color_y1, linewidth=2.5, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color_y1, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color_y1)
    
    # Y2 (Right): DTG (%/°C) - Dashed line
    ax2 = ax1.twinx()
    # Multiply by -1 so peaks point up
    dtg_inv = dtg * -1
    color_y2 = '#c0392b' # Standard red for DTG unless specified otherwise, but user only changed style
    ax2.plot(temp, dtg_inv, color=color_y2, linestyle='--', linewidth=1.5, label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color=color_y2, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color_y2)
    
    # Title
    ax1.set_title(sheet_name, fontsize=14)
    
    # Residue Label (nearest 800°C)
    target_temp = 800.0
    idx_res = np.argmin(np.abs(temp - target_temp))
    residue_val = weight[idx_res]
    residue_temp = temp[idx_res]
    
    # Marker for residue
    ax1.plot(residue_temp, residue_val, 'o', color='black', markersize=5)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(residue_temp, residue_val), 
                 xytext=(10, -15), textcoords='offset points',
                 fontsize=10, color='black',
                 arrowprops=dict(arrowstyle='->', color='black'))
    
    # Peak Label (highest DTG peak above 150°C)
    mask = temp > 150
    if np.any(mask):
        dtg_masked = dtg[mask]
        temp_masked = temp[mask]
        
        # Find index of minimum DTG (which corresponds to max mass loss rate)
        # Since we plotted dtg * -1, the highest visual peak is where dtg is most negative.
        min_dtg_idx_local = np.argmin(dtg_masked)
        peak_temp = temp_masked[min_dtg_idx_local]
        peak_dtg_val = dtg_masked[min_dtg_idx_local]
        
        # Marker for peak on DTG curve (ax2)
        ax2.plot(peak_temp, peak_dtg_val * -1, 'v', color=color_y2, markersize=5)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_dtg_val * -1), 
                     xytext=(-10, 10), textcoords='offset points',
                     fontsize=10, color=color_y2, ha='right',
                     arrowprops=dict(arrowstyle='->', color=color_y2))
    
    # Legend at center left
    lines_1, labels_1 = ax1.get_legend_handles_labels()
    lines_2, labels_2 = ax2.get_legend_handles_labels()
    ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center left', bbox_to_anchor=(0.05, 0.5))
    
    plt.tight_layout()
    
    # Save file
    safe_sheet_name = "".join([i if i.isalnum() or i in ['-', '_'] else '_' for i in sheet_name])
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    plt.savefig(out_path, dpi=300)
    plt.close(fig)

print("Plots saved successfully.")
```
