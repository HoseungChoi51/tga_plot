```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B07_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read only the second sheet (index 1), which is "PVC"
df = pd.read_excel(DATA_FILE, sheet_name=1)

# Extract columns based on standard TGA structure from preview
temp_col = df.columns[1]   # Temperature (°C)
weight_col = df.columns[2] # Weight (%)
dtg_col = df.columns[3]    # Deriv. Weight (%/°C)

temperature = df[temp_col].values
weight = df[weight_col].values
dtg = df[dtg_col].values

# Get sheet name for title
sheet_names = pd.ExcelFile(DATA_FILE).sheet_names
sheet_name = sheet_names[1]

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight on left y-axis
color_weight = '#1f4e9c'
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color_weight, fontsize=12)
line1, = ax1.plot(temperature, weight, color=color_weight, label='Weight')
ax1.tick_params(axis='y', labelcolor=color_weight)

# Create secondary y-axis for DTG
ax2 = ax1.twinx()
color_dtg = '#c0392b'
ax2.set_ylabel('DTG (%/°C)', color=color_dtg, fontsize=12)
# Multiply DTG by -1 so peaks point up
line2, = ax2.plot(temperature, dtg * -1, color=color_dtg, linestyle='--', label='DTG (-1×)')
ax2.tick_params(axis='y', labelcolor=color_dtg)

# Title
ax1.set_title(sheet_name, fontsize=14, fontweight='bold')

# Combine legends
lines = [line1, line2]
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='center left')

# Function to find weight at a specific temperature
def get_weight_at_temp(temp_target):
    idx = np.argmin(np.abs(temperature - temp_target))
    return temperature[idx], weight[idx]

# Mark weights at 300°C and 600°C
temps_to_mark = [300, 600]
for t_mark in temps_to_mark:
    t_act, w_val = get_weight_at_temp(t_mark)
    # Draw marker on weight curve
    ax1.scatter([t_act], [w_val], color='black', zorder=5, s=50)
    # Annotate
    ax1.annotate(f'{w_val:.1f}%', 
                 xy=(t_act, w_val), 
                 xytext=(10, 10), 
                 textcoords='offset points',
                 fontsize=10,
                 bbox=dict(boxstyle='round,pad=0.3', fc='yellow', alpha=0.7),
                 arrowprops=dict(arrowstyle='->', connectionstyle='arc3,rad=0'))

# Adjust layout
plt.tight_layout()

# Save plot
output_path = os.path.join(OUT_DIR, f"{sheet_name}_TGA.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
