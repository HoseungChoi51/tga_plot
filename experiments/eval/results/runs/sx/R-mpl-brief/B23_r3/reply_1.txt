```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B23_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets
xl = pd.ExcelFile(DATA_FILE)
sheets = xl.sheet_names

# Define colors and labels
colors = {
    "Raw": "black",
    "Acid-washed": "blue",
    "Annealed": "red"
}

fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(8, 10), sharex=True)
fig.suptitle("TGA Comparison of Samples", fontsize=14, fontweight='bold')

for sheet_name in sheets:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    
    # Identify columns based on standard TGA structure or guess from preview
    # Preview says: B=Temp, C=Weight, D=Deriv Weight
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    if temp_col not in df.columns:
        # Fallback to positional if names differ slightly but structure is same
        temp_col = df.columns[1]
        weight_col = df.columns[2]
        dtg_col = df.columns[3]
        
    temp = df[temp_col].values
    weight = df[weight_col].values
    dtg = df[dtg_col].values
    
    color = colors.get(sheet_name, "gray")
    label = sheet_name
    
    # Plot Weight (%) on top axis (ax1)
    ax1.plot(temp, weight, color=color, label=label, linewidth=1.5)
    
    # Plot DTG (-DTG) on bottom axis (ax2)
    # Multiply by -1 so peaks point up
    ax2.plot(temp, -dtg, color=color, label=label, linewidth=1.5)

# Configure Top Axis (Weight %)
ax1.set_ylabel("Weight (%)", fontsize=12)
ax1.legend(loc="center left", title=None)
ax1.grid(True, linestyle="--", alpha=0.6)
ax1.tick_params(axis='both', which='major', labelsize=10)

# Configure Bottom Axis (DTG)
ax2.set_xlabel("Temperature (°C)", fontsize=12)
ax2.set_ylabel("DTG (%/°C)", fontsize=12)
ax2.legend(loc="center left", title=None)
ax2.grid(True, linestyle="--", alpha=0.6)
ax2.tick_params(axis='both', which='major', labelsize=10)

# Add Residue and Peak markers for each sample
for sheet_name in sheets:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    if temp_col not in df.columns:
        temp_col = df.columns[1]
        weight_col = df.columns[2]
        dtg_col = df.columns[3]
        
    temp = df[temp_col].values
    weight = df[weight_col].values
    dtg = df[dtg_col].values
    
    color = colors.get(sheet_name, "gray")
    
    # Residue at ~800°C
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = weight[idx_800]
    temp_res = temp[idx_800]
    
    # Mark residue on ax1
    ax1.plot(temp_res, residue_val, 'o', color=color, markersize=6)
    ax1.annotate(f"{residue_val:.1f}%", 
                 xy=(temp_res, residue_val), 
                 xytext=(10, 5), textcoords='offset points',
                 color=color, fontsize=9, ha='left')
                 
    # Peak DTG above 150°C
    mask = temp > 150
    if np.any(mask):
        dtg_masked = dtg[mask]
        temp_masked = temp[mask]
        # Find index of minimum DTG (which becomes max peak after negation)
        min_dtg_idx = np.argmin(dtg_masked)
        peak_temp = temp_masked[min_dtg_idx]
        peak_dtg_val = -dtg_masked[min_dtg_idx] # Value plotted on ax2
        
        # Mark peak on ax2
        ax2.plot(peak_temp, peak_dtg_val, 'v', color=color, markersize=6)
        ax2.annotate(f"{peak_temp:.0f}°C", 
                     xy=(peak_temp, peak_dtg_val), 
                     xytext=(10, 5), textcoords='offset points',
                     color=color, fontsize=9, ha='left')

plt.tight_layout(rect=[0, 0.03, 1, 0.95])

output_path = os.path.join(OUT_DIR, "tga_comparison.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
