```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/13_r3/out"

if not os.path.exists(OUT_DIR):
    os.makedirs(OUT_DIR)

# Read the Excel file
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

for sheet_name in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    
    # Identify columns based on the preview and user request
    # x: Temperature (°C) -> Column B
    # y1: Weight (mg) -> Column D (User requested this instead of default Weight %)
    # y2: Deriv. Weight (%/°C) -> Column G
    
    x_col = None
    y1_col = None
    y2_col = None
    
    for col in df.columns:
        col_str = str(col).lower()
        if 'temperature' in col_str or 'temp' in col_str:
            x_col = col
        elif 'weight (mg)' in col_str or ('weight' in col_str and 'mg' in col_str):
            y1_col = col
        elif 'deriv' in col_str or 'dtg' in col_str:
            y2_col = col
            
    # Fallback to standard names if specific ones aren't found by logic above but exist in data structure
    if x_col is None:
        for col in df.columns:
            if 'Temperature' in str(col):
                x_col = col
                break
                
    if y1_col is None:
        # User specifically asked for Weight (mg)
        for col in df.columns:
            if 'Weight (mg)' in str(col):
                y1_col = col
                break
        if y1_col is None:
             # If strictly "Weight (mg)" isn't there, maybe just "Weight" but we know from preview it's there
             pass 

    if y2_col is None:
        for col in df.columns:
            if 'Deriv' in str(col) or 'DTG' in str(col):
                y2_col = col
                break

    # Ensure we have valid columns
    if x_col is None or y1_col is None or y2_col is None:
        continue
        
    # Extract data
    temp = df[x_col].values
    weight_mg = df[y1_col].values
    dtg = df[y2_col].values
    
    # Clean data: remove NaNs
    mask = ~(np.isnan(temp) | np.isnan(weight_mg) | np.isnan(dtg))
    temp = temp[mask]
    weight_mg = weight_mg[mask]
    dtg = dtg[mask]
    
    if len(temp) == 0:
        continue

    # Create figure
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot y1 (Weight mg) on left axis
    color_y1 = '#1f4e9c'
    line1, = ax1.plot(temp, weight_mg, color=color_y1, label='Weight (mg)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (mg)', fontsize=12, color=color_y1)
    ax1.tick_params(axis='y', labelcolor=color_y1)
    
    # Plot y2 (DTG) on right axis
    ax2 = ax1.twinx()
    color_y2 = '#c0392b'
    # Multiply DTG by -1 so peaks point up
    dtg_inverted = -1 * dtg
    line2, = ax2.plot(temp, dtg_inverted, color=color_y2, linestyle='--', label='DTG (-1*%/\u00b0C)')
    ax2.set_ylabel('DTG (-1*%/\u00b0C)', fontsize=12, color=color_y2)
    ax2.tick_params(axis='y', labelcolor=color_y2)
    
    # Title
    ax1.set_title(sheet_name, fontsize=14)
    
    # Legend
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='center left', bbox_to_anchor=(0.5, 0.5), frameon=True)
    
    # Calculate Residue at ~800°C
    # Find index closest to 800
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = weight_mg[idx_800]
    residue_temp = temp[idx_800]
    
    # Add Residue marker and text
    ax1.plot(residue_temp, residue_val, 'o', color=color_y1, markersize=8)
    ax1.annotate(f'Residue {residue_val:.2f} mg', 
                 xy=(residue_temp, residue_val), 
                 xytext=(residue_temp + 20, residue_val + (ax1.get_ylim()[1]-ax1.get_ylim()[0])*0.1),
                 arrowprops=dict(facecolor='black', shrink=0.05),
                 fontsize=10, color=color_y1)
                 
    # Calculate Peak temperature (highest DTG peak above 150°C)
    # We look for max of inverted DTG (which corresponds to min of original DTG, i.e., steepest mass loss)
    # Only consider temps > 150
    mask_peak = temp > 150
    if np.any(mask_peak):
        dtg_inv_masked = dtg_inverted[mask_peak]
        temp_masked = temp[mask_peak]
        
        # Find global maximum of the inverted DTG in this range
        idx_max = np.argmax(dtg_inv_masked)
        peak_temp = temp_masked[idx_max]
        peak_dtg_inv_val = dtg_inv_masked[idx_max]
        
        # Mark on plot
        # Note: The peak is on the DTG curve (right axis)
        ax2.plot(peak_temp, peak_dtg_inv_val, 'v', color=color_y2, markersize=8)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_dtg_inv_val), 
                     xytext=(peak_temp - 50, peak_dtg_inv_val + (ax2.get_ylim()[1]-ax2.get_ylim()[0])*0.1),
                     arrowprops=dict(facecolor='black', shrink=0.05),
                     fontsize=10, color=color_y2)

    # Grid
    ax1.grid(True, linestyle=':', alpha=0.6)
    
    # Save
    out_path = os.path.join(OUT_DIR, f"{sheet_name}.png")
    plt.tight_layout()
    plt.savefig(out_path, dpi=300)
    plt.close(fig)

print("Plots saved.")
```
