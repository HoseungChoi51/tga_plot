```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from matplotlib import font_manager

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/07_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets from the Excel file
xls = pd.ExcelFile(DATA_FILE)

for sheet_name in xls.sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=1)
    
    # Identify columns based on typical TGA structure
    # Assuming column names contain keywords or use positional logic if needed
    # Based on preview: B=Temperature, C=Weight(%), D=D.W(%/C)
    cols = df.columns.tolist()
    
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for c in cols:
        c_lower = str(c).lower()
        if 'temperature' in c_lower or 'temp' in c_lower:
            temp_col = c
        elif 'weight' in c_lower and '%' in c_lower:
            weight_col = c
        elif 'd.w' in c_lower or 'dtg' in c_lower or ('derivative' in c_lower and 'weight' in c_lower):
            dtg_col = c
            
    # Fallback to positional if not found by name (based on preview order)
    if temp_col is None and len(cols) >= 2:
        temp_col = cols[1]
    if weight_col is None and len(cols) >= 3:
        weight_col = cols[2]
    if dtg_col is None and len(cols) >= 4:
        dtg_col = cols[3]
        
    if temp_col is None or weight_col is None or dtg_col is None:
        continue
        
    x = df[temp_col].values
    y1 = df[weight_col].values
    y2 = df[dtg_col].values
    
    # Filter valid data
    mask = ~(np.isnan(x) | np.isnan(y1) | np.isnan(y2))
    x = x[mask]
    y1 = y1[mask]
    y2 = y2[mask]
    
    if len(x) == 0:
        continue

    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot Weight (%) on left axis
    color1 = '#1f4e9c'
    ax1.plot(x, y1, color=color1, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color1)
    
    # Create twin axis for DTG
    ax2 = ax1.twinx()
    color2 = '#c0392b'
    # Multiply DTG by -1 so peaks point up
    y2_inv = -y2
    ax2.plot(x, y2_inv, color=color2, label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color=color2, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color2)
    
    # Title is the sheet name
    ax1.set_title(sheet_name, fontsize=14)
    
    # Find Residue at ~800°C
    idx_800 = np.argmin(np.abs(x - 800))
    residue_val = y1[idx_800]
    residue_temp = x[idx_800]
    
    # Find Peak DTG above 150°C
    mask_peak = x > 150
    if np.any(mask_peak):
        # We want the highest peak of the inverted DTG (which corresponds to the most negative original DTG)
        # Since we plotted -y2, the max value of -y2 is the peak
        y2_inv_masked = y2_inv[mask_peak]
        x_masked = x[mask_peak]
        
        idx_max = np.argmax(y2_inv_masked)
        peak_temp = x_masked[idx_max]
        peak_val_inv = y2_inv_masked[idx_max]
    else:
        peak_temp = None
        peak_val_inv = None
        
    # Add markers and labels
    # Residue marker
    ax1.plot(residue_temp, residue_val, 'o', color=color1, markersize=5)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(residue_temp, residue_val), 
                 xytext=(10, 10), textcoords='offset points',
                 fontsize=10, color=color1,
                 bbox=dict(boxstyle='round,pad=0.3', fc='white', ec='none', alpha=0.8))
                 
    # Peak marker
    if peak_temp is not None:
        ax2.plot(peak_temp, peak_val_inv, 'o', color=color2, markersize=5)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_val_inv), 
                     xytext=(10, 10), textcoords='offset points',
                     fontsize=10, color=color2,
                     bbox=dict(boxstyle='round,pad=0.3', fc='white', ec='none', alpha=0.8))

    # Legend: Move to upper right, set font size to 9
    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    legend = ax1.legend(lines1 + lines2, labels1 + labels2, loc='upper right', fontsize=9)
    
    # Adjust layout
    fig.tight_layout()
    
    # Save figure
    safe_sheet_name = "".join([c if c.isalnum() or c in (' ', '-', '_') else '_' for c in sheet_name])
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    fig.savefig(out_path, dpi=300)
    plt.close(fig)

print("Plots saved successfully.")
```
