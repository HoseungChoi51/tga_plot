```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/24_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read the Excel file
xls = pd.ExcelFile(DATA_FILE)
sheet_name = "CNT-01"
df = pd.read_excel(xls, sheet_name=sheet_name, header=0)

# Identify columns based on preview
# Temperature (°C), Weight (%), Heat Flow (W/g)
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
heat_flow_col = "Heat Flow (W/g)"

# Clean data: drop rows with NaN in essential columns
df_clean = df.dropna(subset=[temp_col, weight_col, heat_flow_col])

# Extract data
temperature = df_clean[temp_col].values
weight_pct = df_clean[weight_col].values
heat_flow = df_clean[heat_flow_col].values

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight (%) on left y-axis (y1)
color1 = '#1f4e9c'
ax1.plot(temperature, weight_pct, color=color1, label='Weight (%)')
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
ax1.tick_params(axis='y', labelcolor=color1)

# Create second y-axis for Heat Flow
ax2 = ax1.twinx()
color2 = '#c0392b'
ax2.plot(temperature, heat_flow, color=color2, label='Heat flow (W/g)')
ax2.set_ylabel('Heat flow (W/g)', color=color2, fontsize=12)
ax2.tick_params(axis='y', labelcolor=color2)

# Title is the sheet name
ax1.set_title(sheet_name, fontsize=14)

# Residue calculation: weight at measured point nearest 800 °C
target_temp = 800.0
idx_residue = np.argmin(np.abs(temperature - target_temp))
residue_weight = weight_pct[idx_residue]
residue_temp = temperature[idx_residue]

# Add residue marker and label on ax1
ax1.scatter(residue_temp, residue_weight, color=color1, zorder=5)
ax1.annotate(f'Residue {residue_weight:.1f}%', 
             xy=(residue_temp, residue_weight), 
             xytext=(10, 10), textcoords='offset points',
             fontsize=10, color=color1,
             arrowprops=dict(arrowstyle='->', color=color1))

# Legend
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left')

# Grid
ax1.grid(True, linestyle='--', alpha=0.5)

# Save plot
output_path = os.path.join(OUT_DIR, f"{sheet_name}.png")
plt.tight_layout()
plt.savefig(output_path, dpi=300)
plt.close()
```
