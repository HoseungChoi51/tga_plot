```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B01_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read data
df = pd.read_csv(DATA_FILE)

# Identify columns based on the preview
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

x = df[temp_col].values
y1 = df[weight_col].values
y2 = df[dtg_col].values * -1  # Invert DTG so peaks point up

# Create figure and axes
fig, ax1 = plt.subplots(figsize=(10, 6))

# Plot Weight (%) on left y-axis
color1 = '#1f4e9c'
ax1.plot(x, y1, color=color1, label='Weight (%)')
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
ax1.tick_params(axis='y', labelcolor=color1)

# Create second y-axis for DTG
ax2 = ax1.twinx()
color2 = '#c0392b'
ax2.plot(x, y2, color=color2, label='DTG (%/°C)')
ax2.set_ylabel('DTG (%/°C)', color=color2, fontsize=12)
ax2.tick_params(axis='y', labelcolor=color2)

# Define regions and colors
regions = [
    {'start': 120, 'end': 250, 'label': 'H2O', 'color': '#a8dadc'},   # Light blue
    {'start': 430, 'end': 580, 'label': 'CO', 'color': '#ffb3ba'},    # Light pink/red
    {'start': 640, 'end': 830, 'label': 'CO2', 'color': '#bae1ff'}    # Light blue-ish
]

# Shade regions and add labels
for region in regions:
    start = region['start']
    end = region['end']
    label = region['label']
    color = region['color']
    
    # Fill between x range across full y height of ax1
    ax1.axvspan(start, end, alpha=0.3, color=color)
    
    # Add text label above the shaded region
    # Find a good y position for the text (top part of the plot area)
    y_min, y_max = ax1.get_ylim()
    y_text = y_max - 0.05 * (y_max - y_min)
    
    # Center the text horizontally in the region
    x_center = (start + end) / 2
    
    ax1.text(x_center, y_text, label, ha='center', va='bottom', fontsize=12, fontweight='bold')

# Calculate Residue at ~800°C
residue_temp = 800
idx_residue = np.argmin(np.abs(x - residue_temp))
residue_val = y1[idx_residue]
residue_x = x[idx_residue]

# Mark residue on weight curve
ax1.plot(residue_x, residue_val, 'o', color=color1, markersize=8)
ax1.annotate(f'Residue {residue_val:.1f}%', xy=(residue_x, residue_val), 
             xytext=(residue_x + 50, residue_val + 5),
             arrowprops=dict(facecolor='black', shrink=0.05),
             fontsize=10, color=color1)

# Calculate Peak temperature (>150°C)
mask_peak = x > 150
if np.any(mask_peak):
    y2_masked = y2[mask_peak]
    x_masked = x[mask_peak]
    idx_peak_local = np.argmax(y2_masked)
    peak_x = x_masked[idx_peak_local]
    peak_y = y2_masked[idx_peak_local]
    
    # Map back to original indices if needed, but we can just use masked values for plotting marker on ax2
    # We need the corresponding y1 value? No, marker is on DTG curve usually or just temp label.
    # The prompt says "with a marker". Usually on the DTG peak.
    ax2.plot(peak_x, peak_y, 'o', color=color2, markersize=8)
    ax2.annotate(f'{peak_x:.0f} °C', xy=(peak_x, peak_y), 
                 xytext=(peak_x + 50, peak_y + 0.05),
                 arrowprops=dict(facecolor='black', shrink=0.05),
                 fontsize=10, color=color2)

# Legend
lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()
ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center left', bbox_to_anchor=(0.02, 0.5))

# Title
sheet_name = os.path.splitext(os.path.basename(DATA_FILE))[0]
plt.title(sheet_name)

# Grid
ax1.grid(True, linestyle='--', alpha=0.5)

plt.tight_layout()

# Save
output_path = os.path.join(OUT_DIR, f"{sheet_name}.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
