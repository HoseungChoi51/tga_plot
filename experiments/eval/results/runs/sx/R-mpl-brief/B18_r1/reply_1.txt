import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import find_peaks
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B18_r1/out"
SHEET_NAME = "Annealed"

def calculate_onset_temperature(temp, dtg):
    """
    Calculate the onset temperature using the tangent method on the DTG curve.
    1. Find the main peak in DTG (above 150°C).
    2. Find the inflection point before the peak (max slope of DTG before peak).
    3. Draw a tangent at the inflection point and extrapolate to the baseline.
    The intersection with the baseline is the onset temperature.
    """
    # Work with -DTG so peaks are positive
    neg_dtg = -dtg
    
    # Filter for T > 150 to ignore low-temp noise/moisture if any
    mask = temp > 150
    t_filt = temp[mask]
    ndtg_filt = neg_dtg[mask]
    
    if len(t_filt) < 5:
        return None, None, None, None, None

    # Find the highest peak
    peaks, _ = find_peaks(ndtg_filt, height=np.max(ndtg_filt)*0.5)
    if len(peaks) == 0:
        # Fallback: just take the max
        peak_idx = np.argmax(ndtg_filt)
    else:
        # Take the highest peak among found peaks
        peak_heights = ndtg_filt[peaks]
        peak_idx = peaks[np.argmax(peak_heights)]
        
    peak_temp = t_filt[peak_idx]
    
    # Find inflection point before the peak
    # Inflection point is where the derivative of DTG (second derivative of Weight) is zero/max slope
    # We look at the region from start up to the peak
    search_range = slice(0, peak_idx)
    t_before_peak = t_filt[search_range]
    ndtg_before_peak = ndtg_filt[search_range]
    
    if len(t_before_peak) < 3:
        return peak_temp, None, None, None, None
        
    # Calculate numerical derivative of DTG (slope of DTG)
    # Using gradient which handles uneven spacing reasonably well
    d_ndtg = np.gradient(ndtg_before_peak, t_before_peak)
    
    # The onset is associated with the steepest rise in mass loss rate (highest positive slope of -DTG)
    # This corresponds to the maximum value of d_ndtg before the peak
    inflection_idx_local = np.argmax(d_ndtg)
    
    # Ensure we don't pick an artifact at the very beginning if possible
    # But usually, the steepest part is near the onset.
    
    t_inflection = t_before_peak[inflection_idx_local]
    val_inflection = ndtg_before_peak[inflection_idx_local]
    slope_inflection = d_ndtg[inflection_idx_local]
    
    # Define baseline: average of -DTG values before significant decomposition starts
    # Let's take the first 10% of the filtered data or values close to 0
    baseline_vals = ndtg_filt[:int(len(ndtg_filt)*0.1)]
    baseline_val = np.mean(baseline_vals) if len(baseline_vals) > 0 else 0.0
    
    # Equation of tangent line: y - y1 = m(x - x1)
    # y = slope * (x - t_inflection) + val_inflection
    # We want x where y = baseline_val
    # baseline_val = slope * (x_onset - t_inflection) + val_inflection
    # x_onset = (baseline_val - val_inflection) / slope + t_inflection
    
    if abs(slope_inflection) < 1e-9:
        onset_temp = t_inflection
    else:
        onset_temp = (baseline_val - val_inflection) / slope_inflection + t_inflection
        
    return onset_temp, peak_temp, t_inflection, val_inflection, slope_inflection

def plot_tga_sheet(df, sheet_name, out_dir):
    # Extract columns based on standard naming conventions seen in preview
    try:
        temp_col = [c for c in df.columns if 'Temperature' in c][0]
        weight_col = [c for c in df.columns if 'Weight (%)' in c][0]
        dtg_col = [c for c in df.columns if 'Deriv. Weight' in c][0]
    except IndexError:
        print(f"Could not identify columns in sheet {sheet_name}")
        return

    temp = df[temp_col].values
    weight = df[weight_col].values
    dtg = df[dtg_col].values
    
    # Plotting
    fig, ax1 = plt.subplots(figsize=(10, 6))
    
    color_y1 = '#1f4e9c'
    color_y2 = '#c0392b'
    
    ax1.plot(temp, weight, color=color_y1, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color_y1, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color_y1)
    
    ax2 = ax1.twinx()
    # Multiply DTG by -1 to make peaks point up
    ax2.plot(temp, -dtg, color=color_y2, linestyle='--', label='DTG (-%)')
    ax2.set_ylabel('DTG (%/°C)', color=color_y2, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color_y2)
    
    # Title
    ax1.set_title(sheet_name, fontsize=14, fontweight='bold')
    
    # Legend
    lines_1, labels_1 = ax1.get_legend_handles_labels()
    lines_2, labels_2 = ax2.get_legend_handles_labels()
    ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center left')
    
    # Grid
    ax1.grid(True, linestyle=':', alpha=0.6)
    
    # --- Residue Label ---
    # Find weight nearest 800 C
    idx_res = np.argmin(np.abs(temp - 800))
    res_weight = weight[idx_res]
    res_temp = temp[idx_res]
    
    ax1.plot(res_temp, res_weight, 'o', color=color_y1, markersize=8)
    ax1.annotate(f'Residue {res_weight:.1f}%', 
                 xy=(res_temp, res_weight), 
                 xytext=(res_temp - 100, res_weight + 5),
                 arrowprops=dict(facecolor='black', arrowstyle='->'),
                 fontsize=10, color=color_y1)
                 
    # --- Peak Label ---
    # Find highest DTG peak above 150 C
    mask_peak = temp > 150
    if np.any(mask_peak):
        neg_dtg_masked = -dtg[mask_peak]
        temp_masked = temp[mask_peak]
        peak_idx = np.argmax(neg_dtg_masked)
        peak_temp = temp_masked[peak_idx]
        peak_val = neg_dtg_masked[peak_idx]
        
        ax2.plot(peak_temp, peak_val, 'v', color=color_y2, markersize=8)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_val), 
                     xytext=(peak_temp + 20, peak_val + 0.1),
                     arrowprops=dict(facecolor='black', arrowstyle='->'),
                     fontsize=10, color=color_y2)
                     
    # --- Onset Temperature Calculation and Display ---
    onset_temp, p_temp, t_infl, v_infl, slope = calculate_onset_temperature(temp, dtg)
    
    if onset_temp is not None:
        # Mark the onset temperature on the graph
        # Usually marked on the weight curve or as a vertical line
        # Finding the weight at onset temp for marker position
        idx_onset = np.argmin(np.abs(temp - onset_temp))
        w_at_onset = weight[idx_onset]
        
        # Draw vertical dashed line at onset
        ax1.axvline(x=onset_temp, color='green', linestyle='--', linewidth=1.5, label=f'Onset: {onset_temp:.1f} °C')
        
        # Add text annotation
        ax1.text(onset_temp + 10, w_at_onset, f'Onset\n{onset_temp:.1f} °C', 
                 color='green', fontsize=10, va='bottom')
        
        # Update legend to include onset line if desired, or keep separate
        # Here we added it via axvline label, need to update legend
        lines_1, labels_1 = ax1.get_legend_handles_labels()
        lines_2, labels_2 = ax2.get_legend_handles_labels()
        ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center left')

    # Save figure
    filename = f"{sheet_name}_tga.png"
    filepath = os.path.join(out_dir, filename)
    plt.tight_layout()
    plt.savefig(filepath, dpi=300)
    plt.close()
    print(f"Saved plot to {filepath}")

# Main execution
if __name__ == "__main__":
    if not os.path.exists(OUT_DIR):
        os.makedirs(OUT_DIR)
        
    xls = pd.ExcelFile(DATA_FILE)
    
    # Only process the requested sheet
    if SHEET_NAME in xls.sheet_names:
        df = pd.read_excel(xls, sheet_name=SHEET_NAME)
        plot_tga_sheet(df, SHEET_NAME, OUT_DIR)
    else:
        print(f"Sheet '{SHEET_NAME}' not found in file.")
