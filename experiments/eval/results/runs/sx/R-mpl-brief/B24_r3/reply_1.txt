```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import find_peaks

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B24_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read the PE sheet
df = pd.read_excel(DATA_FILE, sheet_name="PE")

# Extract columns based on preview
temp_c = df["Temperature (°C)"].values
weight = df["Weight (%)"].values
dtg = df["Deriv. Weight (%/°C)"].values

# Convert temperature to Kelvin
temp_k = temp_c + 273.15

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight (%) on left y-axis
line1 = ax1.plot(temp_k, weight, color='#1f4e9c', label='Weight (%)')
ax1.set_xlabel('Temperature (K)', fontsize=12)
ax1.set_ylabel('Weight (%)', color='#1f4e9c', fontsize=12)
ax1.tick_params(axis='y', labelcolor='#1f4e9c')

# Create second y-axis for DTG
ax2 = ax1.twinx()
# Multiply DTG by -1 so peaks point up
dtg_inverted = dtg * -1
line2 = ax2.plot(temp_k, dtg_inverted, color='#c0392b', label='DTG (%/°C)')
ax2.set_ylabel('DTG (%/°C)', color='#c0392b', fontsize=12)
ax2.tick_params(axis='y', labelcolor='#c0392b')

# Find residue at nearest 800 °C -> convert to K: 800 + 273.15 = 1073.15 K
target_temp_k = 800 + 273.15
idx_residue = np.argmin(np.abs(temp_k - target_temp_k))
residue_val = weight[idx_residue]
residue_temp_k = temp_k[idx_residue]

# Add residue marker and label
ax1.plot(residue_temp_k, residue_val, 'o', color='#1f4e9c', markersize=8)
ax1.annotate(f'Residue {residue_val:.1f}%', 
             xy=(residue_temp_k, residue_val), 
             xytext=(residue_temp_k - 100, residue_val + 5),
             arrowprops=dict(arrowstyle='->', color='#1f4e9c'),
             fontsize=10, color='#1f4e9c')

# Find peak above 150 °C -> 150 + 273.15 = 423.15 K
target_peak_k = 150 + 273.15
mask = temp_k > target_peak_k
if np.any(mask):
    # Find index of max inverted DTG in the masked region
    dtg_masked = dtg_inverted[mask]
    temp_masked = temp_k[mask]
    
    # Use simple argmax for highest peak
    idx_peak_local = np.argmax(dtg_masked)
    peak_temp_k = temp_masked[idx_peak_local]
    peak_dtg_val = dtg_masked[idx_peak_local]
    
    # Add peak marker and label
    ax2.plot(peak_temp_k, peak_dtg_val, 'o', color='#c0392b', markersize=8)
    ax2.annotate(f'{peak_temp_k:.0f} K', 
                 xy=(peak_temp_k, peak_dtg_val), 
                 xytext=(peak_temp_k + 50, peak_dtg_val + 0.2),
                 arrowprops=dict(arrowstyle='->', color='#c0392b'),
                 fontsize=10, color='#c0392b')

# Title is sheet name
ax1.set_title("PE", fontsize=14, fontweight='bold')

# Legend at center left
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left', bbox_to_anchor=(0.02, 0.5))

plt.tight_layout()

# Save figure
out_path = os.path.join(OUT_DIR, "PE_K.png")
plt.savefig(out_path, dpi=300, bbox_inches='tight')
plt.close()
```
