```python
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B21_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read the "Raw" sheet
df = pd.read_excel(DATA_FILE, sheet_name="Raw")

# Extract columns based on the preview
temp = df["Temperature (°C)"].values
weight = df["Weight (%)"].values
dtg = df["Deriv. Weight (%/°C)"].values

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight (%) on left y-axis
color1 = '#1f4e9c'
ax1.plot(temp, weight, color=color1, label='Weight (%)')
ax1.set_xlabel('Temperature (°C)')
ax1.set_ylabel('Weight (%)', color=color1)
ax1.tick_params(axis='y', labelcolor=color1)

# Invert x-axis so higher temperature is on the left
ax1.invert_xaxis()

# Create second y-axis for DTG
ax2 = ax1.twinx()
color2 = '#c0392b'
# Multiply DTG by -1 to make peaks point up
ax2.plot(temp, dtg * -1, color=color2, linestyle='--', label='DTG (%/°C)')
ax2.set_ylabel('DTG (%/°C)', color=color2)
ax2.tick_params(axis='y', labelcolor=color2)

# Find residue at ~800 °C
idx_800 = np.argmin(np.abs(temp - 800))
residue_val = weight[idx_800]
residue_temp = temp[idx_800]

# Find peak of DTG above 150 °C
mask = temp > 150
if mask.any():
    dtg_masked = dtg[mask]
    temp_masked = temp[mask]
    # Peak is where DTG is most negative (so -DTG is max)
    idx_peak_local = np.argmax(dtg_masked * -1)
    peak_temp = temp_masked[idx_peak_local]
else:
    peak_temp = None

# Add markers and labels for residue
ax1.scatter(residue_temp, residue_val, color=color1, zorder=5)
ax1.annotate(f'Residue {residue_val:.1f}%', 
             xy=(residue_temp, residue_val), 
             xytext=(residue_temp + 50, residue_val + 5),
             arrowprops=dict(arrowstyle='->', color=color1),
             fontsize=10, color=color1)

# Add markers and labels for peak
if peak_temp is not None:
    # Get the corresponding -DTG value at peak_temp
    idx_peak_full = np.argmin(np.abs(temp - peak_temp))
    peak_dtg_val = dtg[idx_peak_full] * -1
    
    ax2.scatter(peak_temp, peak_dtg_val, color=color2, zorder=5)
    ax2.annotate(f'{peak_temp:.0f} °C', 
                 xy=(peak_temp, peak_dtg_val), 
                 xytext=(peak_temp - 50, peak_dtg_val + 0.1),
                 arrowprops=dict(arrowstyle='->', color=color2),
                 fontsize=10, color=color2)

# Title
plt.title('Raw')

# Legend
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left')

# Grid
ax1.grid(True, linestyle=':', alpha=0.5)

# Save figure
out_path = os.path.join(OUT_DIR, "Raw.png")
plt.savefig(out_path, dpi=300, bbox_inches='tight')
plt.close()
```
