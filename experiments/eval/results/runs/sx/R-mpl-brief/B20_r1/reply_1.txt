```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.ticker import MultipleLocator, AutoMinorLocator

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B20_r1/out"
SHEET_NAME = "Acid-washed"

os.makedirs(OUT_DIR, exist_ok=True)

# Read data
df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)

# Identify columns based on the preview
# Assuming standard TGA structure: Time, Temperature, Weight, Deriv. Weight
# We need Temperature (x), Weight (y1), Deriv. Weight (y2)
cols = df.columns.tolist()

# Find temperature column
temp_col = None
for c in cols:
    if 'Temperature' in str(c):
        temp_col = c
        break

# Find weight column
weight_col = None
for c in cols:
    if 'Weight (%)' in str(c):
        weight_col = c
        break

# Find derivative weight column
dtg_col = None
for c in cols:
    if 'Deriv.' in str(c) or 'DTG' in str(c):
        dtg_col = c
        break

if not all([temp_col, weight_col, dtg_col]):
    # Fallback to positional if names don't match exactly but structure is known from preview
    # Preview says: A=Time, B=Temp, C=Weight, D=Deriv
    temp_col = cols[1]
    weight_col = cols[2]
    dtg_col = cols[3]

# Extract data
temperature = df[temp_col].values
weight = df[weight_col].values
dtg = df[dtg_col].values

# Plotting
fig, ax1 = plt.subplots(figsize=(8, 6))

# Y1 axis: Weight (%)
color_y1 = '#1f4e9c'
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color_y1, fontsize=12)
line1, = ax1.plot(temperature, weight, color=color_y1, label='Weight')
ax1.tick_params(axis='y', labelcolor=color_y1)

# Add minor ticks to X-axis (user request)
ax1.xaxis.set_minor_locator(AutoMinorLocator())
ax1.tick_params(which='minor', direction='in', length=4)
ax1.tick_params(which='major', direction='in', length=6)

# Y2 axis: DTG (%/°C), inverted so peaks point up
ax2 = ax1.twinx()
color_y2 = '#c0392b'
ax2.set_ylabel('DTG (%/°C)', color=color_y2, fontsize=12)
# Multiply DTG by -1 so peaks point up
line2, = ax2.plot(temperature, -dtg, color=color_y2, label='DTG')
ax2.tick_params(axis='y', labelcolor=color_y2)

# Title
ax1.set_title(SHEET_NAME, fontsize=14)

# Legend
lines = [line1, line2]
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='center left')

# Calculate Residue at ~800°C
idx_800 = np.argmin(np.abs(temperature - 800))
residue_val = weight[idx_800]
temp_at_residue = temperature[idx_800]
ax1.plot(temp_at_residue, residue_val, 'o', color=color_y1, markersize=5)
ax1.annotate(f'Residue {residue_val:.1f}%', 
             xy=(temp_at_residue, residue_val), 
             xytext=(10, 10), textcoords='offset points',
             arrowprops=dict(arrowstyle='->', color=color_y1),
             color=color_y1, fontsize=10)

# Calculate Peak above 150°C
mask_peak = temperature > 150
if np.any(mask_peak):
    dtg_positive = -dtg[mask_peak] # Since we plotted -dtg, peak is max value
    temps_peak_range = temperature[mask_peak]
    
    idx_max_dtg_local = np.argmax(dtg_positive)
    peak_temp = temps_peak_range[idx_max_dtg_local]
    peak_dtg_val = dtg_positive[idx_max_dtg_local]
    
    # Find corresponding index in original array for plotting marker on y2
    idx_original = np.where((temperature == peak_temp))[0][0]
    
    ax2.plot(peak_temp, peak_dtg_val, 'o', color=color_y2, markersize=5)
    ax2.annotate(f'{peak_temp:.0f} °C', 
                 xy=(peak_temp, peak_dtg_val), 
                 xytext=(-10, 10), textcoords='offset points',
                 arrowprops=dict(arrowstyle='->', color=color_y2),
                 color=color_y2, fontsize=10)

plt.tight_layout()

out_path = os.path.join(OUT_DIR, f"{SHEET_NAME}.png")
plt.savefig(out_path, dpi=300)
plt.close()
```
