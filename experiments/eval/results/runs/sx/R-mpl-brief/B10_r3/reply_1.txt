```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B10_r3/out"
SHEET_NAME = "Annealed"

def plot_tga_black_white():
    # Ensure output directory exists
    os.makedirs(OUT_DIR, exist_ok=True)
    
    # Read the specific sheet
    df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
    
    # Extract columns based on standard TGA structure
    temp = df["Temperature (°C)"]
    weight = df["Weight (%)"]
    dtg = df["Deriv. Weight (%/°C)"]
    
    # Create figure and axis
    fig, ax1 = plt.subplots(figsize=(3.5, 2.6))
    
    # Plot Weight (left y-axis) in black
    line1, = ax1.plot(temp, weight, color='black', linewidth=1.2, label='Weight')
    ax1.set_xlabel('Temperature (°C)', fontsize=8)
    ax1.set_ylabel('Weight (%)', fontsize=8)
    ax1.tick_params(axis='both', labelsize=8)
    
    # Create right y-axis for DTG
    ax2 = ax1.twinx()
    # Multiply DTG by -1 so peaks point up
    dtg_inv = dtg * -1
    line2, = ax2.plot(temp, dtg_inv, color='black', linestyle='--', linewidth=1.2, label='DTG')
    ax2.set_ylabel('DTG (%/°C)', fontsize=8)
    ax2.tick_params(axis='y', labelsize=8)
    
    # Title
    ax1.set_title(SHEET_NAME, fontsize=8)
    
    # Calculate Residue at ~800°C
    idx_res = np.argmin(np.abs(temp - 800))
    residue_val = weight.iloc[idx_res]
    residue_temp = temp.iloc[idx_res]
    
    # Calculate Peak temperature (highest DTG peak above 150°C)
    mask_peak = (temp > 150) & (dtg_inv > 0)
    if np.any(mask_peak):
        dtg_filtered = dtg_inv[mask_peak]
        temp_filtered = temp[mask_peak]
        idx_peak_local = np.argmax(dtg_filtered)
        peak_temp = temp_filtered.iloc[idx_peak_local]
        peak_val = dtg_filtered.iloc[idx_peak_local]
        
        # Find global index in original arrays for plotting marker
        # We need to find the index in the full dataframe that matches this local max
        # A robust way: find the index where temp == peak_temp and dtg_inv is max among those? 
        # Simpler: just use the filtered indices mapped back or search again
        # Let's search in the original data for the max DTG inv above 150
        idx_global_peak = np.where((temp > 150))[0][np.argmax(dtg_inv[temp > 150])]
        peak_temp_final = temp.iloc[idx_global_peak]
        peak_dtg_final = dtg_inv.iloc[idx_global_peak]
        
        # Markers
        ax1.scatter(residue_temp, residue_val, color='black', s=10, zorder=5)
        ax2.scatter(peak_temp_final, peak_dtg_final, color='black', s=10, marker='v', zorder=5)
        
        # Labels
        # Position residue label slightly offset
        ax1.annotate(f'Residue {residue_val:.1f}%', 
                     xy=(residue_temp, residue_val), 
                     xytext=(residue_temp + 50, residue_val + 5),
                     fontsize=8, ha='left', va='bottom',
                     arrowprops=dict(arrowstyle='->', color='black', lw=0.8))
                     
        # Position peak label slightly offset
        ax2.annotate(f'{peak_temp_final:.0f} °C', 
                     xy=(peak_temp_final, peak_dtg_final), 
                     xytext=(peak_temp_final - 100, peak_dtg_final + 0.1),
                     fontsize=8, ha='right', va='bottom',
                     arrowprops=dict(arrowstyle='->', color='black', lw=0.8))
    else:
        # Fallback if no peak found
        ax1.scatter(residue_temp, residue_val, color='black', s=10, zorder=5)
        ax1.annotate(f'Residue {residue_val:.1f}%', 
                     xy=(residue_temp, residue_val), 
                     xytext=(residue_temp + 50, residue_val + 5),
                     fontsize=8, ha='left', va='bottom',
                     arrowprops=dict(arrowstyle='->', color='black', lw=0.8))

    # Legend
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='center left', fontsize=8, frameon=False)
    
    # Grid
    ax1.grid(True, linestyle=':', alpha=0.6)
    
    # Save
    out_path = os.path.join(OUT_DIR, f"{SHEET_NAME}_bw.png")
    plt.savefig(out_path, dpi=600, bbox_inches='tight')
    plt.close()

if __name__ == "__main__":
    plot_tga_black_white()
```
