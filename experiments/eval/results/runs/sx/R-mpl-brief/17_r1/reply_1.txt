```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os
from pathlib import Path

# Configure Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/17_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

def plot_tga_sheet(sheet_name, df):
    # Identify columns based on standard TGA structure or the guess provided in preview logic
    # Assuming standard: Time, Temperature, Weight(%), D.W(%/C)
    cols = df.columns
    
    # Heuristic to find x, y1, y2 if names aren't exact matches to known patterns
    x_col = None
    y1_col = None
    y2_col = None
    
    for c in cols:
        c_lower = str(c).lower()
        if 'temperature' in c_lower or 'temp' in c_lower:
            x_col = c
        elif 'weight' in c_lower and '%' in str(c):
            y1_col = c
        elif 'd.w' in c_lower or 'derivative' in c_lower or 'dtg' in c_lower:
            y2_col = c
            
    # Fallback if specific names not found (based on typical order Time, Temp, Weight, DTG)
    if x_col is None and len(cols) > 1: x_col = cols[1]
    if y1_col is None and len(cols) > 2: y1_col = cols[2]
    if y2_col is None and len(cols) > 3: y2_col = cols[3]
    
    if x_col is None or y1_col is None or y2_col is None:
        return

    data_x = df[x_col].values
    data_y1 = df[y1_col].values
    data_y2 = df[y2_col].values
    
    # Create figure
    fig, ax1 = plt.subplots(figsize=(8, 5))
    
    # Plot Weight (%) on left axis
    color1 = '#1f4e9c'
    ax1.plot(data_x, data_y1, color=color1, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=10)
    ax1.set_ylabel('Weight (%)', color=color1, fontsize=10)
    ax1.tick_params(axis='y', labelcolor=color1)
    ax1.grid(True, linestyle='--', alpha=0.6)
    
    # Create twin axis for DTG
    ax2 = ax1.twinx()
    color2 = '#c0392b'
    # Multiply by -1 so peaks point up
    dtg_plot = data_y2 * -1
    ax2.plot(data_x, dtg_plot, color=color2, label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color=color2, fontsize=10)
    ax2.tick_params(axis='y', labelcolor=color2)
    
    # Title from sheet name
    ax1.set_title(sheet_name, fontsize=12)
    
    # Legend
    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left', bbox_to_anchor=(0.02, 0.5), frameon=False)
    
    # Calculate Residue at nearest 800°C
    # Find index closest to 800
    idx_800 = np.argmin(np.abs(data_x - 800))
    residue_val = data_y1[idx_800]
    temp_800 = data_x[idx_800]
    
    # Add residue marker and text
    ax1.plot(temp_800, residue_val, 'o', color=color1, markersize=5)
    ax1.text(temp_800, residue_val, f'Residue {residue_val:.1f}%', 
             verticalalignment='bottom', horizontalalignment='right', 
             color=color1, fontsize=9, fontweight='bold')
             
    # Calculate Peak temperature of highest DTG peak above 150°C
    mask_temp = data_x > 150
    if np.any(mask_temp):
        # Filter data above 150C
        x_filtered = data_x[mask_temp]
        dtg_filtered = dtg_plot[mask_temp] # Already multiplied by -1
        
        # Find max DTG (which corresponds to most negative original value, i.e., steepest loss)
        # Since we plotted -DTG, the "highest" peak visually is the maximum of this array
        idx_max_dtg = np.argmax(dtg_filtered)
        peak_temp = x_filtered[idx_max_dtg]
        
        # Get corresponding weight at this temperature for context if needed, but user asked for temp label
        # Find index in original arrays to place marker correctly relative to full curve
        idx_peak_full = np.where(data_x == peak_temp)[0][0]
        
        # Marker on DTG curve (ax2)
        ax2.plot(peak_temp, dtg_plot[idx_peak_full], 'v', color=color2, markersize=5)
        ax2.text(peak_temp, dtg_plot[idx_peak_full], f'{peak_temp:.0f} °C', 
                 verticalalignment='top', horizontalalignment='center', 
                 color=color2, fontsize=9, fontweight='bold')
    
    plt.tight_layout()
    
    # Save file
    safe_name = "".join([c if c.isalnum() or c in (' ', '-', '_') else '_' for c in sheet_name]).strip().replace(' ', '_')
    out_path = os.path.join(OUT_DIR, f"{safe_name}.png")
    plt.savefig(out_path, dpi=600)
    plt.close(fig)

# Load Excel file
xls = pd.ExcelFile(DATA_FILE)

for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name)
    plot_tga_sheet(sheet_name, df)
```
