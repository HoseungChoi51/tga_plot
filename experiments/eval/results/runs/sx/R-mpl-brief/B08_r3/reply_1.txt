```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B08_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read the specific sheet requested: "Raw"
df = pd.read_excel(DATA_FILE, sheet_name="Raw")

# Extract columns based on standard TGA structure and preview info
# B: Temperature (°C), C: Weight (%), D: Deriv. Weight (%/°C)
temp_col = df.columns[1]  # Temperature
weight_col = df.columns[2]  # Weight (%)
dtg_col = df.columns[3]   # Deriv. Weight (%/°C)

x = df[temp_col].values
y1 = df[weight_col].values
y2 = df[dtg_col].values

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight (left y-axis)
color1 = '#1f4e9c'
ax1.plot(x, y1, color=color1, label='Weight (%)')
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
ax1.tick_params(axis='y', labelcolor=color1)

# Plot DTG (right y-axis) - multiply by -1 so peaks point up
ax2 = ax1.twinx()
color2 = '#c0392b'
ax2.plot(x, -y2, color=color2, label='DTG (-dW/dT)')
ax2.set_ylabel('DTG (%/°C)', color=color2, fontsize=12)
ax2.tick_params(axis='y', labelcolor=color2)

# Add horizontal reference line at Weight = 50%
ax1.axhline(y=50, color='gray', linestyle='--', linewidth=1, label='Ref: 50% Weight')

# Find Residue at ~800°C for label
idx_800 = np.argmin(np.abs(x - 800))
residue_val = y1[idx_800]
ax1.plot(x[idx_800], residue_val, 'o', color=color1, markersize=5)
ax1.annotate(f'Residue {residue_val:.1f}%', xy=(x[idx_800], residue_val), 
             xytext=(x[idx_800]+20, residue_val+5),
             arrowprops=dict(facecolor=color1, shrink=0.05), fontsize=10, color=color1)

# Find Peak temperature > 150°C for label
mask_peak = x > 150
if np.any(mask_peak):
    # DTG is negative, so we look for minimum value in original data (which becomes max positive when negated)
    # Or simply find min of y2 (most negative) where x > 150
    dtg_vals_masked = y2[mask_peak]
    temp_vals_masked = x[mask_peak]
    
    idx_min_dtg = np.argmin(dtg_vals_masked) # Most negative DTG
    peak_temp = temp_vals_masked[idx_min_dtg]
    peak_dtg_val = dtg_vals_masked[idx_min_dtg]
    
    # Marker on the DTG curve (using ax2 coordinates, which are -y2)
    ax2.plot(peak_temp, -peak_dtg_val, 'o', color=color2, markersize=5)
    ax2.annotate(f'{peak_temp:.0f} °C', xy=(peak_temp, -peak_dtg_val),
                 xytext=(peak_temp-50, -peak_dtg_val + 0.1),
                 arrowprops=dict(facecolor=color2, shrink=0.05), fontsize=10, color=color2)

# Title
title_text = "Raw"
ax1.set_title(title_text, fontsize=14)

# Legend
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left')

# Add memo text in top right corner
memo_text = '공기, 10 °C/min'
ax1.text(0.98, 0.95, memo_text, transform=ax1.transAxes, 
         ha='right', va='top', fontsize=12, bbox=dict(boxstyle='round,pad=0.3', facecolor='yellow', alpha=0.5))

# Layout adjustment
fig.tight_layout()

# Save plot
out_path = os.path.join(OUT_DIR, f"{title_text}.png")
plt.savefig(out_path, dpi=300)
plt.close()
```
