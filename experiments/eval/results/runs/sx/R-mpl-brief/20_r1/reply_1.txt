```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.font_manager as fm

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/20_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

for sheet_name in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)
    
    # Identify columns based on the preview structure
    # Col A: Time, Col B: Temperature, Col C: Weight(%), Col D: D.W(%/C)
    x_col = None
    y1_col = None
    y2_col = None
    
    cols = df.columns.tolist()
    
    for c in cols:
        c_lower = str(c).lower()
        if 'temp' in c_lower:
            x_col = c
        elif 'weight' in c_lower and 'd.w' not in c_lower and 'dtg' not in c_lower:
            y1_col = c
        elif 'd.w' in c_lower or 'dtg' in c_lower or 'derivative' in c_lower:
            y2_col = c
            
    if x_col is None or y1_col is None or y2_col is None:
        continue
        
    temp = df[x_col].values
    weight = df[y1_col].values
    dtg = df[y2_col].values
    
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot Weight (%) on left axis
    color_w = '#1f4e9c'
    ax1.plot(temp, weight, color=color_w, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)')
    ax1.set_ylabel('Weight (%)', color=color_w)
    ax1.tick_params(axis='y', labelcolor=color_w)
    
    # Create twin axis for DTG
    ax2 = ax1.twinx()
    color_dtg = '#c0392b'
    # Multiply by -1 so peaks point up
    ax2.plot(temp, dtg * -1, color=color_dtg, label='DTG (-D.W)')
    ax2.set_ylabel('DTG (%/°C)', color=color_dtg)
    ax2.tick_params(axis='y', labelcolor=color_dtg)
    
    # Shade region 300-450 °C with light gray
    ax1.axvspan(300, 450, alpha=0.3, color='gray', label='300~450 °C')
    
    # Find Residue at nearest 800 °C
    mask_800 = ~np.isnan(temp) & ~np.isnan(weight)
    t_clean = temp[mask_800]
    w_clean = weight[mask_800]
    
    residue_val = None
    residue_t = None
    if len(t_clean) > 0:
        idx_800 = np.argmin(np.abs(t_clean - 800))
        residue_t = t_clean[idx_800]
        residue_val = w_clean[idx_800]
        
        # Marker for residue
        ax1.plot(residue_t, residue_val, 'o', color=color_w, markersize=5)
        ax1.annotate(f'Residue {residue_val:.1f}%', 
                     xy=(residue_t, residue_val), 
                     xytext=(residue_t + 20, residue_val + 5),
                     fontsize=9, color=color_w,
                     arrowprops=dict(facecolor=color_w, shrink=0.05, width=1, headwidth=5))
                     
    # Find Peak above 150 °C
    mask_dtg = ~np.isnan(temp) & ~np.isnan(dtg)
    t_dtg = temp[mask_dtg]
    d_dtg = dtg[mask_dtg]
    
    peak_t = None
    peak_val = None
    if len(t_dtg) > 0:
        # Filter for T > 150
        mask_gt_150 = t_dtg > 150
        if np.any(mask_gt_150):
            t_peak_search = t_dtg[mask_gt_150]
            d_peak_search = d_dtg[mask_gt_150]
            
            # Highest DTG peak means most negative value in original data, 
            # or highest value in inverted data.
            # We want the temperature of the highest peak magnitude.
            # Since we plot -DTG, we look for max(-DTG) which is min(DTG).
            idx_max_peak = np.argmin(d_peak_search)
            peak_t = t_peak_search[idx_max_peak]
            peak_val = d_peak_search[idx_max_peak] # Original DTG value
            
            # Marker for peak on DTG curve (right axis)
            # Note: ax2 plots -dtg, so the visual peak height is -peak_val
            ax2.plot(peak_t, -peak_val, 'o', color=color_dtg, markersize=5)
            ax2.annotate(f'{peak_t:.0f} °C', 
                         xy=(peak_t, -peak_val), 
                         xytext=(peak_t + 20, -peak_val + 0.1),
                         fontsize=9, color=color_dtg,
                         arrowprops=dict(facecolor=color_dtg, shrink=0.05, width=1, headwidth=5))

    # Title
    ax1.set_title(sheet_name)
    
    # Legend
    lines_1, labels_1 = ax1.get_legend_handles_labels()
    lines_2, labels_2 = ax2.get_legend_handles_labels()
    ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center left')
    
    # Grid
    ax1.grid(True, linestyle='--', alpha=0.5)
    
    plt.tight_layout()
    
    safe_sheet_name = "".join([i if i.isalnum() else "_" for i in sheet_name])
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    plt.savefig(out_path, dpi=300)
    plt.close(fig)
```
