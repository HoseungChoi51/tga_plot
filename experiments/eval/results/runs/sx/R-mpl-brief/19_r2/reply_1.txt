```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.font_manager as fm

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/19_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Load all sheets from the Excel file
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

for sheet_name in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)
    
    # Identify columns based on the preview structure
    # Assuming standard TGA format: Time, Temperature, Weight(%), D.W(%/C)
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        col_lower = str(col).lower()
        if 'temperature' in col_lower or 'temp' in col_lower:
            temp_col = col
        elif 'weight' in col_lower and '%' in col_lower:
            weight_col = col
        elif 'd.w' in col_lower or 'dtg' in col_lower or 'derivative' in col_lower:
            dtg_col = col
            
    # Fallback to positional if names are not clear (based on preview: B=Temp, C=Weight, D=DTG)
    if temp_col is None:
        temp_col = df.columns[1]
    if weight_col is None:
        weight_col = df.columns[2]
    if dtg_col is None:
        dtg_col = df.columns[3]
        
    x_data = df[temp_col].values
    y1_data = df[weight_col].values
    y2_data = df[dtg_col].values
    
    # Filter out NaNs
    mask = ~(np.isnan(x_data) | np.isnan(y1_data) | np.isnan(y2_data))
    x_data = x_data[mask]
    y1_data = y1_data[mask]
    y2_data = y2_data[mask]
    
    if len(x_data) == 0:
        continue
        
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot Weight (%) on left axis
    color_y1 = '#1f4e9c'
    line1, = ax1.plot(x_data, y1_data, color=color_y1, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=13, color='gray')
    ax1.set_ylabel('Weight (%)', fontsize=13, color='gray')
    ax1.tick_params(axis='both', which='major', labelsize=13, colors='gray')
    ax1.tick_params(axis='both', which='minor', labelsize=13, colors='gray')
    
    # Create twin axis for DTG
    ax2 = ax1.twinx()
    color_y2 = '#c0392b'
    # Multiply DTG by -1 so peaks point up
    y2_plot_data = y2_data * -1
    line2, = ax2.plot(x_data, y2_plot_data, color=color_y2, label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', fontsize=13, color='gray')
    ax2.tick_params(axis='y', which='major', labelsize=13, colors='gray')
    ax2.tick_params(axis='y', which='minor', labelsize=13, colors='gray')
    
    # Title
    ax1.set_title(sheet_name, fontsize=13, color='gray')
    
    # Calculate Residue at ~800°C
    # Find index closest to 800
    idx_800 = np.argmin(np.abs(x_data - 800))
    residue_val = y1_data[idx_800]
    temp_at_residue = x_data[idx_800]
    
    # Add marker and text for residue
    ax1.plot(temp_at_residue, residue_val, 'o', color=color_y1, markersize=5)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(temp_at_residue, residue_val), 
                 xytext=(temp_at_residue + 20, residue_val + 5),
                 fontsize=13, color='gray',
                 arrowprops=dict(facecolor='gray', shrink=0.05))
                 
    # Calculate Peak temperature of DTG above 150°C
    # We look for the minimum of original DTG (which becomes maximum after *-1)
    # Or simply find max of y2_plot_data where x > 150
    mask_peak = x_data > 150
    if np.any(mask_peak):
        y2_masked = y2_plot_data[mask_peak]
        x_masked = x_data[mask_peak]
        
        idx_max_dtg = np.argmax(y2_masked)
        peak_temp = x_masked[idx_max_dtg]
        peak_val = y2_masked[idx_max_dtg]
        
        # Add marker and text for peak on ax2
        ax2.plot(peak_temp, peak_val, 'o', color=color_y2, markersize=5)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_val), 
                     xytext=(peak_temp + 20, peak_val + 0.05),
                     fontsize=13, color='gray',
                     arrowprops=dict(facecolor='gray', shrink=0.05))
                     
    # Legend
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='center left', fontsize=13, frameon=False)
    
    # Grid
    ax1.grid(True, linestyle='--', alpha=0.5)
    
    # Save figure
    safe_sheet_name = "".join([c if c.isalnum() else '_' for c in sheet_name])
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    plt.tight_layout()
    plt.savefig(out_path, dpi=300)
    plt.close(fig)
```
