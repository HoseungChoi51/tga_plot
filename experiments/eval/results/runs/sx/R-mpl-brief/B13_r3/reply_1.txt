```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import find_peaks
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B13_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read PVC sheet
df = pd.read_excel(DATA_FILE, sheet_name="PVC")

# Extract columns based on preview
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

temps = df[temp_col].values
weights = df[weight_col].values
dtgs = df[dtg_col].values

# Invert DTG for plotting (peaks point up)
dtg_inv = -dtgs

# Find peaks in inverted DTG above 150 °C
mask = temps > 150
temps_masked = temps[mask]
dtg_inv_masked = dtg_inv[mask]

# Use find_peaks to identify local maxima
# distance ensures we don't pick noise; prominence helps separate distinct steps
peaks, properties = find_peaks(dtg_inv_masked, prominence=0.1, distance=20)

if len(peaks) >= 2:
    # Get the two highest peaks
    peak_heights = dtg_inv_masked[peaks]
    sorted_indices = np.argsort(peak_heights)[::-1][:2]
    top_two_peak_indices = peaks[sorted_indices]
    
    # Sort by temperature for consistent labeling order if needed, but usually left-to-right is fine
    peak_temps = temps_masked[top_two_peak_indices]
    peak_temps_sorted = np.sort(peak_temps)
    
    # Create plot
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    color_weight = '#1f4e9c'
    color_dtg = '#c0392b'
    
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color_weight, fontsize=12)
    ax1.plot(temps, weights, color=color_weight, label='Weight')
    ax1.tick_params(axis='y', labelcolor=color_weight)
    
    ax2 = ax1.twinx()
    ax2.set_ylabel('DTG (%/°C)', color=color_dtg, fontsize=12)
    ax2.plot(temps, dtg_inv, color=color_dtg, label='DTG (-x)')
    ax2.tick_params(axis='y', labelcolor=color_dtg)
    
    # Mark the two temperatures where weight drops sharply (highest DTG peaks)
    for i, t_peak in enumerate(peak_temps_sorted):
        # Find index in original array closest to this temp
        idx = np.argmin(np.abs(temps - t_peak))
        
        # Plot marker on DTG curve (right axis)
        ax2.plot(t_peak, dtg_inv[idx], 'o', color=color_dtg, markersize=8, zorder=5)
        
        # Annotate temperature
        # Position text slightly above the peak or offset
        y_pos = dtg_inv[idx] + 0.1 * max(dtg_inv) 
        ax2.annotate(f'{t_peak:.0f} °C', 
                     xy=(t_peak, dtg_inv[idx]), 
                     xytext=(t_peak, y_pos),
                     arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                     fontsize=10, ha='center', va='bottom', color='black')

    # Add residue at 800°C
    idx_800 = np.argmin(np.abs(temps - 800))
    residue_val = weights[idx_800]
    ax1.plot(temps[idx_800], residue_val, 's', color=color_weight, markersize=8, zorder=5)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(temps[idx_800], residue_val), 
                 xytext=(temps[idx_800]+20, residue_val+5),
                 arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                 fontsize=10, ha='left', va='bottom', color=color_weight)

    # Title and Legend
    plt.title("PVC TGA Analysis", fontsize=14)
    
    # Combine legends
    lines_1, labels_1 = ax1.get_legend_handles_labels()
    lines_2, labels_2 = ax2.get_legend_handles_labels()
    ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center left')
    
    plt.tight_layout()
    
    out_path = os.path.join(OUT_DIR, "PVC_sharp_drops.png")
    plt.savefig(out_path, dpi=300)
    plt.close()
else:
    print("Could not find two distinct peaks.")
```
