```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B23_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

# Define colors for each sheet
color_map = {
    "Raw": "black",
    "Acid-washed": "blue",
    "Annealed": "red"
}

fig, ax1 = plt.subplots(figsize=(8, 6))

# We will plot Weight (%) on y1 (left) and DTG on y2 (right)
ax2 = ax1.twinx()

for sheet in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    
    # Identify columns based on preview
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        if 'Temperature' in col or 'Temp' in col:
            temp_col = col
        elif 'Weight' in col and 'Deriv' not in col and 'DTG' not in col:
            weight_col = col
        elif 'Deriv' in col or 'DTG' in col:
            dtg_col = col
            
    if temp_col is None or weight_col is None or dtg_col is None:
        continue
        
    x = df[temp_col].values
    y1 = df[weight_col].values
    y2 = -df[dtg_col].values  # Multiply by -1 so peaks point up
    
    color = color_map.get(sheet, 'gray')
    
    # Plot Weight on left axis
    ax1.plot(x, y1, color=color, label=f"{sheet} Weight")
    
    # Plot DTG on right axis
    ax2.plot(x, y2, color=color, linestyle='--', alpha=0.7, label=f"{sheet} DTG")
    
    # Find residue at ~800°C
    idx_800 = np.argmin(np.abs(x - 800))
    residue_val = y1[idx_800]
    
    # Find peak temperature above 150°C
    mask = x > 150
    if np.any(mask):
        y2_masked = y2[mask]
        x_masked = x[mask]
        max_idx = np.argmax(y2_masked)
        peak_temp = x_masked[max_idx]
        peak_val = y2_masked[max_idx]
        
        # Add markers for residue and peak
        ax1.scatter(x[idx_800], residue_val, color=color, zorder=5, s=30)
        ax2.scatter(peak_temp, peak_val, color=color, marker='v', zorder=5, s=30)
        
        # Annotate residue
        ax1.annotate(f"Residue {residue_val:.1f}%", 
                     xy=(x[idx_800], residue_val), 
                     xytext=(10, 10), textcoords='offset points',
                     fontsize=8, color=color)
                     
        # Annotate peak
        ax2.annotate(f"{peak_temp:.0f} °C", 
                     xy=(peak_temp, peak_val), 
                     xytext=(10, -10), textcoords='offset points',
                     fontsize=8, color=color)

# Labels and Title
ax1.set_xlabel("Temperature (°C)", fontsize=12)
ax1.set_ylabel("Weight (%)", fontsize=12)
ax2.set_ylabel("DTG (%/°C) [inverted]", fontsize=12)
ax1.set_title("TGA Comparison of Samples", fontsize=14)

# Combine legends
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left', bbox_to_anchor=(0.02, 0.5), fontsize=9)

# Grid
ax1.grid(True, linestyle=':', alpha=0.5)

plt.tight_layout()

output_path = os.path.join(OUT_DIR, "tga_overlay.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
