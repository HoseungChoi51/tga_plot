```python
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os

# 데이터 파일 경로 및 출력 디렉토리 설정
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/rheology.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B03_r1/out"

# 출력 디렉토리가 없으면 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 데이터 읽기
df = pd.read_csv(DATA_FILE)

# 열 이름 확인 (공백 제거 등 전처리 필요할 수 있음)
cols = df.columns.tolist()
# 예상되는 컬럼명: 'Angular frequency (rad/s)', "Storage modulus G' (Pa)", "Loss modulus G'' (Pa)"
# 정확한 매칭을 위해 strip 처리 후 검색하거나 인덱스 사용 가능하지만, 여기서는 명시적 이름 사용 시도
freq_col = [c for c in cols if 'frequency' in c.lower()][0]
g_prime_col = [c for c in cols if "G'" in c and 'storage' in c.lower()][0]
g_double_prime_col = [c for c in cols if "G''" in c and 'loss' in c.lower()][0]

x_data = df[freq_col].values
y1_data = df[g_prime_col].values  # Storage modulus G'
y2_data = df[g_double_prime_col].values  # Loss modulus G''

# 플롯 설정
fig, ax1 = plt.subplots(figsize=(8, 6))

# 왼쪽 축: G' (Storage Modulus)
color1 = '#1f4e9c'
ax1.set_xlabel('Angular Frequency (rad/s)', fontsize=12)
ax1.set_ylabel("Storage Modulus G' (Pa)", color=color1, fontsize=12)
line1, = ax1.plot(x_data, y1_data, color=color1, label="G'", linewidth=2)
ax1.tick_params(axis='y', labelcolor=color1)
ax1.set_xscale('log')
ax1.set_yscale('log')

# 오른쪽 축: G'' (Loss Modulus)
ax2 = ax1.twinx()
color2 = '#c0392b'
ax2.set_ylabel("Loss Modulus G'' (Pa)", color=color2, fontsize=12)
line2, = ax2.plot(x_data, y2_data, color=color2, label="G''", linewidth=2)
ax2.tick_params(axis='y', labelcolor=color2)
ax2.set_yscale('log')

# 제목 설정 (파일명 또는 시트명 활용 가능하나, 여기서는 일반적인 제목 사용)
plt.title('Rheology Data: G\' and G\'\' vs Angular Frequency', fontsize=14)

# 범례 추가
lines = [line1, line2]
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='center left')

# 그리드 추가 (로그 스케일에서 보기 좋게)
ax1.grid(True, which="both", ls="--", alpha=0.5)

# 레이아웃 조정
plt.tight_layout()

# 파일 저장
output_path = os.path.join(OUT_DIR, 'rheology_plot.png')
plt.savefig(output_path, dpi=300)
plt.close()
```
