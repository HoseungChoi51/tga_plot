def get_onset_temperature(temp, dtg):
    """
    Calculate the onset temperature using the tangent method on the DTG curve.
    1. Find the peak of DTG (max mass loss rate).
    2. Draw a tangent at the inflection point (peak) on the rising edge of the weight loss.
       Actually, standard TGA onset is often found by drawing a tangent to the steepest slope 
       on the TG curve and extrapolating it to the baseline.
       
    Alternative common method for "onset" from DTG:
    The onset temperature is typically defined as the temperature where the derivative begins 
    to deviate significantly from zero or via the intersection of the tangent at the maximum 
    rate of change with the baseline.
    
    Let's use the standard tangent method on the Weight (%) curve:
    1. Identify the main decomposition region (usually the largest drop).
    2. Find the point of maximum slope (which corresponds to the peak in -DTG).
    3. Draw a tangent line at this point of maximum slope.
    4. Extend this tangent back to intersect the initial baseline (horizontal line at start weight).
    5. The intersection x-coordinate is the onset temperature.
    """
    # Filter data to avoid noise at very beginning if any, but usually starting temp is fine.
    # We focus on the main decomposition event.
    
    # Ensure we are working with arrays
    temp = np.array(temp)
    weight = np.array(weight_col) # Will be passed explicitly
    
    # For simplicity in this specific script context, I will re-fetch inside or pass correctly.
    # Let's assume 'temp' and 'weight' are passed.
    
    # 1. Smooth slightly if needed? No, raw data usually okay for tangent if clean.
    # Find the index of the maximum absolute derivative (steepest slope)
    # Note: DTG is dW/dT. Mass loss means negative DTG. Steepest loss is most negative DTG.
    # So we look for min(DTG) which corresponds to max(-DTG).
    
    idx_peak_dtg = np.argmin(dtg) # Most negative value = steepest descent
    t_peak = temp[idx_peak_dtg]
    
    # Get local neighborhood to calculate precise tangent slope
    window = 5
    i_start = max(0, idx_peak_dtg - window)
    i_end = min(len(temp), idx_peak_dtg + window + 1)
    
    t_local = temp[i_start:i_end]
    w_local = weight[i_start:i_end]
    
    # Fit linear regression to get slope and intercept of the tangent at the peak
    coeffs = np.polyfit(t_local, w_local, 1)
    slope = coeffs[0]
    intercept = coeffs[1]
    
    # Baseline: Average weight before significant decomposition starts
    # Heuristic: Take average of first 10% of points or points before temp drops by 1%
    baseline_weight = np.mean(weight[:10]) 
    
    # Onset is where tangent intersects baseline
    # w = m*t + c  => t = (w - c) / m
    if slope != 0:
        t_onset = (baseline_weight - intercept) / slope
    else:
        t_onset = t_peak
        
    # Sanity check: onset should be less than peak temperature
    if t_onset > t_peak or t_onset < temp[0]:
        # Fallback: simple threshold method (e.g., 1% weight loss)
        diff = baseline_weight - weight
        indices = np.where(diff >= 1.0)[0]
        if len(indices) > 0:
            t_onset = temp[indices[0]]
        else:
            t_onset = t_peak
            
    return t_onset
