import pandas as pd
import numpy as np
import json

def load_data(filepath, is_csv):
    if is_csv:
        df = pd.read_csv(filepath)
    else:
        # The error suggests the .txt file might be comma-separated or have mixed delimiters.
        # Let's try reading it with default settings first, then fallback to tab/space if needed.
        # Given the error '0,0;30,0...', it looks like the entire row was read as a single string 
        # because the delimiter wasn't detected correctly, OR the file format is unusual.
        # However, looking at the preview, it seems standard columns. 
        # Let's try sep=None with engine='python' which auto-detects.
        try:
            df = pd.read_csv(filepath, sep=None, engine='python')
        except Exception:
            # Fallback to tab
            df = pd.read_csv(filepath, sep='\t')
            
    # Clean column names to match expectations
    cols = df.columns.tolist()
    
    # Find temperature and weight columns dynamically
    temp_col = None
    weight_col = None
    deriv_col = None
    
    for c in cols:
        if 'Temperature' in c:
            temp_col = c
        elif 'Weight (%)' in c:
            weight_col = c
        elif 'Deriv' in c:
            deriv_col = c
            
    if not all([temp_col, weight_col, deriv_col]):
        raise ValueError("Could not identify required columns in data file")

    # Convert to numeric, forcing errors to NaN then dropping them just in case
    temp_vals = pd.to_numeric(df[temp_col], errors='coerce').dropna().values
    weight_vals = pd.to_numeric(df[weight_col], errors='coerce').dropna().values
    deriv_vals = pd.to_numeric(df[deriv_col], errors='coerce').dropna().values
    
    # Ensure lengths match (they should if dropna worked on same rows, but safer to align)
    min_len = min(len(temp_vals), len(weight_vals), len(deriv_vals))
    return temp_vals[:min_len], weight_vals[:min_len], deriv_vals[:min_len]
