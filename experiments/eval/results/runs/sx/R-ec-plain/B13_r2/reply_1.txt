import pandas as pd
import numpy as np
import json
from scipy.signal import find_peaks

def get_tga_data(file_path, sheet_name):
    df = pd.read_excel(file_path, sheet_name=sheet_name)
    
    # Identify columns based on the preview
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    deriv_col = "Deriv. Weight (%/°C)"
    
    # Clean data: drop NaNs if any
    df = df.dropna(subset=[temp_col, weight_col, deriv_col])
    
    temps = df[temp_col].values
    weights = df[weight_col].values
    derivs = df[deriv_col].values
    
    return temps, weights, derivs

def main():

    try:
        temps, weights, derivs = get_tga_data(file_path, sheet_name)
        
        # Find peaks in the derivative curve (DTG). 
        # DTG peaks correspond to maximum rate of mass loss (steepest descent in TGA).
        # We look for negative peaks (since mass is decreasing), so we invert the signal or use prominence.
        # The derivative values are negative during decomposition.
        # To find peaks using scipy, we can pass -derivs because find_peaks finds local maxima.
        
        inverted_derivs = -derivs
        
        # Use a reasonable distance and prominence to avoid noise.
        # PVC typically has two main stages: ~250-300°C (dehydrochlorination) and ~450-500°C (cracking).
        # Let's set parameters to find significant peaks.
        peaks, properties = find_peaks(inverted_derivs, prominence=0.1, distance=50)

        peak_weights = weights[peaks]
        peak_derivs = derivs[peaks]
