```python
import pandas as pd
import json
import numpy as np

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

# Read all sheets from the Excel file
xl = pd.ExcelFile(DATA_FILE)

series_list = []
colors = ["#1f77b4", "#ff7f0e", "#2ca02c"]

for idx, sheet_name in enumerate(xl.sheet_names):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)
    
    # Identify columns based on standard TGA naming
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        col_lower = str(col).lower()
        if "temperature" in col_lower or "temp" in col_lower:
            temp_col = col
        elif "weight" in col_lower and "(%)" in col_lower:
            weight_col = col
        elif "d.w" in col_lower or "dtg" in col_lower:
            dtg_col = col
            
    if temp_col is None or weight_col is None:
        continue
        
    temps = df[temp_col].dropna().values
    weights = df[weight_col].dropna().values
    
    # Create Weight(%) series
    data_weight = [[float(t), float(w)] for t, w in zip(temps, weights)]
    series_list.append({
        "name": f"{sheet_name} - Weight(%)",
        "type": "line",
        "data": data_weight,
        "showSymbol": False,
        "yAxisIndex": 0,
        "lineStyle": {"width": 1.5},
        "itemStyle": {"color": colors[idx % len(colors)]}
    })
    
    # Create DTG series if exists
    if dtg_col is not None:
        dtgs = df[dtg_col].dropna().values
        # Align lengths just in case of NaNs causing mismatch (though dropna usually handles it per column)
        # Better approach: filter rows where both exist
        valid_mask = ~df[dtg_col].isna() & ~df[temp_col].isna()
        temps_valid = df.loc[valid_mask, temp_col].values
        dtgs_valid = df.loc[valid_mask, dtg_col].values
        
        data_dtg = [[float(t), float(d)] for t, d in zip(temps_valid, dtgs_valid)]
        series_list.append({
            "name": f"{sheet_name} - D.W(%/C)",
            "type": "line",
            "data": data_dtg,
            "showSymbol": False,
            "yAxisIndex": 1,
            "lineStyle": {"width": 1.5, "type": "dashed"},
            "itemStyle": {"color": colors[idx % len(colors)]}
        })

# Construct ECharts option
option = {
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "cross"}
    },
    "legend": {
        "data": [s["name"] for s in series_list],
        "top": 10,
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 9}
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 10},
        "axisTick": {
            "alignWithLabel": False,
            "inside": False,
            "length": 6
        },
        "axisLabel": {
            "fontSize": 9,
            "fontFamily": "Noto Sans CJK KR"
        },
        "splitLine": {"show": False}
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "position": "left",
            "axisTick": {
                "alignWithLabel": False,
                "inside": False,
                "length": 6
            },
            "axisLabel": {
                "fontSize": 9,
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {"show": True, "lineStyle": {"type": "dashed"}}
        },
        {
            "type": "value",
            "name": "D.W (%/°C)",
            "position": "right",
            "axisTick": {
                "alignWithLabel": False,
                "inside": False,
                "length": 6
            },
            "axisLabel": {
                "fontSize": 9,
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {"show": False}
        }
    ],
    "series": series_list
}

output = {
    "figures": [
        {
            "name": "tga_combined",
            "option": option,
            "width": 800,
            "height": 500,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(output, ensure_ascii=False))
```
