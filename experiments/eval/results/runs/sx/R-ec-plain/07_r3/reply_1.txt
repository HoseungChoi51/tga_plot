```python
import pandas as pd
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

# Read all sheets from the Excel file
xl_file = pd.ExcelFile(DATA_FILE)
sheet_names = xl_file.sheet_names

# Prepare data for each sheet
tga_data = []
for sheet in sheet_names:
    df = pd.read_excel(xl_file, sheet_name=sheet, header=0)
    # Ensure columns are named correctly based on preview
    # Columns: Time, Temperature, Weight(%), D.W(%/C)
    temp = df['Temperature'].tolist()
    weight = df['Weight(%)'].tolist()
    dtg = df['D.W(%/C)'].tolist()
    
    tga_data.append({
        'name': sheet,
        'temp': temp,
        'weight': weight,
        'dtg': dtg
    })

# Define colors for different samples
colors = ['#5470c6', '#91cc75', '#fac858']

series_list = []
legend_data = []

for i, data in enumerate(tga_data):
    sample_name = data['name']
    legend_data.append(sample_name)
    
    # TGA curve (Weight %) - Left Y Axis
    series_list.append({
        "name": f"{sample_name} Weight",
        "type": "line",
        "data": [[t, w] for t, w in zip(data['temp'], data['weight'])],
        "yAxisIndex": 0,
        "showSymbol": False,
        "lineStyle": {
            "color": colors[i % len(colors)],
            "width": 2
        },
        "itemStyle": {
            "color": colors[i % len(colors)]
        }
    })
    
    # DTG curve (D.W %/C) - Right Y Axis
    series_list.append({
        "name": f"{sample_name} DTG",
        "type": "line",
        "data": [[t, d] for t, d in zip(data['temp'], data['dtg'])],
        "yAxisIndex": 1,
        "showSymbol": False,
        "lineStyle": {
            "color": colors[i % len(colors)],
            "width": 2,
            "type": "dashed"
        },
        "itemStyle": {
            "color": colors[i % len(colors)]
        }
    })

option = {
    "title": {
        "text": "TGA/DTG Analysis",
        "left": "center",
        "top": 10,
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 16
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        },
        "formatter": None # Keep default or simple string if needed, but usually fine without custom JS
    },
    "legend": {
        "data": legend_data, # Only show main sample names in legend to avoid clutter, or specific items? 
                             # The user asked to move legend. Usually legends map to series names.
                             # Let's include all series names so they can be toggled individually if desired, 
                             # or just the sample names if we group them. 
                             # Standard ECharts legend maps to series.name.
                             # To keep it clean, let's list all series names.
        "data": [f"{d['name']} Weight" for d in tga_data] + [f"{d['name']} DTG" for d in tga_data],
        "orient": "vertical",
        "right": 10,
        "top": 30,
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 9
        },
        "icon": "roundRect"
    },
    "grid": {
        "left": "8%",
        "right": "25%", # Space for vertical legend on right
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "type": "value",
        "min": 30,
        "max": 900,
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR"
        },
        "splitLine": {
            "show": True,
            "lineStyle": {
                "type": "dashed"
            }
        }
    },
    "yAxis": [
        {
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 40,
            "type": "value",
            "position": "left",
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "show": True
            }
        },
        {
            "name": "DTG (%/°C)",
            "nameLocation": "middle",
            "nameGap": 40,
            "type": "value",
            "position": "right",
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": series_list
}

output = {
    "figures": [
        {
            "name": "tga_analysis",
            "option": option,
            "width": 800,
            "height": 600,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(output))
```
