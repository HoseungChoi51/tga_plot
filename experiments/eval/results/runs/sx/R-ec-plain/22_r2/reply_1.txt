import pandas as pd
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

# Read all sheets from the Excel file
xls = pd.ExcelFile(DATA_FILE)

sheets_data = []
for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name, header=0)
    # Ensure column names are strings and match expected patterns
    cols = [str(c).strip() for c in df.columns]
    df.columns = cols
    
    # Find temperature, weight, and derivative columns by name matching
    temp_col = None
    weight_col = None
    dw_col = None
    
    for col in cols:
        if 'temperature' in col.lower():
            temp_col = col
        elif 'weight' in col.lower() and '%' in col:
            weight_col = col
        elif 'd.w' in col.lower() or 'dw' in col.lower().replace(' ', ''):
            dw_col = col
    
    if temp_col and weight_col and dw_col:
        # Clean data: remove NaN rows
        df_clean = df[[temp_col, weight_col, dw_col]].dropna()
        sheets_data.append({
            'name': sheet_name,
            'temp': df_clean[temp_col].tolist(),
            'weight': df_clean[weight_col].tolist(),
            'dw': df_clean[dw_col].tolist()
        })

# Build ECharts option with two y-axes: left for Weight(%), right for D.W(%/C)
series_list = []
legend_data = []

for i, sd in enumerate(sheets_data):
    label_w = f"{sd['name']} - Weight(%)"
    label_dw = f"{sd['name']} - D.W(%/C)"
    
    legend_data.append(label_w)
    legend_data.append(label_dw)
    
    # TGA curve (Weight %) on yAxis 0
    series_list.append({
        "name": label_w,
        "type": "line",
        "yAxisIndex": 0,
        "data": [[t, w] for t, w in zip(sd['temp'], sd['weight'])],
        "smooth": False,
        "symbol": "none",
        "lineStyle": {
            "width": 1.5
        }
    })
    
    # DTG curve (D.W %/C) on yAxis 1
    series_list.append({
        "name": label_dw,
        "type": "line",
        "yAxisIndex": 1,
        "data": [[t, d] for t, d in zip(sd['temp'], sd['dw'])],
        "smooth": False,
        "symbol": "none",
        "lineStyle": {
            "width": 1.5,
            "color": "#00aa00"  # Green color for DTG lines
        }
    })

option = {
    "backgroundColor": "#ffffff",
    "title": {
        "text": "TGA / DTG Analysis",
        "left": "center",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 16
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        },
        "formatter": "{a0}: ({b}, {c})<br/>{a1}: ({b}, {c})",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "legend": {
        "data": legend_data,
        "bottom": 10,
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 11
        },
        "type": "scroll"
    },
    "grid": {
        "left": 70,
        "right": 80,
        "top": 60,
        "bottom": 90,
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "nameTextStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 12
        },
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 11
        },
        "splitLine": {
            "show": True,
            "lineStyle": {
                "color": "#eee"
            }
        }
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "position": "left",
            "nameTextStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 12
            },
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 11
            },
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "color": "#eee"
                }
            }
        },
        {
            "type": "value",
            "name": "D.W (%/°C)",
            "position": "right",
            "nameTextStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 12,
                "color": "#00aa00"  # Green label text for y2 axis
            },
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 11,
                "color": "#00aa00"  # Green tick labels for y2 axis
            },
            "axisLine": {
                "show": True,
                "lineStyle": {
                    "color": "#00aa00"  # Green axis line for y2
                }
            },
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": series_list
}

output = {
