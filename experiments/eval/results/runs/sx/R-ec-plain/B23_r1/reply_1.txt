```python
import pandas as pd
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"

# Define sheet names and corresponding colors
sheets_info = [
    {"sheet": "Raw", "color": "#000000"},
    {"sheet": "Acid-washed", "color": "#1f77b4"},  # Blue
    {"sheet": "Annealed", "color": "#d62728"}      # Red
]

series_list = []

for info in sheets_info:
    df = pd.read_excel(DATA_FILE, sheet_name=info["sheet"])
    
    # Extract columns based on standard TGA structure from preview
    # x: Temperature (°C), y1: Weight (%), y2: Deriv. Weight (%/°C)
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    deriv_col = "Deriv. Weight (%/°C)"
    
    # Prepare data for Weight (%) - Y axis 0
    weight_data = [[row[temp_col], row[weight_col]] for _, row in df.iterrows()]
    
    # Prepare data for Deriv. Weight (%/°C) - Y axis 1
    deriv_data = [[row[temp_col], row[deriv_col]] for _, row in df.iterrows()]
    
    series_list.append({
        "name": f"{info['sheet']} Weight",
        "type": "line",
        "yAxisIndex": 0,
        "data": weight_data,
        "lineStyle": {
            "color": info["color"],
            "width": 2
        },
        "itemStyle": {
            "color": info["color"]
        },
        "showSymbol": False,
        "smooth": True
    })
    
    series_list.append({
        "name": f"{info['sheet']} DTG",
        "type": "line",
        "yAxisIndex": 1,
        "data": deriv_data,
        "lineStyle": {
            "color": info["color"],
            "width": 2,
            "type": "dashed"
        },
        "itemStyle": {
            "color": info["color"]
        },
        "showSymbol": False,
        "smooth": True
    })

option = {
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "legend": {
        "data": [
            "Raw Weight", "Raw DTG",
            "Acid-washed Weight", "Acid-washed DTG",
            "Annealed Weight", "Annealed DTG"
        ],
        "top": "5%"
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "min": 0,
        "max": 900
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "position": "left",
            "min": 0,
            "max": 100
        },
        {
            "type": "value",
            "name": "DTG (%/°C)",
            "position": "right",
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": series_list
}

output = {
    "figures": [
        {
            "name": "tga_comparison",
            "option": option,
            "width": 800,
            "height": 600,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(output))
```
