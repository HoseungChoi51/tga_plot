import pandas as pd
import numpy as np
import json

# Load data
file_path = "/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt"

# Read the file. The error suggests inconsistent delimiters or extra columns in header/metadata lines.
# Using 'sep=None' with engine='python' allows pandas to infer delimiter and handle irregularities better,
# but it can be slow. Given the specific error "Expected 2 fields... saw 4", it's likely mixed tabs/spaces or 
# metadata lines that don't match the data structure.
# Let's try reading all lines first to inspect or use a more robust method.
# Since we know data starts at row 11 (index 10), let's read raw lines and parse manually if needed,
# or try pandas with on_bad_lines='skip' and sep='\t'.

try:
    # Attempt with tab separator and skipping bad lines just in case headers are messy
    df = pd.read_csv(file_path, sep='\t', header=None, skiprows=10, on_bad_lines='skip')
except Exception:
    # Fallback: Read line by line for the last resort robustness
    data_rows = []
    with open(file_path, 'r') as f:
        lines = f.readlines()
    
    # Data starts at index 10 (row 11)
    for i, line in enumerate(lines[10:], start=10):
        line = line.strip()
        if not line:
            continue
        parts = line.split('\t')
        if len(parts) >= 4:
            try:
                # Take first 4 columns
                row = [float(p) for p in parts[:4]]
                data_rows.append(row)
            except ValueError:
                pass
    
    if data_rows:
        df = pd.DataFrame(data_rows)
    else:
        raise ValueError("No valid data rows found")

# Ensure we have at least 4 columns
if df.shape[1] < 4:
    raise ValueError(f"Data has only {df.shape[1]} columns, expected at least 4")

# Assign column names based on TGA guess

for x, y in zip(x_data, y_mass):
    if not (np.isnan(x) or np.isnan(y)):
        series_tga.append([float(x), float(y)])

for x, y in zip(x_data, y_dtg):

        series_dtg.append([float(x), float(y)])
