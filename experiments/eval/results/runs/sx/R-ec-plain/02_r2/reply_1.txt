import json
import numpy as np

# Load the Excel file
file_path = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
sheet_name = "1006R1-V normal"

# Read data from the specified sheet
df = pd.read_excel(file_path, sheet_name=sheet_name)

# Extract columns based on the description:
# B: Temperature (X-axis)
# C: Weight(%) (Y1-axis)
# D: D.W(%/C) (Y2-axis)

temp_col = 'Temperature'
weight_col = 'Weight(%)'
dw_col = 'D.W(%/C)'

x_data = df[temp_col].tolist()

y2_data = df[dw_col].tolist()

# Find max of Deriv.Weight (D.W)
max_dw = max(y2_data)
max_dw_index = y2_data.index(max_dw)
max_dw_temp = x_data[max_dw_index]

# Find the temperature closest to 800 and its corresponding weight
target_temp = 800
closest_temp_idx = min(range(len(x_data)), key=lambda i: abs(x_data[i] - target_temp))
residue_weight = y1_data[closest_temp_idx]
actual_closest_temp = x_data[closest_temp_idx]
