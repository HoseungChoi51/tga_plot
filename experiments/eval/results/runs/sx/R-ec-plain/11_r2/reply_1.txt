```python
import pandas as pd
import json

def main():
    file_path = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
    
    # Sheet names from preview
    sheets_info = [
        {"name": "1006R1-V normal", "label": "Normal"},
        {"name": "1006R1-V acid", "label": "Acid"},
        {"name": "1006R1-V 750C", "label": "750C"}
    ]
    
    series_list = []
    
    for i, sheet in enumerate(sheets_info):
        try:
            df = pd.read_excel(file_path, sheet_name=sheet["name"])
            
            # Identify columns based on the preview structure
            # Col B: Temperature (x-axis)
            # Col C: Weight(%) (y-axis 1)
            # Col D: D.W(%/C) (y-axis 2)
            
            temp_col = None
            weight_col = None
            dtg_col = None
            
            for col in df.columns:
                col_str = str(col).lower()
                if 'temperature' in col_str:
                    temp_col = col
                elif 'weight' in col_str and '%' in col_str:
                    weight_col = col
                elif 'd.w' in col_str or 'dtg' in col_str or 'derivative' in col_str:
                    dtg_col = col
            
            # Fallback to position if names don't match strictly
            if temp_col is None and len(df.columns) > 1: temp_col = df.columns[1]
            if weight_col is None and len(df.columns) > 2: weight_col = df.columns[2]
            if dtg_col is None and len(df.columns) > 3: dtg_col = df.columns[3]
            
            if temp_col is None or weight_col is None:
                continue
                
            temps = df[temp_col].dropna().tolist()
            weights = df[weight_col].dropna().tolist()
            
            # Prepare data pairs for TGA (Weight vs Temp)
            tga_data = [[t, w] for t, w in zip(temps, weights)]
            
            # Prepare data pairs for DTG (D.W vs Temp)
            dtg_data = []
            if dtg_col is not None:
                dtgs = df[dtg_col].dropna().tolist()
                # Align lengths just in case of NaN dropping differently
                min_len = min(len(temps), len(dtgs))
                dtg_data = [[temps[j], dtgs[j]] for j in range(min_len)]
            
            color_idx = i % 5
            colors = ['#5470c6', '#91cc75', '#fac858', '#ee6666', '#73c0de']
            color = colors[color_idx]
            
            # TGA Series (Left Y Axis)
            series_list.append({
                "name": f"{sheet['label']} TGA",
                "type": "line",
                "data": tga_data,
                "smooth": True,
                "showSymbol": False,
                "yAxisIndex": 0,
                "lineStyle": {"width": 2, "color": color},
                "itemStyle": {"color": color}
            })
            
            # DTG Series (Right Y Axis)
            if dtg_data:
                series_list.append({
                    "name": f"{sheet['label']} DTG",
                    "type": "line",
                    "data": dtg_data,
                    "smooth": True,
                    "showSymbol": False,
                    "yAxisIndex": 1,
                    "lineStyle": {"width": 2, "color": color, "type": "dashed"},
                    "itemStyle": {"color": color}
                })
                
        except Exception as e:
            pass

    option = {
        "title": {
            "text": "TGA/DTG Analysis",
            "left": "center"
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "top": "bottom",
            "data": [s["name"] for s in series_list]
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "min": 30,
            "max": 900,
            "splitLine": {"show": True, "lineStyle": {"type": "dashed"}}
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "min": 0,
                "max": 100,
                "splitLine": {"show": True, "lineStyle": {"type": "dashed"}}
            },
            {
                "type": "value",
                "name": "Derivative Weight (%/°C)",
                "position": "right",
                "splitLine": {"show": False}
            }
        ],
        "series": series_list
    }

    result = {
        "figures": [
            {
                "name": "tga_plot",
                "option": option,
                "width": 800,
                "height": 600,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(result))

if __name__ == "__main__":
    main()
```
