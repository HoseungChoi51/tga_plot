import pandas as pd
import json
import numpy as np

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def main():
    # Read all sheets from the Excel file
    xls = pd.ExcelFile(DATA_FILE)
    
    figures = []
    
    for sheet_name in xls.sheet_names:
        df = pd.read_excel(xls, sheet_name=sheet_name, header=0)
        
        # Extract data columns based on the preview structure
        # Columns: Time, Temperature, Weight(%), D.W(%/C)
        temp = df['Temperature'].values
        weight = df['Weight(%)'].values
        dw = df['D.W(%/C)'].values
        
        # Prepare series data for ECharts
        # Series 1: Weight (%) - left Y axis
        # Series 2: D.W (%/C) - right Y axis (yAxisIndex: 1)
        
        series_data_weight = [[float(t), float(w)] for t, w in zip(temp, weight)]
        series_data_dw = [[float(t), float(d)] for t, d in zip(temp, dw)]
        
        # Determine x-axis range for ticks every 100
        x_min = float(min(temp))
        x_max = float(max(temp))
        
        # Create tick values at intervals of 100
        tick_start = int(np.floor(x_min / 100) * 100)
        tick_end = int(np.ceil(x_max / 100) * 100)
        x_ticks = list(range(tick_start, tick_end + 1, 100))
        
        option = {
            "title": {
                "text": f"TGA & DTG - {sheet_name}",
                "left": "center",
                "top": 10,
                "textStyle": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 16
                }
            },
            "tooltip": {
                "trigger": "axis",
                "axisPointer": {
                    "type": "cross"
                }
            },
            "legend": {
                "data": ["Weight (%)", "D.W (%/C)"],
                "top": 40,
                "textStyle": {
                    "fontFamily": "Noto Sans CJK KR"
                }
            },
            "grid": {
                "left": "8%",
                "right": "8%",
                "bottom": "10%",
                "containLabel": True
            },
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 30,
                "min": x_min,
                "max": x_max,
                "interval": 100,
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "color": "#eee"
                    }
                },
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR"
                }
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "position": "left",
                    "min": 0,
                    "max": 105,
                    "splitLine": {
                        "show": True,
                        "lineStyle": {
                            "color": "#eee"
                        }
                    },
                    "axisLabel": {
                        "fontFamily": "Noto Sans CJK KR"
                    }
                },
                {
                    "type": "value",
                    "name": "D.W (%/C)",
                    "position": "right",
                    "min": -0.05,
                    "max": 1.2,
                    "splitLine": {
                        "show": False
                    },
                    "axisLabel": {
                        "fontFamily": "Noto Sans CJK KR"
                    }
                }
            ],
            "series": [
                {
                    "name": "Weight (%)",
                    "type": "line",
                    "data": series_data_weight,
                    "smooth": True,
                    "symbol": "none",
                    "lineStyle": {
                        "width": 2,
                        "color": "#5470c6"
                    },
                    "itemStyle": {
                        "color": "#5470c6"
                    }
                },
                {
                    "name": "D.W (%/C)",
                    "type": "line",
                    "yAxisIndex": 1,
                    "data": series_data_dw,
                    "smooth": True,
                    "symbol": "none",
                    "lineStyle": {
                        "width": 2,
                        "color": "#ee6666"
                    },
                    "itemStyle": {
                        "color": "#ee6666"
                    }
                }
            ]
        }
        
        figure = {
            "name": f"tga_{sheet_name.replace(' ', '_').replace('-', '_')}",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
        figures.append(figure)
    
    output = {"figures": figures}
    print(json.dumps(output, ensure_ascii=False))

if __name__ == "__main__":
    main()
