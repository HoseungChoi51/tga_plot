```python
import pandas as pd
import json

def main():
    file_path = "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx"
    
    # Define sheet names and their corresponding labels for the legend
    sheets_info = [
        {"name": "1006R1-V normal", "label": "Normal"},
        {"name": "1006R1-V acid", "label": "Acid"},
        {"name": "1006R1-V 750C", "label": "750C"}
    ]
    
    series_list = []
    
    # Colors for different samples
    colors = ["#5470c6", "#91cc75", "#fac858"]
    
    for i, sheet in enumerate(sheets_info):
        try:
            # Read the specific sheet
            df = pd.read_excel(file_path, sheet_name=sheet["name"])
            
            # Extract Temperature (X) and Weight% (Y1)
            # Note: D.W is usually plotted on a secondary axis or separate graph. 
            # For a standard TGA overview, Weight vs Temp is primary.
            # If we want to include DTG, it requires dual axes. Let's create one comprehensive chart with Dual Axes (TGA & DTG).
            
            temp = df['Temperature'].tolist()
            weight = df['Weight(%)'].tolist()
            dtg = df['D.W(%/C)'].tolist()
            
            # Prepare data pairs for ECharts line series
            # Format: [[x1, y1], [x2, y2], ...]
            tga_data = [[t, w] for t, w in zip(temp, weight)]
            dtg_data = [[t, d] for t, d in zip(temp, dtg)]
            
            color = colors[i % len(colors)]
            
            # Add TGA Series (Left Axis)
            series_list.append({
                "name": f"{sheet['label']} (TGA)",
                "type": "line",
                "smooth": True,
                "symbol": "none",
                "lineStyle": {
                    "width": 2,
                    "color": color
                },
                "itemStyle": {
                    "color": color
                },
                "data": tga_data,
                "yAxisIndex": 0
            })
            
            # Add DTG Series (Right Axis) - Optional but good for TGA analysis
            # We use dashed lines for DTG to distinguish from TGA
            series_list.append({
                "name": f"{sheet['label']} (DTG)",
                "type": "line",
                "smooth": True,
                "symbol": "none",
                "lineStyle": {
                    "width": 1.5,
                    "type": "dashed",
                    "color": color,
                    "opacity": 0.7
                },
                "itemStyle": {
                    "color": color
                },
                "data": dtg_data,
                "yAxisIndex": 1
            })
            
        except Exception as e:
            print(f"Error processing sheet {sheet['name']}: {e}")
            continue

    if not series_list:
        return {"figures": []}

    option = {
        "backgroundColor": "#ffffff",
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            },
            "formatter": None # Default formatter handles multiple series well
        },
        "legend": {
            "data": [f"{s['name']}" for s in series_list],
            "top": 10,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 12
            }
        },
        "grid": {
            "left": "8%",
            "right": "8%",
            "bottom": "10%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "min": 0,
            "max": 900,
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "type": "dashed",
                    "color": "#eee"
                }
            },
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "min": 0,
                "max": 105,
                "axisLabel": {
                    "formatter": "{value} %",
                    "fontFamily": "Noto Sans CJK KR"
                },
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "type": "solid",
                        "color": "#ddd"
                    }
                }
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "position": "right",
                "axisLabel": {
                    "formatter": "{value}",
                    "fontFamily": "Noto Sans CJK KR"
                },
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": series_list
    }

    figure = {
        "name": "tga_analysis",
        "option": option,
        "width": 800,
        "height": 600,
        "dpi": 300,
        "formats": ["png"]
    }

    result = {
        "figures": [figure]
    }
    
    print(json.dumps(result))

if __name__ == "__main__":
    main()
```
