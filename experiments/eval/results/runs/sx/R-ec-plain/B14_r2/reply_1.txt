```python
import pandas as pd
import json

def main():
    file_path = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
    
    # Read the PVC sheet
    df = pd.read_excel(file_path, sheet_name="PVC")
    
    # Extract columns based on header names found in preview
    # Columns: 'Time (min)', 'Temperature (°C)', 'Weight (%)', 'Deriv. Weight (%/°C)'
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    deriv_weight_col = "Deriv. Weight (%/°C)"
    
    # Prepare data for ECharts
    # Format: [[x, y1], [x, y2]] is not directly supported in a single series if axes are different types or scales, 
    # but here we have two Y-axes. We need to map x to both series.
    
    # Series 1: Weight (%) - Left Axis
    # Series 2: Deriv. Weight (%/°C) - Right Axis
    
    # Clean data: ensure numeric and handle NaNs if any (though preview looks clean)
    temps = df[temp_col].tolist()
    weights = df[weight_col].tolist()
    deriv_weights = df[deriv_weight_col].tolist()
    
    # Create data arrays for ECharts
    # ECharts expects data as array of objects {value: [x, y]} or just [x, y] if xAxis is value type
    # Since we have multiple series sharing X axis, we can provide separate data arrays for each series
    
    data_weight = []
    data_deriv = []
    
    for t, w, dw in zip(temps, weights, deriv_weights):
        # Filter out non-numeric or infinite values if necessary, though TGA data is usually continuous
        try:
            val_t = float(t)
            val_w = float(w)
            val_dw = float(dw)
            
            # Append to respective series data
            # For line charts with shared X-axis, it's often better to provide data as [x, y] pairs per point
            data_weight.append([val_t, val_w])
            data_deriv.append([val_t, val_dw])
        except (ValueError, TypeError):
            continue
            
    # Define colors
    color_weight = "#5470c6"  # Blue
    color_deriv = "#ee6666"   # Red
    
    # Calculate dimensions
    # Width: 10 inches * 96 px/inch = 960 px
    # Height: Let's assume standard slide aspect ratio or square-ish for large display. 
    # User said "크게" (large) and "가로 10인치". 
    # Standard presentation slide height might be around 7.5 inches -> 720px. 
    # Or maybe 16:9? 10 * 9 / 16 = 5.625 inches -> ~540px.
    # Let's go with a reasonable height for a chart, e.g., 6 inches -> 576px, or match width for square?
    # Usually TGA plots are wider than tall. Let's use 10 inches wide and 6 inches high.
    width_px = int(10 * 96)
    height_px = int(6 * 96)
    
    font_size = 16
    
    option = {
        "backgroundColor": "#ffffff",
        "title": {
            "text": "TGA Analysis: PVC",
            "left": "center",
            "top": "3%",
            "textStyle": {
                "fontSize": font_size + 8,
                "fontWeight": "bold",
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            },
            "formatter": None # Use default formatter which handles multi-series well
        },
        "legend": {
            "data": ["Weight (%)", "Deriv. Weight (%/°C)"],
            "top": "12%",
            "textStyle": {
                "fontSize": font_size,
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "8%",
            "right": "8%",
            "bottom": "12%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "min": 0,
            "max": 900,
            "axisLabel": {
                "fontSize": font_size,
                "fontFamily": "Noto Sans CJK KR"
            },
            "nameTextStyle": {
                "fontSize": font_size,
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "axisLine": {
                    "lineStyle": {
                        "color": color_weight
                    }
                },
                "axisLabel": {
                    "fontSize": font_size,
                    "fontFamily": "Noto Sans CJK KR",
                    "formatter": "{value} %"
                },
                "nameTextStyle": {
                    "fontSize": font_size,
                    "fontFamily": "Noto Sans CJK KR",
                    "padding": [0, 0, 0, 10]
                }
            },
            {
                "type": "value",
                "name": "Deriv. Weight (%/°C)",
                "position": "right",
                "axisLine": {
                    "lineStyle": {
                        "color": color_deriv
                    }
                },
                "axisLabel": {
                    "fontSize": font_size,
                    "fontFamily": "Noto Sans CJK KR",
                    "formatter": "{value}"
                },
                "nameTextStyle": {
                    "fontSize": font_size,
                    "fontFamily": "Noto Sans CJK KR",
                    "padding": [0, 10, 0, 0]
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "data": data_weight,
                "yAxisIndex": 0,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "width": 2,
                    "color": color_weight
                },
                "itemStyle": {
                    "color": color_weight
                }
            },
            {
                "name": "Deriv. Weight (%/°C)",
                "type": "line",
                "data": data_deriv,
                "yAxisIndex": 1,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "width": 2,
                    "color": color_deriv
                },
                "itemStyle": {
                    "color": color_deriv
                }
            }
        ]
    }
    
    figure = {
        "name": "pvc_tga",
        "option": option,
        "width": width_px,
        "height": height_px,
        "dpi": 300,
        "formats": ["png"]
    }
    
    output = {
        "figures": [figure]
    }
    
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
