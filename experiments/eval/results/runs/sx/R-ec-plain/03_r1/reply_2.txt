import pandas as pd
import numpy as np
import json
from scipy.signal import find_peaks

DATA_FILE = "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx"

# Read the specific sheet
df = pd.read_excel(DATA_FILE, sheet_name="1006R1-V acid", header=0)

# Extract columns
temp = df["Temperature"].values
weight = df["Weight(%)"].values
dtg = df["D.W(%/C)"].values

# Filter data for x-axis range 100~800
mask = (temp >= 100) & (temp <= 800)
x_vals = temp[mask]
y1_vals = weight[mask]
y2_vals = dtg[mask]

# Find peaks in DTG (local minima since decomposition is negative)
# We'll find the two most significant peaks (most negative values)
# Use a simple approach: find local minima and pick the two with largest absolute magnitude
inverted_dtg = -y2_vals
peaks, properties = find_peaks(inverted_dtg, prominence=0.05, distance=20)

if len(peaks) < 2:
    # Fallback: just take the two lowest points
    sorted_indices = np.argsort(y2_vals)[:2]
    peak_indices = sorted(sorted_indices)
else:
    # Take the top 2 by prominence or height
    prominences = properties['prominences']
    top_2_idx = np.argsort(prominences)[-2:]
    peak_indices = sorted(peaks[top_2_idx])

peak_points = []
for idx in peak_indices:
    peak_points.append([float(x_vals[idx]), float(y2_vals[idx])])
