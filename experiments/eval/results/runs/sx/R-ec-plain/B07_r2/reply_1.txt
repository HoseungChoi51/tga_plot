import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

# Read the second sheet (PVC)
df = pd.read_excel(DATA_FILE, sheet_name="PVC")

# Extract columns
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"

# Prepare data for TGA curve (Weight vs Temperature)
temps = df[temp_col].tolist()
weights = df[weight_col].tolist()

# Find weight at 300°C and 600°C by interpolation or nearest neighbor
def get_weight_at_temp(temp_target):
    # Use linear interpolation via numpy if available, else simple approach
    import numpy as np
    temps_np = np.array(temps)
    weights_np = np.array(weights)
    # Find index where temp is closest
    idx = np.argmin(np.abs(temps_np - temp_target))
    return float(weights_np[idx])

w_600 = get_weight_at_temp(600)
