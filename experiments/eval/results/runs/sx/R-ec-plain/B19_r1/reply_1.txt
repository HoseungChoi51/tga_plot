```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

def main():
    # Load PVC sheet
    df_pvc = pd.read_excel(DATA_FILE, sheet_name="PVC")
    
    # Ensure numeric types
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    
    df_pvc[temp_col] = pd.to_numeric(df_pvc[temp_col], errors='coerce')
    df_pvc[weight_col] = pd.to_numeric(df_pvc[weight_col], errors='coerce')
    
    # Drop rows with NaN in temperature or weight
    df_pvc_clean = df_pvc.dropna(subset=[temp_col, weight_col])
    
    # Filter data between 200 and 400 °C for calculation
    mask_200_400 = (df_pvc_clean[temp_col] >= 200) & (df_pvc_clean[temp_col] <= 400)
    df_range = df_pvc_clean[mask_200_400]
    
    if not df_range.empty:
        # Find weight at closest to 200°C and 400°C
        w_200_idx = (df_range[temp_col] - 200).abs().idxmin()
        w_400_idx = (df_range[temp_col] - 400).abs().idxmin()
        
        w_at_200 = df_pvc_clean.loc[w_200_idx, weight_col]
        w_at_400 = df_pvc_clean.loc[w_400_idx, weight_col]
        
        mass_loss = w_at_200 - w_at_400
        
        # Prepare data for the chart (all TGA data for context)
        x_data = df_pvc_clean[temp_col].tolist()
        y_weight = df_pvc_clean[weight_col].tolist()
        
        # Highlight points at 200 and 400
        highlight_points = [
            {"coord": [float(df_pvc_clean.loc[w_200_idx, temp_col]), float(w_at_200)], 
             "name": f"200°C ({w_at_200:.2f}%)"},
            {"coord": [float(df_pvc_clean.loc[w_400_idx, temp_col]), float(w_at_400)], 
             "name": f"400°C ({w_at_400:.2f}%)"}
        ]
        
        # Create ECharts option
        option = {
            "title": {
                "text": "PVC TGA Analysis",
                "subtext": f"Mass Loss (200-400°C): {mass_loss:.2f}%",
                "left": "center",
                "top": "5%"
            },
            "tooltip": {
                "trigger": "axis",
                "formatter": "{a}<br/>Temp: {b} °C<br/>Weight: {c} %"
            },
            "grid": {
                "left": "10%",
                "right": "10%",
                "bottom": "15%",
                "containLabel": True
            },
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 30,
                "min": 0,
                "max": 900,
                "splitLine": {"show": True, "lineStyle": {"type": "dashed"}}
            },
            "yAxis": {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "min": 0,
                "max": 105,
                "splitLine": {"show": True, "lineStyle": {"type": "dashed"}}
            },
            "series": [
                {
                    "name": "Weight (%)",
                    "type": "line",
                    "data": [[x, y] for x, y in zip(x_data, y_weight)],
                    "smooth": True,
                    "symbol": "none",
                    "lineStyle": {"width": 2, "color": "#5470c6"},
                    "markPoint": {
                        "data": highlight_points,
                        "itemStyle": {"color": "#ee6666"},
                        "label": {"show": True, "position": "top", "fontSize": 10}
                    },
                    "markArea": {
                        "silent": True,
                        "itemStyle": {
                            "color": "rgba(255, 200, 200, 0.3)"
                        },
                        "data": [
                            [
                                {"xAxis": 200},
                                {"xAxis": 400}
                            ]
                        ]
                    }
                }
            ]
        }
        
        figure = {
            "name": "pvc_tga_mass_loss",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
        
        result = {"figures": [figure]}
        print(json.dumps(result))
    else:
        print(json.dumps({"figures": []}))

if __name__ == "__main__":
    main()
```
