```python
import pandas as pd
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def main():
    # Read all sheets from the Excel file
    xl_file = pd.ExcelFile(DATA_FILE)
    
    figures = []
    
    for sheet_name in xl_file.sheet_names:
        df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)
        
        # Identify columns based on typical TGA data structure
        # We assume column B is Temperature, C is Weight(%), D is DTG (D.W(%/C))
        # Based on the preview, we can map them explicitly or by position/name
        
        # Let's try to find the columns by name if possible, otherwise by index
        cols = df.columns.tolist()
        
        temp_col = None
        weight_col = None
        dtg_col = None
        
        for c in cols:
            if 'Temperature' in str(c):
                temp_col = c
            elif 'Weight' in str(c):
                weight_col = c
            elif 'D.W' in str(c) or 'DTG' in str(c):
                dtg_col = c
                
        # Fallback to indices if names don't match strictly
        if temp_col is None and len(cols) > 1: temp_col = cols[1]
        if weight_col is None and len(cols) > 2: weight_col = cols[2]
        if dtg_col is None and len(cols) > 3: dtg_col = cols[3]
        
        if temp_col is None or weight_col is None:
            continue
            
        # Prepare data
        x_data = df[temp_col].dropna().tolist()
        y1_data = df[weight_col].dropna().tolist()
        
        # Align lengths just in case
        min_len = min(len(x_data), len(y1_data))
        x_data = x_data[:min_len]
        y1_data = y1_data[:min_len]
        
        series_list = []
        
        # Series 1: Weight (%) - Black solid line, width 2.5
        series_list.append({
            "name": "Weight (%)",
            "type": "line",
            "data": [[x, y] for x, y in zip(x_data, y1_data)],
            "lineStyle": {
                "color": "#000000",
                "width": 2.5,
                "type": "solid"
            },
            "showSymbol": False,
            "yAxisIndex": 0
        })
        
        # Series 2: DTG (%/C) - Dashed line
        if dtg_col is not None:
            y2_raw = df[dtg_col].dropna().tolist()
            # Need to align with temperature. 
            # Since dropna might have removed different rows, it's safer to use the original dataframe indices
            # But for simplicity in this context, assuming clean data where NaNs are consistent or rare.
            # A more robust way: filter df first
            
            valid_df = df[[temp_col, weight_col, dtg_col]].dropna()
            
            x_vals = valid_df[temp_col].tolist()
            w_vals = valid_df[weight_col].tolist()
            d_vals = valid_df[dtg_col].tolist()
            
            # Update series 1 data with aligned values
            series_list[0]["data"] = [[x, y] for x, y in zip(x_vals, w_vals)]
            
            series_list.append({
                "name": "DTG (%/C)",
                "type": "line",
                "data": [[x, y] for x, y in zip(x_vals, d_vals)],
                "lineStyle": {
                    "color": "#FF6B6B", # Distinct color for DTG
                    "width": 1.5,
                    "type": "dashed"
                },
                "showSymbol": False,
                "yAxisIndex": 1
            })
            
        option = {
            "title": {
                "text": f"TGA Analysis: {sheet_name}",
                "left": "center",
                "textStyle": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 14
                }
            },
            "tooltip": {
                "trigger": "axis",
                "axisPointer": {
                    "type": "cross"
                }
            },
            "legend": {
                "data": [s["name"] for s in series_list],
                "top": "8%"
            },
            "grid": {
                "left": "10%",
                "right": "10%",
                "bottom": "15%",
                "containLabel": True
            },
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 30,
                "splitLine": {"show": True}
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "position": "left",
                    "splitLine": {"show": True},
                    "min": 0,
                    "max": 110
                },
                {
                    "type": "value",
                    "name": "DTG (%/C)",
                    "position": "right",
                    "splitLine": {"show": False}
                }
            ],
            "series": series_list
        }
        
        fig_entry = {
            "name": f"tga_{sheet_name.replace(' ', '_').replace('-', '_')}",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
        figures.append(fig_entry)
        
    output = {"figures": figures}
    print(json.dumps(output, ensure_ascii=False))

if __name__ == "__main__":
    main()
```
