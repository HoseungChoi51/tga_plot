```python
import pandas as pd
import json

# Load the data
file_path = "/home/chs/Work/tga_plot/eval/inputs_b/isothermal.txt"
df = pd.read_csv(file_path, sep=r'\s+', header=0)

# Extract columns
time_col = 'Time (min)'
temp_col = 'Temperature (°C)'
weight_col = 'Weight (%)'

# Get data arrays
times = df[time_col].tolist()
temps = df[temp_col].tolist()
weights = df[weight_col].tolist()

# Find weight at 140 min
idx_140 = df[df[time_col] == 140.0].index
if len(idx_140) > 0:
    weight_at_140 = df.loc[idx_140[0], weight_col]
else:
    # Interpolate if exact match not found
    weight_at_140 = None

# Prepare markPoint for 140 min
mark_point_data = []
if weight_at_140 is not None:
    mark_point_data.append({
        "coord": [140, weight_at_140],
        "value": f"{weight_at_140:.2f}%",
        "symbolSize": 10,
        "label": {
            "show": True,
            "position": "top",
            "formatter": "{c}"
        }
    })

option = {
    "title": {
        "text": "Isothermal TGA Data",
        "left": "center"
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "legend": {
        "data": ["Weight (%)", "Temperature (°C)"],
        "top": 30
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%"
    },
    "xAxis": {
        "name": "Time (min)",
        "type": "value",
        "min": 0,
        "max": 140,
        "splitLine": {
            "show": False
        }
    },
    "yAxis": [
        {
            "name": "Weight (%)",
            "type": "value",
            "min": 70,
            "max": 105,
            "position": "left"
        },
        {
            "name": "Temperature (°C)",
            "type": "value",
            "min": 20,
            "max": 420,
            "position": "right",
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "yAxisIndex": 0,
            "data": [[t, w] for t, w in zip(times, weights)],
            "smooth": False,
            "symbol": "none",
            "lineStyle": {
                "width": 2,
                "color": "#5470C6"
            },
            "markPoint": {
                "data": mark_point_data,
                "itemStyle": {
                    "color": "#EE6666"
                }
            }
        },
        {
            "name": "Temperature (°C)",
            "type": "line",
            "yAxisIndex": 1,
            "data": [[t, temp] for t, temp in zip(times, temps)],
            "smooth": False,
            "symbol": "none",
            "lineStyle": {
                "width": 2,
                "color": "#91CC75"
            }
        }
    ]
}

result = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(result))
```
