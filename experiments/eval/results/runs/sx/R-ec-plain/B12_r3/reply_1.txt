import pandas as pd
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

# Read the CSV file
df = pd.read_csv(DATA_FILE)

# Filter data for x range 100 to 900 °C
mask = (df["Temperature (°C)"] >= 100) & (df["Temperature (°C)"] <= 900)
filtered_df = df[mask].copy()

# Find residue at 900 °C (or closest point)
residue_mask = (df["Temperature (°C)"] >= 895) & (df["Temperature (°C)"] <= 905)
if not residue_mask.any():
    # Fallback: find the point closest to 900
    idx_closest = (df["Temperature (°C)"] - 900).abs().idxmin()
    residue_temp = df.loc[idx_closest, "Temperature (°C)"]
    residue_weight = df.loc[idx_closest, "Weight (%)"]
else:
    # Take the last point in the range or average? Let's take the one closest to 900 within mask
    sub_df = df[residue_mask]
    idx_closest = (sub_df["Temperature (°C)"] - 900).abs().idxmin()
    residue_temp = df.loc[idx_closest, "Temperature (°C)"]
    residue_weight = df.loc[idx_closest, "Weight (%)"]

# Prepare series data
series_data = []
for _, row in filtered_df.iterrows():
    series_data.append([float(row["Temperature (°C)"]), float(row["Weight (%)"])])

# Build ECharts option
option = {
