# Identify peaks in Heat Flow
# Endothermic peaks: local maxima (positive values)
# Exothermic peaks: local minima (negative values)

def find_peaks(y, x, prominence=0.5):
    """Find local maxima and minima with significant prominence."""
    from scipy.signal import find_peaks
    
    # Find endothermic peaks (maxima)
    max_indices, _ = find_peaks(y, prominence=prominence)
    
    # Find exothermic peaks (minima) by finding peaks of -y
    min_indices, _ = find_peaks(-y, prominence=prominence)
    
    return x[max_indices], y[max_indices], x[min_indices], y[min_indices]

endo_temps, endo_vals, exo_temps, exo_vals = find_peaks(heat_flow, temp, prominence=0.3)

# Sort and limit to top 2 endothermic and top 1 exothermic based on magnitude

    # Sort by value descending (highest peak first)
    idx_endo = np.argsort(endo_vals)[::-1][:2]
    endo_temps = endo_temps[idx_endo]
    endo_vals = endo_vals[idx_endo]
else:
    endo_temps = []
