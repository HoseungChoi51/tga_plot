import pandas as pd
import numpy as np
from scipy.signal import find_peaks
import json

# Load data
df = pd.read_csv("/home/chs/Work/tga_plot/eval/inputs_b/caox.csv")

# Extract columns
temp = df["Temperature (°C)"].values
weight = df["Weight (%)"].values
heat_flow = df["Heat Flow (W/g)"].values

# Find peaks in Heat Flow
# Endothermic peaks: positive values (absorbing heat)
# Exothermic peaks: negative values (releasing heat)

# We need to find 2 endothermic and 1 exothermic peak.
# Let's smooth the data slightly or just use find_peaks with appropriate parameters.
# Since we know there are 3 specific peaks, we can try to identify them based on prominence.

# Invert for finding local minima if needed, but find_peaks finds maxima by default.
# For endothermic (positive), we look for maxima of Heat Flow.
# For exothermic (negative), we look for minima of Heat Flow (or maxima of -Heat Flow).

# Let's try to find all significant peaks first.
# Prominence helps filter out noise.
peaks_endo, props_endo = find_peaks(heat_flow, prominence=0.5) # Adjust prominence as needed
troughs_exo, props_exo = find_peaks(-heat_flow, prominence=0.5)

# Sort by prominence to get the most significant ones
if len(peaks_endo) > 0:
    idx_endo_sorted = np.argsort(props_endo['prominences'])[::-1]
    top_peaks_endo_idx = peaks_endo[idx_endo_sorted[:2]] # Get top 2 endothermic peaks
else:
    top_peaks_endo_idx = []

if len(troughs_exo) > 0:
    idx_exo_sorted = np.argsort(props_exo['prominences'])[::-1]
    top_troughs_exo_idx = troughs_exo[idx_exo_sorted[:1]] # Get top 1 exothermic peak
else:
    top_troughs_exo_idx = []

# Combine peak indices for marking
mark_points = []

for idx in top_peaks_endo_idx:
    mark_points.append({
        "coord": [float(temp[idx]), float(heat_flow[idx])],
        "name": f"Endo Peak ({temp[idx]:.1f}°C)",
        "symbolSize": 10,
        "itemStyle": {"color": "#ff4d4d"} # Red for endothermic
    })

for idx in top_troughs_exo_idx:
    mark_points.append({
        "coord": [float(temp[idx]), float(heat_flow[idx])],
        "name": f"Exo Peak ({temp[idx]:.1f}°C)",
        "symbolSize": 10,
        "itemStyle": {"color": "#4da6ff"} # Blue for exothermic
    })
