import pandas as pd
import numpy as np
from scipy.signal import find_peaks
import json

# Load the 'Annealed' sheet from the Excel file
file_path = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
df = pd.read_excel(file_path, sheet_name="Annealed")

# Extract relevant columns
temperature = df['Temperature (°C)'].values.astype(float)
weight = df['Weight (%)'].values.astype(float)
deriv_weight = df['Deriv. Weight (%/°C)'].values.astype(float)

# Calculate onset temperature using extrapolation method
# Find peaks in derivative weight to identify decomposition stages
peaks, _ = find_peaks(-deriv_weight, height=0.1, distance=50)  # Negative because we want minima in deriv

onset_temperatures = []
for peak_idx in peaks:
    # Get local region around the peak for linear fit
    start_idx = max(0, peak_idx - 30)
    end_idx = min(len(temperature), peak_idx + 30)
    
    temp_region = temperature[start_idx:end_idx]
    deriv_region = deriv_weight[start_idx:end_idx]
    
    # Fit a line to the steepest part of the curve near the peak
    # Use points where derivative is significantly negative
    mask = deriv_region < -0.1 * np.min(deriv_region)  # Relative threshold
    if np.sum(mask) > 2:
        fitted_temp = temp_region[mask]
        fitted_deriv = deriv_region[mask]
        
        # Linear regression
        coeffs = np.polyfit(fitted_temp, fitted_deriv, 1)
        slope, intercept = coeffs
        
        # Onset temperature is where the tangent intersects baseline (derivative = 0)
        if slope != 0:
            onset_temp = float(-intercept / slope)
            if temperature[0] <= onset_temp <= temperature[-1]:
                onset_temperatures.append(onset_temp)

# If no onsets found via peak detection, use a simpler approach: first significant drop
if not onset_temperatures:
    # Find first point where weight drops below 99% and derivative becomes significantly negative
    for i in range(1, len(weight)):
        if weight[i] < 99.0 and deriv_weight[i] < -0.05:
            onset_temperatures.append(float(temperature[i]))
            break

# Prepare data for ECharts

sampled_deriv = [float(d) for d in deriv_weight[::step]]
