```python
import pandas as pd
import json
import numpy as np

def main():
    file_path = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
    
    # Read all sheets
    xls = pd.ExcelFile(file_path)
    sheet_names = xls.sheet_names
    
    figures = []
    
    # Prepare data for each sheet
    series_data = []
    legend_names = []
    
    for sheet in sheet_names:
        df = pd.read_excel(xls, sheet_name=sheet)
        
        # Identify columns based on preview info
        # Col B: Temperature (x), Col C: Weight(%) (y1), Col D: D.W(%/C) (y2)
        temp_col = 'Temperature'
        weight_col = 'Weight(%)'
        dtg_col = 'D.W(%/C)'
        
        # Check if columns exist, otherwise try to find by position or name similarity
        if temp_col not in df.columns:
            # Fallback logic if names differ slightly
            cols = df.columns.tolist()
            # Assuming standard order from preview: Time, Temp, Weight, DTG
            if len(cols) >= 4:
                temp_col = cols[1]
                weight_col = cols[2]
                dtg_col = cols[3]
                
        # Clean data: remove NaN rows
        df_clean = df.dropna(subset=[temp_col, weight_col, dtg_col])
        
        temps = df_clean[temp_col].values
        weights = df_clean[weight_col].values
        dtgs = df_clean[dtg_col].values
        
        # Create pairs for ECharts [x, y] format
        weight_pairs = [[float(t), float(w)] for t, w in zip(temps, weights)]
        dtg_pairs = [[float(t), float(d)] for t, d in zip(temps, dtgs)]
        
        # Determine color based on index or name
        colors = ['#5470c6', '#91cc75', '#fac858', '#ee6666', '#73c0de']
        color_idx = sheet_names.index(sheet) % len(colors)
        color = colors[color_idx]
        
        # Series for Weight (%) - Left Y Axis
        series_data.append({
            "name": f"{sheet} - Weight",
            "type": "line",
            "data": weight_pairs,
            "showSymbol": False,
            "smooth": True,
            "lineStyle": {"width": 2, "color": color},
            "itemStyle": {"color": color},
            "yAxisIndex": 0
        })
        
        # Series for DTG (%/C) - Right Y Axis
        series_data.append({
            "name": f"{sheet} - DTG",
            "type": "line",
            "data": dtg_pairs,
            "showSymbol": False,
            "smooth": True,
            "lineStyle": {"width": 2, "color": color, "type": "dashed"},
            "itemStyle": {"color": color},
            "yAxisIndex": 1
        })
        
        legend_names.append(f"{sheet} - Weight")
        legend_names.append(f"{sheet} - DTG")

    # Construct the ECharts Option
    option = {
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": legend_names,
            "top": 10
        },
        "grid": {
            "left": "8%",
            "right": "8%",
            "bottom": "10%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "min": 0,
            "max": 900,
            "interval": 100,  # User request: x-axis tick interval 100
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "type": "dashed",
                    "color": "#ccc"
                }
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "min": 0,
                "max": 100,
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "type": "dashed",
                        "color": "#eee"
                    }
                }
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "position": "right",
                "min": -0.05,   # User request: y2 axis range start
                "max": 1.2,     # User request: y2 axis range end
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": series_data
    }

    output = {
        "figures": [
            {
                "name": "tga_combined",
                "option": option,
                "width": 800,
                "height": 600,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
