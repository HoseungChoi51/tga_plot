```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 파일 경로 설정
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B08_r1/out"

# 출력 디렉토리 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 데이터 읽기 (Raw 시트만 사용)
df = pd.read_excel(DATA_FILE, sheet_name='Raw')

# 컬럼명 정리
temp_col = 'Temperature (°C)'
weight_col = 'Weight (%)'
deriv_col = 'Deriv. Weight (%/°C)'

# 데이터 추출
temperature = df[temp_col]
weight = df[weight_col]
d_weight = df[deriv_col]

# 그래프 생성
fig, ax1 = plt.subplots(figsize=(8, 6))

# TGA 곡선 (무게 %) - 왼쪽 y축
color1 = 'tab:blue'
ax1.set_xlabel('온도 (°C)', fontsize=12)
ax1.set_ylabel('무게 (%)', color=color1, fontsize=12)
line1 = ax1.plot(temperature, weight, color=color1, label='TGA (무게 %)')
ax1.tick_params(axis='y', labelcolor=color1)
ax1.grid(True, linestyle='--', alpha=0.5)

# 무게 50% 위치 가로 보조선 추가
ax1.axhline(y=50, color='gray', linestyle=':', linewidth=1.5, label='50% 기준선')

# DTG 곡선 (미분 무게) - 오른쪽 y축
ax2 = ax1.twinx()
color2 = 'tab:red'
ax2.set_ylabel('미분 무게 (%/°C)', color=color2, fontsize=12)
line2 = ax2.plot(temperature, d_weight, color=color2, label='DTG (미분 무게 %)')
ax2.tick_params(axis='y', labelcolor=color2)

# 범례 추가
lines = line1 + line2
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='upper right')

# 오른쪽 위에 메모 추가
memo_text = '공기, 10 °C/min'
ax1.text(0.95, 0.95, memo_text, transform=ax1.transAxes, fontsize=11,
         verticalalignment='top', horizontalalignment='right',
         bbox=dict(boxstyle='round,pad=0.3', facecolor='yellow', alpha=0.7))

# 제목 설정
plt.title('CNT 샘플의 열중량 분석(TGA/DTG) 결과', fontsize=14)

# 레이아웃 조정
plt.tight_layout()

# 파일 저장
output_path = os.path.join(OUT_DIR, 'tga_raw_annotated.png')
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
