```python
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# Define file paths and output directory
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B23_r2/out"

# Ensure output directory exists
os.makedirs(OUT_DIR, exist_ok=True)

# Read data from Excel sheets
sheets = {
    "Raw": {"color": "black", "label": "Raw"},
    "Acid-washed": {"color": "blue", "label": "Acid-washed"},
    "Annealed": {"color": "red", "label": "Annealed"}
}

fig, ax1 = plt.subplots(figsize=(10, 6))

for sheet_name, props in sheets.items():
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    
    # Extract columns based on the preview structure
    temp = df["Temperature (°C)"]
    weight = df["Weight (%)"]
    deriv_weight = df["Deriv. Weight (%/°C)"]
    
    # Plot TGA (Weight %) on primary axis
    ax1.plot(temp, weight, color=props["color"], label=f"{props['label']} (TGA)", linewidth=1.5)
    
    # Plot DTG (Deriv. Weight) on secondary axis
    ax2 = ax1.twinx()
    ax2.plot(temp, deriv_weight, color=props["color"], linestyle='--', alpha=0.7, label=f"{props['label']} (DTG)")

# Configure Primary Axis (TGA)
ax1.set_xlabel("Temperature (°C)", fontsize=12)
ax1.set_ylabel("Weight (%)", fontsize=12, color='black')
ax1.tick_params(axis='y', labelcolor='black')

# Configure Secondary Axis (DTG)
ax2.set_ylabel("Deriv. Weight (%/°C)", fontsize=12, color='gray')
ax2.tick_params(axis='y', labelcolor='gray')

# Combine legends from both axes
lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()
ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center right', fontsize=9)

# Title and layout
plt.title("TGA/DTG Analysis of CNT Batches", fontsize=14)
fig.tight_layout()

# Save figure
output_path = os.path.join(OUT_DIR, "tga_combined.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
