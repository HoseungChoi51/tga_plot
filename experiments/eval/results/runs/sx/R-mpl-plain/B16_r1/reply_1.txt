```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# Read data
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B16_r1/out"

df = pd.read_csv(DATA_FILE)

# Extract columns based on the preview
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

# Ensure OUT_DIR exists
os.makedirs(OUT_DIR, exist_ok=True)

# Create figure with two subplots sharing x-axis
fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(8, 8), sharex=True, dpi=300)

# Top plot: Weight (%) vs Temperature
ax1.plot(df[temp_col], df[weight_col], color='blue', label='TGA')
ax1.set_ylabel('무게 (%)')
ax1.set_title('TGA 및 DTG 분석 결과')
ax1.grid(True, linestyle='--', alpha=0.5)
ax1.legend()

# Bottom plot: Deriv. Weight (%/°C) vs Temperature
ax2.plot(df[temp_col], df[dtg_col], color='red', label='DTG')
ax2.set_xlabel('온도 (°C)')
ax2.set_ylabel('미분 무게 (%/°C)')
ax2.grid(True, linestyle='--', alpha=0.5)
ax2.legend()

plt.tight_layout()

# Save figure
output_path = os.path.join(OUT_DIR, "tga_dtg_plot.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
