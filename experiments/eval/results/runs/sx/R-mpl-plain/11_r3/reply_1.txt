```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 설정
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/11_r3/out"
os.makedirs(OUT_DIR, exist_ok=True)

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 데이터 읽기
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

fig, axes = plt.subplots(len(sheet_names), 1, figsize=(8, 4 * len(sheet_names)), sharex=True)

if len(sheet_names) == 1:
    axes = [axes]

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(xls, sheet_name=sheet_name, header=0)
    
    # 컬럼명 정리 (공백 제거 등)
    df.columns = [str(col).strip() for col in df.columns]
    
    # TGA 데이터 추출
    temp_col = 'Temperature'
    weight_col = 'Weight(%)'
    dw_col = 'D.W(%/C)'
    
    if temp_col not in df.columns or weight_col not in df.columns:
        continue
        
    x = df[temp_col]
    y1 = df[weight_col]
    
    ax1 = axes[i]
    
    # TGA 곡선 (Left Y-axis)
    color1 = '#1f77b4'
    ax1.plot(x, y1, color=color1, label='TGA', linewidth=1.5)
    ax1.set_ylabel('Weight (%)', color=color1)
    ax1.tick_params(axis='y', labelcolor=color1)
    ax1.grid(True, linestyle='--', alpha=0.5)
    
    # DTG 곡선 (Right Y-axis) - 피크 표시 없이 단순 선으로만
    if dw_col in df.columns:
        y2 = df[dw_col]
        ax2 = ax1.twinx()
        color2 = '#d62728'
        ax2.plot(x, y2, color=color2, label='DTG', linewidth=1.5, linestyle='--')
        ax2.set_ylabel('DTG (%/°C)', color=color2)
        ax2.tick_params(axis='y', labelcolor=color2)
        
        # 레전드 통합
        lines1, labels1 = ax1.get_legend_handles_labels()
        lines2, labels2 = ax2.get_legend_handles_labels()
        ax1.legend(lines1 + lines2, labels1 + labels2, loc='best')
    else:
        ax1.legend(loc='best')
    
    # 제목 설정
    clean_name = sheet_name.replace('normal', 'Normal').replace('acid', 'Acid-treated').replace('750C', '750°C Calcined')
    ax1.set_title(f'TGA/DTG Curve: {clean_name}', fontsize=12)
    
    # X축 라벨은 마지막 서브플롯에만 표시 (sharex=True이므로)
    if i == len(sheet_names) - 1:
        ax1.set_xlabel('Temperature (°C)')

plt.tight_layout()
output_path = os.path.join(OUT_DIR, "tga_plots.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
