```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 설정
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/17_r1/out"
FIG_SIZE = (8, 5)
DPI = 600

# 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 출력 디렉토리 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 데이터 읽기
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

fig, ax1 = plt.subplots(figsize=FIG_SIZE)

colors = ['#1f77b4', '#ff7f0e', '#2ca02c']

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(xls, sheet_name=sheet_name)
    
    # 열 이름 확인 및 매핑
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        if 'Temperature' in str(col):
            temp_col = col
        elif 'Weight' in str(col) and '%' in str(col):
            weight_col = col
        elif 'D.W' in str(col) or 'DTG' in str(col).upper():
            dtg_col = col
            
    if temp_col is None or weight_col is None:
        continue
        
    x = df[temp_col]
    y_weight = df[weight_col]
    
    # TGA 곡선 (좌측 Y축)
    ax1.plot(x, y_weight, color=colors[i % len(colors)], label=f'{sheet_name} (TGA)', linewidth=1.5)

ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12)
ax1.tick_params(axis='y')

# DTG를 위한 우측 Y축 생성
ax2 = ax1.twinx()

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(xls, sheet_name=sheet_name)
    
    temp_col = None
    dtg_col = None
    
    for col in df.columns:
        if 'Temperature' in str(col):
            temp_col = col
        elif 'D.W' in str(col) or 'DTG' in str(col).upper():
            dtg_col = col
            
    if temp_col is not None and dtg_col is not None:
        x = df[temp_col]
        y_dtg = df[dtg_col]
        
        # DTG 곡선 (우측 Y축, 점선 스타일 등으로 구분 가능하지만 여기서는 색상 통일 후 선종류 변경 또는 투명도 조절 등 고려 가능)
        # 일반적으로 TGA는 실선, DTG는 점선이거나 다른 축을 사용함. 여기서는 색상 동일하게 하되, 라벨링 명확히 함
        ax2.plot(x, y_dtg, color=colors[i % len(colors)], linestyle='--', alpha=0.7, label=f'{sheet_name} (DTG)', linewidth=1.5)

ax2.set_ylabel('dW/dT (%/°C)', fontsize=12)
ax2.tick_params(axis='y')

# 범례 통합
lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()
ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center left', bbox_to_anchor=(1, 0.5), fontsize=9)

plt.title('TGA and DTG Curves Comparison', fontsize=14)
plt.tight_layout()

out_path = os.path.join(OUT_DIR, "tga_plot.png")
plt.savefig(out_path, dpi=DPI)
plt.close()
```
