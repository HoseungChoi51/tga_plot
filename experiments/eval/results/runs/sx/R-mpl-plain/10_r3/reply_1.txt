```python
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os

# 설정
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/10_r3/out"

# 폰트 설정 (한글 지원)
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 출력 디렉토리 확인
os.makedirs(OUT_DIR, exist_ok=True)

# 데이터 읽기
xl_file = pd.ExcelFile(DATA_FILE)
sheet_names = xl_file.sheet_names

fig, ax1 = plt.subplots(figsize=(10, 6))

colors = ['#1f77b4', '#ff7f0e', '#2ca02c'] # 기본 색상 팔레트 활용 또는 직접 지정

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(xl_file, sheet_name=sheet_name)
    
    # 컬럼명 정리 (공백 제거 등)
    df.columns = [col.strip() for col in df.columns]
    
    temp_col = 'Temperature'
    weight_col = 'Weight(%)'
    
    if temp_col not in df.columns or weight_col not in df.columns:
        continue
        
    x = df[temp_col]
    y_weight = df[weight_col]
    
    # 라벨 생성: 시트 이름에서 일부 정보를 추출하거나 그대로 사용
    # 예: "1006R1-V normal" -> "normal", "1006R1-V acid" -> "acid" 등
    label_part = sheet_name.split('-')[-1].strip()
    label = f"{label_part}"
    
    color = colors[i % len(colors)]
    
    # TGA 곡선 (좌측 Y축)
    line1, = ax1.plot(x, y_weight, color=color, linewidth=2, label=f'TGA - {label}')
    
    # DTG 곡선 (우측 Y축)
    dtg_col = 'D.W(%/C)'
    if dtg_col in df.columns:
        y_dtg = df[dtg_col]
        # 우측 축은 하나의 축만 사용하는 것이 깔끔할 수 있지만, 요청에 따라 각 샘플의 DTG를 함께 표시하거나
        # 보통 TGA와 DTG는 한 그래프에 이중축으로 표현한다.
        # 여기서는 가독성을 위해 좌측 축에 TGA, 우측 축에 DTG를 공통으로 쓰되, 색상을 일치시킨다.
        # 하지만 여러 샘플이 있을 때 우측 축 스케일 문제가 생길 수 있다.
        # 일반적으로 TGA 분석 보고서에서는 TGA와 DTG를 각각 별도 그래프로 그리거나, 
        # 하나의 그래프에 TGA(실선)와 DTG(점선)로 겹쳐서 그린다.
        
        # 접근 방식 수정: 
        # 사용자가 "잔류량 표시"를 언급했으므로 TGA 곡선이 메인이다.
        # DTG도 함께 보여주는 것이 일반적이므로, 우측 축을 사용하여 DTG를 점선으로 추가한다.
        
        ax2 = ax1.twinx() # twinx를 루프 안에서 계속 호출하면 안 됨. 밖에서 하나 만들고 관리해야 함.
        
    # 위 로직 수정: twinx는 루프 밖에서 한 번만 생성
    
# 다시 작성: 루프 구조 최적화

fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx() # 우측 축 생성

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(xl_file, sheet_name=sheet_name)
    df.columns = [col.strip() for col in df.columns]
    
    temp_col = 'Temperature'
    weight_col = 'Weight(%)'
    dtg_col = 'D.W(%/C)'
    
    if temp_col not in df.columns or weight_col not in df.columns:
        continue
        
    x = df[temp_col]
    y_weight = df[weight_col]
    
    # 라벨 파싱
    parts = sheet_name.split('-')
    label_suffix = parts[-1].strip() if len(parts) > 1 else sheet_name
    label = f"{label_suffix}"
    
    color = colors[i % len(colors)]
    
    # TGA Plot (Left Axis)
    l1, = ax1.plot(x, y_weight, color=color, linewidth=2, linestyle='-', label=f'TGA ({label})')
    
    # DTG Plot (Right Axis)
    if dtg_col in df.columns:
        y_dtg = df[dtg_col]
        l2, = ax2.plot(x, y_dtg, color=color, linewidth=1.5, linestyle='--', alpha=0.7, label=f'DTG ({label})')

# 잔류량 계산 및 표시 (700 °C 기준)
residual_data = []

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(xl_file, sheet_name=sheet_name)
    df.columns = [col.strip() for col in df.columns]
    
    temp_col = 'Temperature'
    weight_col = 'Weight(%)'
    
    if temp_col not in df.columns or weight_col not in df.columns:
        continue
    
    # 700도에 가장 가까운 데이터 포인트 찾기
    target_temp = 700.0
    idx = (df[temp_col] - target_temp).abs().idxmin()
    
    actual_temp = df.loc[idx, temp_col]
    residual_weight = df.loc[idx, weight_col]
    
    parts = sheet_name.split('-')
    label_suffix = parts[-1].strip() if len(parts) > 1 else sheet_name
    
    residual_data.append({
        'label': label_suffix,
        'temp': actual_temp,
        'weight': residual_weight,
        'color': colors[i % len(colors)]
    })

# 축 설정
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12)
ax2.set_ylabel('Derivative Weight (%/°C)', fontsize=12)

ax1.set_xlim(30, 900) # 데이터 범위에 맞춰 조정 가능하지만 고정값도 무방
ax1.set_ylim(0, 105)

# 범례 추가
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='upper right', fontsize=9)

# 제목
plt.title('TGA/DTG Analysis Results', fontsize=14)

# 잔류량 텍스트 박스 생성
text_lines = ["Residual at ~700°C:"]
for data in residual_data:
    # 소수점 둘째 자리까지 포맷팅
    formatted_weight = f"{data['weight']:.2f}%"
    text_lines.append(f"{data['label']}: {formatted_weight}")

textbox_str = "\n".join(text_lines)

# 텍스트 박스를 그래프 위에 배치 (위치 조정 필요)
# left axis 기준으로 배치하되, 오른쪽 상단 빈 공간이나 왼쪽 하단 등을 고려
# 여기서는 왼쪽 하단에 배치하여 곡선과 겹치지 않도록 시도하거나, 여백을 충분히 두고 배치
props = dict(boxstyle='round', facecolor='wheat', alpha=0.8)
ax1.text(0.05, 0.05, textbox_str, transform=ax1.transAxes, fontsize=10,
         verticalalignment='bottom', bbox=props)

plt.tight_layout()

output_path = os.path.join(OUT_DIR, "tga_plot.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
