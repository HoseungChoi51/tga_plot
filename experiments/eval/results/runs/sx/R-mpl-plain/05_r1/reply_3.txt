```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/CNT/Plot/eval/results/runs/sx/R-mpl-plain/05_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

fig, axes = plt.subplots(len(sheet_names), 1, figsize=(8, 4 * len(sheet_names)), sharex=False)
if len(sheet_names) == 1:
    axes = [axes]

colors_y2 = ['#d62728', '#2ca02c', '#9467bd']  # red, green, purple for y2 lines

for idx, sheet_name in enumerate(sheet_names):
    # header_rows=[1] means row index 1 is the header (0-based). 
    # In pandas read_excel, header=1 means use row at index 1 as header.
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=1)
    
    # Clean column names to remove any potential whitespace or hidden characters
    # Convert columns to string first to ensure .str accessor works if needed, 
    # but usually columns are strings. If they are floats/nan due to empty cells in header, we need to handle it.
    new_cols = []
    for col in df.columns:
        if isinstance(col, str):
            new_cols.append(col.strip())
        else:
            new_cols.append(str(col))
    df.columns = new_cols
    
    # Identify columns based on the preview info which might have slight variations or spaces
    # The preview says: "Temperature", "Weight(%)", "D.W(%/C)"
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        col_lower = str(col).lower()
        if 'temperature' in col_lower:
            temp_col = col
        elif 'weight' in col_lower:
            weight_col = col
        elif 'd.w' in col_lower or 'dtg' in col_lower:
            dtg_col = col
            
    if temp_col is None or weight_col is None or dtg_col is None:
        # Fallback: try reading raw and manually setting headers if name matching fails
        df_raw = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=None)
        # Find the row that looks like headers
        # Preview says header_rows: [1], so row index 1 (2nd row) is header.
        # Data starts at row 2 (index 2).
        header_row_idx = 1
        data_start_idx = 2
        
        if len(df_raw) > header_row_idx:
             headers = df_raw.iloc[header_row_idx].values
             df = df_raw.iloc[data_start_idx:].reset_index(drop=True)
             
             new_headers = []
             for h in headers:
                 if isinstance(h, str):
                     new_headers.append(h.strip())
                 else:
                     new_headers.append(str(h))
             df.columns = new_headers
             
             for col in df.columns:
                 col_lower = str(col).lower()
                 if 'temperature' in col_lower:
                     temp_col = col
                 elif 'weight' in col_lower:
                     weight_col = col
                 elif 'd.w' in col_lower or 'dtg' in col_lower:
                     dtg_col = col

    # Ensure numeric types
    df[temp_col] = pd.to_numeric(df[temp_col], errors='coerce')
    df[weight_col] = pd.to_numeric(df[weight_col], errors='coerce')
    df[dtg_col] = pd.to_numeric(df[dtg_col], errors='coerce')
    
    # Drop NaNs if any occurred during conversion
    df = df.dropna(subset=[temp_col, weight_col, dtg_col])

    temp = df[temp_col]
    weight = df[weight_col]
    dtg = df[dtg_col]
    
    ax = axes[idx]
    ax2 = ax.twinx()
    
    # y1: Weight(%) - black solid line with linewidth 2.5
    line1 = ax.plot(temp, weight, color='black', linewidth=2.5, label='Weight (%)')
    ax.set_ylabel('Weight (%)', fontsize=12)
    ax.set_xlabel('Temperature (°C)', fontsize=12)
    
    # y2: D.W(%/C) - dashed line
    line2 = ax2.plot(temp, dtg, linestyle='--', color=colors_y2[idx % len(colors_y2)], linewidth=1.5, label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', fontsize=12)
    
    ax.set_title(f'TGA - {sheet_name}', fontsize=13)
    
    # Combine legends
    lines = line1 + line2
    labels = [l.get_label() for l in lines]
    ax.legend(lines, labels, loc='best', fontsize=10)
    
    ax.grid(True, linestyle=':', alpha=0.5)

plt.tight_layout()
out_path = os.path.join(OUT_DIR, "TGA_plot.png")
plt.savefig(out_path, dpi=300, bbox_inches='tight')
plt.close()
```
