```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 파일 경로 설정
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B15_r2/out"

# 출력 디렉토리가 없으면 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 시트 이름 목록
sheets = ['PE', 'PVC', 'PET']

# 그래프 설정
fig, ax1 = plt.subplots(figsize=(10, 6))

# 색상 정의
colors = {
    'PE': '#1f77b4',   # Blue
    'PVC': '#ff7f0e',  # Orange
    'PET': '#2ca02c'   # Green
}

# 각 시트 데이터를 읽어와서 겹쳐 그리기
for sheet in sheets:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    
    # TGA 데이터: Temperature (x), Weight (%) (y1)
    temp = df['Temperature (°C)']
    weight = df['Weight (%)']
    
    # Primary axis (TGA Weight %)
    ax1.plot(temp, weight, label=f'{sheet} - Weight %', color=colors[sheet], linewidth=2)

# Primary axis 라벨 및 범위 설정
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12, color='black')
ax1.tick_params(axis='y', labelcolor='black')
ax1.grid(True, linestyle='--', alpha=0.5)

# Secondary axis (DTG) 생성
ax2 = ax1.twinx()

for sheet in sheets:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    temp = df['Temperature (°C)']
    dtg = df['Deriv. Weight (%/°C)']
    
    # DTG는 음수값이므로 시각적 가독성을 위해 부호를 반전하여 plotting하는 경우가 많지만, 
    # 요청사항에 명시되지 않았으므로 원본 값 그대로 사용하되, 범례에서 구분 가능하도록 점선으로 표시하거나 
    # 단순히 겹치지 않도록 처리할 수 있음. 여기서는 명확성을 위해 DTG도 실선으로 그리되, 
    # 사용자가 "겹쳐 그려달라"고 했으므로 TGA와 DTG 모두 포함하되, 
    # 보통 TGA 논문에서는 TGA는 실선, DTG는 점선 또는 다른 축을 사용함.
    # 요청이 "세 시트 겹쳐 그리고" 이므로, 가장 중요한 TGA 곡선을 중심으로 하되, 
    # 만약 DTG도 포함해야 한다면 secondary axis를 사용하는 것이 일반적임.
    # 하지만 요청이 간단하므로, 우선 TGA(무게%)만 주요 곡선으로 보고, 
    # 필요시 DTG는 별도 레이어로 생각할 수 있음. 
    # 그러나 데이터 미리보기에 Deriv Weight가 있으므로, 이를 포함하는 것이 더 완전한 TGA 플롯임.
    # 일반적으로 TGA plot은 TGA와 DTG를 함께 보여주는 것이 표준임.
    
    # DTG 값을 양수로 변환하여 오른쪽 축에 맞추거나, 원본 유지. 
    # 많은 경우 DTG는 음의 피크를 가지므로, 오른쪽 축을 뒤집거나 절대값을 쓰기도 함.
    # 여기서는 원본 값을 사용하되, 오른쪽 축을 사용함.
    ax2.plot(temp, dtg, label=f'{sheet} - DTG', color=colors[sheet], linestyle='--', linewidth=1.5, alpha=0.8)

# Secondary axis 라벨 설정
ax2.set_ylabel('Deriv. Weight (%/°C)', fontsize=12, color='gray')
ax2.tick_params(axis='y', labelcolor='gray')

# 범례 설정: 그래프 바깥 오른쪽에 위치
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()

# 모든 핸들과 레이블을 결합
all_lines = lines1 + lines2
all_labels = labels1 + labels2

# 범례를 그래프 외부 오른쪽 상단에 배치
ax1.legend(all_lines, all_labels, loc='upper left', bbox_to_anchor=(1.05, 1), borderaxespad=0., title="Polymer Types")

# 레이아웃 조정하여 범례가 잘리거나 하지 않도록 함
plt.tight_layout(rect=[0, 0, 0.85, 1]) # 오른쪽 여백 확보

# 파일 저장
output_path = os.path.join(OUT_DIR, 'tga_combined.png')
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
