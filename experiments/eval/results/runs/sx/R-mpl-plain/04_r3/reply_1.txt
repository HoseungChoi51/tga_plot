```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 설정
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/04_r3/out"
SHEET_NAME = "1006R1-V normal"
TITLE_TEXT = "CNT 산화 거동"
TITLE_SIZE = 18
TITLE_COLOR = "navy"

# 출력 디렉토리가 없으면 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 데이터 읽기
df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME, header=1)

# 열 이름 확인 및 매핑 (미리보기 기준: B=Temperature, C=Weight(%), D=D.W(%/C))
# 실제 컬럼명이 공백이나 특수문자를 포함할 수 있으므로 strip 처리
cols = df.columns.tolist()
col_temp = None
col_weight = None
col_dtg = None

for c in cols:
    if 'temperature' in str(c).lower():
        col_temp = c
    elif 'weight' in str(c).lower() and '%' in str(c):
        col_weight = c
    elif 'd.w' in str(c).lower() or 'dtg' in str(c).lower():
        col_dtg = c

# 만약 자동 인식이 실패하면 위치 기반 접근 (B, C, D -> index 1, 2, 3)
if col_temp is None: col_temp = df.columns[1]
if col_weight is None: col_weight = df.columns[2]
if col_dtg is None: col_dtg = df.columns[3]

# 데이터 추출
temp = df[col_temp].values
weight = df[col_weight].values
dtg = df[col_dtg].values

# 그래프 그리기
fig, ax1 = plt.subplots(figsize=(10, 6))

# TGA 곡선 (Weight %) - 왼쪽 축
ax1.plot(temp, weight, color='blue', label='Weight (%)')
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color='blue', fontsize=12)
ax1.tick_params(axis='y', labelcolor='blue')
ax1.grid(True, linestyle='--', alpha=0.5)

# DTG 곡선 (D.W %/C) - 오른쪽 축
ax2 = ax1.twinx()
ax2.plot(temp, dtg, color='red', label='DTG (%/°C)')
ax2.set_ylabel('DTG (%/°C)', color='red', fontsize=12)
ax2.tick_params(axis='y', labelcolor='red')

# 제목 설정
plt.title(TITLE_TEXT, fontsize=TITLE_SIZE, color=TITLE_COLOR)

# 범례 추가
lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()
ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='upper right')

# 레이아웃 조정
fig.tight_layout()

# 저장
output_path = os.path.join(OUT_DIR, "tga_normal.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
