```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set global font and color settings
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['text.color'] = 'gray'
plt.rcParams['axes.labelcolor'] = 'gray'
plt.rcParams['xtick.color'] = 'gray'
plt.rcParams['ytick.color'] = 'gray'
plt.rcParams['font.size'] = 13

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/19_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

fig, axes = plt.subplots(len(sheet_names), 1, figsize=(8, 4 * len(sheet_names)), sharex=True)

if len(sheet_names) == 1:
    axes = [axes]

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    
    # Identify columns based on the preview structure
    # Assuming column names are consistent or we can map by position/name from preview
    # From preview: B=Temperature, C=Weight(%), D=D.W(%/C)
    temp_col = 'Temperature'
    weight_col = 'Weight(%)'
    dtg_col = 'D.W(%/C)'
    
    ax1 = axes[i]
    ax2 = ax1.twinx()
    
    # Plot TGA (Weight %)
    line1, = ax1.plot(df[temp_col], df[weight_col], label='TGA', color='blue')
    ax1.set_ylabel('Weight (%)', color='blue')
    ax1.tick_params(axis='y', labelcolor='blue')
    
    # Plot DTG (D.W %/C)
    line2, = ax2.plot(df[temp_col], df[dtg_col], label='DTG', color='red')
    ax2.set_ylabel('dW/dT (%/°C)', color='red')
    ax2.tick_params(axis='y', labelcolor='red')
    
    # Title for each subplot
    ax1.set_title(f'TGA/DTG Analysis - {sheet_name}')
    
    # Combine legends
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='best')

# Set common x-label
axes[-1].set_xlabel('Temperature (°C)')

plt.tight_layout()
output_path = os.path.join(OUT_DIR, "tga_plots.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
