```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/CNT/Plot/eval/results/runs/sx/R-mpl-plain/07_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets from the Excel file
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

fig, ax1 = plt.subplots(figsize=(8, 6))

colors = ['#1f77b4', '#ff7f0e', '#2ca02c']
labels = []

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    
    # Identify columns based on common TGA naming conventions or position
    # From preview: B=Temperature, C=Weight(%), D=D.W(%/C)
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        if 'temperature' in col.lower():
            temp_col = col
        elif 'weight' in col.lower() and '%' in col:
            weight_col = col
        elif 'd.w' in col.lower() or 'dtg' in col.lower():
            dtg_col = col
            
    # Fallback to positional if names don't match well
    if temp_col is None:
        temp_col = df.columns[1]
    if weight_col is None:
        weight_col = df.columns[2]
    if dtg_col is None:
        dtg_col = df.columns[3]
        
    x = df[temp_col]
    y_weight = df[weight_col]
    y_dtg = df[dtg_col]
    
    label_short = sheet_name.split('-')[-1] if '-' in sheet_name else sheet_name
    labels.append(label_short)
    
    # Plot Weight (%) on primary axis
    line_w, = ax1.plot(x, y_weight, color=colors[i], linewidth=1.5, label=f'{label_short} Weight')
    
    # Create secondary axis for DTG
    ax2 = ax1.twinx()
    line_d, = ax2.plot(x, y_dtg, color=colors[i], linestyle='--', linewidth=1.0, alpha=0.7, label=f'{label_short} DTG')

ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12)
ax2.set_ylabel('DTG (%/°C)', fontsize=12)

# Combine handles and labels for a single legend
lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()

all_lines = lines_1 + lines_2
all_labels = labels_1 + labels_2

# Place legend at upper right with font size 9
legend = ax1.legend(all_lines, all_labels, loc='upper right', fontsize=9, frameon=True)

plt.title('TGA Analysis Results', fontsize=14)
plt.tight_layout()

output_path = os.path.join(OUT_DIR, "tga_plot.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
