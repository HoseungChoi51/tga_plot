```python
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os

# 설정
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/12_r1/out"

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 데이터 읽기
xls = pd.ExcelFile(DATA_FILE)
sheets = xls.sheet_names

# 그림 생성 (3개의 시트를 각각 서브플롯으로 표시하거나, 하나의 그래프에 겹쳐서 표시할 수 있음)
# TGA 데이터는 보통 Weight(%)와 DTG(D.W)를 함께 보는 것이 일반적임.
# 여기서는 각 시트별로 Weight(%) 곡선을 그리고, 450C에 선을 그리는 것을 목표로 함.
# 만약 모든 데이터를 한 그래프에 비교하고 싶다면, 색상과 라벨로 구분해야 함.
# 요청이 "세로 보조선 넣고 '산화 시작'이라고 써줘" 이므로, 일반적인 TGA 플롯 형식을 따름.

fig, axes = plt.subplots(len(sheets), 1, figsize=(8, 6 * len(sheets)), sharex=True)

if len(sheets) == 1:
    axes = [axes]

for idx, sheet_name in enumerate(sheets):
    ax = axes[idx]
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    
    # 열 이름 확인 및 데이터 추출
    temp_col = 'Temperature'
    weight_col = 'Weight(%)'
    dtg_col = 'D.W(%/C)'
    
    x = df[temp_col]
    y_weight = df[weight_col]
    y_dtg = df[dtg_col]
    
    # 메인 축: Weight (%)
    color_main = '#1f77b4'
    line_w, = ax.plot(x, y_weight, color=color_main, linewidth=1.5, label='Weight (%)')
    ax.set_ylabel('Weight (%)', fontsize=12, color=color_main)
    ax.tick_params(axis='y', labelcolor=color_main)
    ax.grid(True, linestyle='--', alpha=0.5)
    
    # 보조 축: DTG (D.W %/C) - 필요에 따라 생략 가능하지만 TGA 분석에서는 유용함
    # 요청사항에는 명시되지 않았으나, 일반적으로 TGA 그래프에 포함됨. 
    # 단순함을 위해 Weight만 강조하되, 필요하면 추가. 여기서는 Weight에 집중하되, 시각적 완결성을 위해 DTG도 얇게 표시해봄.
    # 하지만 요청이 "보조선"과 "텍스트"에 집중되어 있으므로, 주의를 분산시키지 않도록 Weight 곡선 위주로 구성.
    # 실제로는 DTG가 산화 시작점을 더 명확히 보여주지만, 요청은 450도에 고정된 선임.
    
    # 450 °C 세로 보조선
    target_temp = 450
    ax.axvline(x=target_temp, color='red', linestyle='--', linewidth=1.5, label='450 °C')
    
    # 텍스트 추가: '산화 시작'
    # 텍스트 위치는 그래프 상단 근처나 적절한 빈 공간에 배치
    # Y축 범위 계산하여 텍스트 위치 조정
    ymin, ymax = ax.get_ylim()
    text_y_pos = ymin + (ymax - ymin) * 0.85  # 상단에서 15% 아래
    
    ax.text(target_temp + 10, text_y_pos, '산화 시작', 
            color='red', fontsize=12, fontweight='bold',
            verticalalignment='center', horizontalalignment='left')
            
    # 제목 및 라벨
    ax.set_title(f'TGA Analysis: {sheet_name}', fontsize=14)
    if idx == len(sheets) - 1:
        ax.set_xlabel('Temperature (°C)', fontsize=12)
        
    # 범례
    ax.legend(loc='lower left')

plt.tight_layout()

# 저장
output_filename = os.path.join(OUT_DIR, "tga_plot_450C_marker.png")
plt.savefig(output_filename, dpi=300, bbox_inches='tight')
plt.close()
```
