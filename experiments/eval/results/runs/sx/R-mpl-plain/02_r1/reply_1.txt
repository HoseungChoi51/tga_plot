```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/CNT/Plot/eval/results/runs/sx/R-mpl-plain/02_r1/out"

# Ensure output directory exists
os.makedirs(OUT_DIR, exist_ok=True)

# Read the specific sheet
sheet_name = "1006R1-V normal"
df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)

# Extract columns based on the preview info
# B: Temperature (X-axis)
# C: Weight(%) (Y1-axis)
# D: D.W(%/C) (Y2-axis)
temperature = df['Temperature'].values
weight = df['Weight(%)'].values
dw = df['D.W(%/C)'].values

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(10, 6))

# Plot Weight (%) on primary Y-axis (left)
color1 = 'tab:blue'
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
line1 = ax1.plot(temperature, weight, color=color1, label='Weight (%)')
ax1.tick_params(axis='y', labelcolor=color1)
ax1.grid(True, linestyle='--', alpha=0.5)

# Set X-axis ticks from 0 to max temperature with step 200
max_temp = np.max(temperature)
x_ticks = np.arange(0, int(max_temp) + 200, 200)
ax1.set_xticks(x_ticks)
ax1.set_xlim(left=0) # Start from 0 as requested

# Create secondary Y-axis for Deriv.Weight
ax2 = ax1.twinx()
color2 = 'tab:red'
ax2.set_ylabel('Deriv. Weight (%/°C)', color=color2, fontsize=12)
line2 = ax2.plot(temperature, dw, color=color2, label='Deriv. Weight (%/°C)')
ax2.tick_params(axis='y', labelcolor=color2)

# Set Y2-axis ticks with step 0.2
min_dw = np.min(dw)
max_dw = np.max(dw)
# Generate ticks centered around 0 or covering the range nicely with 0.2 steps
# Find a nice start point close to min_dw that is a multiple of 0.2
start_y2 = np.floor(min_dw / 0.2) * 0.2
end_y2 = np.ceil(max_dw / 0.2) * 0.2
y2_ticks = np.arange(start_y2, end_y2 + 0.2, 0.2)
ax2.set_yticks(y2_ticks)

# Combine lines and labels for legend
lines = line1 + line2
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='upper right')

# Title
plt.title(f'TGA Analysis: {sheet_name}', fontsize=14)

# --- Annotation 1: Max value of Deriv.Weight ---
# Find index of maximum derivative weight (note: usually we look for peak degradation rate, which is most negative, 
# but the prompt says "max값". In TGA DTG, the "peak" loss is often the minimum value (most negative). 
# However, strictly "max" means the highest numerical value. 
# Let's check the data context: D.W ranges from -1.027 to 0.002528. The max is ~0.0025. The min is -1.027.
# Usually, users want the main decomposition peak. If they say "max", they might mean the magnitude of the drop or literally the max number.
# Given the context of TGA, the significant feature is the deep negative peak. But I will follow the literal instruction "max값" first.
# If the user meant the main degradation peak, it would be the minimum. 
# Let's assume literal max for now, but if the graph looks weird, consider the absolute max? 
# Actually, looking at typical requests, "Max Deriv Weight" often refers to the peak height of the curve. Since the curve goes down, the "height" of the peak is the absolute value. 
# However, without clarification, I will plot the literal mathematical maximum. 
# Wait, let's re-read carefully: "Deriv.Weight 의 max값을 표시해주고". 
# If I mark the near-zero positive noise, it's useless. The significant event is the minimum (-1.027). 
# I will mark the **minimum** value (the main degradation peak) as it is the standard interpretation of "peak" in this context, 
# OR I can mark the one with the largest absolute deviation from zero. 
# Let's stick to the literal "max" (highest y-value) but also consider that maybe they meant the peak magnitude. 
# Let's calculate both and pick the one that makes sense visually? No, strict adherence. 
# BUT, in many engineering contexts, "Max Rate of Loss" implies the steepest slope (largest magnitude). 
# Let's assume the user wants the main degradation peak marked. That is the MINIMUM value of DW. 
# I will annotate the Minimum value (main peak) because annotating a tiny positive noise spike is rarely useful.
# Correction: I will annotate the value with the largest absolute magnitude (the main peak).

idx_max_abs_dw = np.argmax(np.abs(dw))
temp_at_peak = temperature[idx_max_abs_dw]
val_at_peak = dw[idx_max_abs_dw]

# Annotate the main peak (largest absolute derivative)
ax2.annotate(f'Max |DW|: {val_at_peak:.3f}\nTemp: {temp_at_peak:.1f}°C',
             xy=(temp_at_peak, val_at_peak),
             xytext=(temp_at_peak + 50, val_at_peak - 0.2), # Offset slightly
             arrowprops=dict(facecolor='black', shrink=0.05),
             fontsize=10,
             ha='center',
             va='top')

# --- Annotation 2: Residue at Temperature closest to 800 ---
target_temp = 800
# Find index where temperature is closest to 800
idx_800 = np.argmin(np.abs(temperature - target_temp))
actual_temp_800 = temperature[idx_800]
residue_weight = weight[idx_800]

# Annotate residue
ax1.annotate(f'Residue @ ~{actual_temp_800:.0f}°C:\n{residue_weight:.2f}%',
             xy=(actual_temp_800, residue_weight),
             xytext=(actual_temp_800 - 100, residue_weight + 20), # Offset left and up
             arrowprops=dict(facecolor='green', shrink=0.05),
             fontsize=10,
             ha='right',
             va='bottom',
             color='green')

# Save the plot
output_path = os.path.join(OUT_DIR, f"TGA_{sheet_name.replace(' ', '_')}.png")
plt.tight_layout()
plt.savefig(output_path, dpi=300)
plt.close()

print(f"Saved plot to {output_path}")
```
