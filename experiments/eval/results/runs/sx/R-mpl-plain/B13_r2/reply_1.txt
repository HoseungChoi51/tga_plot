```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import find_peaks

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B13_r2/out"

# Read PVC sheet
df_pvc = pd.read_excel(DATA_FILE, sheet_name="PVC")

# Extract columns
temp = df_pvc["Temperature (°C)"].values
weight = df_pvc["Weight (%)"].values
deriv_weight = df_pvc["Deriv. Weight (%/°C)"].values

# Find peaks in derivative weight (negative peaks indicate mass loss steps)
# We look for significant negative peaks. The derivative is mostly negative or zero.
# A "sharp drop" corresponds to a minimum in the derivative curve (most negative value).
# Using find_peaks on the inverted derivative to find minima as peaks.
inverted_deriv = -deriv_weight
peaks, properties = find_peaks(inverted_deriv, prominence=0.5) # Adjust prominence if needed

# Get the temperatures and weights at these peaks
peak_temps = temp[peaks]
peak_weights = weight[peaks]
peak_deriv_vals = deriv_weight[peaks]

# Sort by temperature just in case
sorted_indices = np.argsort(peak_temps)
peak_temps_sorted = peak_temps[sorted_indices]
peak_weights_sorted = peak_weights[sorted_indices]
peak_deriv_vals_sorted = peak_deriv_vals[sorted_indices]

# Plotting
fig, ax1 = plt.subplots(figsize=(10, 6))

color1 = 'tab:blue'
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
ax1.plot(temp, weight, color=color1, label='Weight (%)')
ax1.tick_params(axis='y', labelcolor=color1)
ax1.grid(True, linestyle='--', alpha=0.5)

ax2 = ax1.twinx()
color2 = 'tab:red'
ax2.set_ylabel('Deriv. Weight (%/°C)', color=color2, fontsize=12)
ax2.plot(temp, deriv_weight, color=color2, label='Deriv. Weight (%/°C)')
ax2.tick_params(axis='y', labelcolor=color2)

# Mark the identified drop points
for i, (t, w, d) in enumerate(zip(peak_temps_sorted, peak_weights_sorted, peak_deriv_vals_sorted)):
    # Annotate on the weight plot
    ax1.annotate(f'{t:.1f} °C', 
                 xy=(t, w), 
                 xytext=(t + 20, w + 10),
                 arrowprops=dict(facecolor='black', shrink=0.05),
                 fontsize=10, ha='left')
    
    # Add a marker on the derivative plot for clarity
    ax2.plot(t, d, 'o', color='black', markersize=8)

ax1.set_title('TGA Data for PVC with Key Decomposition Temperatures', fontsize=14)
fig.tight_layout()

# Save figure
output_path = f"{OUT_DIR}/pvc_tga_drops.png"
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
