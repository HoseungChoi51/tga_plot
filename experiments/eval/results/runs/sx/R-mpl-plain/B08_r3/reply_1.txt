```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 파일 경로 및 출력 디렉토리 설정
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B08_r3/out"

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 데이터 읽기 (Raw 시트)
df = pd.read_excel(DATA_FILE, sheet_name='Raw')

# 컬럼명 확인 및 매핑
# 예상되는 컬럼: 'Time (min)', 'Temperature (°C)', 'Weight (%)', 'Deriv. Weight (%/°C)'
temp_col = 'Temperature (°C)'
weight_col = 'Weight (%)'

if temp_col not in df.columns or weight_col not in df.columns:
    # 만약 컬럼명이 정확히 일치하지 않는 경우 첫 번째와 세 번째 컬럼을 사용하는 등 유연성 확보 가능하나,
    # 미리보기에 따르면 정확한 이름이 있음.
    pass

# 그래프 생성
fig, ax1 = plt.subplots(figsize=(8, 6))

# TGA 곡선 (무게 %)
ax1.plot(df[temp_col], df[weight_col], color='blue', label='Weight (%)')
ax1.set_xlabel('Temperature (°C)')
ax1.set_ylabel('Weight (%)', color='blue')
ax1.tick_params(axis='y', labelcolor='blue')

# 무게 50% 위치에 가로 보조선 추가
ax1.axhline(y=50, color='gray', linestyle='--', linewidth=1)

# 오른쪽 위에 메모 텍스트 추가
ax1.text(0.95, 0.95, '공기, 10 °C/min', transform=ax1.transAxes, 
         fontsize=12, verticalalignment='top', horizontalalignment='right',
         bbox=dict(boxstyle='round,pad=0.5', facecolor='white', alpha=0.8))

# 축 범위 설정 (데이터 범위에 맞춰 약간 여유를 둠)
ax1.set_xlim(df[temp_col].min(), df[temp_col].max())
ax1.set_ylim(0, 110) # 무게가 0~100% 사이이므로 0~110 정도로 설정하여 보조선과 라벨이 잘 보이도록 함

# 그리드 표시
ax1.grid(True, linestyle=':', alpha=0.6)

# 레이아웃 조정
plt.tight_layout()

# 저장
output_path = os.path.join(OUT_DIR, 'tga_raw_annotated.png')
plt.savefig(output_path, dpi=300)
plt.close()
```
