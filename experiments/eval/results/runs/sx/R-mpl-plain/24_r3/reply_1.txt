```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 데이터 파일 경로 및 출력 디렉토리 설정
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/24_r3/out"

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 데이터 읽기
df = pd.read_excel(DATA_FILE, sheet_name="CNT-01")

# 열 이름 확인 및 변수 할당 (미리보기 기준)
x_col = "Temperature (°C)"
y1_col = "Weight (%)"
y2_col = "Heat Flow (W/g)"

# 그래프 생성
fig, ax1 = plt.subplots(figsize=(8, 6))

# Y1축: Weight (%)
color1 = 'tab:red'
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel(y1_col, color=color1, fontsize=12)
ax1.plot(df[x_col], df[y1_col], color=color1, label=y1_col)
ax1.tick_params(axis='y', labelcolor=color1)

# Y2축: Heat Flow (W/g) - 요청에 따라 y2로 설정
ax2 = ax1.twinx()
color2 = 'tab:blue'
ax2.set_ylabel('Heat flow (W/g)', color=color2, fontsize=12) # 라벨 텍스트는 사용자 요청대로
ax2.plot(df[x_col], df[y2_col], color=color2, label=y2_col)
ax2.tick_params(axis='y', labelcolor=color2)

# 제목 및 범례
plt.title('TGA / DSC Data Plot', fontsize=14)
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='upper right')

# 그리드 표시
ax1.grid(True, linestyle='--', alpha=0.5)

# 레이아웃 조정
plt.tight_layout()

# 출력 디렉토리가 없으면 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 파일 저장
output_path = os.path.join(OUT_DIR, "tga_heat_flow.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
