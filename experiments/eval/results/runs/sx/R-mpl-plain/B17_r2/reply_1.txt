```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B17_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

sheets = ["PE", "PVC", "PET"]
colors = ['#1f77b4', '#ff7f0e', '#2ca02c']

fig, ax = plt.subplots(figsize=(8, 6))

for i, sheet in enumerate(sheets):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    
    # Get temperature and weight columns
    temp_col = None
    weight_col = None
    
    for col in df.columns:
        if 'temperature' in col.lower():
            temp_col = col
        elif 'weight' in col.lower() and 'deriv' not in col.lower():
            weight_col = col
            
    if temp_col is None or weight_col is None:
        continue
        
    temps = df[temp_col]
    weights = df[weight_col]
    
    # Offset: shift each curve by 20% * i (so PE=0, PVC=20, PET=40)
    offset = i * 20.0
    shifted_weights = weights + offset
    
    ax.plot(temps, shifted_weights, label=f"{sheet} (+{offset}%)", color=colors[i], linewidth=1.5)

ax.set_xlabel("Temperature (°C)", fontsize=12)
ax.set_ylabel("Weight (%)", fontsize=12)
ax.set_title("TGA Curves with Vertical Offsets", fontsize=14)
ax.legend(loc='upper right')
ax.grid(True, linestyle='--', alpha=0.5)

plt.tight_layout()
output_path = os.path.join(OUT_DIR, "tga_offsets.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
