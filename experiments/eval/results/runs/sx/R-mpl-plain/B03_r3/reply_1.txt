```python
import pandas as pd
import matplotlib.pyplot as plt
from pathlib import Path

# 데이터 파일 경로 및 출력 디렉토리 설정
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/rheology.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B03_r3/out"

# 출력 디렉토리가 없으면 생성
Path(OUT_DIR).mkdir(parents=True, exist_ok=True)

# 데이터 로드
df = pd.read_csv(DATA_FILE)

# 컬럼명 확인 (미리보기 정보 기반)
# col A: Angular frequency (rad/s)
# col B: Storage modulus G' (Pa)
# col C: Loss modulus G'' (Pa)

# 컬럼명이 정확히 일치하지 않을 수 있으므로 위치 기반으로 접근하거나 이름 매칭 시도
cols = df.columns.tolist()

# 일반적인 rheology 데이터 컬럼 인덱스 가정 (A=0, B=1, C=2)
freq_col = cols[0]
g_prime_col = cols[1]
g_double_prime_col = cols[2]

# 데이터 추출
freq = df[freq_col].values
g_prime = df[g_prime_col].values
g_double_prime = df[g_double_prime_col].values

# 그래프 설정
fig, ax1 = plt.subplots(figsize=(8, 6))

# 왼쪽 축: G'
ax1.plot(freq, g_prime, color='blue', marker='o', linestyle='-', label="G' (Storage Modulus)")
ax1.set_xlabel('Angular Frequency (rad/s)', fontsize=12)
ax1.set_ylabel("Storage Modulus G' (Pa)", color='blue', fontsize=12)
ax1.tick_params(axis='y', labelcolor='blue')
ax1.set_xscale('log')
ax1.set_yscale('log')

# 오른쪽 축: G''
ax2 = ax1.twinx()
ax2.plot(freq, g_double_prime, color='red', marker='s', linestyle='--', label="G'' (Loss Modulus)")
ax2.set_ylabel("Loss Modulus G'' (Pa)", color='red', fontsize=12)
ax2.tick_params(axis='y', labelcolor='red')
ax2.set_yscale('log')

# 제목 추가
plt.title('Rheological Properties vs Angular Frequency', fontsize=14)

# 범례 추가
lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()
ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='best')

# 그리드 표시 (로그 스케일에서 보기 좋게)
ax1.grid(True, which="both", ls="--", alpha=0.5)

# 레이아웃 조정
plt.tight_layout()

# 저장
output_path = Path(OUT_DIR) / "rheology_plot.png"
plt.savefig(output_path, dpi=300)
plt.close()
```
