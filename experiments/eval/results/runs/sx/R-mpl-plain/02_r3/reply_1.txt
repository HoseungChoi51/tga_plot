```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/02_r3/out"

# Ensure output directory exists
os.makedirs(OUT_DIR, exist_ok=True)

# Read the specific sheet
sheet_name = "1006R1-V normal"
df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)

# Extract columns based on description: B->X (Temperature), C->Y1 (Weight%), D->Y2 (D.W %/C)
# Assuming column names are exactly as in preview or we can use positional if needed.
# Preview says: A: Time, B: Temperature, C: Weight(%), D: D.W(%/C)
temp_col = df.columns[1]  # Temperature
weight_col = df.columns[2]  # Weight(%)
dw_col = df.columns[3]      # D.W(%/C)

x_data = df[temp_col].values
y1_data = df[weight_col].values
y2_data = df[dw_col].values

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(10, 7))

# Plot TGA curve (Weight vs Temperature)
color1 = 'tab:red'
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
line1, = ax1.plot(x_data, y1_data, color=color1, label='Weight (%)')
ax1.tick_params(axis='y', labelcolor=color1)
ax1.set_xlim(0, 900) # Extend to start at 0 as requested for ticks logic, though data starts at 30
ax1.set_ylim(0, 105)

# Configure X-axis ticks: 0, 200, 400, ...
ax1.set_xticks(np.arange(0, 901, 200))

# Find Residue at ~800°C
target_temp = 800
idx_800 = np.argmin(np.abs(x_data - target_temp))
residue_weight = y1_data[idx_800]
actual_temp_800 = x_data[idx_800]

# Annotate Residue
ax1.annotate(f'Residue {residue_weight:.2f}%', 
             xy=(actual_temp_800, residue_weight), 
             xytext=(actual_temp_800 + 50, residue_weight + 10),
             arrowprops=dict(facecolor='black', shrink=0.05),
             fontsize=10, color='black')

# Create second axis for DTG (Derivative Weight)
ax2 = ax1.twinx()
color2 = 'tab:blue'
ax2.set_ylabel('Deriv. Weight (%/°C)', color=color2, fontsize=12)
line2, = ax2.plot(x_data, y2_data, color=color2, linestyle='--', label='Deriv. Weight (%/°C)')
ax2.tick_params(axis='y', labelcolor=color2)

# Configure Y2 axis ticks: step 0.2
# Determine range for Y2 to set appropriate ticks
min_y2 = np.min(y2_data)
max_y2 = np.max(y2_data)
# Generate ticks from floor(min/0.2)*0.2 to ceil(max/0.2)*0.2 with step 0.2
start_tick = np.floor(min_y2 / 0.2) * 0.2
end_tick = np.ceil(max_y2 / 0.2) * 0.2
ticks_y2 = np.arange(start_tick, end_tick + 0.1, 0.2)
ax2.set_yticks(ticks_y2)

# Find Max Deriv.Weight
max_dw_val = np.max(y2_data)
idx_max_dw = np.argmax(y2_data)
temp_at_max_dw = x_data[idx_max_dw]

# Annotate Max Deriv.Weight
ax2.annotate(f'Max Deriv. Weight\n{max_dw_val:.4f} %/°C @ {temp_at_max_dw:.1f}°C', 
             xy=(temp_at_max_dw, max_dw_val), 
             xytext=(temp_at_max_dw + 50, max_dw_val - 0.2),
             arrowprops=dict(facecolor='black', shrink=0.05),
             fontsize=10, color='black')

# Add Legend
lines = [line1, line2]
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='upper right')

# Title
plt.title(f'TGA & DTG Curve ({sheet_name})', fontsize=14)

# Grid
ax1.grid(True, linestyle=':', alpha=0.6)

# Save plot
output_path = os.path.join(OUT_DIR, "TGA_normal.png")
plt.tight_layout()
plt.savefig(output_path, dpi=300)
plt.close()
```
