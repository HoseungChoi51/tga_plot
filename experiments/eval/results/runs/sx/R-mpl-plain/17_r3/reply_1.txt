```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/17_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets from the Excel file
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

fig, ax1 = plt.subplots(figsize=(8, 5))

colors = ['#1f77b4', '#ff7f0e', '#2ca02c']

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(xls, sheet_name=sheet_name)
    
    # Identify columns based on the preview info
    # Assuming standard TGA format: Temperature, Weight(%), D.W(%/C)
    temp_col = None
    weight_col = None
    dw_col = None
    
    for col in df.columns:
        col_str = str(col).lower()
        if 'temperature' in col_str or 'temp' in col_str:
            temp_col = col
        elif 'weight' in col_str and '%' in col_str:
            weight_col = col
        elif 'd.w' in col_str or 'dw' in col_str or ('derivative' in col_str and 'weight' in col_str):
            dw_col = col
            
    if temp_col is None:
        temp_col = df.columns[1] # Fallback to second column usually temperature
    if weight_col is None:
        weight_col = df.columns[2] # Fallback to third column usually weight
    if dw_col is None:
        dw_col = df.columns[3] # Fallback to fourth column usually derivative
        
    x = df[temp_col]
    y_weight = df[weight_col]
    y_dw = df[dw_col]
    
    color = colors[i % len(colors)]
    
    # Plot Weight (%) on primary axis (left)
    ax1.plot(x, y_weight, label=f'{sheet_name} - Weight', color=color, linestyle='-')
    
    # Create secondary axis for Derivative Weight
    ax2 = ax1.twinx()
    ax2.plot(x, y_dw, label=f'{sheet_name} - dW/dT', color=color, linestyle='--', alpha=0.6)

ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12, color='black')
ax2.set_ylabel('Derivative Weight (%/°C)', fontsize=12, color='gray')

ax1.tick_params(axis='y', labelcolor='black')
ax2.tick_params(axis='y', labelcolor='gray')

# Combine legends
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='best', fontsize=9)

ax1.grid(True, linestyle=':', alpha=0.5)

plt.title('TGA Analysis Results', fontsize=14)
plt.tight_layout()

output_path = os.path.join(OUT_DIR, "tga_plot.png")
plt.savefig(output_path, dpi=600)
plt.close()
```
