```python
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# Read data
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B12_r2/out"

df = pd.read_csv(DATA_FILE)

# Extract temperature and weight columns
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"

temperature = df[temp_col].values
weight = df[weight_col].values

# Filter for x range 100 to 900
mask = (temperature >= 100) & (temperature <= 900)
temp_filtered = temperature[mask]
weight_filtered = weight[mask]

# Find residue at 900°C - use the last point in the filtered data or interpolate
# Since we want exactly at 900, let's find the closest value or interpolate
idx_900 = np.argmin(np.abs(temperature - 900))
residue_temp = temperature[idx_900]
residue_weight = weight[idx_900]

# Create plot
fig, ax = plt.subplots(figsize=(8, 6))

ax.plot(temp_filtered, weight_filtered, color='black', linewidth=1.5, label='TGA Weight')

# Mark the residue at 900°C
ax.scatter([residue_temp], [residue_weight], color='red', zorder=5, s=50)
ax.annotate(f'Residue: {residue_weight:.2f}%\n@ {residue_temp:.0f}°C', 
            xy=(residue_temp, residue_weight),
            xytext=(residue_temp - 150, residue_weight + 5),
            fontsize=10,
            arrowprops=dict(arrowstyle='->', color='red'),
            color='red')

ax.set_xlabel('Temperature (°C)', fontsize=12)
ax.set_ylabel('Weight (%)', fontsize=12)
ax.set_xlim(100, 900)
ax.grid(True, linestyle='--', alpha=0.7)
ax.legend(loc='upper right', fontsize=10)

plt.tight_layout()

# Save figure
os.makedirs(OUT_DIR, exist_ok=True)
output_path = os.path.join(OUT_DIR, "tga_weight_curve.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
