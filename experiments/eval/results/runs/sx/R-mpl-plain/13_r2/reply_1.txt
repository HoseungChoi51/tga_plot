```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'

# File paths
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/13_r2/out"

# Ensure output directory exists
os.makedirs(OUT_DIR, exist_ok=True)

# Read data from the first sheet
df = pd.read_excel(DATA_FILE, sheet_name=0)

# Extract columns needed for plotting
# X-axis: Temperature (°C) -> Column B
# Y1-axis: Weight (mg) -> Column D
x_col = 'Temperature (°C)'
y1_col = 'Weight (mg)'

# Check if columns exist
if x_col not in df.columns or y1_col not in df.columns:
    raise ValueError(f"Columns '{x_col}' and/or '{y1_col}' not found in the data.")

# Create the plot
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight (mg) on the primary y-axis
ax1.plot(df[x_col], df[y1_col], color='blue', label='Weight (mg)')
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (mg)', fontsize=12, color='blue')
ax1.tick_params(axis='y', labelcolor='blue')

# Optional: Add grid for better readability
ax1.grid(True, linestyle='--', alpha=0.7)

# Adjust layout to prevent clipping
plt.tight_layout()

# Save the figure
output_path = os.path.join(OUT_DIR, "tga_weight_mg.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close(fig)
```
