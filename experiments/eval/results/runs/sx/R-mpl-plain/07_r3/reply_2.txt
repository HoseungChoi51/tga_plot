```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/07_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets
xl_file = pd.ExcelFile(DATA_FILE)
sheet_names = xl_file.sheet_names

fig, ax1 = plt.subplots(figsize=(10, 6))

colors = ['#1f77b4', '#ff7f0e', '#2ca02c']
ax2 = None

for idx, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    
    # Identify columns based on the preview info
    temp_col = 'Temperature'
    weight_col = 'Weight(%)'
    dtg_col = 'D.W(%/C)'
    
    temp = df[temp_col]
    weight = df[weight_col]
    dtg = df[dtg_col]
    
    color = colors[idx % len(colors)]
    
    # Plot Weight (%) on primary axis
    line1 = ax1.plot(temp, weight, label=f'{sheet_name} - Weight', color=color, linestyle='-')
    
    # Create secondary axis for DTG if not exists
    if ax2 is None:
        ax2 = ax1.twinx()
    
    line2 = ax2.plot(temp, dtg, label=f'{sheet_name} - DTG', color=color, linestyle='--')

ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12)
if ax2:
    ax2.set_ylabel('DTG (%/°C)', fontsize=12)

# Calculate global limits safely
all_temps = []
for sheet_name in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    all_temps.append(df['Temperature'])

global_min_temp = min(t.min() for t in all_temps)
global_max_temp = max(t.max() for t in all_temps)

ax1.set_xlim(global_min_temp, global_max_temp)
ax1.set_ylim(0, 105)

# Combine lines and labels for legend
lines = []
labels = []
for ax in [ax1, ax2]:
    if ax is not None:
        lines += ax.get_lines()
        labels += [l.get_label() for l in ax.get_lines()]

# Place legend at upper right with font size 9
ax1.legend(lines, labels, loc='upper right', fontsize=9)

ax1.grid(True, linestyle=':', alpha=0.6)

plt.title('TGA and DTG Curves Comparison', fontsize=14)
plt.tight_layout()

out_path = os.path.join(OUT_DIR, "tga_plot.png")
plt.savefig(out_path, dpi=300)
plt.close()
```
