```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# Define colors and styles based on user request: all text gray, size 13
TEXT_COLOR = 'gray'
FONT_SIZE = 13

# Read data from Excel file
file_path = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
out_dir = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/19_r3/out"

# Ensure output directory exists
os.makedirs(out_dir, exist_ok=True)

# Sheet names
sheets = ['1006R1-V normal', '1006R1-V acid', '1006R1-V 750C']

# Create a figure with subplots for each sheet
fig, axes = plt.subplots(len(sheets), 1, figsize=(10, 4 * len(sheets)))

for i, sheet_name in enumerate(sheets):
    # Header row is index 1 (0-based), so header=1 means the second row is the header
    df = pd.read_excel(file_path, sheet_name=sheet_name, header=1)
    
    # Clean column names just in case of extra spaces or hidden characters
    df.columns = [str(c).strip() for c in df.columns]
    
    # Identify columns by position if names don't match exactly due to formatting issues
    # Based on preview: Col A=Time, B=Temperature, C=Weight(%), D=D.W(%/C)
    # We can use iloc to be safe against header name mismatches
    temp = df.iloc[:, 1]  # Temperature
    weight = df.iloc[:, 2] # Weight(%)
    dtg = df.iloc[:, 3]   # D.W(%/C)
    
    ax = axes[i] if len(sheets) > 1 else axes
    
    # Plot TGA (Weight %)
    line1, = ax.plot(temp, weight, color='black', label='TGA (Weight %)')
    ax.set_xlabel('Temperature (°C)', fontsize=FONT_SIZE, color=TEXT_COLOR)
    ax.set_ylabel('Weight (%)', fontsize=FONT_SIZE, color=TEXT_COLOR)
    ax.tick_params(axis='both', which='major', labelsize=FONT_SIZE, colors=TEXT_COLOR)
    ax.tick_params(axis='both', which='minor', labelsize=FONT_SIZE, colors=TEXT_COLOR)
    
    # Add secondary y-axis for DTG
    ax2 = ax.twinx()
    line2, = ax2.plot(temp, dtg, color='red', linestyle='--', label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', fontsize=FONT_SIZE, color=TEXT_COLOR)
    ax2.tick_params(axis='y', which='major', labelsize=FONT_SIZE, colors=TEXT_COLOR)
    ax2.tick_params(axis='y', which='minor', labelsize=FONT_SIZE, colors=TEXT_COLOR)
    
    # Combine legends
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax.legend(lines, labels, loc='upper right', fontsize=FONT_SIZE, frameon=False)
    
    # Title
    ax.set_title(f'TGA/DTG - {sheet_name}', fontsize=FONT_SIZE + 2, color=TEXT_COLOR, pad=10)
    
    # Grid
    ax.grid(True, linestyle=':', alpha=0.5)

# Adjust layout
plt.tight_layout()

# Save figure
output_file = os.path.join(out_dir, "tga_plots.png")
plt.savefig(output_file, dpi=300, bbox_inches='tight')
plt.close()
```
