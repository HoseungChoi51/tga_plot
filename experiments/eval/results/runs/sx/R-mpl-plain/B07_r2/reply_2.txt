```python
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B07_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read the second sheet (PVC)
df = pd.read_excel(DATA_FILE, sheet_name='PVC')

# Extract columns
temp = df['Temperature (°C)'].values
weight = df['Weight (%)'].values

# Create figure with two subplots sharing x-axis
fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(10, 8), sharex=True)

# Plot Weight (%) on top axis
ax1.plot(temp, weight, color='blue', linewidth=1.5, label='Weight (%)')
ax1.set_ylabel('Weight (%)')
ax1.grid(True, linestyle='--', alpha=0.6)
ax1.legend(loc='upper right')

# Find and mark weights at 300°C and 600°C
def get_weight_at_temp(temp_arr, weight_arr, target_temp):
    idx = np.argmin(np.abs(temp_arr - target_temp))
    return temp_arr[idx], weight_arr[idx]

t300, w300 = get_weight_at_temp(temp, weight, 300)
t600, w600 = get_weight_at_temp(temp, weight, 600)

# Mark points on top plot
ax1.scatter([t300, t600], [w300, w600], color='red', zorder=5, s=50)
ax1.annotate(f'{w300:.2f}%', xy=(t300, w300), xytext=(t300 + 20, w300 + 5),
             arrowprops=dict(arrowstyle='->', color='black'), fontsize=10)
ax1.annotate(f'{w600:.2f}%', xy=(t600, w600), xytext=(t600 + 20, w600 + 5),
             arrowprops=dict(arrowstyle='->', color='black'), fontsize=10)

# Plot Deriv. Weight on bottom axis
deriv_weight = df['Deriv. Weight (%/°C)'].values
ax2.plot(temp, deriv_weight, color='green', linewidth=1.5, label='Deriv. Weight (%/°C)')
ax2.set_xlabel('Temperature (°C)')
ax2.set_ylabel('Deriv. Weight (%/°C)')
ax2.grid(True, linestyle='--', alpha=0.6)
ax2.legend(loc='upper right')

# Mark vertical lines at 300 and 600 for reference on both plots
for ax in [ax1, ax2]:
    ax.axvline(x=300, color='gray', linestyle=':', alpha=0.5)
    ax.axvline(x=600, color='gray', linestyle=':', alpha=0.5)

fig.suptitle('TGA Data: PVC (Sheet 2)', fontsize=14)
plt.tight_layout()

output_path = os.path.join(OUT_DIR, "pvc_tga.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
