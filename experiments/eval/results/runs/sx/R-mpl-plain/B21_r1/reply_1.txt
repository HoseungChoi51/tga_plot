```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B21_r1/out"

# Ensure output directory exists
os.makedirs(OUT_DIR, exist_ok=True)

# Read the 'Raw' sheet
df_raw = pd.read_excel(DATA_FILE, sheet_name='Raw')

# Extract columns based on the preview info
# Col B: Temperature (°C), Col C: Weight (%), Col D: Deriv. Weight (%/°C)
temp = df_raw.iloc[:, 1]
weight = df_raw.iloc[:, 2]
deriv_weight = df_raw.iloc[:, 3]

# Create figure with two subplots sharing x-axis
fig, ax1 = plt.subplots(figsize=(10, 6))

# Plot Weight (%) on primary y-axis
color_w = 'tab:blue'
ax1.set_xlabel('Temperature (°C)')
ax1.set_ylabel('Weight (%)', color=color_w)
line1, = ax1.plot(temp, weight, color=color_w, label='Weight (%)')
ax1.tick_params(axis='y', labelcolor=color_w)

# Invert x-axis so higher temperature is on the left
ax1.invert_xaxis()

# Create secondary y-axis for Derivative Weight
ax2 = ax1.twinx()
color_dw = 'tab:red'
ax2.set_ylabel('Deriv. Weight (%/°C)', color=color_dw)
line2, = ax2.plot(temp, deriv_weight, color=color_dw, linestyle='--', label='Deriv. Weight (%/°C)')
ax2.tick_params(axis='y', labelcolor=color_dw)

# Combine legends
lines = [line1, line2]
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='upper right')

# Title and layout
plt.title('TGA Data: Raw Sheet')
fig.tight_layout()

# Save figure
output_path = os.path.join(OUT_DIR, "tga_raw_inverted_x.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
