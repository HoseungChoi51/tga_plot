```python
import pandas as pd
import matplotlib.pyplot as plt
from matplotlib.ticker import AutoMinorLocator
import os

# 파일 경로 및 출력 디렉토리 설정
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B20_r3/out"

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 데이터 로드 (Acid-washed 시트)
df = pd.read_excel(DATA_FILE, sheet_name="Acid-washed")

# 열 이름 추출
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
deriv_col = "Deriv. Weight (%/°C)"

# TGA 플롯 생성 (Twin axes: Weight와 Derivative를 함께 표시하는 것이 일반적이지만, 
# 요청은 단순히 Acid-washed 시트의 데이터를 x축에 보조 눈금을 넣어 그려달라는 것임.
# 보통 TGA는 TG(무게 감소)와 DTG(미분 무게 변화)를 함께 보여주는 것이 표준이므로 둘 다 포함함)

fig, ax1 = plt.subplots(figsize=(8, 6))

# X축: 온도
x = df[temp_col]

# Y축 1: 무게 (%)
y1 = df[weight_col]
color1 = 'tab:red'
ax1.set_xlabel('온도 (°C)', fontsize=12)
ax1.set_ylabel('무게 (%)', color=color1, fontsize=12)
ax1.plot(x, y1, color=color1, label='TG')
ax1.tick_params(axis='y', labelcolor=color1)
ax1.grid(True, linestyle='--', alpha=0.5)

# X축에 보조 눈금(minor ticks) 추가
ax1.xaxis.set_minor_locator(AutoMinorLocator())

# Y축 2: 미분 무게 (%/°C) - 오른쪽 축
ax2 = ax1.twinx()
y2 = df[deriv_col]
color2 = 'tab:blue'
ax2.set_ylabel('미분 무게 (%/°C)', color=color2, fontsize=12)
ax2.plot(x, y2, color=color2, linestyle='--', label='DTG')
ax2.tick_params(axis='y', labelcolor=color2)

# 제목 및 범례
fig.suptitle('Acid-washed 샘플의 TGA 분석 결과', fontsize=14)
lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()
ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center right')

# 레이아웃 조정
plt.tight_layout(rect=[0, 0, 1, 0.95])

# 출력 디렉토리가 없으면 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 파일 저장
output_path = os.path.join(OUT_DIR, "acid_washed_tga.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
