import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"

# Read the 'Raw' sheet
df = pd.read_excel(DATA_FILE, sheet_name="Raw")

# Extract columns
temp = df["Temperature (°C)"].values
weight = df["Weight (%)"].values
dtg = df["Deriv. Weight (%/°C)"].values

# Find the weight at 150 °C for normalization
idx_150 = np.argmin(np.abs(temp - 150))
w150 = weight[idx_150]

# Normalize weight so that weight at 150 °C becomes 100%
if w150 != 0:
    weight_norm = weight / w150 * 100.0
else:
    weight_norm = weight.copy()

# Prepare data for ECharts
# For TGA plot: x = temperature, y1 = normalized weight, y2 = -DTG (so peaks point up)
data_weight = [[float(t), float(w)] for t, w in zip(temp, weight_norm)]
data_dtg = [[float(t), float(-d)] for t, d in zip(temp, dtg)]

# Find residue at nearest 800 °C
idx_800 = np.argmin(np.abs(temp - 800))
residue_val = weight_norm[idx_800]
residue_temp = temp[idx_800]

# Find DTG peak above 150 °C (highest -DTG value means most negative original DTG)
mask_above_150 = temp > 150
if np.any(mask_above_150):
    dtg_above = dtg[mask_above_150]
    temp_above = temp[mask_above_150]
    # Most negative DTG -> highest -DTG peak
    min_idx = np.argmin(dtg_above)
    peak_temp = temp_above[min_idx]
    peak_dtg_original = dtg_above[min_idx]
else:
    peak_temp = None
    peak_dtg_original = None

# Build markPoint for residue
mark_point_data = [
    {
        "name": "Residue",
        "coord": [float(residue_temp), float(residue_val)],
        "value": f"Residue {residue_val:.1f}%",
        "symbolSize": 8,
        "label": {
            "show": True,
            "position": "top",
            "formatter": "{c}",
            "fontSize": 11,
            "color": "#333"
        }
    }
]

# Add peak marker if exists
if peak_temp is not None:
    # The peak on the -DTG curve (y2 axis) has value -peak_dtg_original
    peak_y2_val = -peak_dtg_original
    mark_point_data.append({
        "name": "Peak",
        "coord": [float(peak_temp), float(peak_y2_val)],
        "value": f"{peak_temp:.0f} °C",
        "symbolSize": 8,
        "label": {
            "show": True,
            "position": "top",
            "formatter": "{c}",
            "fontSize": 11,
            "color": "#333"
        },
        "yAxisIndex": 1
    })

option = {
    "title": {
        "text": "Raw",
        "left": "center",
        "textStyle": {
            "fontSize": 14,
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "tooltip": {
        "trigger": "axis"
    },
    "legend": {
        "data": ["Weight (%)", "-DTG (%/°C)"],
        "left": "left",
        "top": "middle",
        "orient": "vertical",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 11
        }
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "12%",
        "top": "15%"
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 11
        },
        "axisTick": {
            "show": True
        },
        "splitLine": {
            "show": False
        }
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 45,
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 11
            },
            "axisTick": {
                "show": True
            },
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "type": "dashed",
                    "opacity": 0.3
                }
            }
        },
        {
            "type": "value",
            "name": "-DTG (%/°C)",
            "nameLocation": "middle",
            "nameGap": 45,
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 11
            },
            "axisTick": {
                "show": True
            },
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "data": data_weight,
            "yAxisIndex": 0,
            "smooth": False,
            "symbol": "none",
            "lineStyle": {
                "color": "#1f4e9c",
                "width": 2
            },
            "itemStyle": {
                "color": "#1f4e9c"
            },
            "markPoint": {
                "data": [mp for mp in mark_point_data if mp.get("yAxisIndex", 0) == 0],
                "symbol": "circle",
                "itemStyle": {
                    "color": "#1f4e9c"
                }
            }
        },
        {
            "name": "-DTG (%/°C)",
            "type": "line",
            "data": data_dtg,
            "yAxisIndex": 1,
            "smooth": False,
            "symbol": "none",
            "lineStyle": {
                "color": "#c0392b",
                "width": 2
            },
            "itemStyle": {
                "color": "#c0392b"
            },
            "markPoint": {
                "data": [mp for mp in mark_point_data if mp.get("yAxisIndex", 0) == 1],
                "symbol": "circle",
                "itemStyle": {
                    "color": "#c0392b"
                }
            }
        }
    ]
}

result = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,

        }
    ]
