```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

# Read the CSV file
df = pd.read_csv(DATA_FILE)

# Identify columns based on the preview
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

# Extract data
temperature = df[temp_col].values
weight = df[weight_col].values
dtg = df[dtg_col].values

# Prepare series data for ECharts
# Format: [[x, y], ...]
weight_data = []
for t, w in zip(temperature, weight):
    if not np.isnan(t) and not np.isnan(w):
        weight_data.append([float(t), float(w)])

dtg_data = []
for t, d in zip(temperature, dtg):
    if not np.isnan(t) and not np.isnan(d):
        # Multiply DTG by -1 so peaks point up
        dtg_data.append([float(t), float(-d)])

# Define the three regions for shading
regions = [
    {"name": "H2O", "start": 120, "end": 250, "color": "rgba(31, 78, 156, 0.15)"},
    {"name": "CO", "start": 430, "end": 580, "color": "rgba(192, 57, 43, 0.15)"},
    {"name": "CO2", "start": 640, "end": 830, "color": "rgba(39, 174, 96, 0.15)"}
]

# Calculate residue at ~800°C
residue_temp = 800.0
idx_residue = np.argmin(np.abs(temperature - residue_temp))
residue_value = weight[idx_residue]
residue_label = f"Residue {residue_value:.1f}%"

# Find peak temperature of highest DTG peak above 150°C
mask_above_150 = temperature > 150
if np.any(mask_above_150):
    dtg_above_150 = dtg[mask_above_150]
    temp_above_150 = temperature[mask_above_150]
    # Since we plot -DTG, the highest peak corresponds to the most negative DTG value
    min_dtg_idx = np.argmin(dtg_above_150)
    peak_temp = temp_above_150[min_dtg_idx]
    peak_label = f"{peak_temp:.0f} °C"
else:
    peak_temp = None
    peak_label = ""

# Build markArea for shaded regions
mark_area_data = []
for reg in regions:
    mark_area_data.append([
        {"xAxis": reg["start"], "itemStyle": {"color": reg["color"]}},
        {"xAxis": reg["end"]}
    ])

# Build markPoint for residue and peak
mark_point_data_weight = []
mark_point_data_dtg = []

# Residue marker
mark_point_data_weight.append({
    "coord": [float(temperature[idx_residue]), float(residue_value)],
    "value": residue_label,
    "symbolSize": 10,
    "label": {
        "show": True,
        "formatter": "{c}",
        "position": "top",
        "fontSize": 12,
        "color": "#333"
    }
})

# Peak marker
if peak_temp is not None:
    # Find index closest to peak_temp in original arrays for exact coord
    idx_peak = np.argmin(np.abs(temperature - peak_temp))
    mark_point_data_dtg.append({
        "coord": [float(temperature[idx_peak]), float(-dtg[idx_peak])],
        "value": peak_label,
        "symbolSize": 10,
        "label": {
            "show": True,
            "formatter": "{c}",
            "position": "top",
            "fontSize": 12,
            "color": "#333"
        }
    })

# Build data objects for region labels (scatter points with labels)
region_labels_data_objects = []
for reg in regions:
    mid_x = (reg["start"] + reg["end"]) / 2.0
    region_labels_data_objects.append({
        "value": [mid_x, 95.0],
        "name": reg["name"],
        "label": {
            "show": True,
            "formatter": "{b}",
            "position": "inside",
            "fontSize": 14,
            "fontWeight": "bold",
            "color": "#333"
        }
    })

option = {
    "title": {
        "text": "caox",
        "left": "center",
        "top": 10,
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 16,
            "fontWeight": "bold"
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "legend": {
        "data": ["Weight (%)", "DTG (%/°C)", "Region Labels"],
        "orient": "vertical",
        "left": "left",
        "top": "middle",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR"
        },
        "splitLine": {
            "show": True,
            "lineStyle": {
                "type": "dashed",
                "color": "#ccc"
            }
        }
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 40,
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "type": "dashed",
                    "color": "#eee"
                }
            }
        },
        {
            "type": "value",
            "name": "DTG (%/°C)",
            "nameLocation": "middle",
            "nameGap": 40,
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "data": weight_data,
            "smooth": False,
            "showSymbol": False,
            "lineStyle": {
                "color": "#1f4e9c",
                "width": 2
            },
            "itemStyle": {
                "color": "#1f4e9c"
            },
            "markArea": {
                "silent": True,
                "data": mark_area_data
            },
            "markPoint": {
                "data": mark_point_data_weight
            }
        },
        {
            "name": "DTG (%/°C)",
            "type": "line",
            "yAxisIndex": 1,
            "data": dtg_data,
            "smooth": False,
            "showSymbol": False,
            "lineStyle": {
                "color": "#c0392b",
                "width": 2
            },
            "itemStyle": {
                "color": "#c0392b"
            },
            "markPoint": {
                "data": mark_point_data_dtg
            }
        },
        {
            "name": "Region Labels",
            "type": "scatter",
            "data": region_labels_data_objects,
            "symbolSize": 0,
            "z": 10
        }
    ]
}

output = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(output))
```
