def get_tga_data(file_path, sheet_name):
    df = pd.read_excel(file_path, sheet_name=sheet_name)
    
    # Identify columns based on typical TGA structure or names from preview
    cols = df.columns.tolist()
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for c in cols:
        cl = str(c).lower()
        if 'temperature' in cl or 'temp' in cl:
            temp_col = c
        elif 'weight' in cl and 'deriv' not in cl and 'dtg' not in cl:
            weight_col = c
        elif 'deriv' in cl or 'dtg' in cl:
            dtg_col = c
            
    if not temp_col or not weight_col or not dtg_col:
        # Fallback to positional if names don't match well, though preview suggests they do
        temp_col = cols[1]
        weight_col = cols[2]
        dtg_col = cols[3]

    x = df[temp_col].dropna().values
    y1 = df[weight_col].dropna().values
    y2_raw = df[dtg_col].dropna().values
    
    # Ensure lengths match after dropping NaNs if necessary (usually aligned in TGA exports)
    min_len = min(len(x), len(y1), len(y2_raw))
    x = x[:min_len]
    y1 = y1[:min_len]
    y2_raw = y2_raw[:min_len]
    
    # Filter data within requested X range 100-700
    mask = (x >= 100) & (x <= 700)
    x_filtered = x[mask]
    y1_filtered = y1[mask]
    y2_raw_filtered = y2_raw[mask]
    
    return x_filtered, y1_filtered, y2_raw_filtered

def process_sheet(file_path, sheet_name):
    df = pd.read_excel(file_path, sheet_name=sheet_name)
    cols = df.columns.tolist()
    
    # Map columns
    temp_col = next((c for c in cols if 'temperature' in str(c).lower()), cols[1])
    weight_col = next((c for c in cols if 'weight' in str(c).lower() and 'deriv' not in str(c).lower()), cols[2])
    dtg_col = next((c for c in cols if 'deriv' in str(c).lower() or 'dtg' in str(c).lower()), cols[3])
    
    # Full data for calculations
    x_full = df[temp_col].dropna().values
    y1_full = df[weight_col].dropna().values
    y2_full = df[dtg_col].dropna().values
    
    # Align lengths
    min_len = min(len(x_full), len(y1_full), len(y2_full))
    x_full = x_full[:min_len]
    y1_full = y1_full[:min_len]
    y2_full = y2_full[:min_len]
    
    # 1. Calculate Residue (nearest 800C)
    idx_800 = np.argmin(np.abs(x_full - 800))
    residue_temp = x_full[idx_800]
    residue_weight = y1_full[idx_800]
    
    # 2. Calculate Peak DTG (highest peak above 150C)
    # DTG is negative during mass loss. "Highest peak" usually means most negative (steepest slope).
    # We want the temperature where |DTG| is max, or DTG is min.
    # Constraint: Above 150 C.
    mask_dtg = (x_full > 150)
    if np.any(mask_dtg):
        x_dtg_range = x_full[mask_dtg]
        y_dtg_range = y2_full[mask_dtg]
        
        # Find index of minimum DTG (most negative)
        idx_min_dtg = np.argmin(y_dtg_range)
        peak_temp = x_dtg_range[idx_min_dtg]
        peak_val = y_dtg_range[idx_min_dtg]
    else:
        peak_temp = None
        peak_val = None
        
    # Prepare Plot Data (Filtered 100-700)
    mask_plot = (x_full >= 100) & (x_full <= 700)
    x_plot = x_full[mask_plot]
    y1_plot = y1_full[mask_plot]
    y2_plot = y2_full[mask_plot]
    
    # Convert to list for JSON
    series_data_y1 = [[float(t), float(w)] for t, w in zip(x_plot, y1_plot)]
    # Y2 needs to be multiplied by -1 so peaks point up
    series_data_y2 = [[float(t), float(-d)] for t, d in zip(x_plot, y2_plot)]
    
    # Markers
    markers = []
    
    # Check if Residue marker is within plot range [100, 700]
    if 100 <= residue_temp <= 700:
        markers.append({
            "coord": [float(residue_temp), float(residue_weight)],
            "value": f"Residue {residue_weight:.1f}%",
            "type": "scatter", # Will handle via markPoint later or separate series? 
            # Better to use markPoint on the main series or a dedicated scatter series.
            # Using a dedicated scatter series for annotations is cleaner in ECharts JSON.
        })
        
    # Check if Peak marker is within plot range [100, 700]
    if peak_temp is not None and 100 <= peak_temp <= 700:
        # The marker on the DTG curve. The DTG curve is plotted as -DTG.
        # So the y-coordinate for the marker on the visual plot is -peak_val.
        markers.append({
            "coord": [float(peak_temp), float(-peak_val)],
            "value": f"{peak_temp:.0f} °C",
            "seriesIndex": 1 # Points to DTG series
        })

    return {
        "name": sheet_name,
        "data_y1": series_data_y1,
        "data_y2": series_data_y2,
        "markers": markers,
        "y1_color": "#ff0000", # Red as requested
        "y1_width": 3          # Bold as requested
    }

def create_option(sheet_info):
    name = sheet_info["name"]
    data_y1 = sheet_info["data_y1"]
    data_y2 = sheet_info["data_y2"]
    markers = sheet_info["markers"]
    
    # Base Option
    option = {
        "title": {
            "text": name,
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontSize": 16,
                "fontWeight": "bold"
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "top": 40,
            "left": "center"
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "min": 100,
            "max": 700,
            "interval": 100,
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "color": "#eee"
                }
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "color": "#eee"
                    }
                }
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "position": "right",
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "data": data_y1,
                "yAxisIndex": 0,
                "showSymbol": False,
                "lineStyle": {
                    "color": "#ff0000", # Red
                    "width": 3         # Bold
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "data": data_y2,
                "yAxisIndex": 1,
                "showSymbol": False,
                "lineStyle": {
                    "color": "#c0392b", # Default dark red/brown for contrast, or keep distinct
                    "width": 1.5
                }
            }
        ]
    }
    
    # Add Markers using markPoint
    # We need to attach markPoints to the respective series.
    # Weight Series (Index 0) gets Residue marker
    # DTG Series (Index 1) gets Peak marker
    
    # Initialize markPoint arrays
    mp_weight = []
    mp_dtg = []
    
    for m in markers:
        coord = m["coord"]
        val_text = m["value"]
        s_idx = m.get("seriesIndex", 0) # Default to weight if not specified (though residue is weight)
        
        # Determine which series this belongs to based on logic above
        # If it was added as residue, it's on Weight. If peak, it's on DTG.
        # My previous loop added "seriesIndex": 1 for peak. Residue didn't have it, defaults to 0.
        
        point_def = {
            "coord": coord,
            "value": val_text,
            "symbolSize": 10,
            "label": {
                "show": True,
                "formatter": "{@[2]} {@[3]}", # This formatter is tricky for custom strings
                # Better to use rich text or simple string replacement in formatter function isn't allowed (no JS functions).
                # We can use `formatter` string template. 
                # ECharts markPoint label formatter: `{b}` is name, `{c}` is value.
                # Here we want the custom string "Residue 8.0%" or "600 C".
                # We can set the `name` property of the markPoint to the desired text and use `{b}`.
                "formatter": "{b}"
            }
        }
        
        # Set the name to the display text
        point_def["name"] = val_text
        
        if s_idx == 0:
            mp_weight.append(point_def)
        else:
            mp_dtg.append(point_def)
            
    if mp_weight:
        option["series"][0]["markPoint"] = {
            "data": mp_weight,
            "itemStyle": {
                "color": "#ff0000"
            },
            "label": {
                "position": "top",
                "color": "#333",
                "backgroundColor": "rgba(255,255,255,0.8)",
                "padding": [2, 4]
            }
        }
        
    if mp_dtg:
        option["series"][1]["markPoint"] = {
            "data": mp_dtg,
            "itemStyle": {
                "color": "#c0392b"
            },
            "label": {
                "position": "top",
                "color": "#333",
                "backgroundColor": "rgba(255,255,255,0.8)",
                "padding": [2, 4]
            }
        }

    return option

def main():
    file_path = DATA_FILE
    sheets_to_plot = ["PE"] # Only PE as requested
    
    figures = []
    
    for sheet in sheets_to_plot:
        try:
            info = process_sheet(file_path, sheet)
            opt = create_option(info)
            fig_entry = {
                "name": f"plot_{sheet.lower()}",
                "option": opt,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            }
            figures.append(fig_entry)
        except Exception as e:
            # In case of error, we might skip or log. For strict JSON output, better to ensure valid structure.
            pass
            
    output = {"figures": figures}
    print(json.dumps(output))

if __name__ == "__main__":
    main()
