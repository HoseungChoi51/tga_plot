def read_data(file_path):
    if file_path.endswith('.csv'):
        return pd.read_csv(file_path)
    elif file_path.endswith('.txt'):
        # Assuming tab or whitespace separated, but header is row 1. 
        # The preview suggests standard CSV-like structure with headers.
        # Let's try sep=None with engine='python' to detect delimiter automatically for txt files that might be csv-like.
        return pd.read_csv(file_path, sep=None, engine='python')
    else:
        raise ValueError("Unsupported file format")

def get_tga_series(df, lot_name):
    # Identify columns based on the preview info provided in the prompt context
    # x = Temperature (°C), y1 = Weight (%), y2 = Deriv. Weight (%/°C)
    
    temp_col = 'Temperature (°C)'
    weight_col = 'Weight (%)'
    dtg_col = 'Deriv. Weight (%/°C)'
    
    # Check if columns exist, otherwise try to find them by pattern if names differ slightly
    if temp_col not in df.columns:
        # Fallback heuristic
        cols = df.columns.tolist()
        temp_col = [c for c in cols if 'Temp' in c][0]
        weight_col = [c for c in cols if 'Weight' in c and 'Deriv' not in c][0]
        dtg_col = [c for c in cols if 'Deriv' in c][0]

    temps = df[temp_col].values
    weights = df[weight_col].values
    dtgs = df[dtg_col].values
    
    # Filter out NaNs just in case
    mask = ~(np.isnan(temps) | np.isnan(weights) | np.isnan(dtgs))
    temps = temps[mask]
    weights = weights[mask]
    dtgs = dtgs[mask]
    
    # Prepare data points for ECharts [x, y]
    # Weight series: [Temp, Weight]
    weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]
    
    # DTG series: [Temp, -DTG] so peaks point up
    dtg_data = [[float(t), float(-d)] for t, d in zip(temps, dtgs)]
    
    # Calculate Residue at ~800°C
    idx_800 = np.argmin(np.abs(temps - 800))
    residue_val = weights[idx_800]
    residue_temp = temps[idx_800]
    
    # Calculate Peak Temp > 150°C
    # We look for the maximum of -DTG (which corresponds to minimum DTG) where Temp > 150
    valid_mask = temps > 150
    if np.any(valid_mask):
        valid_dtgs = dtgs[valid_mask]
        valid_temps = temps[valid_mask]
        # Find index of min DTG (most negative) which is max loss rate
        min_dtg_idx_local = np.argmin(valid_dtgs)
        peak_temp = valid_temps[min_dtg_idx_local]
        peak_weight = weights[np.argmin(np.abs(temps - peak_temp))]
    else:
        peak_temp = None
        peak_weight = None
        
    return {
        "name": lot_name,
        "weight_data": weight_data,
        "dtg_data": dtg_data,
        "residue_val": residue_val,
        "residue_temp": residue_temp,
        "peak_temp": peak_temp,
        "peak_weight": peak_weight
    }

def process_file(f, name, color_w, color_d):
    df = read_data(f)
    temp_col = 'Temperature (°C)'
    weight_col = 'Weight (%)'
    dtg_col = 'Deriv. Weight (%/°C)'
    
    if temp_col not in df.columns:
         cols = df.columns.tolist()
         temp_col = [c for c in cols if 'Temp' in c][0]
         weight_col = [c for c in cols if 'Weight' in c and 'Deriv' not in c][0]
         dtg_col = [c for c in cols if 'Deriv' in c][0]

    temps = df[temp_col].values
    weights = df[weight_col].values
    dtgs = df[dtg_col].values
    
    mask = ~(np.isnan(temps) | np.isnan(weights) | np.isnan(dtgs))
    temps = temps[mask]
    weights = weights[mask]
    dtgs = dtgs[mask]
    
    weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]
    dtg_data = [[float(t), float(-d)] for t, d in zip(temps, dtgs)]
    
    idx_800 = np.argmin(np.abs(temps - 800))
    res_temp = float(temps[idx_800])
    res_val = float(weights[idx_800])
    
    valid_mask = temps > 150
    pk_temp = None
    pk_val = None # This is the plotted value (-DTG)
    
    if np.any(valid_mask):
        v_dtgs = dtgs[valid_mask]
        v_temps = temps[valid_mask]
        min_idx = np.argmin(v_dtgs)
        pk_temp = float(v_temps[min_idx])
        pk_val = float(-v_dtgs[min_idx]) # Plotted value
        
    return {
        "w_series": {
            "name": f"{name} Weight",
            "type": "line",
            "yAxisIndex": 0,
            "data": weight_data,
            "showSymbol": False,
            "lineStyle": {"width": 2, "color": color_w},
            "itemStyle": {"color": color_w}
        },
        "d_series": {
            "name": f"{name} DTG",
            "type": "line",
            "yAxisIndex": 1,
            "data": dtg_data,
            "showSymbol": False,
            "lineStyle": {"width": 2, "color": color_d, "type": "dashed"},
            "itemStyle": {"color": color_d}
        },
        "mp_w": {
            "coord": [res_temp, res_val],
            "name": f"{name} Residue",
            "value": f"Residue {res_val:.1f}%",
            "symbolSize": 8,
            "label": {
                "formatter": "{b}: {c}", # {b} is name, {c} is value? No, {c} is usually the value in series. 
                                         # For markPoint, {c} is the coordinate value. 
                                         # We want to display the custom string.
                                         # Using formatter function isn't allowed (no JS).
                                         # So we rely on 'name' being displayed if label shows name? 
                                         # Default label shows value. 
                                         # Let's set label.show=true and formatter to a static string via 'name'?
                                         # ECharts markPoint label default formatter is '{b}'. If we set name, it shows name.
                                         # But we want "Residue X%". 
                                         # Solution: Put the full text in 'name' and ensure label displays it.
                "position": "top",
                "distance": 5,
                "fontSize": 10,
                "color": color_w
            },
            "itemStyle": {"color": color_w}
        },
        "mp_d": {
            "coord": [pk_temp, pk_val],

                "distance": 5,

            },
            "itemStyle": {"color": color_d}
        } if pk_temp is not None else None
    }
