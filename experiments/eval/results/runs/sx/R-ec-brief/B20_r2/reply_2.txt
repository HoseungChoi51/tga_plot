```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
SHEET_NAME = "Acid-washed"

def main():
    # Read the specific sheet
    df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
    
    # Identify columns based on preview info or standard TGA structure
    # Preview says: A=Time, B=Temperature, C=Weight, D=Deriv Weight
    x_col = "Temperature (°C)"
    y1_col = "Weight (%)"
    y2_col = "Deriv. Weight (%/°C)"
    
    # Extract data
    temp = df[x_col].values
    weight = df[y1_col].values
    dtg = df[y2_col].values
    
    # Prepare data for ECharts
    # Series 1: Weight (%) - Left Y axis
    # We need pairs [x, y]
    weight_data = [[float(t), float(w)] for t, w in zip(temp, weight)]
    
    # Series 2: DTG - Right Y axis
    # Invert DTG so peaks point up (-1 * value)
    dtg_inverted = [-d for d in dtg]
    dtg_data = [[float(t), float(d)] for t, d in zip(temp, dtg_inverted)]
    
    # Calculate Residue at nearest 800°C
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = float(weight[idx_800])
    residue_temp = float(temp[idx_800])
    
    # Calculate Peak temperature (highest DTG peak above 150°C)
    # Since we inverted DTG, we look for max value in inverted DTG where temp > 150
    mask_peak = temp > 150
    if np.any(mask_peak):
        dtg_inv_masked = np.array(dtg_inverted)[mask_peak]
        temp_masked = temp[mask_peak]
        max_idx_local = np.argmax(dtg_inv_masked)
        peak_temp = float(temp_masked[max_idx_local])
        peak_val_inv = float(dtg_inv_masked[max_idx_local])
    else:
        peak_temp = None
        peak_val_inv = None

    # Determine Axis Ranges
    x_min = float(min(temp))
    x_max = float(max(temp))
    
    # Y1 Range (Weight %)
    y1_min = float(min(weight))
    y1_max = float(max(weight))
    # Add some padding
    y1_range = y1_max - y1_min
    y1_min_plot = y1_min - y1_range * 0.05
    y1_max_plot = y1_max + y1_range * 0.05
    
    # Y2 Range (DTG inverted)
    y2_min = float(min(dtg_inverted))
    y2_max = float(max(dtg_inverted))
    y2_range = y2_max - y2_min
    y2_min_plot = y2_min - y2_range * 0.05
    y2_max_plot = y2_max + y2_range * 0.05
    
    # Minor ticks for X axis
    # Major ticks usually every 100 or 50 degrees. Let's set major to 100, minor to 20.
    # Or let echarts handle it with splitNumber and minorSplitLine.
    # The request specifically asks for minor ticks. In ECharts, this is `minorTick` and `minorSplitLine`.
    
    option = {
        "backgroundColor": "#ffffff",
        "title": {
            "text": SHEET_NAME,
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontSize": 16,
                "fontWeight": "bold",
                "color": "#333333"
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "-DTG"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "icon": "rect",
            "itemWidth": 10,
            "itemHeight": 10,
            "textStyle": {
                "fontSize": 12
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "top": "15%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "min": x_min,
            "max": x_max,
            "splitLine": {
                "show": False
            },
            "axisLine": {
                "lineStyle": {
                    "color": "#333333"
                }
            },
            "axisTick": {
                "show": True,
                "length": 6,
                "lineStyle": {
                    "color": "#333333"
                }
            },
            "minorTick": {
                "show": True,
                "length": 4,
                "lineStyle": {
                    "color": "#999999"
                }
            },
            "minorSplitLine": {
                "show": False 
            },
            "axisLabel": {
                "formatter": "{value}",
                "fontSize": 12
            }
        },
        "yAxis": [
            {
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "left",
                "min": y1_min_plot,
                "max": y1_max_plot,
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "color": "#eeeeee"
                    }
                },
                "axisLine": {
                    "lineStyle": {
                        "color": "#1f4e9c"
                    }
                },
                "axisTick": {
                    "show": True,
                    "lineStyle": {
                        "color": "#1f4e9c"
                    }
                },
                "axisLabel": {
                    "formatter": "{value}",
                    "fontSize": 12,
                    "color": "#1f4e9c"
                }
            },
            {
                "name": "DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "right",
                "min": y2_min_plot,
                "max": y2_max_plot,
                "splitLine": {
                    "show": False
                },
                "axisLine": {
                    "lineStyle": {
                        "color": "#c0392b"
                    }
                },
                "axisTick": {
                    "show": True,
                    "lineStyle": {
                        "color": "#c0392b"
                    }
                },
                "axisLabel": {
                    "formatter": "{value}",
                    "fontSize": 12,
                    "color": "#c0392b"
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "data": weight_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [residue_temp, residue_val],
                            "symbolSize": 8,
                            "itemStyle": {
                                "color": "#1f4e9c"
                            },
                            "label": {
                                "show": True,
                                "formatter": f"Residue {residue_val:.1f}%",
                                "position": "right",
                                "distance": 5,
                                "fontSize": 10,
                                "color": "#1f4e9c"
                            }
                        }
                    ]
                }
            },
            {
                "name": "-DTG",
                "type": "line",
                "yAxisIndex": 1,
                "data": dtg_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 2
                },
                "markPoint": {
                    "data": []
                }
            }
        ]
    }
    
    # Add Peak Mark Point if found
    if peak_temp is not None:
        # Find the corresponding index in the original arrays to get exact coords for markpoint
        # We already have peak_temp and peak_val_inv from earlier calculation
        # But we need to make sure the coord exists in the series data roughly
        
        # Re-calculate strictly from masked arrays to ensure consistency
        mask_peak = temp > 150
        dtg_inv_arr = np.array(dtg_inverted)
        temp_arr = np.array(temp)
        
        dtg_inv_masked = dtg_inv_arr[mask_peak]
        temp_masked = temp_arr[mask_peak]
        
        max_idx_local = np.argmax(dtg_inv_masked)
        p_temp = float(temp_masked[max_idx_local])
        p_val = float(dtg_inv_masked[max_idx_local])
        
        option["series"][1]["markPoint"]["data"].append({
            "coord": [p_temp, p_val],
            "symbolSize": 8,
            "itemStyle": {
                "color": "#c0392b"
            },
            "label": {
                "show": True,
                "formatter": f"{p_temp:.0f} °C",
                "position": "top",
                "distance": 5,
                "fontSize": 10,
                "color": "#c0392b"
            }
        })

    output = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
