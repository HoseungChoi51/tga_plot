import pandas as pd
import numpy as np
import json

def get_tga_data(file_path, sheet_name):
    df = pd.read_excel(file_path, sheet_name=sheet_name)
    
    # Identify columns based on standard TGA names or guess from preview
    cols = df.columns.tolist()
    
    x_col = None
    y1_col = None
    y2_col = None
    
    for c in cols:
        c_lower = str(c).lower()
        if 'temperature' in c_lower:
            x_col = c
        elif 'weight (%)' in c_lower and 'deriv' not in c_lower:
            y1_col = c
        elif 'deriv' in c_lower or 'dtg' in c_lower:
            y2_col = c
            
    if x_col is None or y1_col is None or y2_col is None:
        return None
        
    temp = df[x_col].values
    weight = df[y1_col].values
    dtg = df[y2_col].values
    
    return temp, weight, dtg

def find_nearest_point(arr, target_val):
    idx = np.argmin(np.abs(arr - target_val))
    return arr[idx], idx

def main():

    sheets_info = [
        {"name": "Raw", "color": "#000000"},

        {"name": "Annealed", "color": "#c0392b"}   # Red

            })

            neg_dtgs = -dtgs

                "name": f"{sheet_name} DTG",
                "type": "line",
                "data": d_data,
                "yAxisIndex": 1,
                "showSymbol": False,
                "lineStyle": {"width": 1.5, "color": color_map[sheet_name], "opacity": 0.6},

            # Calculate Peak Temp (>150C)
            peak_mask = temps > 150.0
            if np.any(peak_mask):
                # Find index of maximum negative derivative (which becomes max positive after negation)
                # Actually, we want the highest DTG peak magnitude. 
                # Original DTG is negative during loss. So we look for min(DTG) i.e., most negative.
                # Or simply max(-DTG).
                filtered_dtgs = dtgs[peak_mask]
                filtered_temps = temps[peak_mask]
                
                min_dtg_idx_local = np.argmin(filtered_dtgs) # Most negative value
                peak_temp = filtered_temps[min_dtg_idx_local]
            else:
                peak_temp = None
            
            # Add Mark Points for this sheet
            mp_list = []
            
            # Residue Marker
            mp_list.append({
                "coord": [res_temp, res_weight],

                "symbolSize": 8,

                    "show": True,

                    "formatter": "{c}",
                    "fontSize": 10,

                },
                "itemStyle": {"color": color_map[sheet_name]}
            })
            
            # Peak Marker
            if peak_temp is not None:
                # Find corresponding weight at peak temp for coordinate accuracy if needed, 
                # but label just needs temp. We plot marker on Weight curve? 
                # Instruction says: "marker on the curve". Usually residue is on weight, peak is on DTG or Weight?
                # "written like '600 °C', with a marker."
                # Typically peak markers are placed on the DTG curve or projected. 
                # Given y2 is DTG, placing it on y2 makes sense visually for the peak itself.
                # However, mixing axes in markPoint can be tricky if not careful.
                # Let's place the peak marker on the DTG curve (y2 axis).
                
                # Find nearest index for peak_temp to get exact coord
                _, p_idx = find_nearest_point(temps, peak_temp)
                p_dtg_neg = -dtgs[p_idx]
                
                mp_list.append({
                    "coord": [peak_temp, p_dtg_neg],
                    "value": f"{peak_temp:.0f} °C",
                    "symbolSize": 8,
                    "label": {
                        "show": True,
                        "position": "top",
                        "formatter": "{c}",
                        "fontSize": 10,
                        "color": color_map[sheet_name]
                    },
                    "itemStyle": {"color": color_map[sheet_name]}
                })
                
            # Attach mark points to the first series of this group (Weight) or create separate scatter?
            # ECharts allows markPoint on any series. Since coordinates are mixed (some y1, some y2), 
            # it's safer to attach residue to Weight series and Peak to DTG series separately, 
            # OR use a single scatter series for markers. 
            # Simpler approach: Attach to respective series.
            
            # We need to update the series objects we added earlier. 
            # They are at indices -2 (Weight) and -1 (DTG) for this iteration.
            
            # Update Weight series markPoints
            series_list[-2]["markPoint"] = {
                "data": [mp for mp in mp_list if "Residue" in mp["value"]]
            }
            
            # Update DTG series markPoints
            series_list[-1]["markPoint"] = {
                "data": [mp for mp in mp_list if "°C" in mp["value"]]

    # If original min is -1.3, plotted max is 1.3.

    plotted_min_dtg = -max_dtg

        "title": {

            "formatter": function_placeholder # Not allowed in JSON, remove or simplify

        "grid": {

            "right": "10%",
            "bottom": "10%",
            "containLabel": True
        },
        "xAxis": {

            "nameLocation": "middle",

            "max": round(max_temp / 50) * 50 + 50,
            "axisLine": {"onZero": False},
