```python
import json
import pandas as pd
import numpy as np

def main():
    file_path = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
    
    # Read only the 'PVC' sheet as requested
    df = pd.read_excel(file_path, sheet_name='PVC')
    
    # Identify columns based on preview info:
    # B: Temperature (°C), C: Weight (%), D: Deriv. Weight (%/°C)
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    # Extract data and clean NaNs
    temps = df[temp_col].dropna().values
    weights = df[weight_col].dropna().values
    dtgs = df[dtg_col].dropna().values
    
    # Ensure arrays are aligned if dropna was applied differently (usually they are same length in TGA)
    min_len = min(len(temps), len(weights), len(dtgs))
    temps = temps[:min_len]
    weights = weights[:min_len]
    dtgs = dtgs[:min_len]
    
    # Prepare data for ECharts
    # Format: [[x, y], ...]
    weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]
    # DTG is multiplied by -1 so peaks point up
    dtg_data = [[float(t), float(-d)] for t, d in zip(temps, dtgs)]
    
    # Calculate Residue at 800°C
    idx_800 = np.argmin(np.abs(temps - 800))
    residue_val = weights[idx_800]
    residue_temp = temps[idx_800]
    residue_label_text = f"Residue {residue_val:.1f}%"
    
    # Find Peak Temp (>150°C) from original DTG (most negative value)
    mask = temps > 150
    if np.any(mask):
        dtg_masked = dtgs[mask]
        temps_masked = temps[mask]
        peak_idx = np.argmin(dtg_masked) # Most negative DTG corresponds to highest mass loss rate
        peak_temp = temps_masked[peak_idx]
        peak_dtg_val = dtg_masked[peak_idx]
        peak_label_text = f"{peak_temp:.0f} °C"
    else:
        peak_temp = None
        peak_label_text = ""
        
    # Axis Ranges
    x_min = 0
    x_max = 950
    y1_min = 0
    y1_max = 110
    # Y2 range for inverted DTG (-DTG). Original min is ~-1.05, max is ~0.002.
    # Inverted: min is ~-0.002, max is ~1.05.
    # We want some padding.
    dtg_vals_inv = [-d for d in dtgs]
    y2_min = min(dtg_vals_inv) * 1.1 if min(dtg_vals_inv) < 0 else -0.1
    y2_max = max(dtg_vals_inv) * 1.2
    
    # Font settings: Size 16, Korean font
    font_family = "Noto Sans CJK KR"
    font_size = 16
    
    option = {
        "title": {
            "text": "PVC",
            "left": "center",
            "top": "top",
            "textStyle": {
                "fontSize": font_size + 4,
                "fontWeight": "bold",
                "fontFamily": font_family
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "-DTG"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontSize": font_size,
                "fontFamily": font_family
            }
        },
        "grid": {
            "left": "15%",
            "right": "15%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "nameTextStyle": {
                "fontSize": font_size,
                "fontFamily": font_family
            },
            "type": "value",
            "min": x_min,
            "max": x_max,
            "splitLine": {"show": False},
            "axisLabel": {
                "fontSize": font_size,
                "fontFamily": font_family
            }
        },
        "yAxis": [
            {
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "nameTextStyle": {
                    "fontSize": font_size,
                    "fontFamily": font_family
                },
                "type": "value",
                "position": "left",
                "min": y1_min,
                "max": y1_max,
                "splitLine": {"show": True, "lineStyle": {"color": "#ccc"}},
                "axisLabel": {
                    "fontSize": font_size,
                    "fontFamily": font_family
                }
            },
            {
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "nameTextStyle": {
                    "fontSize": font_size,
                    "fontFamily": font_family
                },
                "type": "value",
                "position": "right",
                "min": y2_min,
                "max": y2_max,
                "splitLine": {"show": False},
                "axisLabel": {
                    "fontSize": font_size,
                    "fontFamily": font_family
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "data": weight_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [float(residue_temp), float(residue_val)],
                            "value": residue_label_text,
                            "symbolSize": 10,
                            "itemStyle": {"color": "#1f4e9c"},
                            "label": {
                                "show": True,
                                "formatter": "{b}",
                                "position": "top",
                                "distance": 5,
                                "fontSize": font_size,
                                "fontFamily": font_family,
                                "color": "#1f4e9c"
                            }
                        }
                    ]
                }
            },
            {
                "name": "-DTG",
                "type": "line",
                "yAxisIndex": 1,
                "data": dtg_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [float(peak_temp), float(-peak_dtg_val)] if peak_temp else [0,0],
                            "value": peak_label_text,
                            "symbolSize": 10,
                            "itemStyle": {"color": "#c0392b"},
                            "label": {
                                "show": True,
                                "formatter": "{b}",
                                "position": "top",
                                "distance": 5,
                                "fontSize": font_size,
                                "fontFamily": font_family,
                                "color": "#c0392b"
                            }
                        }
                    ] if peak_temp else []
                }
            }
        ]
    }
    
    # Figure size: Width 10 inches -> 960 px. Height default reasonable e.g. 7 inches -> 672 px.
    width_px = int(10 * 96)
    height_px = int(7 * 96)
    
    result = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": width_px,
                "height": height_px,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
```
