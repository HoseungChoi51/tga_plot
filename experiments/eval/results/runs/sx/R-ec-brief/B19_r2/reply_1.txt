weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

# Filter data for 200-400 C range to calculate mass loss
mask_200_400 = (df[temp_col] >= 200) & (df[temp_col] <= 400)
df_range = df[mask_200_400].copy()

if not df_range.empty:
    # Find weight at 200°C and 400°C using nearest point interpolation logic or direct lookup
    # Since TGA data is discrete, we find the points closest to 200 and 400
    
    # Get index of point closest to 200
    idx_200 = (df_range[temp_col] - 200).abs().idxmin()
    w_200 = df.loc[idx_200, weight_col]
    
    # Get index of point closest to 400
    idx_400 = (df_range[temp_col] - 400).abs().idxmin()
    w_400 = df.loc[idx_400, weight_col]
    
    mass_loss = w_200 - w_400
