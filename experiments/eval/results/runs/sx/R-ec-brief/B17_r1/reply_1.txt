```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

# Read all sheets from the Excel file
xls = pd.ExcelFile(DATA_FILE)
sheets = xls.sheet_names

figures = []

for sheet_name in sheets:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)
    
    # Identify columns based on typical TGA structure
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        col_lower = str(col).lower()
        if 'temperature' in col_lower or 'temp' in col_lower:
            temp_col = col
        elif 'weight' in col_lower and '%' in str(col):
            weight_col = col
        elif 'deriv' in col_lower or 'dtg' in col_lower:
            dtg_col = col
    
    if temp_col is None or weight_col is None or dtg_col is None:
        continue
    
    temps = df[temp_col].values
    weights = df[weight_col].values
    dtgs = df[dtg_col].values
    
    # Prepare data points for ECharts
    # Weight curve (y1)
    weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]
    
    # DTG curve (y2), multiplied by -1 so peaks point up
    dtg_data = [[float(t), float(-d)] for t, d in zip(temps, dtgs)]
    
    # Find residue at 800°C
    idx_800 = np.argmin(np.abs(temps - 800))
    residue_temp = temps[idx_800]
    residue_weight = weights[idx_800]
    
    # Find peak DTG above 150°C
    mask_above_150 = temps > 150
    if np.any(mask_above_150):
        dtg_negated = -dtgs  # We want positive peaks
        dtg_above_150 = dtg_negated[mask_above_150]
        temps_above_150 = temps[mask_above_150]
        peak_idx_local = np.argmax(dtg_above_150)
        peak_temp = temps_above_150[peak_idx_local]
        peak_dtg_val = dtg_above_150[peak_idx_local]
    else:
        peak_temp = None
        peak_dtg_val = None
    
    # Build markPoint for residue and peak
    mark_points = []
    if residue_weight is not None:
        mark_points.append({
            "coord": [float(residue_temp), float(residue_weight)],
            "name": f"Residue {residue_weight:.1f}%",
            "value": f"{residue_weight:.1f}%",
            "symbolSize": 8,
            "label": {
                "show": True,
                "formatter": f"Residue {residue_weight:.1f}%",
                "position": "right",
                "fontSize": 11,
                "color": "#333"
            }
        })
    
    if peak_temp is not None and peak_dtg_val is not None:
        mark_points.append({
            "coord": [float(peak_temp), float(peak_dtg_val)],
            "name": f"{peak_temp:.0f} °C",
            "value": f"{peak_temp:.0f}",
            "symbolSize": 8,
            "label": {
                "show": True,
                "formatter": f"{peak_temp:.0f} °C",
                "position": "top",
                "fontSize": 11,
                "color": "#c0392b"
            },
            "xAxisIndex": 0,
            "yAxisIndex": 1
        })
    
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 14
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 11
            }
        },
        "grid": {
            "left": "12%",
            "right": "12%",
            "bottom": "12%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "min": 30,
            "max": 900,
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 10
            },
            "splitLine": {
                "show": False
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "position": "left",
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 10
                },
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "type": "dashed"
                    }
                }
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "position": "right",
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 10
                },
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "smooth": True,
                "symbol": "none",
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 1.5
                },
                "itemStyle": {
                    "color": "#1f4e9c"
                },
                "data": weight_data,
                "markPoint": {
                    "data": [mp for mp in mark_points if mp.get("yAxisIndex") != 1]
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "smooth": True,
                "symbol": "none",
                "yAxisIndex": 1,
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 1.5
                },
                "itemStyle": {
                    "color": "#c0392b"
                },
                "data": dtg_data,
                "markPoint": {
                    "data": [mp for mp in mark_points if mp.get("yAxisIndex") == 1]
                }
            }
        ]
    }
    
    figures.append({
        "name": f"tga_{sheet_name}",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    })

result = {"figures": figures}
print(json.dumps(result, ensure_ascii=False))
```
