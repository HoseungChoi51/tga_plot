```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def get_korean_font():
    return {
        "fontFamily": "Noto Sans CJK KR",
        "fontSize": 12,
        "color": "#000000"
    }

def make_tga_plot(df, sheet_name):
    # Extract columns based on standard TGA structure from preview
    # Assuming columns: Time, Temperature, Weight(%), D.W(%/C)
    temp_col = None
    weight_col = None
    dtg_col = None
    
    cols = df.columns.tolist()
    
    for c in cols:
        cl = str(c).lower()
        if 'temp' in cl or '온도' in cl:
            temp_col = c
        elif 'weight' in cl or '무게' in cl or '%' in cl and 'w' in cl:
            weight_col = c
        elif 'd.w' in cl or 'dtg' in cl or 'derivative' in cl:
            dtg_col = c
            
    # Fallback to positional if names are ambiguous but structure is known from preview
    if temp_col is None: temp_col = cols[1] # B
    if weight_col is None: weight_col = cols[2] # C
    if dtg_col is None: dtg_col = cols[3] # D

    x_data = df[temp_col].dropna().values
    y1_data = df[weight_col].dropna().values
    
    # Align lengths just in case of NaNs dropping differently
    min_len = min(len(x_data), len(y1_data))
    x_data = x_data[:min_len]
    y1_data = y1_data[:min_len]
    
    # DTG data needs alignment with temperature
    dtg_raw = df[dtg_col].values
    # Create a temporary dataframe to align non-NaN values properly
    valid_mask = ~np.isnan(df[temp_col].values) & ~np.isnan(df[dtg_col].values)
    x_dtg = df[temp_col].values[valid_mask]
    y2_data = -df[dtg_col].values[valid_mask] # Multiply by -1 so peaks point up
    
    # Calculate Residue at 800°C (or max temp if < 800)
    target_temp_residue = 800.0
    if x_data.max() < target_temp_residue:
        target_temp_residue = x_data.max()
        
    idx_residue = np.argmin(np.abs(x_data - target_temp_residue))
    residue_val = y1_data[idx_residue]
    residue_temp = x_data[idx_residue]
    
    # Calculate Peak above 150°C
    mask_peak = x_dtg > 150.0
    if np.any(mask_peak):
        x_dtg_filtered = x_dtg[mask_peak]
        y2_filtered = y2_data[mask_peak]
        idx_peak_local = np.argmax(y2_filtered)
        peak_temp = x_dtg_filtered[idx_peak_local]
        peak_val = y2_filtered[idx_peak_local]
    else:
        # If no peak above 150, take global max
        idx_peak_global = np.argmax(y2_data)
        peak_temp = x_dtg[idx_peak_global]
        peak_val = y2_data[idx_peak_global]

    font_style = get_korean_font()
    
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            **font_style
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"}
        },
        "legend": {
            "data": ["Weight (%)", "-DTG (%/°C)"],
            "top": "middle",
            "left": "left",
            "orient": "vertical",
            **font_style
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            **font_style,
            "axisLine": {"show": True},
            "splitLine": {"show": False}
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                **font_style,
                "axisLine": {"show": True},
                "splitLine": {"show": True, "lineStyle": {"type": "dashed"}}
            },
            {
                "type": "value",
                "name": "-DTG (%/°C)",
                "position": "right",
                **font_style,
                "axisLine": {"show": True},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "smooth": True,
                "symbol": "none",
                "lineStyle": {"width": 2, "color": "#1f4e9c"},
                "itemStyle": {"color": "#1f4e9c"},
                "data": [[float(t), float(w)] for t, w in zip(x_data, y1_data)],
                "markPoint": {
                    "data": [
                        {
                            "coord": [float(residue_temp), float(residue_val)],
                            "label": {
                                "formatter": f"Residue {residue_val:.1f}%",
                                "position": "top",
                                **font_style
                            },
                            "symbol": "circle",
                            "symbolSize": 8,
                            "itemStyle": {"color": "#1f4e9c"}
                        }
                    ]
                },
                "markLine": {
                    "silent": True,
                    "symbol": ["none", "none"],
                    "lineStyle": {
                        "color": "#666666",
                        "type": "dashed",
                        "width": 1.5
                    },
                    "label": {
                        "formatter": "산화 시작",
                        "position": "insideStartTop",
                        **font_style,
                        "backgroundColor": "rgba(255,255,255,0.8)",
                        "padding": [2, 4]
                    },
                    "data": [
                        {
                            "xAxis": 450
                        }
                    ]
                }
            },
            {
                "name": "-DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "smooth": True,
                "symbol": "none",
                "lineStyle": {"width": 2, "color": "#c0392b"},
                "itemStyle": {"color": "#c0392b"},
                "data": [[float(t), float(d)] for t, d in zip(x_dtg, y2_data)],
                "markPoint": {
                    "data": [
                        {
                            "coord": [float(peak_temp), float(peak_val)],
                            "label": {
                                "formatter": f"{int(round(peak_temp))} °C",
                                "position": "top",
                                **font_style
                            },
                            "symbol": "triangle",
                            "symbolSize": 10,
                            "itemStyle": {"color": "#c0392b"}
                        }
                    ]
                }
            }
        ]
    }
    
    return {
        "name": sheet_name.replace(" ", "_"),
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }

figures = []
xls = pd.ExcelFile(DATA_FILE)
for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name)
    fig = make_tga_plot(df, sheet_name)
    figures.append(fig)

output = {"figures": figures}
print(json.dumps(output, ensure_ascii=False))
```
