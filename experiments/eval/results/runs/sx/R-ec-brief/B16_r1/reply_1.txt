```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

# Read the CSV file
df = pd.read_csv(DATA_FILE)

# Identify columns based on the preview
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

# Extract data
temps = df[temp_col].values
weights = df[weight_col].values
dtgs = df[dtg_col].values * -1  # Multiply by -1 to make peaks point up

# Find residue at nearest 800°C
idx_800 = np.argmin(np.abs(temps - 800))
residue_temp = temps[idx_800]
residue_weight = weights[idx_800]
residue_label = f"Residue {residue_weight:.1f}%"

# Find DTG peak above 150°C
mask_above_150 = temps > 150
if np.any(mask_above_150):
    dtgs_filtered = dtgs[mask_above_150]
    temps_filtered = temps[mask_above_150]
    max_dtg_idx = np.argmax(dtgs_filtered)
    peak_temp = temps_filtered[max_dtg_idx]
    peak_dtg_val = dtgs_filtered[max_dtg_idx]
    peak_label = f"{peak_temp:.0f} °C"
else:
    peak_temp = None
    peak_dtg_val = None
    peak_label = ""

# Prepare data for ECharts
weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]
dtg_data = [[float(t), float(d)] for t, d in zip(temps, dtgs)]

# Common axis settings
x_axis_common = {
    "type": "value",
    "name": "Temperature (°C)",
    "nameLocation": "middle",
    "nameGap": 30,
    "axisLabel": {"formatter": "{value}"},
    "splitLine": {"show": True, "lineStyle": {"color": "#eee"}}
}

# Figure 1: Weight plot (top)
fig1_option = {
    "title": {
        "text": "caox - Weight (%)",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR"}
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "cross"}
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": [
        dict(x_axis_common, gridIndex=0)
    ],
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 40,
            "position": "left",
            "axisLabel": {"formatter": "{value}"},
            "splitLine": {"show": True, "lineStyle": {"color": "#eee"}}
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "data": weight_data,
            "smooth": False,
            "symbol": "none",
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "markPoint": {
                "data": [
                    {
                        "coord": [float(residue_temp), float(residue_weight)],
                        "label": {
                            "formatter": residue_label,
                            "position": "top",
                            "fontSize": 12,
                            "color": "#1f4e9c"
                        },
                        "itemStyle": {"color": "#1f4e9c"}
                    }
                ]
            }
        }
    ]
}

# Figure 2: DTG plot (bottom)
fig2_option = {
    "title": {
        "text": "caox - DTG (%/°C)",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR"}
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "cross"}
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": [
        dict(x_axis_common, gridIndex=0)
    ],
    "yAxis": [
        {
            "type": "value",
            "name": "DTG (%/°C)",
            "nameLocation": "middle",
            "nameGap": 40,
            "position": "left",
            "axisLabel": {"formatter": "{value}"},
            "splitLine": {"show": True, "lineStyle": {"color": "#eee"}}
        }
    ],
    "series": [
        {
            "name": "DTG (%/°C)",
            "type": "line",
            "data": dtg_data,
            "smooth": False,
            "symbol": "none",
            "lineStyle": {"color": "#c0392b", "width": 2},
            "markPoint": {
                "data": []
            }
        }
    ]
}

# Add peak marker to DTG plot if available
if peak_temp is not None and peak_dtg_val is not None:
    fig2_option["series"][0]["markPoint"]["data"].append({
        "coord": [float(peak_temp), float(peak_dtg_val)],
        "label": {
            "formatter": peak_label,
            "position": "top",
            "fontSize": 12,
            "color": "#c0392b"
        },
        "itemStyle": {"color": "#c0392b"}
    })

# Build output JSON
output = {
    "figures": [
        {
            "name": "plot1",
            "option": fig1_option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        },
        {
            "name": "plot2",
            "option": fig2_option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(output, ensure_ascii=False))
```
