```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

df = pd.read_csv(DATA_FILE)

# Column names
x_col = "Temperature (°C)"
y1_col = "Weight (%)"
heat_flow_col = "Heat Flow (W/g)"

# Extract data
temps = df[x_col].values.astype(float)
weights = df[y1_col].values.astype(float)
heat_flows = df[heat_flow_col].values.astype(float)

# --- Find peaks in Heat Flow ---
# We need to find local maxima (endothermic, positive peaks) and local minima (exothermic, negative peaks).
# Using a simple peak detection approach. Since scipy is available, we can use find_peaks.
from scipy.signal import find_peaks

# For endothermic peaks (positive values): look for peaks in heat_flows
# Filter out noise by setting a prominence or height threshold if needed, 
# but let's first try standard peak finding on the raw signal.
# Note: TGA/DSC signals can be noisy. A small smoothing might help, but let's stick to raw first.
# The user asked for "two endothermic peaks and one exothermic peak".
# Let's detect all significant peaks and then select them.

# Endothermic (positive) peaks
peaks_endo_idx, properties_endo = find_peaks(heat_flows, distance=50, prominence=0.2) # Adjusted prominence/distance based on typical DSC data
# Exothermic (negative) peaks -> find peaks of -heat_flows
peaks_exo_idx, properties_exo = find_peaks(-heat_flows, distance=50, prominence=0.2)

# Sort peaks by temperature just in case
peaks_endo_temps = temps[peaks_endo_idx]
peaks_exo_temps = temps[peaks_exo_idx]

# Select the top 2 endothermic and top 1 exothermic based on prominence or just order if count matches expectation
# If there are more than expected, we pick the most prominent ones.
if len(peaks_endo_temps) > 2:
    # Get indices sorted by prominence descending
    prom_endo = properties_endo['prominences']
    sorted_indices = np.argsort(prom_endo)[::-1][:2]
    selected_endo_temps = peaks_endo_temps[sorted_indices]
else:
    selected_endo_temps = peaks_endo_temps[:2]

if len(peaks_exo_temps) > 1:
    prom_exo = properties_exo['prominences']
    sorted_indices_exo = np.argsort(prom_exo)[::-1][:1]
    selected_exo_temp = peaks_exo_temps[sorted_indices_exo][0]
elif len(peaks_exo_temps) == 1:
    selected_exo_temp = peaks_exo_temps[0]
else:
    selected_exo_temp = None

# Prepare data for ECharts
# Weight line (y1)
weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]

# Heat Flow line (y2)
heat_flow_data = [[float(t), float(hf)] for t, hf in zip(temps, heat_flows)]

# Labels configuration
labels = []

# Add labels for endothermic peaks
for i, temp in enumerate(selected_endo_temps):
    # Find corresponding heat flow value for marker position
    idx = np.argmin(np.abs(temps - temp))
    hf_val = heat_flows[idx]
    
    labels.append({
        "type": "scatter",
        "name": f"Endo Peak {i+1}",
        "data": [[float(temp), float(hf_val)]],
        "symbolSize": 8,
        "itemStyle": {"color": "#e67e22"}, # Orange for endothermic
        "label": {
            "show": True,
            "formatter": f"{temp:.1f} °C",
            "position": "top",
            "fontSize": 10,
            "color": "#e67e22"
        }
    })

# Add label for exothermic peak
if selected_exo_temp is not None:
    idx = np.argmin(np.abs(temps - selected_exo_temp))
    hf_val = heat_flows[idx]
    
    labels.append({
        "type": "scatter",
        "name": "Exo Peak",
        "data": [[float(selected_exo_temp), float(hf_val)]],
        "symbolSize": 8,
        "itemStyle": {"color": "#3498db"}, # Blue for exothermic
        "label": {
            "show": True,
            "formatter": f"{selected_exo_temp:.1f} °C",
            "position": "bottom",
            "fontSize": 10,
            "color": "#3498db"
        }
    })

# Residue at 800°C
idx_800 = np.argmin(np.abs(temps - 800))
residue_weight = weights[idx_800]
residue_temp = temps[idx_800]

# Add residue marker on Weight curve
labels.append({
    "type": "scatter",
    "name": "Residue",
    "data": [[float(residue_temp), float(residue_weight)]],
    "symbolSize": 8,
    "itemStyle": {"color": "#1f4e9c"},
    "label": {
        "show": True,
        "formatter": f"Residue {residue_weight:.1f}%",
        "position": "right",
        "fontSize": 10,
        "color": "#1f4e9c"
    }
})

# Construct ECharts Option
option = {
    "title": {
        "text": "caox",
        "left": "center",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 16
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "legend": {
        "data": ["Weight (%)", "Heat Flow (W/g)"],
        "left": "left",
        "top": "middle",
        "orient": "vertical",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "position": "left",
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "lineStyle": {
                    "type": "dashed"
                }
            }
        },
        {
            "type": "value",
            "name": "Heat Flow (W/g)",
            "position": "right",
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "yAxisIndex": 0,
            "data": weight_data,
            "lineStyle": {
                "color": "#1f4e9c",
                "width": 2
            },
            "showSymbol": False,
            "smooth": False
        },
        {
            "name": "Heat Flow (W/g)",
            "type": "line",
            "yAxisIndex": 1,
            "data": heat_flow_data,
            "lineStyle": {
                "color": "#c0392b",
                "width": 2
            },
            "showSymbol": False,
            "smooth": False
        }
    ] + labels
}

result = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(result, ensure_ascii=False))
```
