```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def get_tga_data(file_path):
    xls = pd.ExcelFile(file_path)
    figures = []
    
    for sheet_name in xls.sheet_names:
        df = pd.read_excel(xls, sheet_name=sheet_name)
        
        # Identify columns based on the preview and standard TGA naming
        # Time, Temperature, Weight(%), D.W(%/C)
        temp_col = None
        weight_col = None
        dtg_col = None
        
        cols = df.columns.tolist()
        for c in cols:
            c_lower = str(c).lower()
            if 'temperature' in c_lower or 'temp' in c_lower:
                temp_col = c
            elif 'weight' in c_lower and '%' in c_lower:
                weight_col = c
            elif 'd.w' in c_lower or 'dtg' in c_lower or ('derivative' in c_lower and 'weight' in c_lower):
                dtg_col = c
        
        # Fallback if names are slightly different but structure is known from preview
        if temp_col is None and len(cols) >= 2: temp_col = cols[1]
        if weight_col is None and len(cols) >= 3: weight_col = cols[2]
        if dtg_col is None and len(cols) >= 4: dtg_col = cols[3]

        if temp_col is None or weight_col is None or dtg_col is None:
            continue
            
        temps = df[temp_col].values
        weights = df[weight_col].values
        dtgs = df[dtg_col].values
        
        # Prepare data for ECharts
        # Series 1: Weight (%)
        series_weight = []
        for t, w in zip(temps, weights):
            if not (np.isnan(t) or np.isnan(w)):
                series_weight.append([float(t), float(w)])
                
        # Series 2: DTG (%/C) * -1 to make peaks point up
        series_dtg = []
        for t, d in zip(temps, dtgs):
            if not (np.isnan(t) or np.isnan(d)):
                series_dtg.append([float(t), float(-d)])
                
        # Calculate Residue at ~800 C
        residue_val = None
        residue_point = None
        mask_800 = (temps >= 790) & (temps <= 810)
        if np.any(mask_800):
            idx_800 = np.where(mask_800)[0][0]
            residue_val = weights[idx_800]
            residue_point = [float(temps[idx_800]), float(weights[idx_800])]
            
        # Calculate Peak temperature > 150 C
        peak_temp = None
        peak_val = None
        peak_point = None
        mask_peak = temps > 150
        if np.any(mask_peak):
            valid_dtgs = dtgs[mask_peak]
            valid_temps = temps[mask_peak]
            min_idx = np.argmin(valid_dtgs) # Most negative DTG corresponds to max mass loss rate
            peak_temp = valid_temps[min_idx]
            peak_val = valid_dtgs[min_idx]
            peak_point = [float(peak_temp), float(-peak_val)] # Plotting inverted DTG

        # Build Option
        option = {
            "backgroundColor": "#fff",
            "title": {
                "text": sheet_name,
                "left": "center",
                "top": 10,
                "textStyle": {
                    "fontSize": 16,
                    "fontWeight": "bold",
                    "fontFamily": "Noto Sans CJK KR"
                }
            },
            "tooltip": {
                "trigger": "axis",
                "axisPointer": {"type": "cross"},
                "formatter": None # Default formatter works well enough with array data
            },
            "legend": {
                "data": ["Weight (%)", "DTG (%/°C)"],
                "orient": "vertical",
                "left": "left",
                "top": "middle",
                "textStyle": {
                    "fontFamily": "Noto Sans CJK KR"
                }
            },
            "grid": {
                "left": "10%",
                "right": "10%",
                "bottom": "15%",
                "containLabel": True
            },
            "xAxis": {
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 30,
                "type": "value",
                "axisLine": {"lineStyle": {"color": "#333"}},
                "axisTick": {"show": True},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "splitLine": {"show": False}
            },
            "yAxis": [
                {
                    "name": "Weight (%)",
                    "nameLocation": "middle",
                    "nameGap": 40,
                    "type": "value",
                    "position": "left",
                    "axisLine": {"lineStyle": {"color": "#1f4e9c"}},
                    "axisTick": {"show": True},
                    "axisLabel": {"fontFamily": "Noto Sans CJK KR", "formatter": "{value}"},
                    "splitLine": {"show": True, "lineStyle": {"color": "#eee"}}
                },
                {
                    "name": "DTG (%/°C)",
                    "nameLocation": "middle",
                    "nameGap": 40,
                    "type": "value",
                    "position": "right",
                    "axisLine": {"lineStyle": {"color": "#c0392b"}},
                    "axisTick": {"show": True},
                    "axisLabel": {"fontFamily": "Noto Sans CJK KR", "formatter": "{value}"},
                    "splitLine": {"show": False}
                }
            ],
            "series": [
                {
                    "name": "Weight (%)",
                    "type": "line",
                    "yAxisIndex": 0,
                    "data": series_weight,
                    "lineStyle": {"color": "#1f4e9c", "width": 2},
                    "itemStyle": {"color": "#1f4e9c"},
                    "showSymbol": False,
                    "markArea": {
                        "silent": True,
                        "itemStyle": {
                            "color": "rgba(200, 200, 200, 0.3)" # Light gray shading
                        },
                        "data": [
                            [
                                {"xAxis": 300},
                                {"xAxis": 450}
                            ]
                        ]
                    },
                    "markPoint": {
                        "symbolSize": 10,
                        "label": {
                            "show": True,
                            "position": "top",
                            "formatter": function_placeholder_residue(residue_val),
                            "fontFamily": "Noto Sans CJK KR",
                            "color": "#1f4e9c",
                            "fontWeight": "bold"
                        },
                        "data": [
                            {
                                "coord": residue_point if residue_point else [0,0],
                                "symbol": "circle",
                                "itemStyle": {"color": "#1f4e9c"}
                            }
                        ] if residue_point else []
                    }
                },
                {
                    "name": "DTG (%/°C)",
                    "type": "line",
                    "yAxisIndex": 1,
                    "data": series_dtg,
                    "lineStyle": {"color": "#c0392b", "width": 2},
                    "itemStyle": {"color": "#c0392b"},
                    "showSymbol": False,
                    "markPoint": {
                        "symbolSize": 10,
                        "label": {
                            "show": True,
                            "position": "top",
                            "formatter": function_placeholder_peak(peak_temp),
                            "fontFamily": "Noto Sans CJK KR",
                            "color": "#c0392b",
                            "fontWeight": "bold"
                        },
                        "data": [
                            {
                                "coord": peak_point if peak_point else [0,0],
                                "symbol": "diamond",
                                "itemStyle": {"color": "#c0392b"}
                            }
                        ] if peak_point else []
                    }
                }
            ]
        }
        
        figures.append({
            "name": f"plot_{sheet_name.replace(' ', '_')}",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        })
        
    return {"figures": figures}

def function_placeholder_residue(val):
    if val is None: return ""
    return f"Residue {val:.1f}%"

def function_placeholder_peak(temp):
    if temp is None: return ""
    return f"{temp:.0f} °C"

if __name__ == "__main__":
    result = get_tga_data(DATA_FILE)
    print(json.dumps(result))
```
