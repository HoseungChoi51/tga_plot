def get_onset_temperature(temp, weight):
    """
    Calculate the onset temperature using the tangent method.
    1. Find the point of maximum rate of weight loss (max DTG).
    2. Fit a linear regression to the weight data around this peak.
    3. Extrapolate the baseline (initial stable weight) and the tangent line.
    4. The intersection is the onset temperature.
    """
    # Calculate derivative manually if not provided or for consistency with temp spacing
    dt = np.diff(temp)
    dw = np.diff(weight)
    
    # Avoid division by zero
    mask = dt != 0
    dtg = np.zeros_like(weight[:-1])
    dtg[mask] = dw[mask] / dt[mask]
    
    # We look for the most negative DTG value (maximum mass loss rate) above 50C to avoid noise at start
    # Using indices relative to the original array
    valid_indices = np.where(temp[1:] > 50)[0]
    if len(valid_indices) == 0:
        return None
        
    dtg_valid = dtg[valid_indices]
    temp_valid = temp[1:][valid_indices]
    
    # Find index of minimum DTG (most negative)
    min_dtg_idx_local = np.argmin(dtg_valid)
    min_dtg_idx_global = valid_indices[min_dtg_idx_local] + 1 # +1 because dtg corresponds to diff
    
    peak_temp = temp[min_dtg_idx_global]
    
    # Define window for tangent fit (e.g., +/- 20 degrees around peak, or specific points)
    # A common approach is to use points on the steep slope side.
    # Let's take a small window around the peak derivative point to fit the tangent line.
    window_size = 5
    start_idx = max(0, min_dtg_idx_global - window_size)
    end_idx = min(len(temp), min_dtg_idx_global + window_size + 1)
    
    t_fit = temp[start_idx:end_idx]
    w_fit = weight[start_idx:end_idx]
    
    if len(t_fit) < 2:
        return None
        
    # Linear fit: y = mx + c
    m, c = np.polyfit(t_fit, w_fit, 1)
    
    # Baseline weight: average of initial weights before degradation starts significantly
    # Assuming first 10% of data is baseline or just first few points
    baseline_weight = np.mean(weight[:10])
    
    # Intersection: baseline_weight = m * T_onset + c
    if m == 0:
        return None
        
    t_onset = (baseline_weight - c) / m
    
    # Sanity check: onset should be less than peak temp
    if t_onset < peak_temp:
        return t_onset
    else:
        return None
