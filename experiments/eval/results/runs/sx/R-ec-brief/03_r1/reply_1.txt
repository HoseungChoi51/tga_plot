# Read the specific sheet
df = pd.read_excel(DATA_FILE, sheet_name="1006R1-V acid")

# Extract columns based on TGA guess
x_col = "Temperature"
y1_col = "Weight(%)"
y2_col = "D.W(%/C)"

# Prepare data
temp = df[x_col].values
weight = df[y1_col].values
dtg = df[y2_col].values * -1  # Multiply by -1 so peaks point up

# Filter for x-axis range 100~800 for plotting, but keep all for peak detection if needed?
# Usually peak detection is done on the full data or within visible range. 
# The request says "peak display 2", implying finding top 2 peaks.
# Let's filter data for the plot first to ensure markers are in range if they fall there.
mask_plot = (temp >= 100) & (temp <= 800)
temp_plot = temp[mask_plot]
weight_plot = weight[mask_plot]
dtg_plot = dtg[mask_plot]

# Find residue at 800 C
idx_800 = np.argmin(np.abs(temp - 800))
residue_val = weight[idx_800]
residue_temp = temp[idx_800]

# Find top 2 DTG peaks above 150 C
# We look at the filtered data for peaks within the plot range, or global?
# Standard practice: find peaks in the measured range > 150C.
mask_peak_search = (temp > 150)
temp_search = temp[mask_peak_search]
dtg_search = dtg[mask_peak_search]

if len(dtg_search) > 0:

    # Note: dtg here is already multiplied by -1, so we want max values
    top_2_indices_local = np.argsort(dtg_search)[-2:]
    top_2_indices_global = np.where(mask_peak_search)[0][top_2_indices_local]
    
    peak_temps = temp[top_2_indices_global]
    peak_vals = dtg[top_2_indices_global] # These are positive values now
    
    # Sort peaks by temperature for consistent labeling order if needed
    sort_idx = np.argsort(peak_temps)
    peak_temps = peak_temps[sort_idx]
    peak_vals = peak_vals[sort_idx]
else:
    peak_temps = []
    peak_vals = []
