```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
SHEET_NAME = "1006R1-V acid"

# Read the specified sheet
df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)

# Identify columns based on preview info
x_col = "Temperature"
y1_col = "Weight(%)"
y2_col = "D.W(%/C)"

# Extract data
temp = df[x_col].values
weight = df[y1_col].values
dtg = df[y2_col].values * -1  # Invert DTG so peaks point up

# Filter for x-axis range 100~800 is handled by axis min/max in ECharts, 
# but we need to find peaks within this range or generally? 
# User said "x축 범위 100~800", so axis limits are set.
# "피크 표시 2개": Find the top 2 highest DTG peaks above 150°C (standard rule) 
# but also constrained to the visible range if implied, though standard TGA peak detection usually looks at the whole curve or relevant decomposition steps.
# Let's find local maxima in DTG above 150°C.

def find_peaks(x, y, min_height=0.01):
    peaks = []
    for i in range(1, len(y)-1):
        if y[i] > y[i-1] and y[i] > y[i+1]:
            if y[i] > min_height:
                peaks.append((i, x[i], y[i]))
    return peaks

# Find all peaks above 150 C
valid_indices = [i for i, t in enumerate(temp) if t > 150]
if not valid_indices:
    valid_indices = list(range(len(temp)))

peaks = []
for i in valid_indices:
    if i > 0 and i < len(dtg) - 1:
        if dtg[i] > dtg[i-1] and dtg[i] > dtg[i+1]:
            peaks.append({'index': i, 'temp': temp[i], 'height': dtg[i]})

# Sort by height descending and take top 2
peaks.sort(key=lambda p: p['height'], reverse=True)
top_2_peaks = peaks[:2]

# Prepare series data
series_data_y1 = [[float(t), float(w)] for t, w in zip(temp, weight)]
series_data_y2 = [[float(t), float(d)] for t, d in zip(temp, dtg)]

# Calculate Residue at nearest 800 C
target_temp_residue = 800.0
idx_residue = np.argmin(np.abs(temp - target_temp_residue))
residue_val = weight[idx_residue]
residue_temp = temp[idx_residue]

# Markers for residue and peaks
markers_y1 = []
markers_y2 = []

# Add residue marker to y1 (Weight)
markers_y1.append({
    "coord": [float(residue_temp), float(residue_val)],
    "value": f"Residue {residue_val:.1f}%",
    "symbolSize": 10,
    "label": {
        "show": True,
        "formatter": f"Residue {residue_val:.1f}%",
        "position": "right",
        "fontSize": 10,
        "color": "#1f4e9c"
    }
})

# Add peak markers to y2 (DTG)
for p in top_2_peaks:
    markers_y2.append({
        "coord": [float(p['temp']), float(p['height'])],
        "value": f"{p['temp']:.0f} °C",
        "symbolSize": 10,
        "label": {
            "show": True,
            "formatter": f"{p['temp']:.0f} °C",
            "position": "top",
            "fontSize": 10,
            "color": "#c0392b"
        }
    })

# Construct ECharts Option
option = {
    "title": {
        "text": SHEET_NAME,
        "left": "center",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 14
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "legend": {
        "data": ["Weight (%)", "DTG"],
        "orient": "vertical",
        "left": "left",
        "top": "middle",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "type": "value",
        "min": 100,
        "max": 800,
        "splitLine": {
            "show": False
        },
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "yAxis": [
        {
            "name": "Weight (%)",
            "type": "value",
            "position": "left",
            "axisLabel": {
                "formatter": "{value}",
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "lineStyle": {
                    "type": "dashed"
                }
            }
        },
        {
            "name": "DTG (%/°C)",
            "type": "value",
            "position": "right",
            "axisLabel": {
                "formatter": "{value}",
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "yAxisIndex": 0,
            "data": series_data_y1,
            "smooth": True,
            "lineStyle": {
                "color": "#1f4e9c",
                "width": 2
            },
            "itemStyle": {
                "color": "#1f4e9c"
            },
            "markPoint": {
                "data": markers_y1,
                "symbol": "circle"
            },
            "showSymbol": False
        },
        {
            "name": "DTG",
            "type": "line",
            "yAxisIndex": 1,
            "data": series_data_y2,
            "smooth": True,
            "lineStyle": {
                "color": "#c0392b",
                "width": 2
            },
            "itemStyle": {
                "color": "#c0392b"
            },
            "markPoint": {
                "data": markers_y2,
                "symbol": "circle"
            },
            "showSymbol": False
        }
    ]
}

result = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(result, ensure_ascii=False))
```
