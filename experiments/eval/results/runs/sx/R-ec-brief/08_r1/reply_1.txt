```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def generate_charts():
    figures = []
    
    # Read all sheets from the Excel file
    xls = pd.ExcelFile(DATA_FILE)
    
    for sheet_name in xls.sheet_names:
        df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        
        # Identify columns based on typical TGA structure and provided preview
        # Assuming standard column order or names from preview: Time, Temperature, Weight(%), D.W(%/C)
        temp_col = None
        weight_col = None
        dtg_col = None
        
        cols = df.columns.tolist()
        
        # Try to find columns by name similarity if possible, otherwise fallback to index
        for c in cols:
            c_lower = str(c).lower()
            if 'temp' in c_lower:
                temp_col = c
            elif 'weight' in c_lower and 'd.w' not in c_lower and 'derivative' not in c_lower:
                weight_col = c
            elif 'd.w' in c_lower or 'derivative' in c_lower or 'dtg' in c_lower:
                dtg_col = c
                
        # Fallback to indices if names aren't clear (based on preview: B=Temp, C=Weight, D=DTG)
        if temp_col is None and len(cols) >= 2: temp_col = cols[1]
        if weight_col is None and len(cols) >= 3: weight_col = cols[2]
        if dtg_col is None and len(cols) >= 4: dtg_col = cols[3]
        
        if temp_col is None or weight_col is None or dtg_col is None:
            continue
            
        temps = df[temp_col].values
        weights = df[weight_col].values
        dtgs = df[dtg_col].values
        
        # Filter valid data points
        mask = ~np.isnan(temps) & ~np.isnan(weights) & ~np.isnan(dtgs)
        temps = temps[mask]
        weights = weights[mask]
        dtgs = dtgs[mask]
        
        if len(temps) == 0:
            continue

        # Prepare data for ECharts
        # Series 1: Weight (%) - Left Y axis (yAxisIndex: 0)
        data_weight = [[float(t), float(w)] for t, w in zip(temps, weights)]
        
        # Series 2: DTG (%/°C) - Right Y axis (yAxisIndex: 1)
        # Multiply by -1 so peaks point up as per instructions
        data_dtg = [[float(t), float(-d)] for t, d in zip(temps, dtgs)]
        
        # Calculate Residue at nearest 800°C
        # Find index closest to 800
        idx_800 = np.argmin(np.abs(temps - 800))
        residue_val = weights[idx_800]
        residue_temp = temps[idx_800]
        
        # Calculate Peak temperature of highest DTG peak above 150°C
        # Filter DTG for temp > 150
        mask_peak = temps > 150
        if np.any(mask_peak):
            temps_filtered = temps[mask_peak]
            dtgs_filtered = dtgs[mask_peak]
            # Find max absolute value (since original DTG is negative during loss, we look for most negative, i.e., min value)
            # Or simply max of (-DTG) which corresponds to peak height
            dtgs_pos = -dtgs_filtered
            idx_max_peak = np.argmax(dtgs_pos)
            peak_temp = temps_filtered[idx_max_peak]
            peak_val_orig = dtgs_filtered[idx_max_peak]
            peak_val_display = -peak_val_orig # Positive value for display logic if needed, but marker uses coords
        else:
            peak_temp = None
            peak_val_orig = None

        # Build Option
        option = {
            "title": {
                "text": sheet_name,
                "left": "center",
                "textStyle": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 16
                }
            },
            "tooltip": {
                "trigger": "axis",
                "axisPointer": {
                    "type": "cross"
                },
                "formatter": function_placeholder # Cannot use JS functions, so default tooltip will be used. 
                # To customize without JS, we rely on default series name/value display.
            },
            "legend": {
                "data": ["무게 (%)", "미분 무게 (%/°C)"],
                "top": "middle",
                "left": "left",
                "orient": "vertical",
                "textStyle": {
                    "fontFamily": "Noto Sans CJK KR"
                }
            },
            "grid": {
                "left": "10%",
                "right": "10%",
                "bottom": "15%",
                "containLabel": True
            },
            "xAxis": [
                {
                    "name": "온도 (°C)",
                    "nameLocation": "middle",
                    "nameGap": 30,
                    "type": "value",
                    "scale": True,
                    "axisLabel": {
                        "fontFamily": "Noto Sans CJK KR"
                    },
                    "axisNameTextStyle": {
                        "fontFamily": "Noto Sans CJK KR",
                        "fontWeight": "bold"
                    }
                }
            ],
            "yAxis": [
                {
                    "name": "무게 (%)",
                    "nameLocation": "middle",
                    "nameGap": 45,
                    "type": "value",
                    "position": "left",
                    "axisLabel": {
                        "fontFamily": "Noto Sans CJK KR"
                    },
                    "axisNameTextStyle": {
                        "fontFamily": "Noto Sans CJK KR",
                        "fontWeight": "bold"
                    },
                    "splitLine": {
                        "lineStyle": {
                            "type": "dashed"
                        }
                    }
                },
                {
                    "name": "미분 무게 (%/°C)",
                    "nameLocation": "middle",
                    "nameGap": 45,
                    "type": "value",
                    "position": "right",
                    "axisLabel": {
                        "fontFamily": "Noto Sans CJK KR"
                    },
                    "axisNameTextStyle": {
                        "fontFamily": "Noto Sans CJK KR",
                        "fontWeight": "bold"
                    },
                    "splitLine": {
                        "show": False
                    }
                }
            ],
            "series": [
                {
                    "name": "무게 (%)",
                    "type": "line",
                    "yAxisIndex": 0,
                    "data": data_weight,
                    "smooth": False,
                    "symbol": "none",
                    "lineStyle": {
                        "color": "#1f4e9c",
                        "width": 2
                    },
                    "markPoint": {
                        "data": [
                            {
                                "coord": [residue_temp, residue_val],
                                "value": f"Residue {residue_val:.1f}%",
                                "itemStyle": {
                                    "color": "#1f4e9c"
                                },
                                "label": {
                                    "show": True,
                                    "position": "top",
                                    "formatter": "{@[0]}°C\n{@[1]:.1f}%\nResidue",
                                    "fontFamily": "Noto Sans CJK KR",
                                    "backgroundColor": "rgba(255,255,255,0.7)",
                                    "borderColor": "#1f4e9c",
                                    "borderWidth": 1,
                                    "padding": 4
                                }
                            }
                        ]
                    }
                },
                {
                    "name": "미분 무게 (%/°C)",
                    "type": "line",
                    "yAxisIndex": 1,
                    "data": data_dtg,
                    "smooth": False,
                    "symbol": "none",
                    "lineStyle": {
                        "color": "#c0392b",
                        "width": 2
                    },
                    "markPoint": {}
                }
            ]
        }
        
        # Add Peak MarkPoint if found
        if peak_temp is not None:
            # The coordinate needs to match the series data format. 
            # For DTG series, y-value is -original_DTG.
            # We want to mark the peak location.
            # Note: The markPoint coord should be [x, y_in_series_units]
            # Since we plotted -DTG, the peak value in plot units is positive.
            
            # Re-calculate exact value at that temp from filtered arrays to ensure consistency
            # Actually, we just need the coordinate.
            val_at_peak = -dtgs[np.where(temps == peak_temp)[0][0]] if np.any(temps == peak_temp) else -peak_val_orig
            
            option["series"][1]["markPoint"] = {
                "data": [
                    {
                        "coord": [peak_temp, val_at_peak],
                        "value": f"{peak_temp:.0f} °C",
                        "itemStyle": {
                            "color": "#c0392b"
                        },
                        "label": {
                            "show": True,
                            "position": "top",
                            "formatter": "{@[0]:.0f} °C",
                            "fontFamily": "Noto Sans CJK KR",
                            "backgroundColor": "rgba(255,255,255,0.7)",
                            "borderColor": "#c0392b",
                            "borderWidth": 1,
                            "padding": 4
                        }
                    }
                ]
            }

        fig = {
            "name": f"plot_{sheet_name.replace(' ', '_').replace('/', '_')}",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
        figures.append(fig)
        
    return {"figures": figures}

if __name__ == "__main__":
    result = generate_charts()
    print(json.dumps(result, ensure_ascii=False))
```
