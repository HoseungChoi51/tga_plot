```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def get_nearest_value(df, x_col, y_col, target_x):
    """Find the y value at the point nearest to target_x."""
    idx = (df[x_col] - target_x).abs().idxmin()
    return df.loc[idx, y_col], df.loc[idx, x_col]

def find_peak_above_threshold(df, x_col, dtg_col, threshold=150):
    """Find the temperature of the highest DTG peak above a certain temperature."""
    # Filter data above threshold
    filtered_df = df[df[x_col] > threshold].copy()
    if filtered_df.empty:
        return None, None
    
    # Find max absolute value in DTG (since DTG is negative for mass loss)
    # We want the most negative value -> lowest value
    min_dtg_idx = filtered_df[dtg_col].idxmin()
    
    temp = filtered_df.loc[min_dtg_idx, x_col]
    val = filtered_df.loc[min_dtg_idx, dtg_col]
    
    return temp, val

def process_sheet(sheet_name, df):
    # Identify columns based on standard TGA structure provided in preview
    # A: Time, B: Temperature, C: Weight(%), D: D.W(%/C)
    x_col = 'Temperature'
    y1_col = 'Weight(%)'
    y2_col = 'D.W(%/C)'
    
    # Ensure numeric types
    df[x_col] = pd.to_numeric(df[x_col], errors='coerce')
    df[y1_col] = pd.to_numeric(df[y1_col], errors='coerce')
    df[y2_col] = pd.to_numeric(df[y2_col], errors='coerce')
    
    # Drop NaNs
    df_clean = df.dropna(subset=[x_col, y1_col, y2_col])
    
    # Prepare series data
    # Series 1: Weight (%)
    weight_data = [[float(t), float(w)] for t, w in zip(df_clean[x_col], df_clean[y1_col])]
    
    # Series 2: DTG (%/C) multiplied by -1 so peaks point up
    dtg_data = [[float(t), float(-d)] for t, d in zip(df_clean[x_col], df_clean[y2_col])]
    
    # Calculate Residue at 800°C
    residue_val, residue_temp = get_nearest_value(df_clean, x_col, y1_col, 800.0)
    
    # Calculate Peak Temperature (>150°C)
    peak_temp, _ = find_peak_above_threshold(df_clean, x_col, y2_col, 150.0)
    
    # Determine axis ranges
    x_min = float(df_clean[x_col].min())
    x_max = float(df_clean[x_col].max())
    y1_min = float(df_clean[y1_col].min())
    y1_max = float(df_clean[y1_col].max())
    
    # For DTG (y2), since we flipped sign, min becomes max and vice versa
    dtg_vals = [-d for d in df_clean[y2_col]]
    y2_min = min(dtg_vals)
    y2_max = max(dtg_vals)
    
    # Add some padding
    y1_pad = (y1_max - y1_min) * 0.05
    y2_pad = (y2_max - y2_min) * 0.05
    
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "min": x_min,
            "max": x_max,
            "axisLabel": {
                "formatter": "{value}"
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "min": y1_min - y1_pad,
                "max": y1_max + y1_pad,
                "splitLine": {
                    "show": True
                }
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "position": "right",
                "min": y2_min - y2_pad,
                "max": y2_max + y2_pad,
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "data": weight_data,
                "yAxisIndex": 0,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [residue_temp, residue_val],
                            "symbolSize": 8,
                            "itemStyle": {
                                "color": "#1f4e9c"
                            },
                            "label": {
                                "show": True,
                                "formatter": f"Residue {residue_val:.1f}%",
                                "position": "top",
                                "fontFamily": "Noto Sans CJK KR",
                                "fontSize": 10
                            }
                        }
                    ]
                }
            },
            {
                "name": "DTG",
                "type": "line",
                "data": dtg_data,
                "yAxisIndex": 1,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 2
                },
                "markPoint": {
                    "data": []
                }
            }
        ]
    }
    
    # Add Peak Marker if found
    if peak_temp is not None:
        # Find the corresponding Y value for the marker on the DTG curve (which is inverted)
        # We need the value from dtg_data at peak_temp
        # Since dtg_data is list of lists, we can search or recalculate
        # Recalculating from original df_clean is safer for precision match
        idx = (df_clean[x_col] - peak_temp).abs().idxmin()
        dtg_val_original = df_clean.loc[idx, y2_col]
        dtg_val_plot = -dtg_val_original
        
        option["series"][1]["markPoint"]["data"].append({
            "coord": [peak_temp, dtg_val_plot],
            "symbolSize": 8,
            "itemStyle": {
                "color": "#c0392b"
            },
            "label": {
                "show": True,
                "formatter": f"{int(round(peak_temp))} °C",
                "position": "top",
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 10
            }
        })
        
    # Add Vertical Reference Line at 450°C with label "산화 시작"
    # Using markLine on the first series (Weight) to ensure it spans the plot area visually
    # Or better, use graphic component or just add to both? 
    # ECharts markLine on a series draws relative to that series' coordinate system.
    # To make it span both Y axes, it's often easier to put it on one series but style it to go full height.
    # Actually, markLine with xAxis value works well.
    
    ref_line = {
        "type": "line",
        "xAxis": 450,
        "lineStyle": {
            "color": "#666666",
            "type": "dashed",
            "width": 1
        },
        "label": {
            "show": True,
            "formatter": "산화 시작",
            "position": "insideEndTop",
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 10,
            "color": "#666666"
        }
    }
    
    # Add markLine to the first series
    if "markLine" not in option["series"][0]:
        option["series"][0]["markLine"] = {"data": []}
    option["series"][0]["markLine"]["data"].append(ref_line)
    
    return option

figures = []

# Read Excel file
xl_file = pd.ExcelFile(DATA_FILE)
sheet_names = xl_file.sheet_names

for name in sheet_names:
    df = pd.read_excel(xl_file, sheet_name=name, header=0)
    # Basic column cleanup if needed, though headers seem clean
    option = process_sheet(name, df)
    
    figures.append({
        "name": f"plot_{name.replace(' ', '_').replace('/', '_')}",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    })

output = {"figures": figures}
print(json.dumps(output, ensure_ascii=False))
```
