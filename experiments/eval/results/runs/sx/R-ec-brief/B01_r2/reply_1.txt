```python
import pandas as pd
import numpy as np
import json

def main():
    file_path = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
    df = pd.read_csv(file_path)
    
    # Column names from preview
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    # Extract data and sort by temperature just in case
    df_sorted = df.sort_values(by=temp_col)
    temps = df_sorted[temp_col].values
    weights = df_sorted[weight_col].values
    dtgs = df_sorted[dtg_col].values
    
    # Prepare series data for ECharts [x, y] pairs
    # Weight series
    weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]
    
    # DTG series (negated so peaks point up)
    dtg_data = [[float(t), float(-d)] for t, d in zip(temps, dtgs)]
    
    # Define the three regions for shading and labels
    regions = [
        {"name": "H2O", "start": 120, "end": 250, "color": "#FFB3BA"}, # Light Red/Pink
        {"name": "CO", "start": 430, "end": 580, "color": "#BAFFC9"},  # Light Green
        {"name": "CO2", "start": 640, "end": 830, "color": "#BAE1FF"}  # Light Blue
    ]
    
    # Find Residue at ~800°C
    idx_800 = np.argmin(np.abs(temps - 800))
    residue_temp = temps[idx_800]
    residue_weight = weights[idx_800]
    
    # Find Peak DTG above 150°C
    mask_peak = temps > 150
    if np.any(mask_peak):
        # We look for the maximum of the negated DTG (which corresponds to the most negative original DTG)
        # In our dtg_data, we stored -DTG. So we want the max value of -DTG where T > 150.
        dtg_neg = np.array([d[1] for d in dtg_data])
        temps_arr = np.array([d[0] for d in dtg_data])
        
        valid_indices = np.where(temps_arr > 150)[0]
        if len(valid_indices) > 0:
            peak_idx_in_valid = np.argmax(dtg_neg[valid_indices])
            global_peak_idx = valid_indices[peak_idx_in_valid]
            peak_temp = temps_arr[global_peak_idx]
            peak_val = dtg_neg[global_peak_idx]
        else:
            peak_temp = None
            peak_val = None
    else:
        peak_temp = None
        peak_val = None

    # Build markArea data for the regions
    mark_area_data = []
    for reg in regions:
        mark_area_data.append([
            {
                "name": reg["name"],
                "itemStyle": {"color": reg["color"]},
                "label": {
                    "show": True,
                    "position": "insideTop",
                    "formatter": reg["name"],
                    "fontSize": 12,
                    "fontWeight": "bold",
                    "color": "#333"
                }
            },
            {
                "xAxis": reg["start"],
                "yAxis": "max" # Extend to top of chart visually within area definition if possible, or use specific Y
            }
        ])
        # Note: For markArea in ECharts with two axes, it's tricky to span full height easily without knowing Y limits.
        # A common trick is to set yAxis to a very large number or use 'max'/'min' keywords if supported by version.
        # However, standard markArea usually needs explicit coordinates. 
        # Let's find the min/max of weight to set the Y range for the areas to cover the whole plot vertically.
        
    # Calculate Y range for Weight to ensure marks cover the area
    w_min = np.min(weights)
    w_max = np.max(weights)
    # Add some padding
    y_pad = (w_max - w_min) * 0.1
    y_limit_low = w_min - y_pad
    y_limit_high = w_max + y_pad
    
    # Re-build mark_area_data with explicit Y coordinates to fill the vertical space of the left axis (Weight)
    mark_area_data = []
    for reg in regions:
        mark_area_data.append([
            {
                "name": reg["name"],
                "itemStyle": {"color": reg["color"], "opacity": 0.3},
                "label": {
                    "show": True,
                    "position": "insideTop",
                    "formatter": reg["name"],
                    "fontSize": 14,
                    "fontWeight": "bold",
                    "color": "#000"
                }
            },
            {
                "xAxis": reg["start"],
                "yAxis": y_limit_high
            }
        ])
        mark_area_data[-1][1]["xAxis"] = reg["end"]
        mark_area_data[-1][1]["yAxis"] = y_limit_low

    # Build markPoint for Residue and Peak
    mark_point_data = []
    
    # Residue Marker
    mark_point_data.append({
        "coord": [float(residue_temp), float(residue_weight)],
        "value": f"Residue {residue_weight:.1f}%",
        "symbolSize": 8,
        "itemStyle": {"color": "#1f4e9c"},
        "label": {
            "show": True,
            "position": "right",
            "formatter": "{b}",
            "fontSize": 12,
            "color": "#1f4e9c"
        }
    })
    
    # Peak Marker
    if peak_temp is not None:
        # The marker should be on the DTG curve (Right Axis). 
        # Since we are plotting DTG on y2, we need to add this markPoint to the DTG series.
        pass 

    # Constructing the Option
    option = {
        "backgroundColor": "#fff",
        "title": {
            "text": "TGA/DTG Analysis (caox)",
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG (-%/°C)"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "axisLine": {"lineStyle": {"color": "#333"}},
            "splitLine": {"show": False}
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "axisLine": {"lineStyle": {"color": "#1f4e9c"}},
                "axisLabel": {"formatter": "{value} %"},
                "min": round(y_limit_low, 1),
                "max": round(y_limit_high, 1)
            },
            {
                "type": "value",
                "name": "DTG (-%/°C)",
                "position": "right",
                "axisLine": {"lineStyle": {"color": "#c0392b"}},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "data": weight_data,
                "smooth": True,
                "showSymbol": False,
                "lineStyle": {"color": "#1f4e9c", "width": 2},
                "markArea": {
                    "silent": True,
                    "data": mark_area_data
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [float(residue_temp), float(residue_weight)],
                            "value": f"Residue {residue_weight:.1f}%",
                            "symbolSize": 8,
                            "itemStyle": {"color": "#1f4e9c"},
                            "label": {
                                "show": True,
                                "position": "right",
                                "formatter": "{b}",
                                "fontSize": 12,
                                "color": "#1f4e9c",
                                "backgroundColor": "rgba(255,255,255,0.7)",
                                "padding": [2, 4],
                                "borderRadius": 3
                            }
                        }
                    ]
                }
            },
            {
                "name": "DTG (-%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": dtg_data,
                "smooth": True,
                "showSymbol": False,
                "lineStyle": {"color": "#c0392b", "width": 2},
                "markPoint": {
                    "data": [
                        {
                            "coord": [float(peak_temp), float(peak_val)],
                            "value": f"{peak_temp:.0f} °C",
                            "symbolSize": 8,
                            "itemStyle": {"color": "#c0392b"},
                            "label": {
                                "show": True,
                                "position": "top",
                                "formatter": "{b}",
                                "fontSize": 12,
                                "color": "#c0392b",
                                "backgroundColor": "rgba(255,255,255,0.7)",
                                "padding": [2, 4],
                                "borderRadius": 3
                            }
                        }
                    ]
                } if peak_temp is not None else None
            }
        ]
    }
    
    # Clean up None values in series if any
    for s in option["series"]:
        if "markPoint" in s and s["markPoint"] is None:
            del s["markPoint"]

    output = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
