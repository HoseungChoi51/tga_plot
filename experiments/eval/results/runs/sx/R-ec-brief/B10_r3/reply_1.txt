import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"

# Read only the 'Annealed' sheet
df = pd.read_excel(DATA_FILE, sheet_name='Annealed')

# Extract columns based on standard TGA structure identified in preview
temp_col = 'Temperature (°C)'
weight_col = 'Weight (%)'
dtg_col = 'Deriv. Weight (%/°C)'

temps = df[temp_col].values
weights = df[weight_col].values
dtgs = df[dtg_col].values * -1  # Invert DTG so peaks point up

# Prepare data for ECharts series: [[x, y], ...]
weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]
dtg_data = [[float(t), float(d)] for t, d in zip(temps, dtgs)]

# --- Calculate Residue at ~800°C ---
idx_800 = np.argmin(np.abs(temps - 800))
residue_val = weights[idx_800]
residue_temp = temps[idx_800]
residue_label_text = f"Residue {residue_val:.1f}%"

# --- Calculate Peak Temperature (>150°C) ---
mask_high_temp = temps > 150
if np.any(mask_high_temp):
    # Find index of max inverted DTG among points > 150°C
    # Note: dtgs array is already inverted (-1 * original)
    dtg_subset = dtgs[mask_high_temp]
    temp_subset = temps[mask_high_temp]
    
    peak_idx_local = np.argmax(dtg_subset)
    peak_temp = temp_subset[peak_idx_local]
    peak_dtg_val = dtg_subset[peak_idx_local] # This is the positive value after inversion
    
    peak_label_text = f"{peak_temp:.0f} °C"
else:
    peak_temp = None
    peak_dtg_val = None
    peak_label_text = ""

# --- Plot Configuration ---

color_black = "#000000"

        "nameTextStyle": {
            "fontSize": font_size,

        }
    },
    "yAxis": [
        {
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "position": "left",
            "min": 0,
            "max": 105,
            "axisLine": {"lineStyle": {"color": color_black}},
            "axisTick": {"show": True, "length": 4, "lineStyle": {"color": color_black}},
            "axisLabel": {
                "fontSize": font_size,

        {

            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "position": "right",
            # Auto scale usually works well, but we can set min/max if needed. 
            # Since it's inverted, min will be negative? No, we inverted values to make peaks positive.
            # Original DTG was negative during loss. We multiplied by -1. So peaks are positive.
            # Let's let echarts handle the range or set a reasonable one.
