```python
import pandas as pd
import numpy as np
import json

def main():
    file_path = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
    df = pd.read_csv(file_path)
    
    # Standard TGA columns based on preview
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    # Prepare data for ECharts
    temps = df[temp_col].values
    weights = df[weight_col].values
    dtgs = df[dtg_col].values
    
    # Invert DTG so peaks point up
    dtgs_inv = -dtgs
    
    # Series 1: Weight (%)
    series_weight = {
        "name": "Weight (%)",
        "type": "line",
        "yAxisIndex": 0,
        "data": [[float(t), float(w)] for t, w in zip(temps, weights)],
        "itemStyle": {"color": "#1f4e9c"},
        "lineStyle": {"color": "#1f4e9c", "width": 2},
        "showSymbol": False
    }
    
    # Series 2: DTG (-DTG)
    series_dtg = {
        "name": "DTG (-)",
        "type": "line",
        "yAxisIndex": 1,
        "data": [[float(t), float(d)] for t, d in zip(temps, dtgs_inv)],
        "itemStyle": {"color": "#c0392b"},
        "lineStyle": {"color": "#c0392b", "width": 2},
        "showSymbol": False
    }
    
    # Regions to shade and label
    regions = [
        {"start": 120, "end": 250, "label": "H2O", "color": "rgba(255, 165, 0, 0.2)"},   # Orange-ish
        {"start": 430, "end": 580, "label": "CO", "color": "rgba(0, 191, 255, 0.2)"},    # DeepSkyBlue-ish
        {"start": 640, "end": 830, "label": "CO2", "color": "rgba(50, 205, 50, 0.2)"}     # LimeGreen-ish
    ]
    
    mark_areas = []
    mark_points_labels = []
    
    for reg in regions:
        s = reg["start"]
        e = reg["end"]
        c = reg["color"]
        lbl = reg["label"]
        
        # Add area to a dummy series or the first series? 
        # Best practice for multiple areas with different colors is usually one series per area or using markArea on one series if possible, 
        # but markArea items can have individual styles. Let's put them on the Weight series.
        mark_areas.append([
            {
                "xAxis": s,
                "itemStyle": {"color": c}
            },
            {
                "xAxis": e
            }
        ])
        
        # Calculate Y position for text label (top of the chart area roughly)
        # We'll place it near the top of the plot. Since we don't know exact max y dynamically easily without more calc,
        # we can use 'insideTop' or calculate based on max weight. Max weight is ~100.
        # Let's find the average weight in that region to place the label reasonably, or just fixed high value.
        # Fixed high value relative to axis is easier. Max weight is 100. Let's put at 95%.
        mid_t = (s + e) / 2
        
        mark_points_labels.append({
            "coord": [mid_t, 95], # Place text near top
            "value": lbl,
            "symbolSize": 0,
            "label": {
                "show": True,
                "position": "top",
                "formatter": lbl,
                "fontSize": 14,
                "fontWeight": "bold",
                "color": "#333"
            }
        })

    # Apply markArea to the weight series
    series_weight["markArea"] = {
        "silent": True,
        "data": mark_areas
    }
    
    # Apply markPoint labels to the weight series as well
    series_weight["markPoint"] = {
        "data": mark_points_labels
    }

    # Residue Label Calculation (Nearest to 800°C)
    idx_800 = np.argmin(np.abs(temps - 800))
    residue_temp = temps[idx_800]
    residue_weight = weights[idx_800]
    
    residue_label = {
        "coord": [residue_temp, residue_weight],
        "value": f"Residue {residue_weight:.1f}%",
        "symbolSize": 6,
        "itemStyle": {"color": "#1f4e9c"},
        "label": {
            "show": True,
            "position": "bottom",
            "formatter": f"Residue {residue_weight:.1f}%",
            "fontSize": 12,
            "color": "#1f4e9c"
        }
    }
    
    # Peak Label Calculation (Highest DTG peak > 150°C)
    mask = temps > 150
    dtg_masked = dtgs_inv[mask]
    temps_masked = temps[mask]
    
    if len(dtg_masked) > 0:
        max_dtg_idx = np.argmax(dtg_masked)
        peak_temp = temps_masked[max_dtg_idx]
        peak_val = dtg_masked[max_dtg_idx]
        
        # Find corresponding weight for coordinate plotting if needed, but peak label usually just points to temp on x-axis or curve.
        # The prompt says "with a marker". Usually markers are on the curve.
        # Let's find the index in original array closest to peak_temp
        idx_peak_orig = np.argmin(np.abs(temps - peak_temp))
        peak_weight_at_temp = weights[idx_peak_orig]
        
        peak_label = {
            "coord": [peak_temp, peak_weight_at_temp], # Marker on weight curve? Or DTG curve? 
            # Prompt: "written like '600 °C', with a marker." 
            # Usually peak refers to DTG peak. Placing marker on DTG curve might be confusing visually if scales differ greatly.
            # However, standard TGA plots often put the peak temp label near the DTG peak.
            # Let's put the marker on the DTG curve since it's the "DTG peak".
            # But wait, series_weight has the markPoints defined above for H2O etc.
            # If I add to series_weight, it plots on Weight curve.
            # If I add to series_dtg, it plots on DTG curve.
            # The request asks for "Peak ... with a marker". It implies marking the location of the peak.
            # Let's add it to the DTG series for accuracy of what "peak" means physically, OR just a vertical line/text.
            # The prompt example format: `{"coord": [temp, val], ...}`.
            # Let's stick to putting the residue on Weight and Peak on DTG.
            
            # Re-evaluating: The previous markPoints were added to series_weight.
            # I will create a separate markPoint configuration for the DTG series for the temperature peak.
            
            series_dtg["markPoint"] = {
                "data": [{
                    "coord": [peak_temp, peak_val],
                    "value": f"{peak_temp:.0f} °C",
                    "symbolSize": 6,
                    "itemStyle": {"color": "#c0392b"},
                    "label": {
                        "show": True,
                        "position": "top",
                        "formatter": f"{peak_temp:.0f} °C",
                        "fontSize": 12,
                        "color": "#c0392b"
                    }
                }]
            }
    else:
        series_dtg["markPoint"] = {}

    # Add residue to series_weight markPoint list (appending to existing ones)
    series_weight["markPoint"]["data"].append(residue_label)

    option = {
        "title": {
            "text": "caox",
            "left": "center",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"}
        },
        "legend": {
            "data": ["Weight (%)", "DTG (-)"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR"
                },
                "splitLine": {"show": True, "lineStyle": {"type": "dashed"}}
            },
            {
                "type": "value",
                "name": "DTG (-)",
                "position": "right",
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR"
                },
                "splitLine": {"show": False}
            }
        ],
        "series": [
            series_weight,
            series_dtg
        ]
    }

    output = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
