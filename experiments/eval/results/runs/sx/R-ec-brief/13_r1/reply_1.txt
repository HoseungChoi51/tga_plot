```python
import pandas as pd
import numpy as np
import json

def main():
    file_path = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"
    
    # Read the Excel file
    xls = pd.ExcelFile(file_path)
    sheet_name = "CNT-01"
    df = pd.read_excel(xls, sheet_name=sheet_name)
    
    # Identify columns based on user request and data preview
    x_col = "Temperature (°C)"
    y1_col = "Weight (mg)"  # User requested Weight (mg) for y1
    y2_col = "Deriv. Weight (%/°C)"
    
    # Extract data
    temp = df[x_col].values
    weight_mg = df[y1_col].values
    dtg = df[y2_col].values
    
    # Prepare series data
    # Y1: Weight (mg)
    series_y1_data = [[float(t), float(w)] for t, w in zip(temp, weight_mg)]
    
    # Y2: DTG multiplied by -1 so peaks point up
    dtg_neg = -dtg
    series_y2_data = [[float(t), float(d)] for t, d in zip(temp, dtg_neg)]
    
    # Calculate Residue at nearest 800°C
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = weight_mg[idx_800]
    residue_temp = temp[idx_800]
    
    # Calculate Peak: highest DTG peak above 150°C
    # We look for the maximum of the original DTG (which is negative, so most negative value) 
    # OR since we plotted -DTG, we look for max of -DTG.
    # The prompt says: "peak: the temperature of the highest DTG peak above 150 °C"
    # Usually "highest peak" in TGA context refers to the magnitude of mass loss rate.
    # Since DTG is negative during loss, the "highest" peak visually (if inverted) or mathematically (most negative) is the target.
    # Let's find the index where -DTG is maximum (which corresponds to minimum DTG) for temps > 150.
    mask = temp > 150
    if np.any(mask):
        dtg_subset = dtg[mask]
        temp_subset = temp[mask]
        # Find index of minimum DTG (most negative) -> largest mass loss rate
        min_dtg_idx_local = np.argmin(dtg_subset)
        peak_temp = temp_subset[min_dtg_idx_local]
        peak_val_mg = weight_mg[temp == peak_temp][0] # Get corresponding weight mg for marker position? 
        # Actually, markers are usually placed on the curve they relate to or just labeled.
        # For DTG peak label, it's often placed near the DTG curve or just annotated.
        # Standard practice in these prompts: Marker on the curve. 
        # The prompt says "with a marker". It doesn't specify which curve. 
        # Usually, residue marker is on Weight curve. Peak marker is often on DTG curve or Weight curve at that temp.
        # Given y1 is Weight(mg), let's put residue marker on y1.
        # For peak, let's put it on y2 (DTG) or just annotate. Let's place it on the y2 axis value for clarity if possible, 
        # but ECharts markPoint on series 'DTG' makes sense.
        
        peak_y2_val = -dtg[temp == peak_temp][0]
    else:
        peak_temp = None
        peak_y2_val = None

    # Construct ECharts Option
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": "2%",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (mg)", "DTG"],
            "orient": "vertical",
            "left": "5%",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "10%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "axisLine": {"onZero": False},
            "splitLine": {"show": False}
        },
        "yAxis": [
            {
                "name": "Weight (mg)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "left",
                "axisLine": {"onZero": False},
                "splitLine": {"show": True, "lineStyle": {"type": "dashed"}}
            },
            {
                "name": "DTG (-%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "right",
                "axisLine": {"onZero": False},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (mg)",
                "type": "line",
                "yAxisIndex": 0,
                "data": series_y1_data,
                "itemStyle": {"color": "#1f4e9c"},
                "lineStyle": {"width": 2},
                "showSymbol": False,
                "markPoint": {
                    "data": [
                        {
                            "coord": [residue_temp, residue_val],
                            "value": f"Residue {residue_val:.1f} mg",
                            "symbolSize": 8,
                            "label": {
                                "formatter": "{b}",
                                "position": "top",
                                "distance": 10,
                                "backgroundColor": "rgba(255,255,255,0.7)",
                                "padding": [2, 4],
                                "borderRadius": 2,
                                "fontSize": 10,
                                "fontFamily": "Noto Sans CJK KR"
                            }
                        }
                    ]
                }
            },
            {
                "name": "DTG",
                "type": "line",
                "yAxisIndex": 1,
                "data": series_y2_data,
                "itemStyle": {"color": "#c0392b"},
                "lineStyle": {"width": 2},
                "showSymbol": False,
                "markPoint": {
                    "data": []
                }
            }
        ]
    }
    
    # Add peak marker if found
    if peak_temp is not None:
        option["series"][1]["markPoint"]["data"].append({
            "coord": [peak_temp, peak_y2_val],
            "value": f"{peak_temp:.0f} °C",
            "symbolSize": 8,
            "label": {
                "formatter": "{b}",
                "position": "top",
                "distance": 10,
                "backgroundColor": "rgba(255,255,255,0.7)",
                "padding": [2, 4],
                "borderRadius": 2,
                "fontSize": 10,
                "fontFamily": "Noto Sans CJK KR"
            }
        })

    result = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
```
