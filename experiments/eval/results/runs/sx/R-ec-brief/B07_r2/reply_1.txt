import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

def get_nearest_weight(df, temp_col, weight_col, target_temp):
    """Find the weight value at the temperature closest to target_temp."""
    idx = (df[temp_col] - target_temp).abs().idxmin()
    return df.loc[idx, weight_col], df.loc[idx, temp_col]

# Read only the second sheet ("PVC")
sheet_name = "PVC"
df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)

# Identify columns based on preview info
x_col = "Temperature (°C)"
y1_col = "Weight (%)"
y2_col = "Deriv. Weight (%/°C)"

# Prepare data for ECharts
# Sort by temperature just in case
df_sorted = df.sort_values(by=x_col)

temp_data = df_sorted[x_col].tolist()
weight_data = df_sorted[y1_col].tolist()
dtg_data = [-v for v in df_sorted[y2_col].tolist()] # Multiply by -1 so peaks point up

# Calculate residue and peak labels (standard TGA requirements)
# Residue: nearest to 800 C
res_val, res_temp = get_nearest_weight(df_sorted, x_col, y1_col, 800)
res_label_text = f"Residue {res_val:.1f}%"

# Peak: highest DTG peak above 150 C
# Filter for temp > 150
df_peak = df_sorted[df_sorted[x_col] > 150]
if not df_peak.empty:
    # Find index of max absolute DTG (which is min original DTG since it's negative loss)
    # But we multiplied by -1 for plotting, so we look for max in dtg_data corresponding to this subset
    # Let's work with original data logic first to be safe
    min_dtg_idx = df_peak[y2_col].idxmin() # Most negative = biggest mass loss rate
    peak_temp = df_sorted.loc[min_dtg_idx, x_col]
    peak_weight = df_sorted.loc[min_dtg_idx, y1_col]
else:
    peak_temp = None
    peak_weight = None

# Specific user request: Mark weights at 300 C and 600 C
w_300, t_300_actual = get_nearest_weight(df_sorted, x_col, y1_col, 300)
w_600, t_600_actual = get_nearest_weight(df_sorted, x_col, y1_col, 600)

# Build markPoints for specific temperatures
mark_points = []

# Standard Residue Marker
mark_points.append({
    "coord": [float(res_temp), float(res_val)],
    "value": res_label_text,
    "symbolSize": 10,
    "itemStyle": {"color": "#1f4e9c"},
    "label": {
        "show": True,
        "position": "right",
        "formatter": "{b}",
        "fontSize": 12,
        "fontWeight": "bold",
        "color": "#1f4e9c"
    }
})

# Standard Peak Marker
if peak_temp is not None:
    mark_points.append({
        "coord": [float(peak_temp), float(peak_weight)],
        "value": f"{peak_temp:.0f} °C",
        "symbolSize": 10,
        "itemStyle": {"color": "#c0392b"},
        "label": {
            "show": True,
            "position": "top",
            "formatter": "{b}",
            "fontSize": 12,
            "fontWeight": "bold",
            "color": "#c0392b"
        }
    })

# User Requested Markers: 300 C and 600 C
mark_points.append({
    "coord": [float(t_300_actual), float(w_300)],
    "value": f"300°C\n{w_300:.1f}%",
    "symbolSize": 8,
    "itemStyle": {"color": "#27ae60"},
    "label": {
        "show": True,
        "position": "bottom",
        "formatter": "{b}",
        "fontSize": 11,
        "color": "#27ae60"
    }
})

mark_points.append({
    "coord": [float(t_600_actual), float(w_600)],
    "value": f"600°C\n{w_600:.1f}%",
    "symbolSize": 8,
    "itemStyle": {"color": "#27ae60"},
    "label": {
        "show": True,
        "position": "bottom",
        "formatter": "{b}",
        "fontSize": 11,
        "color": "#27ae60"
    }
})

# Construct ECharts Option
option = {
    "title": {
        "text": sheet_name,
        "left": "center",
        "top": 10,
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",

    },
    "tooltip": {

        "axisPointer": {

        }
    },
    "legend": {

        "orient": "vertical",
        "left": "left",
        "top": "middle",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "type": "value",
        "min": 30,
        "max": 900,
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "yAxis": [
        {
            "name": "Weight (%)",
            "type": "value",
            "position": "left",
            "axisLine": {
                "lineStyle": {
                    "color": "#1f4e9c"
                }
            },
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR",
                "color": "#1f4e9c"
            },
            "splitLine": {
                "lineStyle": {
                    "type": "dashed",
                    "opacity": 0.5
                }
            }
        },
        {
            "name": "DTG (-%/°C)",
            "type": "value",
            "position": "right",
            "axisLine": {
                "lineStyle": {
                    "color": "#c0392b"
                }
            },
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR",
                "color": "#c0392b"
            },
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "yAxisIndex": 0,
            "data": list(zip(temp_data, weight_data)),
            "smooth": True,
            "symbol": "none",
            "lineStyle": {
                "color": "#1f4e9c",
                "width": 2
            },
            "markPoint": {
                "data": mark_points
            }
        },
        {
            "name": "DTG (-%/°C)",
            "type": "line",
            "yAxisIndex": 1,
            "data": list(zip(temp_data, dtg_data)),
            "smooth": True,
            "symbol": "none",
            "lineStyle": {
                "color": "#c0392b",

            }
        }
    ]
}

result = {
    "figures": [
        {
