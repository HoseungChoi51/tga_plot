```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

# Read the specific sheet requested by user: "1006R1-V normal"
df = pd.read_excel(DATA_FILE, sheet_name="1006R1-V normal")

# Extract columns based on preview info
x_data = df["Temperature"].values
y1_data = df["Weight(%)"].values
# y2 (DTG) is ignored per user request "y2축 없이 무게만 그려줘"

# Calculate Residue at nearest 800°C
# Find index of temperature closest to 800
idx_800 = np.argmin(np.abs(x_data - 800))
residue_temp = x_data[idx_800]
residue_weight = y1_data[idx_800]

# Prepare data for ECharts
data_points = []
for t, w in zip(x_data, y1_data):
    if not (np.isnan(t) or np.isnan(w)):
        data_points.append([float(t), float(w)])

# Determine axis ranges
x_min = min(x_data)
x_max = max(x_data)
y1_min = min(y1_data)
y1_max = max(y1_data)

# Add some padding to axes
x_range_pad = (x_max - x_min) * 0.05
y1_range_pad = (y1_max - y1_min) * 0.05

x_axis_min = x_min - x_range_pad
x_axis_max = x_max + x_range_pad
y1_axis_min = y1_min - y1_range_pad
y1_axis_max = y1_max + y1_range_pad

# Create markPoint for residue
mark_point_residue = {
    "coord": [float(residue_temp), float(residue_weight)],
    "value": f"Residue {residue_weight:.1f}%",
    "symbolSize": 10,
    "itemStyle": {"color": "#1f4e9c"},
    "label": {
        "show": True,
        "position": "top",
        "formatter": "{b}",
        "fontSize": 12,
        "color": "#1f4e9c"
    }
}

option = {
    "title": {
        "text": "1006R1-V normal",
        "left": "center",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 16
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "top": "15%",
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "min": x_axis_min,
        "max": x_axis_max,
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR"
        },
        "splitLine": {
            "show": True,
            "lineStyle": {
                "type": "dashed"
            }
        }
    },
    "yAxis": {
        "type": "value",
        "name": "Weight (%)",
        "nameLocation": "middle",
        "nameGap": 40,
        "min": y1_axis_min,
        "max": y1_axis_max,
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR"
        },
        "splitLine": {
            "show": True,
            "lineStyle": {
                "type": "dashed"
            }
        }
    },
    "series": [
        {
            "name": "Weight",
            "type": "line",
            "data": data_points,
            "smooth": False,
            "showSymbol": False,
            "lineStyle": {
                "color": "#1f4e9c",
                "width": 2
            },
            "markPoint": {
                "data": [mark_point_residue]
            }
        }
    ]
}

output = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(output, ensure_ascii=False))
```
