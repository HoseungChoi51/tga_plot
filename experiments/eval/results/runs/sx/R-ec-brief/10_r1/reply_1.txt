```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def make_tga_plot(df, sheet_name):
    # Identify columns based on standard TGA naming or guess from preview logic
    cols = df.columns.tolist()
    
    # Heuristic to find columns if names aren't exact matches to the preview but follow patterns
    x_col = None
    y1_col = None
    y2_col = None
    
    for c in cols:
        cl = str(c).lower()
        if 'temp' in cl:
            x_col = c
        elif 'weight' in cl and '%' in cl and 'd.w' not in cl and 'dtg' not in cl and 'derivative' not in cl:
            y1_col = c
        elif ('d.w' in cl or 'dtg' in cl or 'derivative' in cl) and '%' in cl:
            y2_col = c
            
    # Fallback to positional if heuristics fail (based on typical A=Time, B=Temp, C=Weight, D=DTG)
    if x_col is None and len(cols) >= 2: x_col = cols[1]
    if y1_col is None and len(cols) >= 3: y1_col = cols[2]
    if y2_col is None and len(cols) >= 4: y2_col = cols[3]

    if x_col is None or y1_col is None or y2_col is None:
        return None

    # Extract data
    temp = df[x_col].values
    weight = df[y1_col].values
    dtg_raw = df[y2_col].values
    
    # Prepare series data
    # Weight series: [x, y]
    weight_data = []
    for t, w in zip(temp, weight):
        if not (np.isnan(t) or np.isnan(w)):
            weight_data.append([float(t), float(w)])
            
    # DTG series: [x, -y] (multiply by -1 so peaks point up)
    dtg_data = []
    for t, d in zip(temp, dtg_raw):
        if not (np.isnan(t) or np.isnan(d)):
            dtg_data.append([float(t), float(-d)])
            
    # Find Residue at 700 °C
    residue_val = None
    residue_point = None
    target_temp = 700.0
    
    # Find index closest to 700
    valid_mask = ~np.isnan(temp) & ~np.isnan(weight)
    valid_temps = temp[valid_mask]
    valid_weights = weight[valid_mask]
    
    if len(valid_temps) > 0:
        idx = np.argmin(np.abs(valid_temps - target_temp))
        residue_val = valid_weights[idx]
        residue_temp_actual = valid_temps[idx]
        residue_point = [float(residue_temp_actual), float(residue_val)]
        
    # Find Peak above 150 °C
    peak_temp = None
    peak_val = None
    peak_point = None
    
    # Filter DTG data for temps > 150
    dtg_valid_mask = ~np.isnan(temp) & ~np.isnan(dtg_raw)
    dtg_temps = temp[dtg_valid_mask]
    dtg_vals = dtg_raw[dtg_valid_mask]
    
    mask_above_150 = dtg_temps > 150
    if np.any(mask_above_150):
        filtered_temps = dtg_temps[mask_above_150]
        filtered_vals = dtg_vals[mask_above_150]
        
        # Peak is the most negative value (largest mass loss rate)
        # Since we plot -DTG, the highest point corresponds to the most negative raw DTG
        min_idx = np.argmin(filtered_vals) # Most negative raw DTG
        
        peak_temp = filtered_temps[min_idx]
        peak_val_raw = filtered_vals[min_idx]
        peak_val_plotted = -peak_val_raw
        peak_point = [float(peak_temp), float(peak_val_plotted)]

    # Build ECharts Option
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": "2%",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"},
            "formatter": "{a0}: <br/> Temp: {b} °C <br/> Weight: {c0} % <br/> DTG: {c1} %/°C"
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "left": "left",
            "top": "middle",
            "orient": "vertical",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "8%",
            "right": "8%",
            "bottom": "10%",
            "top": "15%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 25,
            "type": "value",
            "scale": True,
            "axisLine": {"lineStyle": {"color": "#333"}},
            "splitLine": {"show": False},
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "yAxis": [
            {
                "name": "Weight (%)",
                "type": "value",
                "position": "left",
                "axisLine": {"lineStyle": {"color": "#1f4e9c"}},
                "axisLabel": {
                    "formatter": "{value}",
                    "fontFamily": "Noto Sans CJK KR",
                    "color": "#1f4e9c"
                },
                "splitLine": {"lineStyle": {"color": "#eee"}}
            },
            {
                "name": "DTG (%/°C)",
                "type": "value",
                "position": "right",
                "axisLine": {"lineStyle": {"color": "#c0392b"}},
                "axisLabel": {
                    "formatter": "{value}",
                    "fontFamily": "Noto Sans CJK KR",
                    "color": "#c0392b"
                },
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "data": weight_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {"width": 2, "color": "#1f4e9c"},
                "markPoint": {
                    "data": [
                        {
                            "coord": residue_point,
                            "symbol": "circle",
                            "symbolSize": 8,
                            "itemStyle": {"color": "#1f4e9c"},
                            "label": {
                                "show": True,
                                "formatter": f"Residue {residue_val:.2f}%",
                                "position": "top",
                                "distance": 5,
                                "fontFamily": "Noto Sans CJK KR",
                                "color": "#1f4e9c",
                                "backgroundColor": "rgba(255,255,255,0.8)",
                                "padding": [2, 4],
                                "borderRadius": 2
                            }
                        }
                    ]
                } if residue_point else {}
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": dtg_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {"width": 2, "color": "#c0392b"},
                "markPoint": {
                    "data": [
                        {
                            "coord": peak_point,
                            "symbol": "triangle",
                            "symbolSize": 8,
                            "itemStyle": {"color": "#c0392b"},
                            "label": {
                                "show": True,
                                "formatter": f"{int(round(peak_temp))} °C",
                                "position": "top",
                                "distance": 5,
                                "fontFamily": "Noto Sans CJK KR",
                                "color": "#c0392b",
                                "backgroundColor": "rgba(255,255,255,0.8)",
                                "padding": [2, 4],
                                "borderRadius": 2
                            }
                        }
                    ]
                } if peak_point else {}
            }
        ]
    }
    
    return {
        "name": sheet_name.replace(" ", "_"),
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }

def main():
    xls = pd.ExcelFile(DATA_FILE)
    figures = []
    
    for sheet_name in xls.sheet_names:
        df = pd.read_excel(xls, sheet_name=sheet_name)
        fig = make_tga_plot(df, sheet_name)
        if fig:
            figures.append(fig)
            
    output = {"figures": figures}
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
