```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def get_nearest_value(temp_arr, weight_arr, target_temp):
    """Find the weight value at the temperature nearest to target_temp."""
    idx = np.argmin(np.abs(temp_arr - target_temp))
    return temp_arr[idx], weight_arr[idx]

def find_dtg_peak(temp_arr, dtg_arr, min_temp=150.0):
    """Find the temperature of the highest DTG peak (most negative) above min_temp."""
    mask = temp_arr > min_temp
    if not np.any(mask):
        return None, None
    
    # DTG is negative during mass loss, so we look for the minimum value (most negative)
    # which corresponds to the maximum absolute rate of loss.
    filtered_dtg = dtg_arr[mask]
    filtered_temp = temp_arr[mask]
    
    min_idx = np.argmin(filtered_dtg)
    peak_temp = filtered_temp[min_idx]
    peak_val = filtered_dtg[min_idx]
    
    return peak_temp, peak_val

def process_sheet(df, sheet_name):
    # Identify columns based on standard TGA structure
    # Assuming: B=Temperature, C=Weight(%), D=D.W(%/C) based on preview
    cols = df.columns.tolist()
    
    # Heuristic to find columns if names vary slightly, but using preview info:
    # Col A: Time, Col B: Temperature, Col C: Weight(%), Col D: D.W(%/C)
    # We will try to map by name first, then by position if needed.
    
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for c in cols:
        cl = str(c).lower()
        if 'temp' in cl or 'temperature' in cl:
            temp_col = c
        elif 'weight' in cl and '%' in cl and 'd.w' not in cl and 'derivative' not in cl:
            weight_col = c
        elif 'd.w' in cl or 'derivative' in cl or 'dtg' in cl:
            dtg_col = c
            
    # Fallback to positions if names don't match well
    if temp_col is None and len(cols) >= 2: temp_col = cols[1]
    if weight_col is None and len(cols) >= 3: weight_col = cols[2]
    if dtg_col is None and len(cols) >= 4: dtg_col = cols[3]
    
    if temp_col is None or weight_col is None or dtg_col is None:
        return None

    data = df[[temp_col, weight_col, dtg_col]].dropna()
    
    temps = data[temp_col].values.astype(float)
    weights = data[weight_col].values.astype(float)
    dtgs = data[dtg_col].values.astype(float)
    
    # Prepare series data for ECharts [x, y] format
    weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]
    # DTG multiplied by -1 so peaks point up
    dtg_data = [[float(t), float(-d)] for t, d in zip(temps, dtgs)]
    
    # Calculate Residue at ~800°C
    res_temp, res_weight = get_nearest_value(temps, weights, 800.0)
    residue_label_text = f"Residue {res_weight:.1f}%"
    
    # Calculate Peak DTG > 150°C
    peak_temp, peak_val = find_dtg_peak(temps, dtgs, min_temp=150.0)
    peak_label_text = ""
    peak_marker_coord = []
    
    if peak_temp is not None:
        peak_label_text = f"{peak_temp:.0f} °C"
        # The marker should be on the DTG curve (right axis). 
        # In ECharts markPoint, coord uses the same x-axis, but y needs to match the series yAxisIndex.
        # Since we inverted DTG (-dtg), the value on the chart is -peak_val.
        peak_marker_coord = [peak_temp, -peak_val]

    # Axis ranges
    temp_min = np.floor(temps.min() / 50) * 50
    temp_max = np.ceil(temps.max() / 50) * 50
    
    weight_min = np.floor(weights.min() / 10) * 10
    weight_max = np.ceil(weights.max() / 10) * 10
    
    # For DTG (inverted), max is positive peak height, min is 0 or slight negative noise
    # Let's set y2 range from 0 to slightly above max inverted DTG
    inv_dtg_vals = [-d for d in dtgs]
    dtg_max = max(inv_dtg_vals)
    dtg_min = min(inv_dtg_vals)
    # Ensure 0 is included or close to it for baseline
    y2_min = min(0, np.floor(dtg_min * 10) / 10)
    y2_max = np.ceil(dtg_max * 10) / 10 + 0.1 # padding
    
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 14
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "top": 10,
            "right": 10,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 9
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "top": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "min": temp_min,
            "max": temp_max,
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 10
            },
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "type": "dashed",
                    "color": "#ccc"
                }
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "min": weight_min,
                "max": weight_max,
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 10
                },
                "splitLine": {
                    "show": False
                }
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "min": y2_min,
                "max": y2_max,
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 10
                },
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "data": weight_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [res_temp, res_weight],
                            "symbol": "circle",
                            "symbolSize": 6,
                            "itemStyle": {
                                "color": "#1f4e9c"
                            },
                            "label": {
                                "show": True,
                                "formatter": residue_label_text,
                                "position": "top",
                                "distance": 5,
                                "fontFamily": "Noto Sans CJK KR",
                                "fontSize": 10,
                                "color": "#1f4e9c"
                            }
                        }
                    ]
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": dtg_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": peak_marker_coord,
                            "symbol": "triangle",
                            "symbolSize": 8,
                            "itemStyle": {
                                "color": "#c0392b"
                            },
                            "label": {
                                "show": True,
                                "formatter": peak_label_text,
                                "position": "top",
                                "distance": 5,
                                "fontFamily": "Noto Sans CJK KR",
                                "fontSize": 10,
                                "color": "#c0392b"
                            }
                        }
                    ] if peak_temp is not None else []
                }
            }
        ]
    }
    
    return {
        "name": sheet_name.replace(" ", "_"),
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }

def main():
    xl_file = pd.ExcelFile(DATA_FILE)
    figures = []
    
    for sheet_name in xl_file.sheet_names:
        df = pd.read_excel(xl_file, sheet_name=sheet_name)
        fig = process_sheet(df, sheet_name)
        if fig:
            figures.append(fig)
            
    output = {"figures": figures}
    print(json.dumps(output, ensure_ascii=False))

if __name__ == "__main__":
    main()
```
