```python
import pandas as pd
import json

def main():
    file_path = "/home/chs/Work/tga_plot/tests/inputs/blank_rows_on_top.xlsx"
    
    # Read the Excel file, specifying the header row (0-indexed, so row 8 is index 7)
    df = pd.read_excel(file_path, sheet_name="Sheet1", header=7)
    
    # Extract columns based on user request: C -> Temperature (x), D -> Weight (y1), E -> DTG (y2)
    # Note: The preview shows column names 'Temperature', 'Weight(%)', 'D.W(%/C)'
    x_col = "Temperature"
    y1_col = "Weight(%)"
    y2_col = "D.W(%/C)"
    
    # Ensure data exists and drop NaNs for plotting
    plot_df = df[[x_col, y1_col, y2_col]].dropna()
    
    if plot_df.empty:
        print(json.dumps({"figures": []}))
        return
        
    temps = plot_df[x_col].tolist()
    weights = plot_df[y1_col].tolist()
    dtgs = plot_df[y2_col].tolist()
    
    # Calculate Residue at nearest 800°C
    target_temp = 800.0
    idx_residue = (plot_df[x_col] - target_temp).abs().idxmin()
    residue_val = plot_df.loc[idx_residue, y1_col]
    residue_temp = plot_df.loc[idx_residue, x_col]
    
    # Calculate Peak temperature (highest DTG peak above 150°C)
    # DTG is negative during mass loss, so we look for the minimum value (most negative)
    # which corresponds to the highest rate of loss. 
    # User instruction: "multiplied by -1 so its peaks point up". 
    # So in the raw data, the peak is the minimum value.
    mask_peak = plot_df[x_col] > 150.0
    subset_peak = plot_df[mask_peak]
    
    peak_temp = None
    peak_weight_at_peak = None
    
    if not subset_peak.empty:
        # Find the row with the minimum DTG value (most negative) in this range
        idx_peak = subset_peak[y2_col].idxmin()
        peak_temp = plot_df.loc[idx_peak, x_col]
        peak_weight_at_peak = plot_df.loc[idx_peak, y1_col]
        
    # Prepare series data
    # Series 1: Weight (%)
    weight_data = [[t, w] for t, w in zip(temps, weights)]
    
    # Series 2: DTG (%/°C) multiplied by -1
    dtg_data = [[t, d * -1] for t, d in zip(temps, dtgs)]
    
    # Markers for annotations
    markers = []
    
    # Residue marker
    markers.append({
        "coord": [residue_temp, residue_val],
        "symbolSize": 8,
        "itemStyle": {"color": "#1f4e9c"}
    })
    
    # Peak marker
    if peak_temp is not None:
        # Marker is placed on the DTG curve (right axis)
        # The value plotted on right axis is dtg * -1
        peak_dtg_plotted = plot_df.loc[subset_peak[y2_col].idxmin(), y2_col] * -1
        markers.append({
            "coord": [peak_temp, peak_dtg_plotted],
            "symbolSize": 8,
            "itemStyle": {"color": "#c0392b"}
        })

    option = {
        "title": {
            "text": "Sheet1",
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "axisLine": {
                "lineStyle": {
                    "color": "#333"
                }
            },
            "splitLine": {
                "show": False
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "axisLine": {
                    "show": True,
                    "lineStyle": {
                        "color": "#1f4e9c"
                    }
                },
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "type": "dashed",
                        "color": "#eee"
                    }
                }
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "position": "right",
                "axisLine": {
                    "show": True,
                    "lineStyle": {
                        "color": "#c0392b"
                    }
                },
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "data": weight_data,
                "smooth": False,
                "showSymbol": False,
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [residue_temp, residue_val],
                            "symbol": "circle",
                            "symbolSize": 8,
                            "itemStyle": {"color": "#1f4e9c"},
                            "label": {
                                "show": True,
                                "formatter": f"Residue {residue_val:.1f}%",
                                "position": "top",
                                "distance": 5,
                                "color": "#1f4e9c",
                                "fontSize": 10,
                                "fontFamily": "Noto Sans CJK KR"
                            }
                        }
                    ]
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": dtg_data,
                "smooth": False,
                "showSymbol": False,
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [peak_temp, peak_dtg_plotted],
                            "symbol": "circle",
                            "symbolSize": 8,
                            "itemStyle": {"color": "#c0392b"},
                            "label": {
                                "show": True,
                                "formatter": f"{peak_temp:.0f} °C",
                                "position": "top",
                                "distance": 5,
                                "color": "#c0392b",
                                "fontSize": 10,
                                "fontFamily": "Noto Sans CJK KR"
                            }
                        }
                    ] if peak_temp is not None else []
                }
            }
        ]
    }
    
    figure = {
        "name": "plot1",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }
    
    output = {"figures": [figure]}
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
