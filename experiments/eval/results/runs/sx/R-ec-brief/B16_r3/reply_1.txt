import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

def main():
    # Read the CSV file
    df = pd.read_csv(DATA_FILE)
    
    # Extract columns based on header names
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    # Get data arrays
    temperature = df[temp_col].values
    weight = df[weight_col].values
    dtg = df[dtg_col].values
    
    # Calculate DTG inverted for display (peaks point up)
    dtg_inverted = -dtg
    
    # Find residue at 800°C (nearest measured point)
    idx_800 = np.argmin(np.abs(temperature - 800))
    residue_weight = weight[idx_800]
    residue_temp = temperature[idx_800]
    
    # Find highest DTG peak above 150°C
    mask_above_150 = temperature > 150
    if np.any(mask_above_150):
        dtg_for_peak = dtg.copy()
        dtg_for_peak[~mask_above_150] = 0  # Set values below 150 to 0 so they don't affect max
        # We want the most negative DTG value (which becomes the highest positive after inversion)
        min_dtg_idx = np.argmin(dtg_for_peak)
        peak_temp = temperature[min_dtg_idx]
        peak_dtg_value = dtg[min_dtg_idx]
    else:
        peak_temp = None
        peak_dtg_value = None
    
    # Prepare data for ECharts
    # For line charts, we need [x, y] pairs
    weight_data = [[float(t), float(w)] for t, w in zip(temperature, weight)]
    dtg_data = [[float(t), float(d)] for t, d in zip(temperature, dtg_inverted)]

    residue_marker = [[float(residue_temp), float(residue_weight)]]

    if peak_temp is not None:

        peak_marker = []
