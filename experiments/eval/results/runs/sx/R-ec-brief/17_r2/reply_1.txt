def get_nearest_value(df, x_col, y_col, target_x):
    """Find the y value at the point nearest to target_x."""
    if df.empty:
        return None, None
    idx = (df[x_col] - target_x).abs().idxmin()
    return df.loc[idx, x_col], df.loc[idx, y_col]

def find_peak_above_threshold(df, x_col, dtg_col, threshold=150.0):
    """Find the temperature of the highest DTG peak above threshold.
       DTG is negative during mass loss, so we look for the most negative value (minimum) 
       or rather, since we plot -DTG, we want the max of -DTG which corresponds to min of DTG.
       The prompt says 'highest DTG peak', usually meaning the largest magnitude of mass loss rate.
       Since DTG is negative, the 'peak' in a standard TGA plot (where DTG is plotted as -DTG or inverted) 
       is the minimum value of the raw DTG column.
    """
    filtered_df = df[df[x_col] > threshold]
    if filtered_df.empty:
        return None, None
    
    # Find the index where DTG is minimum (most negative), which corresponds to the peak mass loss rate
    idx_min_dtg = filtered_df[dtg_col].idxmin()
    peak_temp = filtered_df.loc[idx_min_dtg, x_col]
    peak_val = filtered_df.loc[idx_min_dtg, dtg_col]
    
    return peak_temp, peak_val

        temp = row[x_col]
        weight = row[y1_col]
        dtg = row[y2_col]
        
        # Handle NaNs
        if pd.isna(temp) or pd.isna(weight) or pd.isna(dtg):
            continue
            
        weight_data.append([float(temp), float(weight)])
        # Multiply DTG by -1 so peaks point up
        dtg_data.append([float(temp), float(-dtg)])
        
    # Calculate Residue (nearest 800°C)
    residue_temp, residue_weight = get_nearest_value(df, x_col, y1_col, 800.0)
    
    # Calculate Peak (highest DTG peak above 150°C)
    # We need the temperature where the absolute value of DTG is maximum (or DTG is minimum)
    peak_temp, peak_dtg_raw = find_peak_above_threshold(df, x_col, y2_col, 150.0)

    markers = []
    annotations = []

        markers.append({

                "show": True,
                "position": "right",

                "distance": 10,

                "color": "#333"

    if peak_temp is not None and peak_dtg_raw is not None:
        # The peak marker should be on the DTG curve (y2 axis)

        peak_y_val = -peak_dtg_raw
        markers.append({
            "coord": [peak_temp, peak_y_val],

            "itemStyle": {"color": "#c0392b"},

            "type": "line",

            "yAxisIndex": 1,
            "lineStyle": {"color": "#c0392b", "width": 2},
            "itemStyle": {"color": "#c0392b"},
            "showSymbol": False,
            "smooth": False
        }

    # Add scatter series for markers if they exist
    if residue_weight is not None and residue_temp is not None:
        series_list.append({
            "name": "Residue Marker",
            "type": "scatter",
            "data": [[residue_temp, residue_weight]],
            "yAxisIndex": 0,
            "symbolSize": 8,
            "itemStyle": {"color": "#1f4e9c", "borderColor": "#fff", "borderWidth": 1},
            "label": {
                "show": True,
                "position": "right",
                "formatter": f"Residue {residue_weight:.1f}%",
                "fontSize": 10,
                "color": "#333",
                "backgroundColor": "rgba(255,255,255,0.7)",
                "padding": [2, 4]
            }
        })
        
    if peak_temp is not None and peak_dtg_raw is not None:
        series_list.append({
            "name": "Peak Marker",
            "type": "scatter",
            "data": [[peak_temp, -peak_dtg_raw]],
            "yAxisIndex": 1,
            "symbolSize": 8,
            "itemStyle": {"color": "#c0392b", "borderColor": "#fff", "borderWidth": 1},
            "label": {
                "show": True,
                "position": "top",
                "formatter": f"{peak_temp:.0f} °C",
                "fontSize": 10,
                "color": "#333",
                "backgroundColor": "rgba(255,255,255,0.7)",
                "padding": [2, 4]
            }
        })

    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            },
            "formatter": function_placeholder # Not allowed in plain JSON, remove or simplify
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",

    # Remove tooltip formatter function placeholder issue by simplifying tooltip

    return option

def main():
    figures = []
    
    try:
        xl_file = pd.ExcelFile(DATA_FILE)
        sheet_names = xl_file.sheet_names
        
        for sheet_name in sheet_names:
            df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
            
            # Clean column names just in case
            df.columns = [str(c).strip() for c in df.columns]
            
            # Check if required columns exist
            if "Temperature" not in df.columns or "Weight(%)" not in df.columns or "D.W(%/C)" not in df.columns:
                # Try to map based on position if names are slightly different but structure matches preview
                # Preview said A: Time, B: Temp, C: Weight, D: DW
                cols = list(df.columns)
                if len(cols) >= 4:
                    df.rename(columns={cols[1]: "Temperature", cols[2]: "Weight(%)", cols[3]: "D.W(%/C)"}, inplace=True)
                else:
                    continue
                    
            option = create_tga_chart_option(sheet_name, df)
            
            # Figure size: 8 inches wide, 5 inches high. 
            # CSS pixels at 96 px/inch -> width = 8 * 96 = 768, height = 5 * 96 = 480
            fig_width_px = int(8 * 96)
            fig_height_px = int(5 * 96)
            
            figure_entry = {
                "name": sheet_name.replace(" ", "_"),
                "option": option,
                "width": fig_width_px,
                "height": fig_height_px,
                "dpi": 600,
                "formats": ["png"]
            }
            figures.append(figure_entry)
            
    except Exception as e:
        # Fallback empty response if error occurs, though prompt implies valid input
        pass

    output = {"figures": figures}
    print(json.dumps(output))

if __name__ == "__main__":
    main()
