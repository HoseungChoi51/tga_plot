def find_column(df, keys, exclude=None):
    for col in df.columns:
        name = str(col).lower()
        if exclude and any(e in name for e in exclude):
            continue
        if any(k in name for k in keys):
            return col
    return None

def get_columns(df):
    xcol = find_column(df, ["temperature", "temp"])
    y1col = find_column(df, ["weight", "mass"], exclude=["deriv", "dtg", "derivative"])

        if y1col is None: y1col = df.columns[2]
        if y2col is None: y2col = df.columns[3]
    return xcol, y1col, y2col
