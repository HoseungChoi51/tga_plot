```python
import json
import math
import os

import numpy as np
import pandas as pd
from scipy.signal import find_peaks

DATA_FILE = globals().get("DATA_FILE")
if not DATA_FILE:
    data_files = globals().get("DATA_FILES")
    if isinstance(data_files, (list, tuple)) and data_files:
        DATA_FILE = data_files[0]
    elif isinstance(data_files, str):
        DATA_FILE = data_files
    else:
        DATA_FILE = os.environ.get("DATA_FILE", "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx")

SHEET_NAME = "1006R1-V acid"

try:
    df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
except Exception:
    xl = pd.ExcelFile(DATA_FILE)
    SHEET_NAME = next((s for s in xl.sheet_names if "acid" in s.lower()), xl.sheet_names[0])
    df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)


def find_col(frame, keys):
    for col in frame.columns:
        s = str(col).strip().lower()
        for key in keys:
            if key in s:
                return col
    return None


xcol = find_col(df, ["temperature", "temp", "degc", "°c"])
y1col = find_col(df, ["weight", "wt", "mass"])
y2col = find_col(df, ["d.w", "dtg", "derivative", "dw", "d w"])

if xcol is None:
    xcol = df.columns[1] if len(df.columns) > 1 else df.columns[0]
if y1col is None:
    y1col = df.columns[2] if len(df.columns) > 2 else df.columns[0]

cols = [xcol, y1col] + ([y2col] if y2col is not None else [])
sub = df[cols].copy()
sub.columns = ["x", "y1"] + (["y2"] if y2col is not None else [])
sub = sub.apply(pd.to_numeric, errors="coerce")
sub = sub.dropna(subset=["x", "y1"])
if y2col is not None:
    sub = sub.dropna(subset=["y2"])
sub = sub.sort_values("x")

mask = (sub["x"] >= 100) & (sub["x"] <= 800)
if int(mask.sum()) >= 2:
    sub = sub[mask]

xarr = sub["x"].to_numpy(dtype=float)
warr = sub["y1"].to_numpy(dtype=float)
dwarr = sub["y2"].to_numpy(dtype=float) if y2col is not None else np.zeros(len(sub), dtype=float)

peaks_idx = []

if len(xarr) >= 2 and y2col is not None:
    yneg = -dwarr
    distance = max(1, len(xarr) // 100)

    for prom in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0001]:
        idx, props = find_peaks(yneg, prominence=prom, distance=distance)
        if len(idx) >= 2:
            if "prominences" in props and len(props["prominences"]) == len(idx):
                order = np.argsort(props["prominences"])[::-1]
            else:
                order = np.argsort(yneg[idx])[::-1]
            peaks_idx = [int(i) for i in idx[order[:2]]]
            break

    if len(peaks_idx) < 2:
        chosen = []
        min_dist = max(1, len(xarr) // 100)
        order = np.argsort(dwarr)

        for i in order:
            i = int(i)
            if all(abs(i - j) >= min_dist for j in chosen):
                chosen.append(i)
            if len(chosen) == 2:
                break

        if len(chosen) < 2:
            min_dist = max(1, len(xarr) // 200)
            for i in order:
                i = int(i)
                if i not in chosen and all(abs(i - j) >= min_dist for j in chosen):
                    chosen.append(i)
                if len(chosen) == 2:
                    break

        if len(chosen) < 2 and len(xarr) > 0:
            if not chosen:
                chosen.append(int(np.argmin(dwarr)))
            if len(chosen) == 1:
                d = np.abs(xarr - xarr[chosen[0]])
                chosen.append(int(np.argmax(d)))

        peaks_idx = chosen[:2]

elif len(xarr) > 0:
    peaks_idx = [int(np.argmin(dwarr))]
    if len(xarr) > 1:
        d = np.abs(xarr - xarr[peaks_idx[0]])
        peaks_idx.append(int(np.argmax(d)))

peaks_idx = sorted(peaks_idx, key=lambda i: xarr[i])


def r(v, n):
    try:
        f = float(v)
        if not math.isfinite(f):
            return None
        return round(f, n)
    except Exception:
        return None


weight_data = [[r(xarr[i], 3), r(warr[i], 4)] for i in range(len(xarr))]
dtg_data = [[r(xarr[i], 3), r(dwarr[i], 6)] for i in range(len(xarr))]
weight_data = [d for d in weight_data if d[0] is not None and d[1] is not None]
dtg_data = [d for d in dtg_data if d[0] is not None and d[1] is not None]

mark_data = []
for n, i in enumerate(peaks_idx, start=1):
    tx = r(xarr[i], 3)
    ty = r(dwarr[i], 6)
    if tx is None or ty is None:
        continue

    label_pos = "top" if n == 1 else "bottom"
    mark_data.append(
        {
            "coord": [tx, ty],
            "symbol": "pin",
            "symbolSize": 40,
            "itemStyle": {
                "color": "#d62728",
                "borderColor": "#ffffff",
                "borderWidth": 1,
            },
            "label": {
                "show": True,
                "position": label_pos,
                "formatter": f"P{n}\n{tx:.1f}C",
                "fontSize": 10,
                "lineHeight": 12,
                "color": "#d62728",
            },
            "tooltip": {
                "formatter": f"Peak {n}\nTemperature: {tx:.1f} C\nD.W: {ty:.3f} %/C"
            },
        }
    )

if len(dwarr) > 0:
    min_dw = float(np.min(dwarr))
    max_dw = float(np.max(dwarr))
else:
    min_dw = 0.0
    max_dw = 0.0

range_dw = max_dw - min_dw
if not math.isfinite(range_dw) or range_dw == 0:
    range_dw = abs(min_dw) + abs(max_dw) + 1.0

margin = range_dw * 0.08
y2_min = min(0.0, min_dw) - margin
y2_max = max(0.0, max_dw) + margin

option = {
    "backgroundColor": "#ffffff",
    "animation": False,
    "textStyle": {
        "fontFamily": "Noto Sans CJK KR, sans-serif"
    },
    "title": {
        "text": f"TGA/DTG - {SHEET_NAME}",
        "left": "center",
        "top": 8,
        "textStyle": {
            "fontSize": 16,
            "fontWeight": "bold"
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "legend": {
        "top": 36,
        "data": ["Weight (%)", "D.W (%/C)"]
    },
    "grid": {
        "left": 70,
        "right": 80,
        "top": 90,
        "bottom": 55
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "min": 100,
        "max": 800,
        "axisLine": {
            "lineStyle": {
                "color": "#333333"
            }
        },
        "axisTick": {
            "lineStyle": {
                "color": "#333333"
            }
        },
        "splitLine": {
            "show": True,
            "lineStyle": {
                "color": "#eeeeee"
            }
        }
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 40,
            "min": 0,
            "max": 100,
            "position": "left",
            "axisLine": {
                "show": True,
                "lineStyle": {
                    "color": "#1f77b4"
                }
            },
            "axisTick": {
                "show": True,
                "lineStyle": {
                    "color": "#1f77b4"
                }
            },
            "axisLabel": {
                "formatter": "{value}"
            },
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "color": "#eeeeee"
                }
            }
        },
        {
            "type": "value",
            "name": "D.W (%/C)",
            "nameLocation": "middle",
            "nameGap": 40,
            "min": y2_min,
            "max": y2_max,
            "position": "right",
            "axisLine": {
                "show": True,
                "lineStyle": {
                    "color": "#d62728"
                }
            },
            "axisTick": {
                "show": True,
                "lineStyle": {
                    "color": "#d62728"
                }
            },
            "axisLabel": {
                "formatter": "{value}"
            },
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "xAxisIndex": 0,
            "yAxisIndex": 0,
            "showSymbol": False,
            "smooth": False,
            "lineStyle": {
                "width": 2,
                "color": "#1f77b4"
            },
            "itemStyle": {
                "color": "#1f77b4"
            },
            "data": weight_data
        },
        {
            "name": "D.W (%/C)",
            "type": "line",
            "xAxisIndex": 0,

            "showSymbol": False,
            "smooth": False,

            },

            "markPoint": {
                "symbol": "pin",
                "symbolSize": 40,
                "data": mark_data
            }
        }
    ]
}

result = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 800,
            "height": 500,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(result, ensure_ascii=False))
```
