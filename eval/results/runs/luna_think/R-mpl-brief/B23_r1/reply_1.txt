```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib.lines import Line2D

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna_think/R-mpl-brief/B23_r1/out"

SHEETS = [
    ("Raw", "black"),
    ("Acid-washed", "blue"),
    ("Annealed", "red"),
]

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx()

legend_handles = []
for sheet_name, color in SHEETS:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    df.columns = [str(c).strip() for c in df.columns]

    temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
    weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
    dtg = pd.to_numeric(df["Deriv. Weight (%/°C)"], errors="coerce")
    valid = temperature.notna() & weight.notna() & dtg.notna()
    temperature = temperature[valid].to_numpy()
    weight = weight[valid].to_numpy()
    dtg = dtg[valid].to_numpy()

    order = np.argsort(temperature)
    temperature, weight, dtg = temperature[order], weight[order], dtg[order]

    # 실선은 중량, 점선은 부호를 반전한 DTG를 나타냅니다.
    ax1.plot(temperature, weight, color=color, linewidth=1.8)
    ax2.plot(temperature, -dtg, color=color, linewidth=1.4, linestyle="--")

    residue_idx = int(np.argmin(np.abs(temperature - 800)))
    ax1.scatter(temperature[residue_idx], weight[residue_idx], color=color,
                edgecolor="white", linewidth=0.8, s=45, zorder=5)
    ax1.annotate(
        f"{sheet_name} 잔류량 {weight[residue_idx]:.1f}%",
        (temperature[residue_idx], weight[residue_idx]),
        xytext=(5, 7), textcoords="offset points", color=color, fontsize=8,
    )

    peak_candidates = np.flatnonzero(temperature > 150)
    if peak_candidates.size:
        peak_idx = peak_candidates[np.argmax(-dtg[peak_candidates])]
        ax2.scatter(temperature[peak_idx], -dtg[peak_idx], color=color,
                    edgecolor="white", linewidth=0.8, s=45, zorder=5)
        ax2.annotate(
            f"{sheet_name} {temperature[peak_idx]:.0f} °C",
            (temperature[peak_idx], -dtg[peak_idx]),
            xytext=(5, -13), textcoords="offset points", color=color, fontsize=8,
        )

    legend_handles.append(Line2D([0], [0], color=color, linewidth=2, label=sheet_name))

ax1.set_title("세 시료 TGA 비교")
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("−Derivative Weight (%/°C)")
ax1.legend(handles=legend_handles, loc="center left", frameon=True)
ax1.grid(True, alpha=0.25)
fig.tight_layout()

fig.savefig(out_dir / "tga_overlay.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
