```python
import os
import re
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna_think/R-mpl-brief/B11_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheet_name = "PET"
df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)

# Select columns by their names; fall back to the standard TGA column positions.
def find_column(pattern, fallback_index):
    matches = [col for col in df.columns if re.search(pattern, str(col), flags=re.IGNORECASE)]
    return matches[0] if matches else df.columns[fallback_index]

x_col = find_column(r"temperature", 1)
weight_col = find_column(r"weight\s*\(", 2)
dtg_col = find_column(r"deriv", 3)

x = pd.to_numeric(df[x_col], errors="coerce").to_numpy()
weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy()
dtg = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy()

valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
x, weight, dtg = x[valid], weight[valid], dtg[valid]

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(x, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
dtg_positive = -dtg
dtg_line, = ax2.plot(x, dtg_positive, color="#c0392b", linewidth=1.6, label="DTG (%/°C)")

# Residue: measured weight at the point nearest 800 °C.
residue_index = int(np.argmin(np.abs(x - 800)))
ax1.plot(x[residue_index], weight[residue_index], marker="o", color="#1f4e9c", zorder=5)
ax1.annotate(
    f"Residue {weight[residue_index]:.2f}%",
    xy=(x[residue_index], weight[residue_index]),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
)

# DTG peak: highest inverted DTG value above 150 °C.
eligible = np.flatnonzero(x > 150)
if eligible.size:
    peak_index = eligible[np.argmax(dtg_positive[eligible])]
    ax2.plot(x[peak_index], dtg_positive[peak_index], marker="o", color="#c0392b", zorder=5)
    ax2.annotate(
        f"{x[peak_index]:.0f} °C",
        xy=(x[peak_index], dtg_positive[peak_index]),
        xytext=(8, 10),
        textcoords="offset points",
        color="#c0392b",
    )

ax1.set_title("PET 열분해")
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("DTG (%/°C)")
ax1.grid(True, linestyle="--", alpha=0.5)
ax1.legend([weight_line, dtg_line], ["Weight (%)", "DTG (%/°C)"], loc="lower left")

fig.tight_layout()
output_path = os.path.join(OUT_DIR, "PET_열분해.png")
fig.savefig(output_path, dpi=300, bbox_inches="tight")
plt.close(fig)
```
