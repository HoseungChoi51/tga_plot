```python
import json
import re
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def clean(value):
    if pd.isna(value):
        return None
    value = float(value)
    return value if np.isfinite(value) else None

def make_option(sheet_name, df):
    cols = {str(c).strip().lower(): c for c in df.columns}
    def find_col(names, fallback_index):
        for name in names:
            for key, col in cols.items():
                if name in key:
                    return col
        return df.columns[fallback_index] if len(df.columns) > fallback_index else None

    xcol = find_col(["temperature", "temp"], 1)
    wcol = find_col(["weight"], 2)
    dcol = find_col(["d.w", "dtg", "derivative"], 3)

    x = pd.to_numeric(df[xcol], errors="coerce").to_numpy(dtype=float)
    weight = pd.to_numeric(df[wcol], errors="coerce").to_numpy(dtype=float)
    dtg = pd.to_numeric(df[dcol], errors="coerce").to_numpy(dtype=float)

    valid_w = np.isfinite(x) & np.isfinite(weight)
    valid_d = np.isfinite(x) & np.isfinite(dtg)
    weight_data = [[clean(a), clean(b)] for a, b in zip(x[valid_w], weight[valid_w])]
    dtg_data = [[clean(a), clean(-b)] for a, b in zip(x[valid_d], dtg[valid_d])]

    residue = None
    if valid_w.any():
        indexes = np.where(valid_w)[0]
        idx = indexes[np.argmin(np.abs(x[indexes] - 800))]
        residue = (x[idx], weight[idx])

    peak = None
    candidates = np.where(valid_d & (x > 150))[0]
    if len(candidates):
        idx = candidates[np.argmax(-dtg[candidates])]
        peak = (x[idx], -dtg[idx])

    weight_series = {
        "name": "Weight (%)",
        "type": "line",
        "showSymbol": False,
        "yAxisIndex": 0,
        "data": weight_data,
        "lineStyle": {"color": "#1f4e9c", "width": 2},
        "itemStyle": {"color": "#1f4e9c"},
        "markArea": {
            "silent": True,
            "itemStyle": {"color": "rgba(180, 180, 180, 0.28)"},
            "data": [[{"xAxis": 300}, {"xAxis": 450}]]
        }
    }
    dtg_series = {
        "name": "DTG (-D.W)",
        "type": "line",
        "showSymbol": False,
        "yAxisIndex": 1,
        "data": dtg_data,
        "lineStyle": {"color": "#c0392b", "width": 2},
        "itemStyle": {"color": "#c0392b"}
    }

    series = [weight_series, dtg_series]
    if residue is not None:
        series.append({
            "name": "Residue",
            "type": "scatter",
            "yAxisIndex": 0,
            "symbolSize": 9,
            "data": [[clean(residue[0]), clean(residue[1])]],
            "itemStyle": {"color": "#1f4e9c"},
            "label": {
                "show": True,
                "formatter": "Residue {:.1f}%".format(residue[1]),
                "position": "right",
                "color": "#1f4e9c"
            }
        })
    if peak is not None:
        series.append({
            "name": "DTG peak",
            "type": "scatter",
            "yAxisIndex": 1,
            "symbolSize": 9,
            "data": [[clean(peak[0]), clean(peak[1])]],
            "itemStyle": {"color": "#c0392b"},
            "label": {
                "show": True,
                "formatter": "{:.0f} °C".format(peak[0]),
                "position": "top",
                "color": "#c0392b"
            }
        })

    return {
        "animation": False,
        "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        "title": {"text": str(sheet_name), "left": "center"},
        "tooltip": {"trigger": "axis"},
        "legend": {"data": ["Weight (%)", "DTG (-D.W)"], "left": "center"},
        "grid": {"left": 75, "right": 75, "top": 75, "bottom": 65},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 38,
            "scale": True
        },
        "yAxis": [
            {"type": "value", "name": "Weight (%)", "nameLocation": "middle", "nameGap": 50},
            {"type": "value", "name": "DTG (-D.W) (%/°C)", "nameLocation": "middle", "nameGap": 55}
        ],
        "series": series
    }

book = pd.ExcelFile(DATA_FILE)
figures = []
for sheet in book.sheet_names:
    frame = pd.read_excel(DATA_FILE, sheet_name=sheet)
    option = make_option(sheet, frame)
    safe_name = re.sub(r"[^\w.-]+", "_", str(sheet), flags=re.UNICODE).strip("_") or "plot"
    figures.append({
        "name": safe_name,
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    })

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
```
