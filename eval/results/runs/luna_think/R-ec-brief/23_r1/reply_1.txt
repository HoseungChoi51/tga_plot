def read_numeric_rows(path):
    rows = []
    for line in Path(path).read_text(encoding="utf-8", errors="replace").splitlines():
        stripped = line.strip()
        if not stripped:
            continue
        # Instrument exports commonly use tabs, commas, semicolons, or whitespace.
        parts = re.split(r"[\t,; ]+", stripped)
        values = []
        for part in parts:
            try:
                values.append(float(part))
            except ValueError:
                values = []
                break
        if len(values) >= 4:
            rows.append(values[:4])
    if not rows:
        raise ValueError(f"No numeric TGA rows found in {path}")
    return np.asarray(rows, dtype=float)


data = read_numeric_rows(DATA_FILE)
