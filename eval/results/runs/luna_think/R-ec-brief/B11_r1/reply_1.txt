```python
import json
import os

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"


def main():
    df = pd.read_excel(DATA_FILE, sheet_name="PET")
    df.columns = [str(c).strip() for c in df.columns]

    def find_column(keywords, fallback):
        for col in df.columns:
            if any(word in col.lower() for word in keywords):
                return col
        return fallback

    x_col = find_column(["temperature", "temp"], df.columns[1])
    weight_col = find_column(["weight"], df.columns[2])
    dtg_col = find_column(["deriv", "dtg"], df.columns[3])

    x = pd.to_numeric(df[x_col], errors="coerce").to_numpy(dtype=float)
    weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy(dtype=float)
    dtg = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy(dtype=float)
    valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
    x, weight, dtg = x[valid], weight[valid], dtg[valid]

    if not len(x):
        raise ValueError("PET 시트에 유효한 수치 데이터가 없습니다.")

    residue_idx = int(np.argmin(np.abs(x - 800)))
    residue_x = float(x[residue_idx])
    residue = float(weight[residue_idx])

    peak_candidates = np.flatnonzero(x > 150)
    if len(peak_candidates):
        peak_idx = int(peak_candidates[np.argmax(-dtg[peak_candidates])])
    else:
        peak_idx = int(np.argmax(-dtg))
    peak_x = float(x[peak_idx])
    peak_y = float(-dtg[peak_idx])

    figure = {
        "name": "PET",
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"],
        "option": {
            "animation": False,
            "backgroundColor": "#ffffff",
            "title": {
                "text": "PET 열분해",
                "left": "center",
                "top": 12,
                "textStyle": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 16,
                    "fontWeight": "normal",
                    "color": "#222222",
                },
            },
            "tooltip": {"trigger": "axis"},
            "legend": {
                "data": ["Weight (%)", "-DTG (%/°C)"],
                "left": 68,
                "bottom": 24,
                "orient": "vertical",
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11},
            },
            "grid": {"left": 68, "right": 68, "top": 58, "bottom": 68},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 36,
                "axisLine": {"show": True},
                "axisTick": {"show": True},
                "axisLabel": {"show": True},
                "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "axisLine": {"show": True},
                    "axisTick": {"show": True},
                    "axisLabel": {"show": True},
                    "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}},
                },
                {
                    "type": "value",
                    "name": "-DTG (%/°C)",
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "axisLine": {"show": True},
                    "axisTick": {"show": True},
                    "axisLabel": {"show": True},
                    "splitLine": {"show": False},
                },
            ],
            "series": [
                {
                    "name": "Weight (%)",
                    "type": "line",
                    "showSymbol": False,
                    "smooth": False,
                    "yAxisIndex": 0,
                    "lineStyle": {"width": 2, "color": "#1f4e9c"},
                    "itemStyle": {"color": "#1f4e9c"},
                    "data": [[float(a), float(b)] for a, b in zip(x, weight)],
                    "markPoint": {
                        "symbol": "circle",
                        "symbolSize": 9,
                        "itemStyle": {"color": "#1f4e9c"},
                        "label": {
                            "show": True,
                            "formatter": f"Residue {residue:.2f}%",
                            "position": "top",
                            "fontFamily": "Noto Sans CJK KR",
                            "fontSize": 11,
                        },
                        "data": [{"coord": [residue_x, residue]}],
                    },
                },
                {
                    "name": "-DTG (%/°C)",
                    "type": "line",
                    "showSymbol": False,
                    "smooth": False,
                    "yAxisIndex": 1,
                    "lineStyle": {"width": 2, "color": "#c0392b"},
                    "itemStyle": {"color": "#c0392b"},
                    "data": [[float(a), float(-b)] for a, b in zip(x, dtg)],
                    "markPoint": {
                        "symbol": "circle",
                        "symbolSize": 9,
                        "itemStyle": {"color": "#c0392b"},
                        "label": {
                            "show": True,
                            "formatter": f"{peak_x:.0f} °C",
                            "position": "top",
                            "fontFamily": "Noto Sans CJK KR",
                            "fontSize": 11,
                        },
                        "data": [{"coord": [peak_x, peak_y]}],
                    },
                },
            ],
        },
    }
    print(json.dumps({"figures": [figure]}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
