```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 설정
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B16_r2/out"

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 데이터 로드
df = pd.read_csv(DATA_FILE)

# 열 이름 정리 (공백 제거 등)
df.columns = df.columns.str.strip()

# 필요한 컬럼 확인 및 추출
# x축: Temperature (°C)
# y1 (위): Weight (%)
# y2 (아래): Deriv. Weight (%/°C)

x_col = "Temperature (°C)"
y_tga_col = "Weight (%)"
y_dtg_col = "Deriv. Weight (%/°C)"

if x_col not in df.columns or y_tga_col not in df.columns or y_dtg_col not in df.columns:
    raise ValueError(f"필요한 컬럼을 찾을 수 없습니다. 사용 가능한 컬럼: {df.columns.tolist()}")

x_data = df[x_col]
y_tga_data = df[y_tga_col]
y_dtg_data = df[y_dtg_col]

# 그래프 그리기
fig, (ax1, ax2) = plt.subplots(2, 1, sharex=True, figsize=(10, 8))

# 위쪽 그래프: TGA (Weight %)
ax1.plot(x_data, y_tga_data, color='blue', label='TGA')
ax1.set_ylabel('무게 (%)')
ax1.grid(True, linestyle='--', alpha=0.6)
ax1.legend(loc='upper right')
ax1.set_title('TGA / DTG 분석 결과')

# 아래쪽 그래프: DTG (Deriv. Weight %/°C)
ax2.plot(x_data, y_dtg_data, color='red', label='DTG')
ax2.set_xlabel('온도 (°C)')
ax2.set_ylabel('미분 무게 (%/°C)')
ax2.grid(True, linestyle='--', alpha=0.6)
ax2.legend(loc='best')

# 레이아웃 조정
plt.tight_layout()

# 출력 디렉토리 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 파일 저장
output_path = os.path.join(OUT_DIR, "tga_dtg_plot.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
