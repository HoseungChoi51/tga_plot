```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# File paths
DATA_FILES = [
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_A.csv",
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_B.txt"
]
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B05_r2/out"

# Ensure output directory exists
os.makedirs(OUT_DIR, exist_ok=True)

# Read data files
df_a = pd.read_csv(DATA_FILES[0])
# For the .txt file, try reading with default separator first, then fallback to tab if needed
try:
    df_b = pd.read_csv(DATA_FILES[1], sep='\t')
except Exception:
    # If tab fails, try space or other common separators, or just default (comma)
    try:
        df_b = pd.read_csv(DATA_FILES[1], sep=None, engine='python')
    except Exception:
        df_b = pd.read_csv(DATA_FILES[1])

# Helper function to find column by partial match or exact match
def get_column(df, target_name):
    # Try exact match first
    if target_name in df.columns:
        return df[target_name]
    
    # Try case-insensitive or stripped match
    for col in df.columns:
        if col.strip().lower() == target_name.strip().lower():
            return df[col]
        
    # Try partial match (e.g., if degree symbol encoding is weird)
    # Specifically looking for Temperature, Weight, Deriv
    keywords = {
        "Temperature": ["temp", "temperature"],
        "Weight (%)": ["weight", "wt"],
        "Deriv. Weight (%/°C)": ["deriv", "derivative", "dtg"]
    }
    
    search_terms = []
    if "Temperature" in target_name:
        search_terms = keywords["Temperature"]
    elif "Weight" in target_name and "Deriv" not in target_name:
        search_terms = keywords["Weight (%)"]
    elif "Deriv" in target_name:
        search_terms = keywords["Deriv. Weight (%/°C)"]
        
    for term in search_terms:
        for col in df.columns:
            if term.lower() in col.lower():
                return df[col]
                
    raise KeyError(f"Column '{target_name}' not found in dataframe columns: {df.columns.tolist()}")

# Extract columns based on the preview info
# x: Temperature (°C), y1: Weight (%), y2: Deriv. Weight (%/°C)
try:
    temp_a = get_column(df_a, 'Temperature (°C)')
    weight_a = get_column(df_a, 'Weight (%)')
    dwt_a = get_column(df_a, 'Deriv. Weight (%/°C)')
    
    temp_b = get_column(df_b, 'Temperature (°C)')
    weight_b = get_column(df_b, 'Weight (%)')
    dwt_b = get_column(df_b, 'Deriv. Weight (%/°C)')
except KeyError as e:
    print(f"Error finding columns: {e}")
    print(f"DF A Columns: {df_a.columns.tolist()}")
    print(f"DF B Columns: {df_b.columns.tolist()}")
    raise e

# Create figure with two subplots sharing x-axis
fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(10, 8), sharex=True, gridspec_kw={'height_ratios': [3, 1]})

# Plot TGA (Weight %)
ax1.plot(temp_a, weight_a, label='Lot A', color='blue')
ax1.plot(temp_b, weight_b, label='Lot B', color='red')
ax1.set_ylabel('Weight (%)')
ax1.legend(loc='upper right')
ax1.grid(True, linestyle='--', alpha=0.6)

# Plot DTG (Derivative Weight)
ax2.plot(temp_a, dwt_a, label='Lot A', color='blue')
ax2.plot(temp_b, dwt_b, label='Lot B', color='red')
ax2.set_xlabel('Temperature (°C)')
ax2.set_ylabel('Deriv. Weight (%/°C)')
ax2.legend(loc='upper right')
ax2.grid(True, linestyle='--', alpha=0.6)

# Adjust layout and save
plt.tight_layout()
output_path = os.path.join(OUT_DIR, "TGA_comparison.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
