```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 데이터 파일 경로와 출력 디렉토리 설정
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/09_r1/out"

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 엑셀 파일 읽기 (모든 시트)
xl = pd.ExcelFile(DATA_FILE)
sheet_names = xl.sheet_names

# 그래프 생성
fig, ax1 = plt.subplots(figsize=(10, 6))

# 색상 정의
colors = ['#1f77b4', '#ff7f0e', '#2ca02c']

for i, sheet_name in enumerate(sheet_names):
    df = xl.parse(sheet_name)
    
    # TGA 곡선 (Weight %) - 왼쪽 y축
    color = colors[i % len(colors)]
    line1 = ax1.plot(df['Temperature'], df['Weight(%)'], label=f'{sheet_name} TGA', color=color)
    
    # DTG 곡선 (D.W %/C) - 오른쪽 y축
    ax2 = ax1.twinx()
    line2 = ax2.plot(df['Temperature'], df['D.W(%/C)'], linestyle='--', label=f'{sheet_name} DTG', color=color, alpha=0.7)

# 축 레이블 설정
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12)
ax2.set_ylabel('dW/dT (%/°C)', fontsize=12)

# 틱 방향 바깥쪽으로 설정
ax1.tick_params(axis='both', which='both', direction='out', labelsize=9)
ax2.tick_params(axis='both', which='both', direction='out', labelsize=9)

# 범례 추가
lines_labels = ax1.get_legend_handles_labels()
lines_labels2 = ax2.get_legend_handles_labels()
all_lines = lines_labels[0] + lines_labels2[0]
all_labels = lines_labels[1] + lines_labels2[1]
ax1.legend(all_lines, all_labels, loc='center right', fontsize=9)

plt.title('TGA and DTG Analysis Results', fontsize=14)
plt.tight_layout()

# 저장
output_path = os.path.join(OUT_DIR, "tga_plot.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
