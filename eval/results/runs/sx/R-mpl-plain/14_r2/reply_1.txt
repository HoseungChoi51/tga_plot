```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 파일 경로 및 출력 디렉토리 설정
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/korean_excel.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/14_r2/out"

# 데이터 로드 (CSV 파일이므로 read_csv 사용, 필요시 sep 파라미터 조정 가능)
try:
    df = pd.read_csv(DATA_FILE)
except Exception as e:
    # 만약 CSV가 아닌 다른 형식이나 인코딩 문제일 경우를 대비해 utf-8-sig 시도
    try:
        df = pd.read_csv(DATA_FILE, encoding='utf-8-sig')
    except:
        raise e

# 컬럼명 확인 및 매핑 (미리보기 기준)
# A: 시간(분), B: 온도(°C), C: 무게(%), D: 미분무게(%/°C)
col_temp = "온도(°C)"
col_weight = "무게(%)"
col_dtemp = "미분무게(%/°C)"

# 그래프 그리기 준비
plt.figure(figsize=(10, 6))

# 왼쪽 Y축: 무게 (%)
ax1 = plt.gca()
line1, = ax1.plot(df[col_temp], df[col_weight], label='무게 (%)', color='blue')
ax1.set_xlabel('온도 (°C)', fontsize=12)
ax1.set_ylabel('무게 (%)', color='blue', fontsize=12)
ax1.tick_params(axis='y', labelcolor='blue')

# 오른쪽 Y축: 미분무게 (%/°C)
ax2 = ax1.twinx()
line2, = ax2.plot(df[col_temp], df[col_dtemp], label='미분무게 (%/°C)', color='red', linestyle='--')
ax2.set_ylabel('미분무게 (%/°C)', color='red', fontsize=12)
ax2.tick_params(axis='y', labelcolor='red')

# 제목 및 범례
plt.title('TGA 결과 그래프', fontsize=14)
lines = [line1, line2]
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='best')

# 레이아웃 자동 조절
plt.tight_layout()

# 출력 디렉토리가 없으면 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 저장 경로 정의
png_path = os.path.join(OUT_DIR, "tga_plot.png")
pdf_path = os.path.join(OUT_DIR, "tga_plot.pdf")

# PNG 저장 (300 dpi)
plt.savefig(png_path, dpi=300, bbox_inches='tight')

# PDF 저장
plt.savefig(pdf_path, bbox_inches='tight')

plt.close()
```
