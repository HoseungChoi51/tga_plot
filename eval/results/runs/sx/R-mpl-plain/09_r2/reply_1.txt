```python
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os

# 데이터 파일 경로 및 출력 디렉토리 설정
DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/09_r2/out"

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 출력 디렉토리가 없으면 생성
os.makedirs(OUT_DIR, exist_ok=True)

# Excel 파일 읽기 (모든 시트)
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

# TGA 플롯 그리기
fig, ax1 = plt.subplots(figsize=(8, 6))

colors = ['#1f77b4', '#ff7f0e', '#2ca02c'] # 파랑, 주황, 초록

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=1)
    
    # 컬럼 이름 정리 (공백 제거 등)
    df.columns = [str(col).strip() for col in df.columns]
    
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        if 'Temperature' in col or 'Temp' in col:
            temp_col = col
        elif 'Weight' in col and '%' in col:
            weight_col = col
        elif 'D.W' in col or 'DTG' in col or 'Derivative' in col:
            dtg_col = col
            
    if temp_col is None or weight_col is None:
        continue
        
    x_data = df[temp_col].values
    y_weight = df[weight_col].values
    
    color = colors[i % len(colors)]
    
    # Weight(%)을 왼쪽 Y축에 plotting
    ax1.plot(x_data, y_weight, label=f'{sheet_name} (TGA)', color=color, linewidth=1.5)
    
    # DTG가 있는 경우 오른쪽 Y축에 plotting
    if dtg_col:
        ax2 = ax1.twinx()
        y_dtg = df[dtg_col].values
        ax2.plot(x_data, y_dtg, label=f'{sheet_name} (DTG)', color=color, linestyle='--', linewidth=1.5, alpha=0.7)
        
        # 오른쪽 축 라벨은 첫 번째 시트에서만 표시하거나, 구분되도록 설정
        if i == 0:
            ax2.set_ylabel('DTG (%/°C)', fontsize=10)
            ax2.tick_params(axis='y', direction='out', labelsize=9)
            
# 축 라벨 설정
ax1.set_xlabel('Temperature (°C)', fontsize=10)
ax1.set_ylabel('Weight (%)', fontsize=10)

# 틱 방향 바깥쪽으로, 틱 라벨 크기 9로 설정
ax1.tick_params(axis='both', direction='out', labelsize=9)
if 'ax2' in locals():
    ax2.tick_params(axis='both', direction='out', labelsize=9)

# 범례 추가
lines_1, labels_1 = ax1.get_legend_handles_labels()
if 'ax2' in locals():
    lines_2, labels_2 = ax2.get_legend_handles_labels()
    lines = lines_1 + lines_2
    labels = labels_1 + labels_2
else:
    lines = lines_1
    labels = labels_1

ax1.legend(lines, labels, loc='center left', bbox_to_anchor=(1, 0.5), fontsize=9)

plt.tight_layout()

# 파일 저장
output_path = os.path.join(OUT_DIR, "tga_plot.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
