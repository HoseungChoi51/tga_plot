```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/12_r2/out"

if not os.path.exists(OUT_DIR):
    os.makedirs(OUT_DIR)

# Read the Excel file
xls = pd.ExcelFile(DATA_FILE)

fig, axes = plt.subplots(len(xls.sheet_names), 1, figsize=(8, 4 * len(xls.sheet_names)), sharex=True)

# If there is only one sheet, axes is not an array
if len(xls.sheet_names) == 1:
    axes = [axes]

colors_tga = ['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728', '#9467bd', '#8c564b', '#e377c2', '#7f7f7f', '#bcbd22', '#17becf']
colors_dta = ['#aec7e8', '#ffbb78', '#98df8a', '#ff9896', '#c5b0d5', '#c49c94', '#f7b6d2', '#c7c7c7', '#dbdb8d', '#9edae5']

for i, sheet_name in enumerate(xls.sheet_names):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=1)
    
    # Identify columns based on the preview info
    # Column B: Temperature, Column C: Weight(%), Column D: D.W(%/C)
    temp_col = df.columns[1]
    weight_col = df.columns[2]
    dtg_col = df.columns[3]
    
    ax1 = axes[i]
    ax2 = ax1.twinx()
    
    # Plot TGA (Weight %)
    line1, = ax1.plot(df[temp_col], df[weight_col], color=colors_tga[i % len(colors_tga)], label='Weight (%)')
    ax1.set_ylabel('Weight (%)', color=colors_tga[i % len(colors_tga)])
    ax1.tick_params(axis='y', labelcolor=colors_tga[i % len(colors_tga)])
    
    # Plot DTG (D.W %/C)
    line2, = ax2.plot(df[temp_col], df[dtg_col], color=colors_dta[i % len(colors_dta)], linestyle='--', label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color=colors_dta[i % len(colors_dta)])
    ax2.tick_params(axis='y', labelcolor=colors_dta[i % len(colors_dta)])
    
    # Add vertical line at 450 °C
    ax1.axvline(x=450, color='red', linestyle=':', linewidth=1.5)
    
    # Add text annotation
    # Find y-position for the text to avoid overlapping with data too much if possible, 
    # but placing it near the top or a specific relative position is safer.
    # Using transform coordinates might be easier for consistent placement.
    ax1.text(455, ax1.get_ylim()[1] * 0.8, '산화 시작', rotation=90, va='top', ha='left', fontsize=10, color='red')
    
    # Titles and Labels
    ax1.set_title(f'TGA/DTG - {sheet_name}')
    ax1.set_xlabel('Temperature (°C)')
    
    # Combine legends
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='best')

plt.tight_layout()
output_path = os.path.join(OUT_DIR, "tga_plot.png")
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
