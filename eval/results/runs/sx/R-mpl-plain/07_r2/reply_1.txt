```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/07_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets from the Excel file
xl_file = pd.ExcelFile(DATA_FILE)
sheet_names = xl_file.sheet_names

fig, ax1 = plt.subplots(figsize=(8, 6))

colors = ['#1f77b4', '#ff7f0e', '#2ca02c']
labels_map = {
    "1006R1-V normal": "Normal",
    "1006R1-V acid": "Acid",
    "1006R1-V 750C": "750C"
}

for idx, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)
    
    # Identify columns based on preview info: Temperature (B), Weight% (C), D.W (D)
    temp_col = 'Temperature'
    weight_col = 'Weight(%)'
    dtg_col = 'D.W(%/C)'
    
    x = df[temp_col]
    y1 = df[weight_col]
    y2 = df[dtg_col]
    
    label = labels_map.get(sheet_name, sheet_name)
    color = colors[idx % len(colors)]
    
    # Plot TG curve (left axis)
    line1, = ax1.plot(x, y1, color=color, linestyle='-', linewidth=1.5, label=f'{label} TG')
    
    # Create twin axis for DTG if not already created for this sample? 
    # Usually, one plot per figure or shared axes. Let's use a single figure with two y-axes.
    # To keep it clean, we'll plot all TGs on left and all DTGs on right.
    
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12, color='black')
ax1.tick_params(axis='y', labelcolor='black')

# Create twin axis for DTG
ax2 = ax1.twinx()

for idx, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)
    temp_col = 'Temperature'
    dtg_col = 'D.W(%/C)'
    
    x = df[temp_col]
    y2 = df[dtg_col]
    
    label = labels_map.get(sheet_name, sheet_name)
    color = colors[idx % len(colors)]
    
    # Plot DTG curve (right axis) with dashed line to distinguish
    line2, = ax2.plot(x, y2, color=color, linestyle='--', linewidth=1.5, alpha=0.7, label=f'{label} DTG')

ax2.set_ylabel('Derivative Weight (%/°C)', fontsize=12, color='gray')
ax2.tick_params(axis='y', labelcolor='gray')

# Combine handles and labels for legend
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()

# Place legend at upper right with font size 9
ax1.legend(lines1 + lines2, labels1 + labels2, loc='upper right', fontsize=9)

plt.title('TGA Analysis Results', fontsize=14)
fig.tight_layout()

output_path = os.path.join(OUT_DIR, 'tga_plot.png')
plt.savefig(output_path, dpi=300)
plt.close()
```
