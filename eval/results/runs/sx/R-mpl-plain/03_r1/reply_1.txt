```python
import pandas as pd
import matplotlib.pyplot as plt
import os
import numpy as np
from scipy.signal import find_peaks

# 설정
DATA_FILE = "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/CNT/Plot/eval/results/runs/sx/R-mpl-plain/03_r1/out"
SHEET_NAME = "1006R1-V acid"

# 폰트 설정 (한글 지원)
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

def main():
    # 데이터 로드
    df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME, header=0)
    
    # 컬럼명 정리 (공백 제거 등)
    df.columns = [str(c).strip() for c in df.columns]
    
    # 필요한 컬럼 확인 및 추출
    x_col = 'Temperature'
    y1_col = 'Weight(%)'
    y2_col = 'D.W(%/C)'
    
    if x_col not in df.columns or y1_col not in df.columns or y2_col not in df.columns:
        raise ValueError(f"필요한 컬럼을 찾을 수 없습니다. Available: {df.columns.tolist()}")
        
    temp = df[x_col].values
    weight = df[y1_col].values
    dwt = df[y2_col].values
    
    # 데이터 정렬 (온도 오름차순)
    sort_idx = np.argsort(temp)
    temp_sorted = temp[sort_idx]
    weight_sorted = weight[sort_idx]
    dwt_sorted = dwt[sort_idx]
    
    # X축 범위 필터링 (100 ~ 800)
    mask = (temp_sorted >= 100) & (temp_sorted <= 800)
    temp_plot = temp_sorted[mask]
    weight_plot = weight_sorted[mask]
    dwt_plot = dwt_sorted[mask]
    
    # 피크 탐지 (DTG 곡선에서 최소값인 피크를 찾기 위해 부호 반전 후 find_peaks 사용)
    # DTG는 보통 음수 값으로 분해가 일어나므로, 가장 낮은 점(음수 최대 크기)이 피크임.
    # find_peaks는 양수 피크를 찾으므로, -dwt_plot에 대해 피크를 찾음.
    peaks_indices, _ = find_peaks(-dwt_plot, prominence=0.05) # prominence로 노이즈 필터링 가능
    
    # 만약 피크가 너무 많이 잡히면 prominent 한 것만 상위 2개 선택하거나, 
    # 사용자가 "피크 표시 2개"라고 했으므로, 가장 prominent한 2개를 선택하는 것이 안전함.
    if len(peaks_indices) > 2:
        # prominence 기준으로 정렬하여 상위 2개 인덱스 가져오기
        # find_peaks의 properties에 'prominences'를 요청해야 함
        peaks_indices_props, props = find_peaks(-dwt_plot, prominence=True)
        prominences = props['prominences']
        top_2_idx = np.argsort(prominences)[-2:]
        final_peak_indices = peaks_indices_props[top_2_idx]
    else:
        final_peak_indices = peaks_indices
        
    peak_temps = temp_plot[final_peak_indices]
    peak_dwt_values = dwt_plot[final_peak_indices]
    
    # 플롯 생성
    fig, ax1 = plt.subplots(figsize=(10, 7))
    
    # TGA 곡선 (Weight %) - 왼쪽 Y축
    color_tga = 'tab:red'
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color_tga, fontsize=12)
    line1, = ax1.plot(temp_plot, weight_plot, color=color_tga, label='TGA')
    ax1.tick_params(axis='y', labelcolor=color_tga)
    ax1.grid(True, linestyle='--', alpha=0.5)
    
    # DTG 곡선 (D.W %/C) - 오른쪽 Y축
    ax2 = ax1.twinx()
    color_dtg = 'tab:blue'
    ax2.set_ylabel('D.W (%/°C)', color=color_dtg, fontsize=12)
    line2, = ax2.plot(temp_plot, dwt_plot, color=color_dtg, linestyle='--', label='DTG')
    ax2.tick_params(axis='y', labelcolor=color_dtg)
    
    # X축 범위 설정
    ax1.set_xlim(100, 800)
    
    # 피크 표시
    for i, (t_val, d_val) in enumerate(zip(peak_temps, peak_dwt_values)):
        # 피크 위치 마커
        ax2.plot(t_val, d_val, 'o', color='black', markersize=8, zorder=5)
        # 텍스트 라벨
        # y위치를 약간 조정하여 가독성 확보
        offset = 0.05 if d_val < 0 else -0.05
        ax2.annotate(f'{t_val:.1f}°C', 
                     xy=(t_val, d_val), 
                     xytext=(t_val + 20, d_val + offset),
                     arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=4),
                     fontsize=10, ha='left', va='bottom' if d_val<0 else 'top')

    # 범례
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='upper right')
    
    # 제목
    plt.title(f'TGA/DTG Analysis: {SHEET_NAME}', fontsize=14, pad=15)
    
    # 저장
    if not os.path.exists(OUT_DIR):
        os.makedirs(OUT_DIR)
        
    out_path = os.path.join(OUT_DIR, "TGA_acid_plot.png")
    plt.tight_layout()
    plt.savefig(out_path, dpi=300)
    plt.close()

if __name__ == "__main__":
    main()
```
