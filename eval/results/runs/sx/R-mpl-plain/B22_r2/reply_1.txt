```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# 설정
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/B22_r2/out"
SHEET_NAME = "Raw"

# 데이터 읽기
df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)

# 열 이름 확인 및 할당 (미리보기 정보 기반)
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"

# 150 °C에서의 무게 값 찾기
# 온도 오차 허용 범위 내에서 가장 가까운 점 또는 보간을 고려할 수 있지만,
# 일반적으로 TGA 데이터는 밀집되어 있으므로 가장 가까운 점을 사용하거나 선형보간을 사용합니다.
# 여기서는 간단하게 150도에 가장 가까운 데이터 포인트의 무게를 기준으로 삼습니다.
# 더 정확성을 위해선 np.interp를 사용할 수도 있습니다.

temps = df[temp_col].values
weights = df[weight_col].values

# 150도에서의 기준 무게 계산 (선형 보간 사용 권장)
ref_temp = 150.0
try:
    # numpy의 interp는 x가 증가하는 순서여야 함. TGA 데이터는 보통 증가함.
    ref_weight = np.interp(ref_temp, temps, weights)
except NameError:
    import numpy as np
    ref_weight = np.interp(ref_temp, temps, weights)

# 정규화된 무게 (%) 계산: (현재 무게 / 150도 무게) * 100
normalized_weights = (weights / ref_weight) * 100

# 그래프 그리기
plt.figure(figsize=(8, 6))
plt.plot(temps, normalized_weights, label=f'{SHEET_NAME} (Normalized at {ref_temp}°C)', color='blue')

# 축 라벨 및 제목
plt.xlabel('Temperature (°C)')
plt.ylabel('Weight (%)')
plt.title(f'TGA Curve - {SHEET_NAME} Sheet (Dry Basis @ {ref_temp}°C)')
plt.grid(True, linestyle='--', alpha=0.7)
plt.legend()

# 출력 디렉토리 생성
os.makedirs(OUT_DIR, exist_ok=True)

# 파일 저장
output_path = os.path.join(OUT_DIR, 'tga_raw_normalized_150C.png')
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
