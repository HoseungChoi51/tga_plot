import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-plain/08_r1/out"

# Read all sheets from the Excel file

sheet_names = xl_file.sheet_names

fig, ax1 = plt.subplots(figsize=(10, 6))

colors = ['#1f77b4', '#ff7f0e', '#2ca02c']

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(xl_file, sheet_name=sheet_name)
    
    # Identify columns based on standard TGA naming conventions provided in preview
    # Temperature is x-axis
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        col_lower = str(col).lower()
        if 'temperature' in col_lower or 'temp' in col_lower:
            temp_col = col
        elif 'weight' in col_lower and '%' in col_lower and 'd.' not in col_lower and 'dw' not in col_lower:
            # Heuristic to distinguish Weight(%) from D.W(%/C)
            if 'd.w' not in col_lower and 'dw' not in col_lower:
                weight_col = col
        elif 'd.w' in col_lower or 'dw' in col_lower or 'derivative' in col_lower:
            dtg_col = col
            
    # Fallback if names are slightly different but order is known (A=Time, B=Temp, C=Weight, D=DTG)
    if temp_col is None and len(df.columns) >= 2:
        temp_col = df.columns[1]
    if weight_col is None and len(df.columns) >= 3:
        weight_col = df.columns[2]
    if dtg_col is None and len(df.columns) >= 4:
        dtg_col = df.columns[3]

    if temp_col and weight_col:
        color = colors[i % len(colors)]
        
        # Plot Weight (%) on primary y-axis (ax1)
        line1 = ax1.plot(df[temp_col], df[weight_col], label=f'{sheet_name} - 무게', color=color, linewidth=2)
        
        # Create secondary y-axis for DTG if column exists
        if dtg_col:
            ax2 = ax1.twinx()
            # Offset the position of the second y-axis so they don't overlap completely if needed, 
            # but standard twin usually works fine. Let's keep it simple.
            line2 = ax2.plot(df[temp_col], df[dtg_col], label=f'{sheet_name} - 미분 무게', color=color, linestyle='--', linewidth=1.5, alpha=0.7)
            
            # Combine legends if we have multiple lines per axis group? 
            # Usually better to combine handles and labels at the end.
            
# Labels requested by user
ax1.set_xlabel('온도 (°C)', fontsize=12)
ax1.set_ylabel('무게 (%)', fontsize=12, color='black')
ax1.tick_params(axis='y', labelcolor='black')

# For the secondary axis label
# We need to access ax2. Since ax2 is created inside the loop only if dtg_col exists, 
# we should ensure ax2 exists for labeling.

    # No, better to check if any sheet had dtg. Based on preview, all do.

# Actually, let's restructure the plotting part slightly to be cleaner regarding axes creation
plt.close(fig) # Close previous fig to restart cleanly with correct axis handling

fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx() # Always create twinx since all sheets have DTG data in preview

all_lines_1 = []
all_labels_1 = []
all_lines_2 = []
all_labels_2 = []

for i, sheet_name in enumerate(sheet_names):
    df = pd.read_excel(xl_file, sheet_name=sheet_name)
    
    temp_col = df.columns[1] # B
    weight_col = df.columns[2] # C
    dtg_col = df.columns[3] # D
    
    color = colors[i % len(colors)]
    
    l1, = ax1.plot(df[temp_col], df[weight_col], label=f'{sheet_name}', color=color, linewidth=2)
    l2, = ax2.plot(df[temp_col], df[dtg_col], color=color, linestyle='--', linewidth=1.5, alpha=0.7)
    
    all_lines_1.append(l1)
    all_labels_1.append(f'{sheet_name} (무게)')
    all_lines_2.append(l2)
    all_labels_2.append(f'{sheet_name} (미분)')

# Set labels as requested
ax1.set_xlabel('온도 (°C)', fontsize=12)
ax1.set_ylabel('무게 (%)', fontsize=12)
ax2.set_ylabel('미분 무게 (%/°C)', fontsize=12)

# Combine legends
lines = all_lines_1 + all_lines_2
labels = all_labels_1 + all_labels_2
ax1.legend(lines, labels, loc='center left', bbox_to_anchor=(1, 0.5), title='시트 이름')

plt.title('TGA 분석 결과', fontsize=14)
