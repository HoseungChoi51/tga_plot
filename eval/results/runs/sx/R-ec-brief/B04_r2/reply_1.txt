```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/xrd.csv"

df = pd.read_csv(DATA_FILE)

x_col = "2Theta (deg)"
y_col = "Intensity (counts)"

# Sort by 2Theta to ensure proper line plot order
df = df.sort_values(by=x_col).reset_index(drop=True)

x_data = df[x_col].tolist()
y_data = df[y_col].tolist()

# Find top 5 strongest peaks
# Use scipy.signal.find_peaks if available, otherwise simple local maxima detection
try:
    from scipy.signal import find_peaks
    # prominence helps avoid noise peaks; height threshold optional
    peaks, properties = find_peaks(y_data, prominence=10, distance=10)
    peak_indices = peaks
except Exception:
    # Fallback: simple local maxima
    peak_indices = []
    for i in range(1, len(y_data)-1):
        if y_data[i] > y_data[i-1] and y_data[i] > y_data[i+1]:
            peak_indices.append(i)

if len(peak_indices) == 0:
    # If no peaks found via algorithm, just take top 5 highest intensities
    top5_idx = np.argsort(y_data)[-5:]
else:
    # Sort peaks by intensity descending and take top 5
    peak_intensities = [y_data[i] for i in peak_indices]
    sorted_peak_indices = sorted(range(len(peak_indices)), key=lambda k: peak_intensities[k], reverse=True)
    top5_idx = [peak_indices[j] for j in sorted_peak_indices[:5]]

# Prepare scatter data for the 5 peaks
scatter_points = []
for idx in top5_idx:
    theta_val = x_data[idx]
    int_val = y_data[idx]
    scatter_points.append({
        "value": [theta_val, int_val],
        "label": {
            "show": True,
            "formatter": f"{theta_val:.2f}",
            "position": "top",
            "fontSize": 10,
            "color": "#c0392b"
        }
    })

option = {
    "title": {
        "text": "XRD Pattern",
        "left": "center",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 14
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "cross"}
    },
    "grid": {
        "left": "10%",
        "right": "5%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "2θ (deg)",
        "nameLocation": "middle",
        "nameGap": 30,
        "min": 10,
        "max": 80,
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 10
        },
        "splitLine": {
            "show": True,
            "lineStyle": {"type": "dashed", "opacity": 0.5}
        }
    },
    "yAxis": {
        "type": "value",
        "name": "Intensity (counts)",
        "nameLocation": "middle",
        "nameGap": 40,
        "axisLabel": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 10
        },
        "splitLine": {
            "show": True,
            "lineStyle": {"type": "dashed", "opacity": 0.5}
        }
    },
    "series": [
        {
            "name": "Intensity",
            "type": "line",
            "data": list(zip(x_data, y_data)),
            "smooth": False,
            "symbol": "none",
            "lineStyle": {
                "color": "#1f4e9c",
                "width": 1.5
            },
            "areaStyle": {
                "color": "rgba(31, 78, 156, 0.1)"
            }
        },
        {
            "name": "Top Peaks",
            "type": "scatter",
            "data": scatter_points,
            "symbolSize": 8,
            "itemStyle": {
                "color": "#c0392b"
            },
            "zlevel": 10
        }
    ]
}

result = {
    "figures": [
        {
            "name": "xrd_pattern",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(result))
```
