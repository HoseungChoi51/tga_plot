import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
SHEET_NAME = "Annealed"

# Read the specific sheet
df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)

# Identify columns based on preview and standard TGA structure
# A: Time, B: Temperature, C: Weight, D: Deriv. Weight
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

# Extract data
temps = df[temp_col].values
weights = df[weight_col].values
dtgs = df[dtg_col].values

# Prepare series data for ECharts
# We need to pair temperature with weight and DTG
# For scatter/line in ECharts, we can use [x, y] pairs or separate arrays if using xAxis type 'value'
# Using [x, y] format is robust for value axes

weight_data = [[t, w] for t, w in zip(temps, weights)]
# Multiply DTG by -1 so peaks point up as per instructions
dtg_data = [[t, d * -1] for t, d in zip(temps, dtgs)]

# Calculate Residue at ~800°C
# Find index closest to 800
idx_800 = np.argmin(np.abs(temps - 800))
residue_val = weights[idx_800]
residue_temp = temps[idx_800]
residue_str = f"Residue {residue_val:.1f}%"

# Calculate Peak above 150°C
# Filter DTG for temp > 150
mask_peak = temps > 150
if np.any(mask_peak):
    # DTG is negative during mass loss. The "highest peak" usually means the most negative value (max mass loss rate).
    # However, we plotted -DTG. So we look for the maximum of (-DTG) which corresponds to the minimum of original DTG.
    dtg_filtered = dtgs[mask_peak]
    temps_filtered = temps[mask_peak]
    
    # Index of minimum original DTG (which is max of -DTG)
    min_dtg_idx_in_filtered = np.argmin(dtg_filtered)
    peak_temp = temps_filtered[min_dtg_idx_in_filtered]
    peak_val_original = dtg_filtered[min_dtg_idx_in_filtered]
    peak_val_plotted = peak_val_original * -1
    
    # Find corresponding weight at this peak temp for context (optional, but good for debugging)
    idx_peak = np.argmin(np.abs(temps - peak_temp))
    weight_at_peak = weights[idx_peak]
else:
    peak_temp = None
    peak_val_plotted = None
    peak_str = ""
    weight_at_peak = None

peak_str = f"{peak_temp:.0f} °C" if peak_temp is not None else ""
