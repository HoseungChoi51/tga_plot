import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def get_nearest_value(x_vals, y_vals, target_x):
    """Find the y value at the x point nearest to target_x."""
    if not len(x_vals):
        return None, None, None
    idx = np.argmin(np.abs(np.array(x_vals) - target_x))
    return x_vals[idx], y_vals[idx], idx

def find_peak_temp(x_vals, dtg_vals, min_temp=150):
    """Find temperature of highest DTG peak (most negative DTG) above min_temp."""
    mask = np.array(x_vals) > min_temp
    if not np.any(mask):
        return None, None, None
    filtered_x = np.array(x_vals)[mask]
    filtered_dtg = np.array(dtg_vals)[mask]
    # Highest peak means most negative DTG value
    idx = np.argmin(filtered_dtg)
    return filtered_x[idx], filtered_dtg[idx], idx

def make_tga_option(df, sheet_name):
    # Identify columns
    cols = df.columns.tolist()
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for c in cols:
        cl = str(c).lower()
        if 'temp' in cl or '온도' in cl:
            temp_col = c
        elif 'weight' in cl or '무게' in cl or 'wt' in cl:
            weight_col = c
        elif 'd.w' in cl or 'dtg' in cl or 'derivative' in cl or '미분' in cl:
            dtg_col = c
            
    if temp_col is None or weight_col is None or dtg_col is None:
        return None
        
    x = df[temp_col].dropna().values
    y1 = df[weight_col].dropna().values
    y2_raw = df[dtg_col].dropna().values
    
    # Align lengths by dropping NaNs properly from all series together
    valid_mask = ~(df[temp_col].isna() | df[weight_col].isna() | df[dtg_col].isna())
    x = df.loc[valid_mask, temp_col].values
    y1 = df.loc[valid_mask, weight_col].values
    y2_raw = df.loc[valid_mask, dtg_col].values
    
    # Negate DTG so peaks point up
    y2 = -y2_raw
    
    # Calculate Residue at 800C
    res_x, res_y, _ = get_nearest_value(x, y1, 800.0)
    
    # Calculate Peak Temperature (>150C)
    peak_x, peak_dtg_val, _ = find_peak_temp(x, y2_raw, 150)
    
    # Prepare data points for ECharts
    # Format: [x, y]
    data_weight = [[float(a), float(b)] for a, b in zip(x, y1)]
    data_dtg = [[float(a), float(b)] for a, b in zip(x, y2)]
    
    # Axis ranges with some padding
    x_min = float(min(x))
    x_max = float(max(x))
    y1_min = float(min(y1))
    y1_max = float(max(y1))
    y2_min = float(min(y2))
    y2_max = float(max(y2))
    
    # Add padding
    x_pad = (x_max - x_min) * 0.05
    y1_pad = (y1_max - y1_min) * 0.1
    y2_pad = (y2_max - y2_min) * 0.1
    
    axis_x_range = [max(0, x_min - x_pad), x_max + x_pad]
    axis_y1_range = [max(0, y1_min - y1_pad), y1_max + y1_pad]
    # For right axis, ensure it covers 0 nicely
    axis_y2_range = [min(-y2_pad, y2_min), y2_max + y2_pad]
    
    gray_color = "#808080"
    font_size = 13
    
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": 10,
            "textStyle": {
                "color": gray_color,
                "fontSize": font_size + 2,
                "fontWeight": "bold"
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"},
            "textStyle": {"color": gray_color}
        },
        "legend": {
            "data": ["Weight (%)", "-DTG (%/°C)"],
            "orient": "vertical",
            "left": 20,
            "top": "middle",
            "textStyle": {
                "color": gray_color,
                "fontSize": font_size
            }
        },
        "grid": {
            "left": "15%",
            "right": "15%",
            "bottom": "15%",
            "top": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "min": axis_x_range[0],
            "max": axis_x_range[1],
            "axisLine": {"lineStyle": {"color": gray_color}},
            "axisTick": {"show": True, "lineStyle": {"color": gray_color}},
            "axisLabel": {"color": gray_color, "fontSize": font_size},
            "splitLine": {"show": False}
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "min": axis_y1_range[0],
                "max": axis_y1_range[1],
                "position": "left",
                "axisLine": {"lineStyle": {"color": gray_color}},
                "axisTick": {"show": True, "lineStyle": {"color": gray_color}},
                "axisLabel": {"color": gray_color, "fontSize": font_size},
                "splitLine": {"show": True, "lineStyle": {"color": "#e0e0e0"}}
            },
            {
                "type": "value",
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "min": axis_y2_range[0],
                "max": axis_y2_range[1],
                "position": "right",
                "axisLine": {"lineStyle": {"color": gray_color}},
                "axisTick": {"show": True, "lineStyle": {"color": gray_color}},
                "axisLabel": {"color": gray_color, "fontSize": font_size},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "data": data_weight,
                "smooth": True,
                "symbol": "none",
                "lineStyle": {"color": "#1f4e9c", "width": 2},
                "itemStyle": {"color": "#1f4e9c"}
            },
            {
                "name": "-DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": data_dtg,
                "smooth": True,
                "symbol": "none",
                "lineStyle": {"color": "#c0392b", "width": 2},
                "itemStyle": {"color": "#c0392b"}
            }
        ]
    }
    
    # Add Residue Label and Marker
    if res_x is not None:
        residue_text = f"Residue {res_y:.1f}%"
        option["graphic"] = []
        
        # Residue Marker on Weight Curve
        option["graphic"].append({
            "type": "circle",
            "shape": {"cx": 0, "cy": 0, "r": 4},
            "style": {"fill": "#1f4e9c", "stroke": "#fff", "lineWidth": 1},
            "position": [0, 0],
            "z": 10,
            "key": f"res_marker_{sheet_name}",
            "convertToPixel": {
                "seriesName": "Weight (%)"
            },
            "data": [res_x, res_y] 
        })
        
        # Residue Text
        option["graphic"].append({
            "type": "text",
            "style": {
                "text": residue_text,
                "fill": gray_color,
                "fontSize": font_size,
                "fontFamily": "Noto Sans CJK KR"
            },
            "position": [0, 0],
            "z": 10,
            "key": f"res_text_{sheet_name}",
            "convertToPixel": {
                "seriesName": "Weight (%)"
            },
            "data": [res_x, res_y],
            "offset": [10, -10]
        })

    # Add Peak Label and Marker
    if peak_x is not None:
        peak_text = f"{peak_x:.0f} °C"
        
        # Peak Marker on DTG Curve
        # Note: The marker should be on the -DTG curve (y2)
        # We need to find the corresponding y2 value for this peak_x
        # Since we found peak based on raw DTG min, the negated value is max
        peak_y_neg = -find_peak_temp(x, y2_raw, 150)[1]
        
        option.setdefault("graphic", [])
        
        option["graphic"].append({
            "type": "circle",
            "shape": {"cx": 0, "cy": 0, "r": 4},
            "style": {"fill": "#c0392b", "stroke": "#fff", "lineWidth": 1},
            "position": [0, 0],
            "z": 10,
            "key": f"peak_marker_{sheet_name}",
            "convertToPixel": {
                "seriesName": "-DTG (%/°C)"
            },
            "data": [peak_x, peak_y_neg]
        })
        
        # Peak Text
        option["graphic"].append({
            "type": "text",
            "style": {
                "text": peak_text,
                "fill": gray_color,
                "fontSize": font_size,
                "fontFamily": "Noto Sans CJK KR"
            },
            "position": [0, 0],
            "z": 10,
            "key": f"peak_text_{sheet_name}",
            "convertToPixel": {
                "seriesName": "-DTG (%/°C)"
            },
            "data": [peak_x, peak_y_neg],
            "offset": [10, -10]
        })

    return option

def main():
    xls = pd.ExcelFile(DATA_FILE)
    figures = []
    
    for sheet_name in xls.sheet_names:
        df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        opt = make_tga_option(df, sheet_name)
        if opt:
            figures.append({
                "name": sheet_name.replace(" ", "_"),
                "option": opt,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            })
            
    result = {"figures": figures}
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
