```python
import pandas as pd
import numpy as np
from scipy.signal import find_peaks
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

# Read data
df = pd.read_csv(DATA_FILE)

# Identify columns
x_col = "Temperature (°C)"
y1_col = "Weight (%)"
heat_flow_col = "Heat Flow (W/g)"

# Extract data
temp = df[x_col].values
weight = df[y1_col].values
heat_flow = df[heat_flow_col].values

# Prepare series data for TGA (Weight %) and Heat Flow
tga_data = [[float(t), float(w)] for t, w in zip(temp, weight)]
hf_data = [[float(t), float(hf)] for t, hf in zip(temp, heat_flow)]

# Find peaks in Heat Flow
# Endothermic peaks: positive values (assuming standard convention where endo is up)
# Exothermic peaks: negative values
# We need to filter by temperature > 150°C as per typical TGA peak analysis rules mentioned in prompt context, 
# though user didn't explicitly restrict temp range for peaks, standard practice applies.
mask_temp = temp > 150
hf_filtered = heat_flow[mask_temp]
temp_filtered = temp[mask_temp]

if len(hf_filtered) == 0:
    hf_filtered = heat_flow
    temp_filtered = temp

# Normalize or just use raw? Raw is better for identifying sign.
# Find endothermic peaks (positive maxima)
# Distance parameter helps avoid noise. Prominence helps ignore small bumps.
peaks_endo_idx, _ = find_peaks(hf_filtered, prominence=0.2, distance=10)
peaks_endo_temps = temp_filtered[peaks_endo_idx]
peaks_endo_vals = hf_filtered[peaks_endo_idx]

# Sort by value descending to get the two most prominent if more than 2 found, or just take top 2
if len(peaks_endo_temps) >= 2:
    # Get indices of top 2 highest peaks
    sorted_indices = np.argsort(peaks_endo_vals)[::-1][:2]
    selected_endo_temps = peaks_endo_temps[sorted_indices]
else:
    selected_endo_temps = peaks_endo_temps[:2]

# Find exothermic peaks (negative minima -> look for peaks in -HF)
neg_hf = -hf_filtered
peaks_exo_idx, _ = find_peaks(neg_hf, prominence=0.2, distance=10)
peaks_exo_temps = temp_filtered[peaks_exo_idx]
peaks_exo_vals = neg_hf[peaks_exo_idx] # These are positive magnitudes

if len(peaks_exo_temps) >= 1:
    # Get index of the single most prominent exothermic peak
    best_exo_idx = np.argmax(peaks_exo_vals)
    selected_exo_temp = peaks_exo_temps[best_exo_idx]
else:
    selected_exo_temp = None

# Calculate Residue at ~800 C
idx_800 = np.argmin(np.abs(temp - 800))
residue_weight = weight[idx_800]
residue_temp = temp[idx_800]

# Build ECharts Option
option = {
    "backgroundColor": "#ffffff",
    "title": {
        "text": "caox",
        "left": "center",
        "top": 10,
        "textStyle": {
            "fontSize": 16,
            "fontWeight": "bold",
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "cross"},
        "formatter": function_placeholder = "{a0} <br/> Temp: {b0} °C<br/> Weight: {c0}%<br/>Heat Flow: {d0} W/g" # Not strictly supported in pure JSON without JS functions, so we rely on default tooltip or simple formatting
    },
    "legend": {
        "data": ["Weight (%)", "Heat Flow (W/g)"],
        "top": 40,
        "left": "center",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "axisLine": {"lineStyle": {"color": "#333"}},
        "axisTick": {"show": True},
        "splitLine": {"show": False},
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"}
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "position": "left",
            "axisLine": {"lineStyle": {"color": "#1f4e9c"}},
            "axisTick": {"show": True},
            "splitLine": {"show": True, "lineStyle": {"type": "dashed", "color": "#ccc"}},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"}
        },
        {
            "type": "value",
            "name": "Heat Flow (W/g)",
            "position": "right",
            "axisLine": {"lineStyle": {"color": "#c0392b"}},
            "axisTick": {"show": True},
            "splitLine": {"show": False},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"}
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "yAxisIndex": 0,
            "data": tga_data,
            "smooth": False,
            "symbol": "none",
            "lineStyle": {
                "color": "#1f4e9c",
                "width": 2
            },
            "markPoint": {
                "data": [
                    {
                        "coord": [float(residue_temp), float(residue_weight)],
                        "value": f"Residue {residue_weight:.1f}%",
                        "symbolSize": 8,
                        "itemStyle": {"color": "#1f4e9c"},
                        "label": {
                            "show": True,
                            "formatter": "{b}",
                            "position": "top",
                            "distance": 10,
                            "fontFamily": "Noto Sans CJK KR",
                            "fontSize": 10
                        }
                    }
                ]
            }
        },
        {
            "name": "Heat Flow (W/g)",
            "type": "line",
            "yAxisIndex": 1,
            "data": hf_data,
            "smooth": False,
            "symbol": "none",
            "lineStyle": {
                "color": "#c0392b",
                "width": 2
            },
            "markPoint": {
                "data": []
            }
        }
    ]
}

# Add MarkPoints for Peaks to the Heat Flow series
peak_mark_points = []

# Endothermic peaks
for i, t_peak in enumerate(selected_endo_temps):
    # Find closest actual data point for accurate coord
    idx = np.argmin(np.abs(temp - t_peak))
    peak_val = heat_flow[idx]
    peak_mark_points.append({
        "coord": [float(temp[idx]), float(peak_val)],
        "value": f"{int(round(temp[idx]))}°C",
        "symbolSize": 8,
        "itemStyle": {"color": "#27ae60"}, # Green for endo
        "label": {
            "show": True,
            "formatter": "{b}",
            "position": "top",
            "distance": 5,
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 10,
            "color": "#27ae60"
        }
    })

# Exothermic peak
if selected_exo_temp is not None:
    idx = np.argmin(np.abs(temp - selected_exo_temp))
    peak_val = heat_flow[idx]
    peak_mark_points.append({
        "coord": [float(temp[idx]), float(peak_val)],
        "value": f"{int(round(temp[idx]))}°C",
        "symbolSize": 8,
        "itemStyle": {"color": "#e67e22"}, # Orange for exo
        "label": {
            "show": True,
            "formatter": "{b}",
            "position": "bottom",
            "distance": 5,
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 10,
            "color": "#e67e22"
        }
    })

option["series"][1]["markPoint"]["data"] = peak_mark_points

# Construct final output
output = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(output))
```
