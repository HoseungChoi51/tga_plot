import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def make_tga_plot(df, sheet_name):
    # Identify columns based on standard TGA structure from preview
    x_col = 'Temperature'
    y1_col = 'Weight(%)'
    y2_col = 'D.W(%/C)'
    
    if x_col not in df.columns or y1_col not in df.columns or y2_col not in df.columns:
        return None
        
    temp = df[x_col].values
    weight = df[y1_col].values
    dtg = df[y2_col].values
    
    # Prepare data for ECharts
    # Weight line (y1)
    weight_data = [[float(t), float(w)] for t, w in zip(temp, weight)]
    
    # DTG line (y2), multiplied by -1 to point peaks up
    dtg_data = [[float(t), float(-d)] for t, d in zip(temp, dtg)]
    
    # Calculate Residue at 800°C
    # Find index closest to 800
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = weight[idx_800]
    residue_temp = temp[idx_800]
    
    # Calculate Peak temperature (highest DTG peak above 150°C)
    # Filter for temp > 150 and find max of -dtg (which is the positive peak value)
    mask = temp > 150
    if np.any(mask):
        dtg_neg = -dtg[mask]
        temps_filtered = temp[mask]
        max_idx = np.argmax(dtg_neg)
        peak_temp = temps_filtered[max_idx]
        peak_val = dtg_neg[max_idx] # This is the height of the inverted peak
    else:
        peak_temp = None
        peak_val = None

    # Build Option
    option = {
        "backgroundColor": "#fff",
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"}
        },
        "legend": {
            "data": ["Weight (%)", "-DTG (%/°C)"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "scale": False,
            "axisLine": {"lineStyle": {"color": "#333"}},
            "axisTick": {"alignWithLabel": True},
            "splitLine": {"show": True, "lineStyle": {"color": "#eee"}}
        },
        "yAxis": [
            {
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "left",
                "axisLine": {"lineStyle": {"color": "#000"}}, # Black axis line for consistency with black curve
                "splitLine": {"show": True, "lineStyle": {"color": "#eee"}}
            },
            {
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "right",
                "axisLine": {"lineStyle": {"color": "#c0392b"}},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "data": weight_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#000000",  # Black color
                    "width": 2.5         # Width 2.5
                },
                "markPoint": {
                    "data": []
                }
            },
            {
                "name": "-DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": dtg_data,
                "smooth": False,
                "symbol": "none",
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 1.5,
                    "type": "dashed"     # Dashed line
                }
            }
        ]
    }
    
    # Add Markers for Residue and Peak
    mark_points = []
    
    # Residue Marker
    mark_points.append({
        "coord": [residue_temp, residue_val],
        "name": f"Residue {residue_val:.1f}%",
        "symbolSize": 10,
        "itemStyle": {"color": "#000000"},
        "label": {
            "show": True,
            "formatter": "{b}",
            "position": "top",
            "distance": 5,
            "color": "#000000",
            "fontSize": 10,
            "fontFamily": "Noto Sans CJK KR"
        }
    })
    
    # Peak Marker
    if peak_temp is not None:
        # The peak marker should be on the DTG series visually, but often marked on the graph generally.
        # We place it on the second series (index 1) so it aligns with y2 axis values? 
        # Actually, markPoint coord uses the data coordinate system. 
        # For the second series, the Y value is -dtg.
        # So we need the -dtg value at peak_temp.
        # Since peak_val was calculated as max(-dtg), that IS the y-value for the second series.
        
        mark_points_dtg = [{
            "coord": [peak_temp, peak_val],
            "name": f"{peak_temp:.0f} °C",
            "symbolSize": 10,
            "itemStyle": {"color": "#c0392b"},
            "label": {
                "show": True,
                "formatter": "{b}",
                "position": "top",
                "distance": 5,
                "color": "#c0392b",
                "fontSize": 10,
                "fontFamily": "Noto Sans CJK KR"
            }
        }]
        
        # Assign markers to respective series
        option["series"][0]["markPoint"] = {"data": [mp for mp in mark_points if "Residue" in mp["name"]]}
        option["series"][1]["markPoint"] = {"data": mark_points_dtg}
    else:
        option["series"][0]["markPoint"] = {"data": mark_points}
        option["series"][1]["markPoint"] = {"data": []}

    return option

figures = []
xl_file = pd.ExcelFile(DATA_FILE)
for sheet_name in xl_file.sheet_names:
    df = pd.read_excel(xl_file, sheet_name=sheet_name)
    opt = make_tga_plot(df, sheet_name)
    if opt:
        figures.append({
            "name": f"plot_{sheet_name.replace(' ', '_')}",
            "option": opt,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        })

output = {"figures": figures}
print(json.dumps(output))
