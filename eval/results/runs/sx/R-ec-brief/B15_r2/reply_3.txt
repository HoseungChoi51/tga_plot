```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

# Read all sheets
xls = pd.ExcelFile(DATA_FILE)
sheets_data = {}
for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name)
    # Standardize column names based on preview
    cols = list(df.columns)
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for c in cols:
        cl = str(c).lower()
        if 'temperature' in cl or 'temp' in cl:
            temp_col = c
        elif 'weight' in cl and 'deriv' not in cl:
            weight_col = c
        elif 'deriv' in cl or 'dtg' in cl:
            dtg_col = c
            
    if temp_col is None or weight_col is None or dtg_col is None:
        continue
        
    data = {
        'temp': df[temp_col].values,
        'weight': df[weight_col].values,
        'dtg': df[dtg_col].values
    }
    sheets_data[sheet_name] = data

if not sheets_data:
    print(json.dumps({"figures": []}))
else:
    # Prepare series data for combined plot
    series_list = []
    
    # Colors for different sheets
    colors = ['#1f4e9c', '#c0392b', '#27ae60']
    
    # We need to find the global min/max for axes to make a single coherent plot
    all_temps = []
    all_weights = []
    all_dtgs = []
    
    for name, d in sheets_data.items():
        all_temps.extend(d['temp'].tolist())
        all_weights.extend(d['weight'].tolist())
        all_dtgs.extend(d['dtg'].tolist())
        
    x_min = float(min(all_temps))
    x_max = float(max(all_temps))
    y1_min = float(min(all_weights) * 0.95) # slight padding
    y1_max = float(max(all_weights) * 1.05)
    # DTG is negative, we multiply by -1 for display so peaks are up.
    # Min of original DTG becomes Max of displayed DTG.
    dtg_min_orig = float(min(all_dtgs))
    dtg_max_orig = float(max(all_dtgs))
    # Displayed DTG range: from -dtg_max_orig (which is near 0 or small positive) to -dtg_min_orig (large positive)
    # But usually we want symmetric-ish or just covering the peaks.
    # Let's set y2 axis to cover the inverted values.
    y2_min = float(-dtg_max_orig * 1.1) # If max orig is ~0, this is ~0. 
    y2_max = float(-dtg_min_orig * 1.1)
    
    # Ensure y2_min isn't too high if there are no significant drops (unlikely for TGA)
    if y2_min > 0:
        y2_min = 0.0
        
    # Calculate labels for each sheet
    annotations = []
    
    for idx, (name, d) in enumerate(sheets_data.items()):
        temps = d['temp']
        weights = d['weight']
        dtgs = d['dtg']
        
        # 1. Residue at nearest 800°C
        idx_800 = int(np.argmin(np.abs(temps - 800)))
        residue_val = float(weights[idx_800])
        residue_temp = float(temps[idx_800])
        
        # 2. Peak DTG above 150°C
        mask_peak = temps > 150
        if np.any(mask_peak):
            filtered_dtgs = dtgs[mask_peak]
            filtered_temps = temps[mask_peak]
            # Find minimum DTG (most negative = highest mass loss rate)
            min_dtg_idx_local = int(np.argmin(filtered_dtgs))
            peak_temp = float(filtered_temps[min_dtg_idx_local])
            peak_dtg_val = float(filtered_dtgs[min_dtg_idx_local])
        else:
            peak_temp = None
            peak_dtg_val = None
            
        color = colors[idx % len(colors)]
        
        # Add Weight Series (yAxisIndex 0)
        series_list.append({
            "name": f"{name} Weight",
            "type": "line",
            "xAxisIndex": 0,
            "yAxisIndex": 0,
            "data": [[float(t), float(w)] for t, w in zip(temps, weights)],
            "showSymbol": False,
            "lineStyle": {"color": color, "width": 2},
            "itemStyle": {"color": color}
        })
        
        # Add DTG Series (yAxisIndex 1), multiplied by -1
        series_list.append({
            "name": f"{name} DTG",
            "type": "line",
            "xAxisIndex": 0,
            "yAxisIndex": 1,
            "data": [[float(t), float(-d)] for t, d in zip(temps, dtgs)],
            "showSymbol": False,
            "lineStyle": {"color": color, "width": 2, "type": "dashed"},
            "itemStyle": {"color": color}
        })
        
        # Annotations
        # Residue marker
        annotations.append({
            "coord": [residue_temp, residue_val],
            "value": f"Residue {residue_val:.1f}% ({name})",
            "symbolSize": 8,
            "symbol": "circle",
            "itemStyle": {"color": color},
            "label": {
                "show": True,
                "formatter": "{@[0]}°C\n{@[1]:.1f}%", 
                "position": "top",
                "distance": 5,
                "fontSize": 10,
                "color": "#333",
                "backgroundColor": "rgba(255,255,255,0.7)",
                "padding": [2, 4],
                "borderRadius": 2
            },
            "emphasis": {
                "disabled": True
            }
        })
        
        if peak_temp is not None:
            # Peak marker on DTG curve (inverted)
            annotations.append({
                "coord": [peak_temp, -peak_dtg_val],
                "value": f"{peak_temp:.0f}°C",
                "symbolSize": 8,
                "symbol": "triangle",
                "itemStyle": {"color": color},
                "label": {
                    "show": True,
                    "formatter": f"{peak_temp:.0f}°C",
                    "position": "top",
                    "distance": 5,
                    "fontSize": 10,
                    "color": "#333",
                    "backgroundColor": "rgba(255,255,255,0.7)",
                    "padding": [2, 4],
                    "borderRadius": 2
                },
                "emphasis": {
                    "disabled": True
                }
            })

    # Construct ECharts Option
    option = {
        "title": {
            "text": "TGA Analysis: PE, PVC, PET",
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16,
                "fontWeight": "bold"
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": [f"{name} Weight" for name in sheets_data.keys()] + [f"{name} DTG" for name in sheets_data.keys()],
            "orient": "vertical",
            "right": 10,
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 12
            }
        },
        "grid": {
            "left": "8%",
            "right": "20%", # Space for legend outside right
            "bottom": "10%",
            "containLabel": True
        },
        "xAxis": [
            {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 30,
                "min": x_min,
                "max": x_max,
                "axisLine": {"onZero": False},
                "splitLine": {"show": False},
                "axisTick": {"show": True},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"}
            }
        ],
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "min": y1_min,
                "max": y1_max,
                "position": "left",
                "axisLine": {"show": True},
                "splitLine": {"show": True, "lineStyle": {"type": "dashed"}},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"}
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "min": y2_min,
                "max": y2_max,
                "position": "right",
                "axisLine": {"show": True},
                "splitLine": {"show": False},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"}
            }
        ],
        "series": series_list + [
            {
                "name": "Annotations",
                "type": "scatter",
                "markPoint": {
                    "data": annotations,
                    "silent": True
                }
            }
        ]
    }
    
    figure = {
        "name": "tga_combined",
        "option": option,
        "width": 800,
        "height": 500,
        "dpi": 300,
        "formats": ["png"]
    }
    
    result = {"figures": [figure]}
    print(json.dumps(result))
```
