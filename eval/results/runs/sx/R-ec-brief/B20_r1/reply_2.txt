```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"

def main():
    # Read the specific sheet 'Acid-washed'
    df = pd.read_excel(DATA_FILE, sheet_name='Acid-washed')
    
    # Identify columns based on the preview and standard TGA naming
    # Columns: A=Time, B=Temperature, C=Weight, D=Deriv. Weight
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    # Extract data
    temps = df[temp_col].values
    weights = df[weight_col].values
    dtgs = df[dtg_col].values
    
    # Prepare series data for ECharts
    # Format: [[x, y1], [x, y2]] is not directly supported in one series if axes differ.
    # We need two separate series linked to different Y axes.
    
    # Series 1: Weight (%) -> Left Y Axis (yAxisIndex: 0)
    data_weight = [[float(t), float(w)] for t, w in zip(temps, weights)]
    
    # Series 2: DTG * -1 -> Right Y Axis (yAxisIndex: 1)
    # Multiply by -1 so peaks point up
    data_dtg = [[float(t), float(d) * -1] for t, d in zip(temps, dtgs)]
    
    # Find Residue at ~800°C
    # Find index closest to 800
    idx_800 = np.argmin(np.abs(temps - 800))
    residue_val = weights[idx_800]
    residue_temp = temps[idx_800]
    
    # Find Peak temperature of DTG above 150°C
    # Filter for temp > 150
    mask = temps > 150
    dtg_filtered = dtgs[mask]
    temps_filtered = temps[mask]
    
    # Find the minimum of original DTG (most negative value) which corresponds to the highest mass loss rate peak
    if len(dtg_filtered) > 0:
        peak_idx = np.argmin(dtg_filtered)
        peak_val_original = dtg_filtered[peak_idx]
        peak_temp = temps_filtered[peak_idx]
    else:
        peak_val_original = 0
        peak_temp = 0
        
    # Construct Option
    option = {
        "title": {
            "text": "Acid-washed",
            "left": "center",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "-DTG (%/°C)"],
            "top": "10%",
            "left": "center",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "type": "value",
            "position": "bottom",
            "splitLine": {
                "show": False
            },
            "minorTick": {
                "show": True,
                "length": 4,
                "lineStyle": {
                    "color": "#ccc"
                }
            },
            "minorSplitLine": {
                "show": False
            },
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "nameTextStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "yAxis": [
            {
                "name": "Weight (%)",
                "type": "value",
                "position": "left",
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "type": "dashed",
                        "color": "#eee"
                    }
                },
                "axisLabel": {
                    "formatter": "{value} %",
                    "fontFamily": "Noto Sans CJK KR"
                },
                "nameTextStyle": {
                    "fontFamily": "Noto Sans CJK KR"
                }
            },
            {
                "name": "-DTG (%/°C)",
                "type": "value",
                "position": "right",
                "splitLine": {
                    "show": False
                },
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR"
                },
                "nameTextStyle": {
                    "fontFamily": "Noto Sans CJK KR"
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "data": data_weight,
                "smooth": True,
                "symbol": "none",
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [float(residue_temp), float(residue_val)],
                            "label": {
                                "formatter": f"Residue {float(residue_val):.1f}%",
                                "position": "top",
                                "fontFamily": "Noto Sans CJK KR",
                                "fontSize": 12
                            },
                            "symbolSize": 8,
                            "itemStyle": {
                                "color": "#1f4e9c"
                            }
                        }
                    ]
                }
            },
            {
                "name": "-DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": data_dtg,
                "smooth": True,
                "symbol": "none",
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [float(peak_temp), float(-peak_val_original)], # Plotting -DTG, so coord y is -original_dtg
                            "label": {
                                "formatter": f"{int(round(float(peak_temp)))} °C",
                                "position": "top",
                                "fontFamily": "Noto Sans CJK KR",
                                "fontSize": 12
                            },
                            "symbolSize": 8,
                            "itemStyle": {
                                "color": "#c0392b"
                            }
                        }
                    ]
                }
            }
        ]
    }
    
    result = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(result))

if __name__ == "__main__":
    main()
```
