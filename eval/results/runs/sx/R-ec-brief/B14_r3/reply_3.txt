import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

def make_tga_chart(df, sheet_name, width_in=10, height_in=None, font_size=16):
    """
    Create an ECharts option for a TGA plot.
    """
    # Extract columns based on standard TGA structure from the preview
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    # Clean data: drop NaNs
    df_clean = df[[temp_col, weight_col, dtg_col]].dropna()
    
    temps = df_clean[temp_col].values
    weights = df_clean[weight_col].values
    dtgs = df_clean[dtg_col].values
    
    # Calculate Residue at 800°C
    idx_800 = np.argmin(np.abs(temps - 800))
    residue_val = weights[idx_800]
    residue_temp = temps[idx_800]
    
    # Calculate Peak DTG above 150°C
    mask_peak = temps > 150
    if not np.any(mask_peak):
        peak_temp = None
        peak_dtg = None
    else:
        dtgs_filtered = dtgs[mask_peak]
        temps_filtered = temps[mask_peak]
        # Find the most negative DTG value (highest mass loss rate)
        min_idx = np.argmin(dtgs_filtered)
        peak_temp = temps_filtered[min_idx]
        peak_dtg = dtgs_filtered[min_idx]
        
    # Prepare data series
    data_weight = [[float(t), float(w)] for t, w in zip(temps, weights)]
    
    # Multiply DTG by -1 so peaks point up
    data_dtg = [[float(t), float(d) * -1] for t, d in zip(temps, dtgs)]
    
    # Determine axis ranges
    min_t = float(np.min(temps))
    max_t = float(np.max(temps))
    min_w = float(np.min(weights))
    max_w = float(np.max(weights))
    
    neg_dtgs = [-d for d in dtgs]
    min_dtg_flip = float(np.min(neg_dtgs))
    max_dtg_flip = float(np.max(neg_dtgs))
    
    pad_t = (max_t - min_t) * 0.05
    pad_w = (max_w - min_w) * 0.05
    pad_d = (max_dtg_flip - min_dtg_flip) * 0.05
    
    x_min = max(0, min_t - pad_t)
    x_max = max_t + pad_t
    y1_min = min_w - pad_w
    y1_max = max_w + pad_w
    y2_min = min_dtg_flip - pad_d
    y2_max = max_dtg_flip + pad_d
    
    # Build ECharts Option
    option = {
        "backgroundColor": "#ffffff",
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": "3%",
            "textStyle": {
                "fontSize": font_size + 4,
                "fontWeight": "bold",
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "left": "left",
            "top": "middle",
            "orient": "vertical",
            "textStyle": {
                "fontSize": font_size,
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "12%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "min": x_min,
            "max": x_max,
            "axisLine": {"lineStyle": {"color": "#333"}},
            "axisTick": {"show": True},
            "axisLabel": {
                "fontSize": font_size,
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {"show": False}
        },
        "yAxis": [
            {
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "left",
                "min": y1_min,
                "max": y1_max,
                "axisLine": {"lineStyle": {"color": "#1f4e9c"}},
                "axisTick": {"show": True},
                "axisLabel": {
                    "fontSize": font_size,
                    "fontFamily": "Noto Sans CJK KR"
                },
                "splitLine": {"show": True, "lineStyle": {"color": "#eee"}}
            },
            {
                "name": "DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "right",
                "min": y2_min,
                "max": y2_max,
                "axisLine": {"lineStyle": {"color": "#c0392b"}},
                "axisTick": {"show": True},
                "axisLabel": {
                    "fontSize": font_size,
                    "fontFamily": "Noto Sans CJK KR"
                },
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "data": data_weight,
                "smooth": True,
                "symbol": "none",
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "markPoint": {
                    "data": [
                        {
                            "coord": [residue_temp, residue_val],
                            "label": {
                                "formatter": f"Residue {residue_val:.1f}%",
                                "position": "top",
                                "fontSize": font_size,
                                "color": "#1f4e9c",
                                "fontFamily": "Noto Sans CJK KR"
                            },
                            "itemStyle": {
                                "color": "#1f4e9c"
                            }
                        }
                    ]
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": data_dtg,
                "smooth": True,
                "symbol": "none",
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 2
                },
                "markPoint": {
                    "data": []
                }
            }
        ]
    }
    
    # Add peak marker to DTG series if found
    if peak_temp is not None:
        peak_val_flipped = peak_dtg * -1
        option["series"][1]["markPoint"]["data"].append({
            "coord": [peak_temp, peak_val_flipped],
            "label": {
                "formatter": f"{int(round(peak_temp))} °C",
                "position": "top",
                "fontSize": font_size,
                "color": "#c0392b",
                "fontFamily": "Noto Sans CJK KR"
            },
            "itemStyle": {
                "color": "#c0392b"
            }
        })

    return option

# Main Execution
try:
    xl_file = pd.ExcelFile(DATA_FILE)
    sheets_to_plot = ["PVC"]
    
    figures = []
    
    for sheet_name in sheets_to_plot:
        if sheet_name not in xl_file.sheet_names:
            continue
            
        df = pd.read_excel(xl_file, sheet_name=sheet_name)
        
        # User requested: Width 10 inches. 
        # CSS pixels at 96 px/inch -> 10 * 96 = 960 px.
        # Height: Not specified, using 7.5 inches (standard slide ratio) -> 7.5 * 96 = 720 px.
        width_px = int(10 * 96)
        height_px = int(7.5 * 96)
        
        option = make_tga_chart(df, sheet_name, width_in=10, font_size=16)
        
        fig_entry = {
            "name": "plot_pvc",
            "option": option,
            "width": width_px,
            "height": height_px,
            "dpi": 300,
            "formats": ["png"]
        }
        figures.append(fig_entry)
        
    output = {"figures": figures}
    print(json.dumps(output, ensure_ascii=False))

except Exception as e:
    import sys
    print(f"Error: {str(e)}", file=sys.stderr)
    print(json.dumps({"figures": []}))
