```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def generate_tga_plot(df, sheet_name):
    # Identify columns based on standard TGA structure or guess
    cols = df.columns.tolist()
    
    # Try to find Temperature (x), Weight (y1), DTG (y2)
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for c in cols:
        c_lower = str(c).lower()
        if 'temp' in c_lower:
            temp_col = c
        elif 'weight' in c_lower and '%' in c_lower:
            weight_col = c
        elif 'd.w' in c_lower or 'dtg' in c_lower or 'derivative' in c_lower:
            dtg_col = c
            
    # Fallback if names are ambiguous but positions match standard
    if temp_col is None and len(cols) >= 2:
        temp_col = cols[1] # Usually B
    if weight_col is None and len(cols) >= 3:
        weight_col = cols[2] # Usually C
    if dtg_col is None and len(cols) >= 4:
        dtg_col = cols[3] # Usually D
        
    if not all([temp_col, weight_col, dtg_col]):
        return None

    x_data = df[temp_col].values
    y1_data = df[weight_col].values
    y2_data = df[dtg_col].values
    
    # Filter valid data
    mask = ~np.isnan(x_data) & ~np.isnan(y1_data) & ~np.isnan(y2_data)
    x_clean = x_data[mask]
    y1_clean = y1_data[mask]
    y2_clean = y2_data[mask]
    
    if len(x_clean) == 0:
        return None

    # Calculate Residue at nearest 800C
    idx_800 = np.argmin(np.abs(x_clean - 800.0))
    residue_val = y1_clean[idx_800]
    residue_temp = x_clean[idx_800]
    
    # Calculate Peak temperature of DTG above 150C
    # We want the highest peak of -DTG (so lowest value of original DTG)
    mask_peak = x_clean > 150.0
    if np.any(mask_peak):
        x_peak_region = x_clean[mask_peak]
        y2_peak_region = y2_clean[mask_peak]
        
        # Find index of minimum DTG (most negative) which corresponds to max mass loss rate
        min_idx = np.argmin(y2_peak_region)
        peak_temp = x_peak_region[min_idx]
        peak_dtg_val = y2_peak_region[min_idx]
    else:
        peak_temp = None
        peak_dtg_val = None

    # Prepare ECharts Option
    
    # Axis configuration with ticks outside and label size 9
    axis_common = {
        "axisTick": {"show": True, "length": 6, "lineStyle": {"color": "#ccc"}},
        "axisLabel": {"fontSize": 9, "color": "#333"},
        "axisLine": {"lineStyle": {"color": "#ccc"}}
    }
    
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": 10,
            "textStyle": {"fontSize": 14, "fontWeight": "bold", "fontFamily": "Noto Sans CJK KR"}
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"}
        },
        "legend": {
            "data": ["Weight (%)", "-DTG (%/°C)"],
            "top": 40,
            "left": "center",
            "orient": "vertical",
            "icon": "roundRect",
            "itemWidth": 15,
            "itemHeight": 8,
            "textStyle": {"fontSize": 10}
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "top": "20%",
            "containLabel": True
        },
        "xAxis": [
            {
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 30,
                "type": "value",
                "min": 0,
                "max": 950,
                "splitLine": {"show": False},
                **axis_common
            }
        ],
        "yAxis": [
            {
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "left",
                "min": 0,
                "max": 105,
                "splitLine": {"show": True, "lineStyle": {"type": "dashed", "color": "#eee"}},
                **axis_common
            },
            {
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "right",
                "inverse": False, # We will invert data manually or handle sign
                "splitLine": {"show": False},
                **axis_common
            }
        ],
        "series": []
    }
    
    # Series 1: Weight
    series_weight = {
        "name": "Weight (%)",
        "type": "line",
        "smooth": False,
        "symbol": "none",
        "lineStyle": {"color": "#1f4e9c", "width": 2},
        "data": [[float(x), float(y)] for x, y in zip(x_clean, y1_clean)],
        "markPoint": {
            "data": [
                {
                    "coord": [float(residue_temp), float(residue_val)],
                    "symbol": "circle",
                    "symbolSize": 8,
                    "itemStyle": {"color": "#1f4e9c"},
                    "label": {
                        "show": True,
                        "formatter": f"Residue {residue_val:.1f}%",
                        "position": "top",
                        "distance": 5,
                        "fontSize": 9,
                        "color": "#1f4e9c",
                        "backgroundColor": "rgba(255,255,255,0.7)",
                        "padding": [2, 4],
                        "borderRadius": 2
                    }
                }
            ]
        }
    }
    
    # Series 2: -DTG
    # Invert DTG so peaks point up
    y2_inverted = [-val for val in y2_clean]
    
    series_dtg = {
        "name": "-DTG (%/°C)",
        "type": "line",
        "yAxisIndex": 1,
        "smooth": False,
        "symbol": "none",
        "lineStyle": {"color": "#c0392b", "width": 2},
        "data": [[float(x), float(y)] for x, y in zip(x_clean, y2_inverted)],
        "markPoint": {}
    }
    
    if peak_temp is not None:
        # Find the corresponding inverted value for the marker position
        # The marker should be on the curve. 
        # Original DTG at peak_temp was peak_dtg_val (negative).
        # Inverted DTG is -peak_dtg_val (positive).
        inv_peak_val = -peak_dtg_val
        
        series_dtg["markPoint"]["data"] = [
            {
                "coord": [float(peak_temp), float(inv_peak_val)],
                "symbol": "triangle",
                "symbolSize": 10,
                "itemStyle": {"color": "#c0392b"},
                "label": {
                    "show": True,
                    "formatter": f"{int(round(peak_temp))} °C",
                    "position": "top",
                    "distance": 5,
                    "fontSize": 9,
                    "color": "#c0392b",
                    "backgroundColor": "rgba(255,255,255,0.7)",
                    "padding": [2, 4],
                    "borderRadius": 2
                }
            }
        ]
        
    option["series"].append(series_weight)
    option["series"].append(series_dtg)
    
    return option

figures = []
xl = pd.ExcelFile(DATA_FILE)
for sheet in xl.sheet_names:
    df = pd.read_excel(xl, sheet_name=sheet)
    opt = generate_tga_plot(df, sheet)
    if opt:
        figures.append({
            "name": f"plot_{sheet.replace(' ', '_').replace('/', '_')}",
            "option": opt,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        })

print(json.dumps({"figures": figures}))
```
