import pandas as pd
import json

def main():
    file_path = "/home/chs/Work/tga_plot/eval/inputs_b/isothermal.txt"
    
    # Read the data file. It's a text file, likely tab or space separated.
    # Based on the preview, it has headers in row 1 (index 0) and data starts at row 2 (index 1).
    # The separator might be tabs or multiple spaces. Using regex for whitespace.
    try:
        df = pd.read_csv(file_path, sep=r'\s+', header=0)
    except Exception:
        # Fallback if standard read fails, maybe different encoding or structure
        df = pd.read_csv(file_path, sep='\t', header=0)
        
    # Clean column names to handle potential hidden characters or spacing issues
    df.columns = [col.strip() for col in df.columns]
    
    # Identify columns based on the user request and data preview
    # Time (min) -> x-axis
    # Weight (%) -> left y-axis (y1)
    # Temperature (°C) -> right y-axis (y2)
    
    time_col = "Time (min)"
    weight_col = "Weight (%)"
    temp_col = "Temperature (°C)"
    
    # Check if columns exist, otherwise try partial matching or index-based access if needed
    if time_col not in df.columns:
        # Try to find closest match
        cols = list(df.columns)
        time_col = next((c for c in cols if 'time' in c.lower()), cols[0])
        weight_col = next((c for c in cols if 'weight' in c.lower()), cols[2] if len(cols)>2 else cols[1])
        temp_col = next((c for c in cols if 'temp' in c.lower()), cols[1])

    # Extract data
    times = df[time_col].tolist()
    weights = df[weight_col].tolist()
    temps = df[temp_col].tolist()
    
    # Find the weight at 140 minutes for the label
    # The max time is 140.0, so we look for the last point or closest to 140
    target_time = 140.0
    # Get the index of the point closest to 140 min
    idx_140 = (df[time_col] - target_time).abs().idxmin()
    weight_at_140 = float(df.loc[idx_140, weight_col])
    time_at_140 = float(df.loc[idx_140, time_col])
    
    # Prepare series data
    # Series 1: Weight (Left Y Axis)
    # Data format for line chart with x/y axes: [[x, y], [x, y], ...]
    weight_data = [[float(t), float(w)] for t, w in zip(times, weights)]

        },
        "grid": {
            "left": "10%",

            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Time (min)",
            "nameLocation": "middle",
            "nameGap": 30,

            "splitLine": {
                "show": True,

        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "position": "left",
                "min": 70,
                "max": 105,
                "splitLine": {
                    "show": True,

                        "color": "#eee"

        "series": [
            {
                "name": "Weight (%)",
