```python
import pandas as pd
import numpy as np
import json

def main():
    file_path = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"
    
    # Read the Excel file
    xls = pd.ExcelFile(file_path)
    sheet_name = "CNT-01"
    df = pd.read_excel(xls, sheet_name=sheet_name)
    
    # Identify columns based on user request and data preview
    x_col = "Temperature (°C)"
    y1_col = "Weight (mg)"  # User requested Weight (mg) for y1
    y2_col = "Deriv. Weight (%/°C)"
    
    # Extract data
    temp = df[x_col].values
    weight_mg = df[y1_col].values
    dtg = df[y2_col].values
    
    # Prepare series data
    # Y1: Weight (mg)
    series_y1_data = [[float(t), float(w)] for t, w in zip(temp, weight_mg)]
    
    # Y2: DTG multiplied by -1 so peaks point up
    dtg_neg = -dtg
    series_y2_data = [[float(t), float(d)] for t, d in zip(temp, dtg_neg)]
    
    # Calculate Residue at nearest 800°C
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = float(weight_mg[idx_800])
    residue_temp = float(temp[idx_800])
    
    # Calculate Peak: highest DTG peak above 150°C
    mask = temp > 150
    peak_temp = None
    peak_y2_val = None
    
    if np.any(mask):
        dtg_subset = dtg[mask]
        temp_subset = temp[mask]
        # Find index of minimum DTG (most negative) -> largest mass loss rate
        min_dtg_idx_local = np.argmin(dtg_subset)
        peak_temp = float(temp_subset[min_dtg_idx_local])
        
        # Get corresponding DTG value (inverted) for marker position
        # We need to find the exact index in the original array to get the correct inverted value
        # Or just use the value from the subset logic carefully.
        # Since we plotted -dtg, the peak is max(-dtg).
        # Let's find the index in the full array where temp == peak_temp
        peak_indices = np.where(temp == peak_temp)[0]
        if len(peak_indices) > 0:
            peak_idx = peak_indices[0]
            peak_y2_val = float(-dtg[peak_idx])

    # Construct ECharts Option
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": "2%",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (mg)", "DTG"],
            "orient": "vertical",
            "left": "5%",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "10%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "axisLine": {"onZero": False},
            "splitLine": {"show": False}
        },
        "yAxis": [
            {
                "name": "Weight (mg)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "left",
                "axisLine": {"onZero": False},
                "splitLine": {"show": True, "lineStyle": {"type": "dashed"}}
            },
            {
                "name": "DTG (-%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "type": "value",
                "position": "right",
                "axisLine": {"onZero": False},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (mg)",
                "type": "line",
                "yAxisIndex": 0,
                "data": series_y1_data,
                "itemStyle": {"color": "#1f4e9c"},
                "lineStyle": {"width": 2},
                "showSymbol": False,
                "markPoint": {
                    "data": []
                }
            },
            {
                "name": "DTG",
                "type": "line",
                "yAxisIndex": 1,
                "data": series_y2_data,
                "itemStyle": {"color": "#c0392b"},
                "lineStyle": {"width": 2},
                "showSymbol": False,
                "markPoint": {
                    "data": []
                }
            }
        ]
    }
    
    # Add residue marker
    option["series"][0]["markPoint"]["data"].append({
        "coord": [residue_temp, residue_val],
        "value": f"Residue {residue_val:.1f} mg",
        "symbolSize": 8,
        "label": {
            "formatter": "{b}",
            "position": "top",
            "distance": 10,
            "backgroundColor": "rgba(255,255,255,0.7)",
            "padding": [2, 4],
            "borderRadius": 2,
            "fontSize": 10,
            "fontFamily": "Noto Sans CJK KR"
        }
    })

    # Add peak marker if found
    if peak_temp is not None and peak_y2_val is not None:
        option["series"][1]["markPoint"]["data"].append({
            "coord": [peak_temp, peak_y2_val],
            "value": f"{peak_temp:.0f} °C",
            "symbolSize": 8,
            "label": {
                "formatter": "{b}",
                "position": "top",
                "distance": 10,
                "backgroundColor": "rgba(255,255,255,0.7)",
                "padding": [2, 4],
                "borderRadius": 2,
                "fontSize": 10,
                "fontFamily": "Noto Sans CJK KR"
            }
        })

    result = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main()
```
