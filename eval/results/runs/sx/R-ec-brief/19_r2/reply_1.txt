def get_nearest_value(x_vals, y_vals, target_x):
    if not len(x_vals):
        return None, None
    idx = np.argmin(np.abs(np.array(x_vals) - target_x))
    return x_vals[idx], y_vals[idx]

def find_peak_temp(x_vals, dtg_vals):
    # Peak is highest DTG (most negative original value -> largest positive after negation? 
    # Wait, user said "multiplied by -1 so its peaks point up". 
    # Original DTG is negative while mass is lost. So -DTG is positive.
    # We want the temperature of the highest peak in the *plotted* curve (-DTG).
    # This corresponds to the most negative value in the original DTG column.
    
    # Filter for T > 150
    mask = np.array(x_vals) > 150
    if not np.any(mask):
        return None, None
    
    filtered_x = np.array(x_vals)[mask]
    filtered_dtg_original = np.array(dtg_vals)[mask]
    
    # Find index of minimum original DTG (which becomes maximum plotted DTG)
    min_idx = np.argmin(filtered_dtg_original)
    peak_x = filtered_x[min_idx]
    peak_y_plotted = -filtered_dtg_original[min_idx]
    
    return peak_x, peak_y_plotted
