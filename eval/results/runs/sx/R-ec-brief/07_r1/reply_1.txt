```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/CNT/Plot/tests/inputs/TGA.xlsx"

def find_nearest_value(temp_arr, weight_arr, target_temp):
    """Find the weight value at the temperature nearest to target_temp."""
    idx = np.argmin(np.abs(temp_arr - target_temp))
    return temp_arr[idx], weight_arr[idx]

def find_dtg_peak(temp_arr, dtg_arr, min_temp=150.0):
    """Find the highest DTG peak (most negative original value, so max of -dtg) above min_temp."""
    mask = temp_arr > min_temp
    if not np.any(mask):
        return None, None
    
    t_masked = temp_arr[mask]
    dtg_masked = dtg_arr[mask]
    
    # We want the peak where mass loss is greatest -> most negative dtg -> max(-dtg)
    # Since we plot -dtg, the peak in the plotted curve corresponds to max(-dtg).
    # So we find index where -dtg is maximum.
    neg_dtg = -dtg_masked
    idx_max = np.argmax(neg_dtg)
    
    peak_temp = t_masked[idx_max]
    peak_val_neg = neg_dtg[idx_max] # This is the height on the y2 axis (which is -dtg)
    
    return peak_temp, peak_val_neg

# Read Excel file
xl = pd.ExcelFile(DATA_FILE)
sheets = xl.sheet_names

figures = []

for sheet_name in sheets:
    df = pd.read_excel(xl, sheet_name=sheet_name)
    
    # Identify columns based on standard TGA structure and preview info
    # Assuming B=Temperature, C=Weight(%), D=D.W(%/C)
    # Let's verify column names loosely or use positions if names are ambiguous
    cols = df.columns.tolist()
    
    # Heuristic to find columns
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for c in cols:
        c_lower = str(c).lower()
        if 'temp' in c_lower:
            temp_col = c
        elif 'weight' in c_lower or 'wt' in c_lower:
            weight_col = c
        elif 'd.w' in c_lower or 'derivative' in c_lower or 'dtg' in c_lower:
            dtg_col = c
            
    # Fallback to indices if heuristics fail based on preview
    if temp_col is None and len(cols) >= 2: temp_col = cols[1]
    if weight_col is None and len(cols) >= 3: weight_col = cols[2]
    if dtg_col is None and len(cols) >= 4: dtg_col = cols[3]
    
    if temp_col is None or weight_col is None or dtg_col is None:
        continue
        
    # Extract data
    temps = df[temp_col].values
    weights = df[weight_col].values
    dtgs = df[dtg_col].values
    
    # Clean NaNs
    valid_mask = ~(np.isnan(temps) | np.isnan(weights) | np.isnan(dtgs))
    temps = temps[valid_mask]
    weights = weights[valid_mask]
    dtgs = dtgs[valid_mask]
    
    if len(temps) == 0:
        continue
        
    # Calculate Residue at ~800C
    res_temp, res_weight = find_nearest_value(temps, weights, 800.0)
    
    # Calculate DTG Peak above 150C
    peak_temp, peak_height_neg = find_dtg_peak(temps, dtgs, min_temp=150.0)
    
    # Prepare series data
    # Series 1: Weight (%) - Left Y Axis
    series_weight_data = [[float(t), float(w)] for t, w in zip(temps, weights)]
    
    # Series 2: DTG (%/C) multiplied by -1 - Right Y Axis
    # Note: The prompt says "multiplied by -1 so its peaks point up"
    series_dtg_data = [[float(t), float(-d)] for t, d in zip(temps, dtgs)]
    
    # Determine Axis Ranges dynamically to look nice
    # X Axis
    x_min = min(temps.min(), 30)
    x_max = max(temps.max(), 900)
    
    # Y1 (Weight)
    y1_min = 0
    y1_max = 100
    
    # Y2 (DTG * -1)
    # Find range of -dtg
    neg_dtgs = -dtgs
    y2_min = min(neg_dtgs.min(), 0)
    y2_max = max(neg_dtgs.max(), 0)
    
    # Add some padding to Y2
    y2_range = y2_max - y2_min
    if y2_range == 0: y2_range = 1
    y2_min -= y2_range * 0.1
    y2_max += y2_range * 0.1
    
    # Build Markers for Annotations
    markers = []
    
    # Residue Marker
    # Only add if within chart bounds roughly
    if res_temp <= x_max and res_weight <= y1_max:
        markers.append({
            "coord": [res_temp, res_weight],
            "symbol": "circle",
            "symbolSize": 8,
            "itemStyle": {
                "color": "#1f4e9c"
            },
            "label": {
                "show": True,
                "formatter": f"Residue {res_weight:.1f}%",
                "position": "right",
                "fontSize": 10,
                "color": "#1f4e9c"
            }
        })
        
    # Peak Marker
    if peak_temp is not None and peak_height_neg is not None:
        if peak_temp <= x_max and peak_height_neg <= y2_max:
             markers.append({
                "coord": [peak_temp, peak_height_neg],
                "symbol": "triangle",
                "symbolSize": 8,
                "itemStyle": {
                    "color": "#c0392b"
                },
                "label": {
                    "show": True,
                    "formatter": f"{int(peak_temp)} °C",
                    "position": "top",
                    "fontSize": 10,
                    "color": "#c0392b"
                }
            })

    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 14
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            },
            "formatter": "{a0}: {c0}<br/>{a1}: {c1}"
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "top": 20,
            "right": 20,
            "textStyle": {
                "fontSize": 9,
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "8%",
            "right": "8%",
            "bottom": "10%",
            "top": "15%",
            "containLabel": true
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "min": x_min,
            "max": x_max,
            "splitLine": {
                "show": false
            },
            "axisLine": {
                "lineStyle": {
                    "color": "#666"
                }
            },
            "axisTick": {
                "alignWithLabel": true
            },
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 10
            }
        },
        "yAxis": [
            {
                "name": "Weight (%)",
                "type": "value",
                "min": y1_min,
                "max": y1_max,
                "position": "left",
                "axisLine": {
                    "lineStyle": {
                        "color": "#1f4e9c"
                    }
                },
                "axisTick": {
                    "show": true
                },
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 10,
                    "color": "#1f4e9c"
                },
                "splitLine": {
                    "show": false
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "value",
                "min": y2_min,
                "max": y2_max,
                "position": "right",
                "axisLine": {
                    "lineStyle": {
                        "color": "#c0392b"
                    }
                },
                "axisTick": {
                    "show": true
                },
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 10,
                    "color": "#c0392b"
                },
                "splitLine": {
                    "show": false
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "data": series_weight_data,
                "smooth": false,
                "symbol": "none",
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 1.5
                },
                "markPoint": {
                    "data": markers
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": series_dtg_data,
                "smooth": false,
                "symbol": "none",
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 1.5
                }
            }
        ]
    }
    
    figures.append({
        "name": f"plot_{sheet_name.replace(' ', '_')}",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    })

output = {"figures": figures}
print(json.dumps(output, ensure_ascii=False))
```
