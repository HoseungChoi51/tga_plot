```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def main():
    # Read the specific sheet
    df = pd.read_excel(DATA_FILE, sheet_name="1006R1-V acid")
    
    # Identify columns based on preview: B=Temperature, C=Weight(%), D=D.W(%/C)
    temp_col = "Temperature"
    weight_col = "Weight(%)"
    dtg_col = "D.W(%/C)"
    
    # Extract data
    temps = df[temp_col].values
    weights = df[weight_col].values
    dtgs = df[dtg_col].values
    
    # Filter for x-axis range 100~800
    mask = (temps >= 100) & (temps <= 800)
    t_filtered = temps[mask]
    w_filtered = weights[mask]
    dtg_filtered = dtgs[mask]
    
    # Prepare series data
    # Weight line (y1)
    weight_data = [[float(t), float(w)] for t, w in zip(t_filtered, w_filtered)]
    
    # DTG line (y2), multiplied by -1 so peaks point up
    dtg_data = [[float(t), float(-d)] for t, d in zip(t_filtered, dtg_filtered)]
    
    # Find residue at nearest 800°C
    # Since we filtered up to 800, the last point is likely near 800. 
    # We find the index closest to 800 in the original filtered set or just take the value at max temp if it's close.
    # The prompt says "nearest 800 °C".
    idx_800 = np.argmin(np.abs(temps - 800))
    residue_val = weights[idx_800]
    residue_temp = temps[idx_800]
    
    # Find DTG peaks above 150°C
    # We need the top 2 peaks. A peak in DTG (negative values becoming positive after inversion) 
    # corresponds to a local maximum in the inverted DTG signal.
    inv_dtg = -dtgs
    valid_mask_peaks = temps > 150
    
    # Simple peak detection: find local maxima in inv_dtg where temp > 150
    peaks = []
    for i in range(1, len(inv_dtg)-1):
        if not valid_mask_peaks[i]:
            continue
        if inv_dtg[i] > inv_dtg[i-1] and inv_dtg[i] > inv_dtg[i+1]:
            # Check if it's a significant peak (above some threshold to avoid noise)
            if inv_dtg[i] > 0.01: 
                peaks.append((i, inv_dtg[i], temps[i]))
    
    # Sort peaks by height (descending) and take top 2
    peaks.sort(key=lambda x: x[1], reverse=True)
    top_2_peaks = peaks[:2]
    
    # Prepare mark points for ECharts
    mark_points_weight = [
        {
            "coord": [float(residue_temp), float(residue_val)],
            "value": f"Residue {residue_val:.1f}%",
            "symbolSize": 8,
            "itemStyle": {"color": "#1f4e9c"},
            "label": {
                "show": True,
                "formatter": "{b}",
                "position": "top",
                "fontSize": 12,
                "color": "#333"
            }
        }
    ]
    
    mark_points_dtg = []
    for _, height, temp in top_2_peaks:
        mark_points_dtg.append({
            "coord": [float(temp), float(height)], # Height is already inverted (-DTG)
            "value": f"{temp:.0f} °C",
            "symbolSize": 8,
            "itemStyle": {"color": "#c0392b"},
            "label": {
                "show": True,
                "formatter": "{b}",
                "position": "top",
                "fontSize": 12,
                "color": "#333"
            }
        })

    option = {
        "title": {
            "text": "1006R1-V acid",
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 16,
                "fontWeight": "bold"
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "left": "left",
            "top": "middle",
            "orient": "vertical",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "type": "value",
            "min": 100,
            "max": 800,
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "type": "dashed",
                    "color": "#ccc"
                }
            }
        },
        "yAxis": [
            {
                "name": "Weight (%)",
                "type": "value",
                "position": "left",
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR"
                },
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "type": "dashed",
                        "color": "#eee"
                    }
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "value",
                "position": "right",
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR"
                },
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "yAxisIndex": 0,
                "data": weight_data,
                "smooth": True,
                "showSymbol": False,
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "markPoint": {
                    "data": mark_points_weight
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": dtg_data,
                "smooth": True,
                "showSymbol": False,
                "lineStyle": {
                    "color": "#c0392b",
                    "width": 2
                },
                "markPoint": {
                    "data": mark_points_dtg
                }
            }
        ]
    }
    
    result = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(result))

if __name__ == "__main__":
    main()
```
