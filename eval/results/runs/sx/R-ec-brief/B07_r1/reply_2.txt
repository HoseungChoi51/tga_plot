import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

# Read the second sheet only (index 1)
df = pd.read_excel(DATA_FILE, sheet_name=1)

sheet_name = "PVC"  # From the preview data

# Extract columns based on TGA guess
x_col = "Temperature (°C)"
y1_col = "Weight (%)"
y2_col = "Deriv. Weight (%/°C)"

temp = df[x_col].values.astype(float)
weight = df[y1_col].values.astype(float)
dtg = df[y2_col].values.astype(float)

# Prepare data for ECharts: [temperature, weight] and [temperature, -dtg]
weight_data = [[float(t), float(w)] for t, w in zip(temp, weight)]
dtg_data = [[float(t), float(-d)] for t, d in zip(temp, dtg)]

# Find weight at 300°C and 600°C
def find_weight_at_temp(target_temp, temps, weights):
    idx = int(np.argmin(np.abs(temps - target_temp)))
    return float(weights[idx]), float(temps[idx])

w_300, t_300_actual = find_weight_at_temp(300, temp, weight)
w_600, t_600_actual = find_weight_at_temp(600, temp, weight)

# Axis ranges
x_min = float(min(temp))
x_max = float(max(temp))
y1_min = float(min(weight) * 0.95)
y1_max = float(max(weight) * 1.05)
y2_min = float(min(-dtg) * 1.1) if min(-dtg) > 0 else float(min(-dtg) * 0.9)
y2_max = float(max(-dtg) * 1.1) if max(-dtg) > 0 else float(max(-dtg) * 0.9)

        {
