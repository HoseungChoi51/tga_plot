import pandas as pd

def find_onset_temperature(temp, weight):
    """
    Find the onset temperature of decomposition.
    Method: 
    1. Calculate DTG (derivative of weight with respect to temperature).
    2. Find the maximum rate of weight loss (peak DTG magnitude) in the main decomposition region.
    3. Draw a tangent line at this peak point.
    4. Extrapolate this tangent line to intersect with the baseline weight (initial stable weight).
    The intersection temperature is the onset temperature.
    """
    # Sort by temperature just in case
    df = pd.DataFrame({'T': temp, 'W': weight}).sort_values('T').reset_index(drop=True)
    
    # Calculate DTG using central difference approximation for smoother results
    dtg = np.gradient(df['W'].values, df['T'].values)
    dtg_abs = -dtg  # Make positive
    
    # Identify the main decomposition peak. 
    # We look for the global minimum of weight change (max absolute DTG) after some initial noise removal.
    # Typically ignore the first few degrees if there's moisture loss, but here we look for the major event.
    # Let's assume the main event is the largest negative slope.
    
    # Find index of max DTG (most negative weight change)
    # Exclude very beginning if needed, but usually global max abs derivative works for single step or dominant step.
    idx_peak = np.argmax(dtg_abs)
    
    # To be more robust, let's ensure we are in the active decomposition zone.
    # If the peak is at T < 100, it might be moisture. But let's stick to the standard tangent method on the steepest part.
    
    # Get local points around the peak to fit a tangent line
    # Use a window of +/- 5-10 points depending on data density
    n_points = len(df)
    window = 10
    start_idx = max(0, idx_peak - window)
    end_idx = min(n_points, idx_peak + window + 1)
    
    t_local = df['T'].values[start_idx:end_idx]
    w_local = df['W'].values[start_idx:end_idx]
    
    # Fit a linear regression to the local segment to get the slope and intercept of the tangent
    # y = mx + c
    m, c = np.polyfit(t_local, w_local, 1)
    
    # Determine the baseline weight (initial weight before decomposition starts)
    # Usually the average weight in the first 10% of the temperature range where weight is stable
    initial_temp_limit = df['T'].min() + (df['T'].max() - df['T'].min()) * 0.1
    mask_initial = df['T'] <= initial_temp_limit
    if not mask_initial.any():
        baseline_weight = df['W'].iloc[0]
    else:
        baseline_weight = df.loc[mask_initial, 'W'].mean()
        
    # Calculate onset temperature: intersection of tangent line and baseline weight
    # W_baseline = m * T_onset + c
    # T_onset = (W_baseline - c) / m
    if m == 0:
        return None
        
    t_onset = (baseline_weight - c) / m
    
    # Sanity check: Onset should be lower than the peak temperature
    if t_onset > df['T'].iloc[idx_peak]:
        # Fallback or adjustment if the geometry is weird (e.g., multi-step)
        # For simple cases, this shouldn't happen.
        pass
        
    return float(t_onset)
