```python
import pandas as pd
import json
import numpy as np

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"

# Read the specified sheet "Raw"
df = pd.read_excel(DATA_FILE, sheet_name="Raw")

# Identify columns based on preview info
x_col = "Temperature (°C)"
y1_col = "Weight (%)"
y2_col = "Deriv. Weight (%/°C)"

# Extract data
temp = df[x_col].values
weight = df[y1_col].values
dtg = df[y2_col].values

# Flip x-axis: high temperature on the left means we reverse the order of points
# ECharts handles axis direction via 'inverse': True on xAxis, but for line series
# with explicit data pairs [x, y], reversing the array ensures correct drawing order if needed,
# though usually inverse on axis is enough. However, to be safe and consistent with "flipping",
# we will use 'inverse': true on the xAxis option.
# The prompt asks to flip the x-axis direction so that high temp is on the left.
# In ECharts, setting xAxis.inverse = true achieves this visually.

# Prepare data for ECharts series
# Series 1: Weight (%) - Left Y Axis (y1)
series_weight_data = [[t, w] for t, w in zip(temp, weight)]

# Series 2: DTG (%/°C) - Right Y Axis (y2)
# Multiply by -1 so peaks point up as per standard TGA plot convention in instructions
dtg_inverted = [-d for d in dtg]
series_dtg_data = [[t, d_inv] for t, d_inv in zip(temp, dtg_inverted)]

# Calculate Residue at nearest 800 °C
idx_800 = np.argmin(np.abs(temp - 800))
residue_val = weight[idx_800]
residue_temp = temp[idx_800]
residue_label_text = f"Residue {residue_val:.1f}%"

# Calculate Peak temperature of highest DTG peak above 150 °C
# Filter for temp > 150
mask = temp > 150
if np.any(mask):
    # Find index of maximum absolute value in DTG (since it's negative, max abs is min value)
    # But we inverted DTG for plotting, so we look for max in dtg_inverted within mask
    dtg_filtered = np.array(dtg_inverted)[mask]
    temp_filtered = np.array(temp)[mask]
    
    idx_max_peak_local = np.argmax(dtg_filtered)
    peak_temp = temp_filtered[idx_max_peak_local]
    peak_val_inv = dtg_filtered[idx_max_peak_local]
    peak_label_text = f"{peak_temp:.0f} °C"
else:
    peak_temp = None
    peak_label_text = ""

# Define colors
color_weight = "#1f4e9c"
color_dtg = "#c0392b"

# Build ECharts Option
option = {
    "title": {
        "text": "Raw",
        "left": "center",
        "top": "2%",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 16,
            "fontWeight": "bold"
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        },
        "formatter": function_placeholder = None # Not allowed in JSON output directly if strict, but ECharts accepts string templates or null. 
        # We'll stick to default tooltip behavior which works well with axis trigger.
    },
    "legend": {
        "data": ["Weight (%)", "-DTG (%/°C)"],
        "orient": "vertical",
        "left": "10%",
        "top": "middle",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "type": "value",
        "inverse": True, # High temp on left
        "position": "bottom",
        "axisLine": {"lineStyle": {"color": "#333"}},
        "axisTick": {"show": True},
        "splitLine": {"show": False},
        "nameTextStyle": {
            "fontFamily": "Noto Sans CJK KR"
        }
    },
    "yAxis": [
        {
            "name": "Weight (%)",
            "type": "value",
            "position": "left",
            "offset": 0,
            "axisLine": {"lineStyle": {"color": color_weight}},
            "axisTick": {"show": True},
            "splitLine": {"show": True, "lineStyle": {"color": "#eee"}},
            "nameTextStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "color": color_weight
            }
        },
        {
            "name": "-DTG (%/°C)",
            "type": "value",
            "position": "right",
            "offset": 0,
            "axisLine": {"lineStyle": {"color": color_dtg}},
            "axisTick": {"show": True},
            "splitLine": {"show": False},
            "nameTextStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "color": color_dtg
            }
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "yAxisIndex": 0,
            "data": series_weight_data,
            "smooth": False,
            "symbol": "none",
            "lineStyle": {
                "color": color_weight,
                "width": 2
            },
            "markPoint": {
                "data": [
                    {
                        "coord": [residue_temp, residue_val],
                        "label": {
                            "show": True,
                            "formatter": residue_label_text,
                            "position": "top",
                            "color": color_weight,
                            "fontSize": 12,
                            "fontFamily": "Noto Sans CJK KR"
                        },
                        "itemStyle": {
                            "color": color_weight
                        },
                        "symbolSize": 8
                    }
                ]
            }
        },
        {
            "name": "-DTG (%/°C)",
            "type": "line",
            "yAxisIndex": 1,
            "data": series_dtg_data,
            "smooth": False,
            "symbol": "none",
            "lineStyle": {
                "color": color_dtg,
                "width": 2
            },
            "markPoint": {
                "data": []
            }
        }
    ]
}

# Add peak marker to DTG series if found
if peak_temp is not None:
    # Find the corresponding y-value (inverted DTG) for the peak temp
    # Since there might be multiple points near the peak, find the one closest to peak_temp
    idx_peak = np.argmin(np.abs(temp - peak_temp))
    peak_y_val = dtg_inverted[idx_peak]
    
    option["series"][1]["markPoint"]["data"].append({
        "coord": [peak_temp, peak_y_val],
        "label": {
            "show": True,
            "formatter": peak_label_text,
            "position": "top",
            "color": color_dtg,
            "fontSize": 12,
            "fontFamily": "Noto Sans CJK KR"
        },
        "itemStyle": {
            "color": color_dtg
        },
        "symbolSize": 8
    })

# Construct final JSON structure
result = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(result, ensure_ascii=False))
```
