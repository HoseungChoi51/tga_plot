import pandas as pd

def load_data(filepath):
    if filepath.endswith('.csv'):
        df = pd.read_csv(filepath)
    elif filepath.endswith('.txt'):
        # Assuming tab or whitespace separated, but standard txt often uses comma or tab. 
        # Given the preview structure looks like CSV, let's try sep=None with engine='python' to auto-detect, 
        # or assume comma if it looks like csv content. The preview says "columns" which implies a delimiter was found.
        # Let's try read_csv with sep=None for .txt as well, or check first line.
        try:
            df = pd.read_csv(filepath, sep=None, engine='python')
        except Exception:
            df = pd.read_csv(filepath)
    else:
        raise ValueError("Unsupported file format")
    
    # Identify columns based on names in preview
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        c_lower = str(col).lower()
        if 'temperature' in c_lower:
            temp_col = col
        elif 'weight (%)' in c_lower or ('weight' in c_lower and '%' in c_lower):
            weight_col = col
        elif 'deriv' in c_lower or 'dtg' in c_lower:
            dtg_col = col
            
    return df[temp_col], df[weight_col], df[dtg_col]

def process_tga(temp, weight, dtg):
    # Sort by temperature just in case
    order = np.argsort(temp)
    temp = np.array(temp)[order]
    weight = np.array(weight)[order]
    dtg = np.array(dtg)[order]
    
    # Residue at 800 C
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = weight[idx_800]
    residue_temp = temp[idx_800]
    
    # Peak DTG above 150 C
    mask_peak = temp > 150
    if not np.any(mask_peak):
        peak_temp = None
        peak_dtg = None
    else:
        # DTG is negative during mass loss, so we look for the most negative value (minimum)
        # Or absolute maximum? Standard TGA plots show DTG inverted (positive peaks).
        # The prompt says: "multiplied by -1 so its peaks point up".
        # So we find the minimum of original DTG (most negative) to get the highest peak after inversion.
        dtg_masked = dtg[mask_peak]
        temp_masked = temp[mask_peak]
        
        min_idx = np.argmin(dtg_masked)
        peak_temp = temp_masked[min_idx]
        peak_dtg_orig = dtg_masked[min_idx]
        
    return temp, weight, dtg, residue_temp, residue_val, peak_temp, peak_dtg_orig
