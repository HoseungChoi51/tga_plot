import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def find_nearest_value(df, col_x, target_x):
    """Find the row index and value where col_x is nearest to target_x"""
    if df.empty:
        return None, None
    idx = (df[col_x] - target_x).abs().idxmin()
    return idx, df.loc[idx, col_x]

def get_residue_label(df, x_col, y1_col):
    """Get residue weight at 800C"""
    idx, _ = find_nearest_value(df, x_col, 800.0)
    if idx is not None:
        val = float(df.loc[idx, y1_col])
        x_val = float(df.loc[idx, x_col])
        return f"Residue {val:.1f}%", x_val, val
    return None, None, None

def get_peak_label(df, x_col, dtg_col):
    """Get temperature of highest DTG peak above 150C"""
    # Filter for temp > 150
    mask = df[x_col] > 150
    filtered = df[mask].copy()
    if filtered.empty:
        return None, None, None
    
    # DTG is negative during mass loss, so we look for the most negative value (lowest)
    # which corresponds to the largest magnitude of mass loss rate.
    min_dtg_idx = filtered[dtg_col].idxmin()
    peak_temp = float(filtered.loc[min_dtg_idx, x_col])
    peak_val = float(filtered.loc[min_dtg_idx, dtg_col])
    
    return f"{peak_temp:.0f} °C", peak_temp, peak_val

def create_tga_plot(sheet_name, df, x_col, y1_col, y2_col):
    """Create ECharts option for a TGA plot"""
    
    # Extract data and convert to native Python types for JSON serialization
    x_data = [float(v) for v in df[x_col].tolist()]
    y1_data = [float(v) for v in df[y1_col].tolist()]
    y2_raw = [float(v) for v in df[y2_col].tolist()]
    # Multiply y2 by -1 so peaks point up
    y2_data = [-v for v in y2_raw]
    
    # Calculate labels
    res_label_text, res_x, res_y = get_residue_label(df, x_col, y1_col)
    peak_label_text, peak_x, peak_y = get_peak_label(df, x_col, y2_col)
    
    # Prepare mark points
    mark_points_weight = []
    mark_points_dtg = []
    
    if res_label_text:
        mark_points_weight.append({
            "name": "Residue",
            "coord": [res_x, res_y],
            "value": res_label_text,
            "symbolSize": 6,
            "label": {
                "show": True,
                "formatter": "{b}",
                "position": "top",
                "color": "#333",
                "fontSize": 10
            }
        })
    
    if peak_label_text:
        # For the peak marker on the DTG curve (which is inverted), 
        # we need to place it on the inverted y2 axis.
        inv_peak_y = -peak_y
        mark_points_dtg.append({
            "name": "Peak",
            "coord": [peak_x, inv_peak_y],
            "value": peak_label_text,
            "symbolSize": 6,
            "label": {
                "show": True,
                "formatter": "{b}",
                "position": "top",
                "color": "#333",
                "fontSize": 10
            }
        })

    # Determine axis ranges dynamically or use reasonable defaults
    # X-axis: usually 0 to max temp + buffer
    x_min = 0
    x_max = int(np.ceil(max(x_data) / 100) * 100) if x_data else 1000
    
    # Y1 axis (Weight %): 0 to 100 or slightly more
    y1_min = 0
    y1_max = 110
    
    # Y2 axis (DTG inverted): Find min/max of inverted data
    if y2_data:
        y2_min = min(y2_data)
        y2_max = max(y2_data)
        # Add some padding
        pad = (y2_max - y2_min) * 0.1 if (y2_max - y2_min) != 0 else 0.1
        y2_axis_min = y2_min - pad
        y2_axis_max = y2_max + pad
    else:
        y2_axis_min = -1
        y2_axis_max = 1

    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 14
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "min": x_min,
            "max": x_max,
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "show": False
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 40,
                "min": y1_min,
                "max": y1_max,
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR"
                },
                "splitLine": {
                    "lineStyle": {
                        "type": "dashed"
                    }
                }
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 40,
                "min": y2_axis_min,
                "max": y2_axis_max,
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR",
                    "color": "green"  # User request: y2 axis label color green
                },
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "data": [[x, y] for x, y in zip(x_data, y1_data)],
                "smooth": False,
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "itemStyle": {
                    "color": "#1f4e9c"
                },
                "showSymbol": False,
                "markPoint": {
                    "data": mark_points_weight
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "data": [[x, y] for x, y in zip(x_data, y2_data)],
                "smooth": False,
                "lineStyle": {
                    "color": "green",  # User request: y2 line color green
                    "width": 2
                },
                "itemStyle": {
                    "color": "green"   # Ensure markers match if shown
                },
                "showSymbol": False,
                "markPoint": {
                    "data": mark_points_dtg
                }
            }
        ]
    }
    
    return option

def main():
    figures = []
    
    try:
        xl = pd.ExcelFile(DATA_FILE)
        sheet_names = xl.sheet_names
        
        for sheet in sheet_names:
            df = pd.read_excel(DATA_FILE, sheet_name=sheet, header=0)
            
            # Identify columns based on standard TGA structure from preview
            cols = df.columns.tolist()
            
            x_col = None
            y1_col = None
            y2_col = None
            
            for c in cols:
                cl = str(c).lower()
                if 'temp' in cl:
                    x_col = c
                elif 'weight' in cl and '%' in cl and 'd.w' not in cl and 'derivative' not in cl:
                    y1_col = c
                elif 'd.w' in cl or 'derivative' in cl or ('dtg' in cl):
                    y2_col = c
            
            # Fallback if specific names not found exactly
            if x_col is None and len(cols) >= 2: x_col = cols[1] # Usually Temp is 2nd col
            if y1_col is None and len(cols) >= 3: y1_col = cols[2] # Weight is 3rd
            if y2_col is None and len(cols) >= 4: y2_col = cols[3] # DTG is 4th
            
            if x_col and y1_col and y2_col:
                # Clean data: drop rows with NaN in critical columns
                df_clean = df.dropna(subset=[x_col, y1_col, y2_col])
                
                if not df_clean.empty:
                    option = create_tga_plot(sheet, df_clean, x_col, y1_col, y2_col)
                    
                    fig_entry = {
                        "name": f"plot_{sheet.replace(' ', '_').replace('/', '_')}",
                        "option": option,
                        "width": 614,
                        "height": 442,
                        "dpi": 300,
                        "formats": ["png"]
                    }
                    figures.append(fig_entry)
                    
    except Exception as e:
        # If error occurs, print empty figures or handle gracefully
        pass

    output = {"figures": figures}
    print(json.dumps(output))

if __name__ == "__main__":
    main()
