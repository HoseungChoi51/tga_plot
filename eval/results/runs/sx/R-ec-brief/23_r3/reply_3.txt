import pandas as pd
import numpy as np
import json
import chardet

def load_data(file_path):
    # Detect encoding first to handle UTF-16/UTF-8 with BOM etc.
    with open(file_path, 'rb') as f:
        raw_data = f.read()
    
    result = chardet.detect(raw_data)
    encoding = result['encoding'] or 'utf-8'
    
    df = None
    
    # Try reading with detected encoding and whitespace separator
    try:
        df = pd.read_csv(file_path, sep=r'\s+', header=None, skiprows=10, engine='python', encoding=encoding)
    except Exception:
        pass
        
    if df is None:
        # Fallback to utf-8-sig which handles BOM
        try:
            df = pd.read_csv(file_path, sep=r'\s+', header=None, skiprows=10, engine='python', encoding='utf-8-sig')
        except Exception:
            pass
            
    if df is None:
        # Fallback to latin-1
        try:
            df = pd.read_csv(file_path, sep=r'\s+', header=None, skiprows=10, engine='python', encoding='latin-1')
        except Exception:
            pass

    if df is None:
        raise ValueError("Could not read file with standard methods")

    # Clean columns: remove empty rows if any caused by trailing newlines or bad parsing
    df = df.dropna(how='all')
    
    # Identify columns based on TGA guess from preview:
    # Col B (index 1) -> Temperature
    # Col C (index 2) -> Weight %
    # Col D (index 3) -> DTG
    
    # Ensure we have at least 4 columns
    if df.shape[1] < 4:
         # Fallback if structure is different, but preview says A,B,C,D
         raise ValueError(f"Expected at least 4 columns, got {df.shape[1]}")

    temp_col = 1
    weight_col = 2
    dtg_col = 3
    
    # Convert to numeric, forcing errors to NaN then dropping them
    temps = pd.to_numeric(df[temp_col], errors='coerce').values.astype(float)
    weights = pd.to_numeric(df[weight_col], errors='coerce').values.astype(float)
    dtgs = pd.to_numeric(df[dtg_col], errors='coerce').values.astype(float)
    
    # Filter out rows where conversion failed (NaNs)
    mask = ~(np.isnan(temps) | np.isnan(weights) | np.isnan(dtgs))
    temps = temps[mask]
    weights = weights[mask]
    dtgs = dtgs[mask]
    
    return temps, weights, dtgs

def find_residue(temps, weights, target_temp=800.0):
    """Find the weight at the temperature nearest to target_temp."""
    if len(temps) == 0:
        return None, None
        
    idx = np.argmin(np.abs(temps - target_temp))
    res_weight = weights[idx]
    res_temp = temps[idx]
    return res_temp, res_weight

def find_peak(temps, dtgs, min_temp=150.0):
    """Find the temperature of the highest DTG peak above min_temp.
    Note: DTG is negative during mass loss. We look for the most negative value 
    (which corresponds to the maximum rate of loss), but the prompt says 
    'multiplied by -1 so its peaks point up'. So we find the max of (-DTG) 
    which is equivalent to finding the min of DTG.
    """
    if len(temps) == 0:
        return None, None
        
    mask = temps > min_temp
    if not np.any(mask):
        return None, None
        
    filtered_temps = temps[mask]
    filtered_dtgs = dtgs[mask]
    
    # Peak is where -DTG is maximum, i.e., DTG is minimum
    idx_min = np.argmin(filtered_dtgs)
    peak_temp = filtered_temps[idx_min]
    peak_val = filtered_dtgs[idx_min]
    
    return peak_temp, peak_val
