import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def make_tga_plot(df, sheet_name):
    """Create a TGA plot option for ECharts."""
    # Identify columns
    x_col = None
    y1_col = None
    y2_col = None
    
    for col in df.columns:
        if 'temperature' in col.lower():
            x_col = col
        elif 'weight' in col.lower() and '%' in col:
            y1_col = col
        elif 'd.w' in col.lower() or 'dtg' in col.lower():
            y2_col = col
    
    if x_col is None:
        x_col = df.columns[1]  # Assume second column is temperature
    if y1_col is None:
        y1_col = df.columns[2]  # Assume third column is weight
    if y2_col is None:
        y2_col = df.columns[3]  # Assume fourth column is DTG
    
    temp = df[x_col].values
    weight = df[y1_col].values
    dtg = df[y2_col].values
    
    # Multiply DTG by -1 so peaks point up
    dtg_neg = [-v for v in dtg]
    
    # Prepare data points
    weight_data = [[float(t), float(w)] for t, w in zip(temp, weight)]
    dtg_data = [[float(t), float(d)] for t, d in zip(temp, dtg_neg)]
    
    # Find residue at ~800°C
    mask_800 = (temp >= 795) & (temp <= 805)
    if mask_800.any():
        idx_800 = np.where(mask_800)[0][0]
        residue_temp = float(temp[idx_800])
        residue_weight = float(weight[idx_800])
    else:
        # Find nearest to 800
        idx_800 = np.argmin(np.abs(temp - 800))
        residue_temp = float(temp[idx_800])
        residue_weight = float(weight[idx_800])
    
    # Find highest DTG peak above 150°C
    mask_above_150 = temp > 150
    if mask_above_150.any():
        dtg_filtered = dtg[mask_above_150]
        temp_filtered = temp[mask_above_150]
        # Highest peak means most negative original DTG, which is largest positive after negation
        max_dtg_idx_local = np.argmax(dtg_filtered)
        peak_temp = float(temp_filtered[max_dtg_idx_local])
        peak_dtg_val = float(dtg_neg[mask_above_150][max_dtg_idx_local])
    else:
        peak_temp = None
        peak_dtg_val = None
    
    # Build markPoint data for residue and peak
    mark_point_data = []
    mark_point_data.append({
        "name": f"Residue {residue_weight:.1f}%",
        "coord": [residue_temp, residue_weight],
        "value": f"Residue {residue_weight:.1f}%"
    })
    
    if peak_temp is not None:
        mark_point_data.append({
            "name": f"{peak_temp:.0f} °C",
            "coord": [peak_temp, peak_dtg_val],
            "value": f"{peak_temp:.0f} °C"
        })
    
    option = {
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": 10,
            "textStyle": {
                "fontSize": 16,
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "DTG (%/°C)"],
            "orient": "vertical",
            "left": "left",
            "top": "middle",
            "textStyle": {
                "fontFamily": "Noto Sans CJK KR"
            }
        },
        "grid": {
            "left": "15%",
            "right": "15%",
            "bottom": "15%",
            "top": "15%"
        },
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 30,
            "nameTextStyle": {
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 12
            },
            "axisLabel": {
                "fontFamily": "Noto Sans CJK KR"
            },
            "splitLine": {
                "show": True,
                "lineStyle": {
                    "color": "#e0e0e0"
                }
            }
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 45,
                "nameTextStyle": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 12,
                    "color": "#1f4e9c"
                },
                "position": "left",
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR",
                    "color": "#1f4e9c"
                },
                "axisLine": {
                    "lineStyle": {
                        "color": "#1f4e9c"
                    }
                },
                "splitLine": {
                    "show": True,
                    "lineStyle": {
                        "color": "#e0e0e0"
                    }
                }
            },
            {
                "type": "value",
                "name": "DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 45,
                "nameTextStyle": {
                    "fontFamily": "Noto Sans CJK KR",
                    "fontSize": 12,
                    "color": "#2ecc71"
                },
                "position": "right",
                "axisLabel": {
                    "fontFamily": "Noto Sans CJK KR",
                    "color": "#2ecc71"
                },
                "axisLine": {
                    "lineStyle": {
                        "color": "#2ecc71"
                    }
                },
                "splitLine": {
                    "show": False
                }
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "data": weight_data,
                "yAxisIndex": 0,
                "lineStyle": {
                    "color": "#1f4e9c",
                    "width": 2
                },
                "itemStyle": {
                    "color": "#1f4e9c"
                },
                "symbol": "none",
                "markPoint": {
                    "data": mark_point_data,
                    "label": {
                        "show": True,
                        "formatter": "{b}",
                        "fontSize": 11,
                        "fontFamily": "Noto Sans CJK KR",
                        "position": "top"
                    },
                    "symbolSize": 8
                }
            },
            {
                "name": "DTG (%/°C)",
                "type": "line",
                "data": dtg_data,
                "yAxisIndex": 1,
                "lineStyle": {
                    "color": "#2ecc71",
                    "width": 2
                },
                "itemStyle": {
                    "color": "#2ecc71"
                },
                "symbol": "none"
            }
        ]
    }
    
    return option

figures = []

xls = pd.ExcelFile(DATA_FILE)
for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name)
    option = make_tga_plot(df, sheet_name)
    figures.append({
        "name": f"plot_{sheet_name.replace(' ', '_')}",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    })

output = {"figures": figures}
print(json.dumps(output, ensure_ascii=False))
