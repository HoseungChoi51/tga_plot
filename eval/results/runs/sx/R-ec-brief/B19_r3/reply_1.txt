```python
import pandas as pd
import numpy as np
import json

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

# Read the PVC sheet
df_pvc = pd.read_excel(DATA_FILE, sheet_name="PVC")

# Extract relevant columns
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

# Filter data between 200 and 400 °C for mass loss calculation
mask_200_400 = (df_pvc[temp_col] >= 200) & (df_pvc[temp_col] <= 400)
df_range = df_pvc[mask_200_400].copy()

# Find weight at ~200°C and ~400°C using nearest points
idx_200 = (df_pvc[temp_col] - 200).abs().idxmin()
idx_400 = (df_pvc[temp_col] - 400).abs().idxmin()

w_200 = df_pvc.loc[idx_200, weight_col]
w_400 = df_pvc.loc[idx_400, weight_col]
mass_loss_pct = w_200 - w_400

# Full TGA plot for PVC with annotation
# Prepare data arrays
temps = df_pvc[temp_col].tolist()
weights = df_pvc[weight_col].tolist()
dtgs = [-v for v in df_pvc[dtg_col].tolist()]  # multiply by -1 so peaks point up

# Residue: weight nearest 800°C
idx_res = (df_pvc[temp_col] - 800).abs().idxmin()
residue_temp = df_pvc.loc[idx_res, temp_col]
residue_weight = df_pvc.loc[idx_res, weight_col]

# Peak: highest DTG peak above 150°C
mask_peak = df_pvc[temp_col] > 150
if mask_peak.any():
    dtg_vals = df_pvc.loc[mask_peak, dtg_col]
    idx_peak = dtg_vals.idxmin()  # most negative = largest peak
    peak_temp = df_pvc.loc[idx_peak, temp_col]
else:
    peak_temp = None

# Build ECharts option
option = {
    "title": {
        "text": "PVC",
        "left": "center",
        "top": 10,
        "textStyle": {"fontSize": 16, "fontFamily": "Noto Sans CJK KR"}
    },
    "tooltip": {"trigger": "axis"},
    "legend": {
        "data": ["Weight (%)", "DTG (-%/°C)"],
        "orient": "vertical",
        "left": 30,
        "top": "middle"
    },
    "grid": {
        "left": 70,
        "right": 70,
        "bottom": 50,
        "top": 60
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 30,
        "min": 30,
        "max": 900
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 40,
            "position": "left",
            "min": 0,
            "max": 110
        },
        {
            "type": "value",
            "name": "DTG (-%/°C)",
            "nameLocation": "middle",
            "nameGap": 40,
            "position": "right",
            "splitLine": {"show": False}
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "data": [[t, w] for t, w in zip(temps, weights)],
            "yAxisIndex": 0,
            "lineStyle": {"color": "#1f4e9c", "width": 1.5},
            "symbol": "none",
            "markPoint": {
                "data": [
                    {
                        "coord": [residue_temp, residue_weight],
                        "symbolSize": 8,
                        "itemStyle": {"color": "#1f4e9c"},
                        "label": {
                            "show": True,
                            "formatter": f"Residue {residue_weight:.1f}%",
                            "position": "right",
                            "fontSize": 11,
                            "fontFamily": "Noto Sans CJK KR",
                            "color": "#1f4e9c"
                        }
                    }
                ]
            },
            "markArea": {
                "silent": True,
                "itemStyle": {
                    "color": "rgba(255, 165, 0, 0.15)",
                    "borderColor": "rgba(255, 165, 0, 0.6)",
                    "borderWidth": 1
                },
                "label": {
                    "show": True,
                    "position": "insideTop",
                    "formatter": f"Mass Loss\n{mass_loss_pct:.2f}%",
                    "fontSize": 12,
                    "fontFamily": "Noto Sans CJK KR",
                    "color": "#d35400",
                    "fontWeight": "bold"
                },
                "data": [
                    [
                        {"xAxis": 200},
                        {"xAxis": 400}
                    ]
                ]
            }
        },
        {
            "name": "DTG (-%/°C)",
            "type": "line",
            "data": [[t, d] for t, d in zip(temps, dtgs)],
            "yAxisIndex": 1,
            "lineStyle": {"color": "#c0392b", "width": 1.5},
            "symbol": "none",
            "markPoint": {
                "data": []
            }
        }
    ]
}

# Add peak marker if exists
if peak_temp is not None:
    # Find the corresponding DTG value (already negated)
    idx_pk = (df_pvc[temp_col] - peak_temp).abs().idxmin()
    dtg_val_neg = -df_pvc.loc[idx_pk, dtg_col]
    option["series"][1]["markPoint"] = {
        "data": [
            {
                "coord": [peak_temp, dtg_val_neg],
                "symbolSize": 8,
                "itemStyle": {"color": "#c0392b"},
                "label": {
                    "show": True,
                    "formatter": f"{peak_temp:.0f} °C",
                    "position": "top",
                    "fontSize": 11,
                    "fontFamily": "Noto Sans CJK KR",
                    "color": "#c0392b"
                }
            }
        ]
    }

result = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(result, ensure_ascii=False))
```
