def get_nearest_point(temp_arr, weight_arr, target_temp):
    idx = np.argmin(np.abs(temp_arr - target_temp))
    return temp_arr[idx], weight_arr[idx]

def find_peak(temp_arr, dtg_arr, min_temp=150):
    mask = temp_arr > min_temp
    if not np.any(mask):
        return None, None
    # DTG is negative during mass loss. We want the most negative value (highest magnitude of mass loss rate).
    # The prompt says "multiplied by -1 so its peaks point up", implying we look for the minimum in original data or maximum in inverted data.
    # Let's work with the inverted DTG for peak finding logic to match visual "up" peaks.
    inv_dtg = -dtg_arr[mask]
    temps_masked = temp_arr[mask]
    
    max_idx = np.argmax(inv_dtg)
    peak_temp = temps_masked[max_idx]
    peak_val_inv = inv_dtg[max_idx]
    peak_val_orig = dtg_arr[mask][max_idx]
    
    return peak_temp, peak_val_orig

def main():

    # Read Excel file
    xls = pd.ExcelFile(DATA_FILE)
    sheet_names = xls.sheet_names
    
    for sheet_name in sheet_names:
        df = pd.read_excel(xls, sheet_name=sheet_name)
        # Drop rows where essential columns are NaN
        df = df.dropna(subset=[df.columns[1], df.columns[2], df.columns[3]])
        
        if len(df) == 0:
            continue
            
        option = create_tga_option(df, sheet_name)
        
        if "error" in option:
            continue
            
        fig_entry = {
            "name": sheet_name.replace(" ", "_"),
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
        figures.append(fig_entry)
        
    output = {"figures": figures}
    print(json.dumps(output, ensure_ascii=False))

if __name__ == "__main__":
