```python
import pandas as pd
import json

def main():
    file_path = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
    
    # Read the second sheet ("PVC")
    df = pd.read_excel(file_path, sheet_name="PVC")
    
    # Extract columns: Temperature (°C), Weight (%), Deriv. Weight (%/°C)
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    
    # Prepare data for TGA curve (Weight vs Temp)
    temps = df[temp_col].tolist()
    weights = df[weight_col].tolist()
    
    # Find weights at specific temperatures by interpolation or nearest neighbor
    def get_weight_at_temp(temp_target):
        # Find the index closest to the target temperature
        idx = (df[temp_col] - temp_target).abs().idxmin()
        return float(df.loc[idx, weight_col]), float(df.loc[idx, temp_col])

    w_300, t_300 = get_weight_at_temp(300)
    w_600, t_600 = get_weight_at_temp(600)
    
    # Create scatter points for the marked values
    mark_points_data = [
        {"coord": [t_300, w_300], "name": f"{w_300:.2f}%"},
        {"coord": [t_600, w_600], "name": f"{w_600:.2f}%"}
    ]
    
    option = {
        "title": {
            "text": "TGA 데이터: PVC",
            "subtext": f"300°C 무게: {w_300:.2f}%, 600°C 무게: {w_600:.2f}%",
            "left": "center"
        },
        "tooltip": {
            "trigger": "axis",
            "formatter": "{a}<br/>온도: {b} °C<br/>무게: {c} %"
        },
        "legend": {
            "data": ["무게 (%)"],
            "top": "10%"
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "type": "value",
            "name": "온도 (°C)",
            "min": 0,
            "max": 900
        },
        "yAxis": {
            "type": "value",
            "name": "무게 (%)",
            "min": 0,
            "max": 110
        },
        "series": [
            {
                "name": "무게 (%)",
                "type": "line",
                "data": [[float(t), float(w)] for t, w in zip(temps, weights)],
                "showSymbol": False,
                "smooth": True,
                "markPoint": {
                    "symbol": "circle",
                    "symbolSize": 10,
                    "itemStyle": {
                        "color": "#FF4500"
                    },
                    "label": {
                        "show": True,
                        "formatter": "{b}" 
                    },
                    "data": mark_points_data
                },
                "markLine": {
                    "silent": True,
                    "symbol": "none",
                    "lineStyle": {
                        "type": "dashed",
                        "color": "#888"
                    },
                    "data": [
                        {"xAxis": 300},
                        {"xAxis": 600}
                    ],
                    "label": {
                        "formatter": "{b}"
                    }
                }
            }
        ]
    }
    
    output = {
        "figures": [
            {
                "name": "pvc_tga",
                "option": option,
                "width": 800,
                "height": 600,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
