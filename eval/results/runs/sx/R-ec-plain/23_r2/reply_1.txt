# Read the data file
# Based on preview, data starts at row 11 (0-indexed would be row 10 in pandas if header_rows is none found but data_start_row is 11)
# The preview says "data_start_row": 11. Usually this means the 11th line of the file.
# Let's try to read it with skiprows=10 (since pandas is 0-indexed for lines) and no header.

try:
    df = pd.read_csv(DATA_FILE, sep=r'\s+', header=None, skiprows=10)
except Exception as e:
    # Try other separators if whitespace fails
    try:
        df = pd.read_csv(DATA_FILE, sep=',', header=None, skiprows=10)
    except:
        df = pd.read_csv(DATA_FILE, sep='\t', header=None, skiprows=10)

# Ensure we have enough columns
if df.shape[1] < 4:
    raise ValueError("Data does not have enough columns")

# Assign column names based on TGA guess: x=B, y1=C, y2=D
# Columns are A(0), B(1), C(2), D(3)
x_col = df.iloc[:, 1].values  # Temperature or Time? Guess says x is B. In TGA, usually Temp is X.
y1_col = df.iloc[:, 2].values # Mass %
y2_col = df.iloc[:, 3].values # Derivative (DTG)

# Clean data: remove NaNs if any
mask = ~np.isnan(x_col) & ~np.isnan(y1_col) & ~np.isnan(y2_col)
x_clean = x_col[mask]
y1_clean = y1_col[mask]
y2_clean = y2_col[mask]
