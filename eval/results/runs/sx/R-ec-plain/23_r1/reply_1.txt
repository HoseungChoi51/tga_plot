```python
import pandas as pd
import json

# Load the data file
file_path = "/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt"

# The preview indicates data starts at row 11 (0-indexed row 10) and there are no headers.
# Columns: A(0), B(Temperature?), C(Mass %), D(Derivative?)
# TGA guess suggests x=B, y1=C, y2=D.

df = pd.read_csv(file_path, sep='\t', header=None, skiprows=10)

# Assign column names based on context
df.columns = ['col_A', 'Temperature', 'Mass_Percent', 'Derivative']

# Extract data for plotting
x_data = df['Temperature'].tolist()
y_mass = df['Mass_Percent'].tolist()
y_deriv = df['Derivative'].tolist()

# Find the final residue (last mass percent value)
residue_value = y_mass[-1] if len(y_mass) > 0 else 0.0
residue_text = f"Residue: {residue_value:.2f}%"

# Construct ECharts option
option = {
    "title": {
        "text": "TGA Curve",
        "subtext": residue_text,
        "left": "center"
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "legend": {
        "data": ["Mass (%)", "Derivative (%/°C)"],
        "top": "bottom"
    },
    "grid": {
        "left": "3%",
        "right": "4%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "min": min(x_data),
        "max": max(x_data)
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Mass (%)",
            "position": "left",
            "min": 0,
            "max": 100
        },
        {
            "type": "value",
            "name": "Derivative (%/°C)",
            "position": "right"
        }
    ],
    "series": [
        {
            "name": "Mass (%)",
            "type": "line",
            "data": [[x, y] for x, y in zip(x_data, y_mass)],
            "smooth": False,
            "showSymbol": False,
            "yAxisIndex": 0
        },
        {
            "name": "Derivative (%/°C)",
            "type": "line",
            "data": [[x, y] for x, y in zip(x_data, y_deriv)],
            "smooth": False,
            "showSymbol": False,
            "yAxisIndex": 1,
            "lineStyle": {
                "width": 1,
                "color": "#91cc75"
            }
        }
    ]
}

output = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(output))
```
