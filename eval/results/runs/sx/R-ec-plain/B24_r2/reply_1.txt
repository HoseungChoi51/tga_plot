```python
import pandas as pd
import json
import numpy as np

def main():
    file_path = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
    
    # Read the PE sheet
    df = pd.read_excel(file_path, sheet_name="PE")
    
    # Extract data
    temp_c = df["Temperature (°C)"].values
    weight = df["Weight (%)"].values
    deriv_weight = df["Deriv. Weight (%/°C)"].values
    
    # Convert temperature to Kelvin
    temp_k = temp_c + 273.15
    
    # Downsample if too many points for better performance in ECharts
    step = max(1, len(temp_k) // 500)
    temp_k_sampled = temp_k[::step].tolist()
    weight_sampled = weight[::step].tolist()
    deriv_weight_sampled = deriv_weight[::step].tolist()
    
    # Prepare series data
    # TGA typically plots Weight on left Y axis and Derivative on right Y axis or separate
    # The request asks to plot with Temperature in Kelvin.
    # We will create a dual-axis chart: Left Y for Weight (%), Right Y for Deriv. Weight (%/°C)
    # X Axis is Temperature (K)
    
    option = {
        "title": {
            "text": "TGA Analysis of PE",
            "subtext": "Temperature in Kelvin",
            "left": "center"
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "Deriv. Weight (%/°C)"],
            "top": "10%"
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": [
            {
                "name": "Temperature (K)",
                "type": "value",
                "min": min(temp_k_sampled),
                "max": max(temp_k_sampled),
                "splitLine": {"show": False}
            }
        ],
        "yAxis": [
            {
                "name": "Weight (%)",
                "type": "value",
                "position": "left",
                "min": 0,
                "max": 100
            },
            {
                "name": "Deriv. Weight (%/°C)",
                "type": "value",
                "position": "right",
                "offset": 0,
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "xAxisIndex": 0,
                "yAxisIndex": 0,
                "data": [[t, w] for t, w in zip(temp_k_sampled, weight_sampled)],
                "smooth": True,
                "symbol": "none",
                "lineStyle": {"width": 2},
                "itemStyle": {"color": "#5470c6"}
            },
            {
                "name": "Deriv. Weight (%/°C)",
                "type": "line",
                "xAxisIndex": 0,
                "yAxisIndex": 1,
                "data": [[t, d] for t, d in zip(temp_k_sampled, deriv_weight_sampled)],
                "smooth": True,
                "symbol": "none",
                "lineStyle": {"width": 2, "type": "dashed"},
                "itemStyle": {"color": "#ee6666"}
            }
        ]
    }
    
    figure = {
        "name": "plot1",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }
    
    output = {"figures": [figure]}
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
