```python
import pandas as pd
import json
import numpy as np

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

def main():
    # Read the PE sheet
    df = pd.read_excel(DATA_FILE, sheet_name="PE")
    
    # Extract data columns based on preview info
    # Col B: Temperature (°C), Col C: Weight (%), Col D: Deriv. Weight (%/°C)
    temp = df['Temperature (°C)'].values
    weight = df['Weight (%)'].values
    deriv_weight = df['Deriv. Weight (%/°C)'].values
    
    # Filter data for x-axis range 100-700 to reduce points and match request
    mask = (temp >= 100) & (temp <= 700)
    temp_filtered = temp[mask]
    weight_filtered = weight[mask]
    deriv_weight_filtered = deriv_weight[mask]
    
    # Prepare series data
    # Series 1: Weight (%) - Red color, bold line (width 3)
    # Series 2: Deriv. Weight (%/°C) - Default or distinct color, secondary axis
    
    series_data_1 = []
    for t, w in zip(temp_filtered, weight_filtered):
        if not np.isnan(t) and not np.isnan(w):
            series_data_1.append([float(t), float(w)])
            
    series_data_2 = []
    for t, dw in zip(temp_filtered, deriv_weight_filtered):
        if not np.isnan(t) and not np.isnan(dw):
            series_data_2.append([float(t), float(dw)])
            
    option = {
        "title": {
            "text": "TGA Data for PE",
            "left": "center"
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {
                "type": "cross"
            }
        },
        "legend": {
            "data": ["Weight (%)", "Deriv. Weight (%/°C)"],
            "top": 30
        },
        "grid": {
            "left": "10%",
            "right": "10%",
            "bottom": "15%",
            "containLabel": True
        },
        "xAxis": {
            "name": "Temperature (°C)",
            "type": "value",
            "min": 100,
            "max": 700,
            "interval": 100,
            "axisLine": {"onZero": False},
            "splitLine": {"show": True}
        },
        "yAxis": [
            {
                "name": "Weight (%)",
                "type": "value",
                "position": "left",
                "axisLine": {"lineStyle": {"color": "#d81e06"}},
                "axisLabel": {"formatter": "{value} %"}
            },
            {
                "name": "Deriv. Weight (%/°C)",
                "type": "value",
                "position": "right",
                "offset": 0,
                "axisLine": {"lineStyle": {"color": "#5470c6"}},
                "splitLine": {"show": False}
            }
        ],
        "series": [
            {
                "name": "Weight (%)",
                "type": "line",
                "smooth": True,
                "symbol": "none",
                "itemStyle": {
                    "color": "#d81e06"
                },
                "lineStyle": {
                    "width": 3,
                    "color": "#d81e06"
                },
                "data": series_data_1
            },
            {
                "name": "Deriv. Weight (%/°C)",
                "type": "line",
                "yAxisIndex": 1,
                "smooth": True,
                "symbol": "none",
                "itemStyle": {
                    "color": "#5470c6"
                },
                "lineStyle": {
                    "width": 2,
                    "color": "#5470c6"
                },
                "data": series_data_2
            }
        ]
    }
    
    output = {
        "figures": [
            {
                "name": "plot1",
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"]
            }
        ]
    }
    
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
