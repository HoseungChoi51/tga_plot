```python
import pandas as pd
import json

# Read the data file
df = pd.read_csv("/home/chs/Work/tga_plot/eval/inputs_b/isothermal.txt", sep=None, engine='python', skiprows=1)

# Clean column names if needed
df.columns = df.columns.str.strip()

# Identify columns by matching keywords to handle potential whitespace or encoding issues
cols = df.columns.tolist()
time_col = next((c for c in cols if 'Time' in c), None)
temp_col = next((c for c in cols if 'Temperature' in c), None)
weight_col = next((c for c in cols if 'Weight' in c), None)

if not all([time_col, temp_col, weight_col]):
    raise ValueError("Could not identify required columns: Time, Temperature, Weight")

# Extract data
time_data = df[time_col].astype(float).tolist()
temp_data = df[temp_col].astype(float).tolist()
weight_data = df[weight_col].astype(float).tolist()

# Find weight at 140 min
idx_140 = df[df[time_col] == 140.0].index
if not idx_140.empty:
    weight_at_140 = float(df.loc[idx_140[0], weight_col])
else:
    # If exact 140 not found, find closest
    closest_idx = (df[time_col] - 140.0).abs().idxmin()
    weight_at_140 = float(df.loc[closest_idx, weight_col])
    time_at_140 = float(df.loc[closest_idx, time_col])

# Prepare series data for ECharts
# Weight series (left y-axis)
weight_series_data = [[t, w] for t, w in zip(time_data, weight_data)]
# Temperature series (right y-axis)
temp_series_data = [[t, temp] for t, temp in zip(time_data, temp_data)]

# Mark point for weight at 140 min
mark_point_data = [{"coord": [140.0, weight_at_140], "name": f"Weight at 140 min: {weight_at_140:.2f}%"}]

option = {
    "title": {
        "text": "TGA Isothermal Analysis",
        "left": "center"
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "legend": {
        "data": ["Weight (%)", "Temperature (°C)"],
        "top": 30
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "type": "value",
        "name": "Time (min)",
        "nameLocation": "middle",
        "nameGap": 30,
        "min": 0,
        "max": 140
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "position": "left",
            "min": 70,
            "max": 105
        },
        {
            "type": "value",
            "name": "Temperature (°C)",
            "position": "right",
            "min": 25,
            "max": 420
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "yAxisIndex": 0,
            "data": weight_series_data,
            "smooth": True,
            "symbol": "none",
            "lineStyle": {"width": 2},
            "markPoint": {
                "data": mark_point_data,
                "label": {
                    "show": True,
                    "formatter": "{b}",
                    "fontSize": 12
                }
            }
        },
        {
            "name": "Temperature (°C)",
            "type": "line",
            "yAxisIndex": 1,
            "data": temp_series_data,
            "smooth": True,
            "symbol": "none",
            "lineStyle": {"width": 2, "color": "#ff6b6b"},
            "itemStyle": {"color": "#ff6b6b"}
        }
    ]
}

result = {
    "figures": [
        {
            "name": "plot1",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(result, ensure_ascii=False))
```
