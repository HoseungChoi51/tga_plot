import pandas as pd
import json

def read_file(filepath):
    if filepath.endswith('.csv'):
        return pd.read_csv(filepath)
    elif filepath.endswith('.txt'):
        # Assuming tab or space separated, but pandas can often guess. 
        # Given the preview looks like a standard delimited file, we try sep=None with engine='python' for robustness
        return pd.read_csv(filepath, sep=None, engine='python')
    else:
        raise ValueError("Unsupported file format")

# Read data

df_b = read_file("/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_B.txt")

# Prepare data for ECharts
# We need to handle NaN values properly if any, though preview suggests clean data.
