```python
import pandas as pd
import json

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def main():
    # Read all sheets from the Excel file
    xl_file = pd.ExcelFile(DATA_FILE)
    
    figures = []
    
    # Define common style for text color and size
    common_text_style = {
        "color": "#808080",  # Gray color
        "fontSize": 13
    }
    
    # Process each sheet
    for sheet_name in xl_file.sheet_names:
        df = pd.read_excel(xl_file, sheet_name=sheet_name)
        
        # Extract data columns based on the preview structure
        # Temperature (X), Weight(%) (Y1), D.W(%/C) (Y2)
        temp = df['Temperature'].tolist()
        weight = df['Weight(%)'].tolist()
        dw = df['D.W(%/C)'].tolist()
        
        # Prepare series data for ECharts
        # Series 1: Weight (%) - Left Y axis
        # Series 2: D.W (%/C) - Right Y axis
        
        series_weight = {
            "name": "Weight (%)",
            "type": "line",
            "data": [[t, w] for t, w in zip(temp, weight)],
            "showSymbol": False,
            "lineStyle": {"width": 2},
            "yAxisIndex": 0
        }
        
        series_dw = {
            "name": "D.W (%/C)",
            "type": "line",
            "data": [[t, d] for t, d in zip(temp, dw)],
            "showSymbol": False,
            "lineStyle": {"width": 2},
            "yAxisIndex": 1
        }
        
        option = {
            "title": {
                "text": f"TGA: {sheet_name}",
                "left": "center",
                "textStyle": common_text_style
            },
            "tooltip": {
                "trigger": "axis",
                "axisPointer": {"type": "cross"}
            },
            "legend": {
                "data": ["Weight (%)", "D.W (%/C)"],
                "top": 30,
                "textStyle": common_text_style
            },
            "grid": {
                "left": "10%",
                "right": "10%",
                "bottom": "15%",
                "containLabel": True
            },
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 30,
                "axisLabel": {"color": "#808080", "fontSize": 13},
                "nameTextStyle": common_text_style,
                "splitLine": {"show": False}
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "position": "left",
                    "axisLabel": {"color": "#808080", "fontSize": 13},
                    "nameTextStyle": common_text_style,
                    "splitLine": {"lineStyle": {"color": "#e0e0e0"}}
                },
                {
                    "type": "value",
                    "name": "D.W (%/C)",
                    "position": "right",
                    "axisLabel": {"color": "#808080", "fontSize": 13},
                    "nameTextStyle": common_text_style,
                    "splitLine": {"show": False}
                }
            ],
            "series": [series_weight, series_dw],
            "animation": False
        }
        
        figure = {
            "name": f"tga_{sheet_name.replace(' ', '_')}",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
        
        figures.append(figure)
    
    output = {"figures": figures}
    print(json.dumps(output))

if __name__ == "__main__":
    main()
```
