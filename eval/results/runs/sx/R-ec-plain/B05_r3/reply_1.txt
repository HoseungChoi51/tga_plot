import pandas as pd
import numpy as np
import json

def load_data(filepath, is_csv):
    if is_csv:
        df = pd.read_csv(filepath)
    else:
        # Assuming tab or whitespace separated for .txt based on typical TGA exports, 
        # but pandas default sep=None with engine='python' handles many cases. 
        # Given the preview shows columns A,B,C,D, it's likely tab-separated or similar.
        try:
            df = pd.read_csv(filepath, sep='\t')
        except Exception:
            df = pd.read_csv(filepath, sep=r'\s+')
    
    # Clean column names to match expectations
    cols = df.columns.tolist()
    # Find temperature and weight columns dynamically if names vary slightly, 
    # but based on preview they are consistent.
    temp_col = [c for c in cols if 'Temperature' in c][0]
    weight_col = [c for c in cols if 'Weight (%)' in c][0]
    deriv_col = [c for c in cols if 'Deriv' in c][0]
    
    return df[temp_col].values, df[weight_col].values, df[deriv_col].values
