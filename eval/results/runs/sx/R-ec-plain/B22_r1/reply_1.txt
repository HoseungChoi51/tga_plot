```python
import pandas as pd
import json
import numpy as np

# Load the data from the specified Excel file and sheet
file_path = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
sheet_name = "Raw"
df = pd.read_excel(file_path, sheet_name=sheet_name)

# Rename columns for easier access based on the preview
# Assuming standard column order: Time, Temperature, Weight, Deriv. Weight
cols = df.columns.tolist()
time_col = cols[0]
temp_col = cols[1]
weight_col = cols[2]
deriv_col = cols[3]

# Extract relevant data
temps = df[temp_col].values
weights = df[weight_col].values
deriv_weights = df[deriv_col].values

# Find the weight at 150 °C for normalization
# We need to find the temperature closest to 150 °C
idx_150 = np.argmin(np.abs(temps - 150))
weight_at_150 = weights[idx_150]

# Normalize weights so that the weight at 150 °C is 100%
normalized_weights = (weights / weight_at_150) * 100.0

# Prepare data for ECharts
# Series 1: Normalized Weight (%) vs Temperature (°C)
data_weight = [[float(t), float(w)] for t, w in zip(temps, normalized_weights)]

# Series 2: Derivative Weight (%/°C) vs Temperature (°C)
# Note: The derivative is usually plotted against temperature as well.
# It might need its own Y-axis or be scaled. Let's use a secondary Y-axis for clarity if needed, 
# but typically TGA plots show TG on left and DTG on right.
data_deriv = [[float(t), float(d)] for t, d in zip(temps, deriv_weights)]

# Construct the ECharts option
option = {
    "title": {
        "text": "TGA Data of Raw Sample (Normalized to 100% at 150°C)",
        "left": "center",
        "textStyle": {
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 16
        }
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {
            "type": "cross"
        }
    },
    "legend": {
        "data": ["Weight (%)", "Deriv. Weight (%/°C)"],
        "top": 40
    },
    "grid": {
        "left": "10%",
        "right": "10%",
        "bottom": "15%",
        "containLabel": True
    },
    "xAxis": {
        "name": "Temperature (°C)",
        "type": "value",
        "min": 0,
        "max": 900,
        "nameLocation": "middle",
        "nameGap": 30
    },
    "yAxis": [
        {
            "name": "Weight (%)",
            "type": "value",
            "position": "left",
            "nameLocation": "middle",
            "nameGap": 40
        },
        {
            "name": "Deriv. Weight (%/°C)",
            "type": "value",
            "position": "right",
            "nameLocation": "middle",
            "nameGap": 40,
            "splitLine": {
                "show": False
            }
        }
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "smooth": True,
            "symbol": "none",
            "lineStyle": {
                "width": 2
            },
            "data": data_weight,
            "yAxisIndex": 0
        },
        {
            "name": "Deriv. Weight (%/°C)",
            "type": "line",
            "smooth": True,
            "symbol": "none",
            "lineStyle": {
                "width": 2,
                "type": "dashed"
            },
            "itemStyle": {
                "color": "#FF7F0E" # Orange for DTG
            },
            "data": data_deriv,
            "yAxisIndex": 1
        }
    ]
}

# Wrap in the required JSON structure
result = {
    "figures": [
        {
            "name": "tga_raw_normalized",
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"]
        }
    ]
}

print(json.dumps(result, ensure_ascii=False))
```
