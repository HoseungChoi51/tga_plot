```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B14_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read the PVC sheet
df = pd.read_excel(DATA_FILE, sheet_name="PVC")

# Extract columns based on standard TGA structure identified in preview
# B: Temperature (°C), C: Weight (%), D: Deriv. Weight (%/°C)
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

if temp_col not in df.columns or weight_col not in df.columns or dtg_col not in df.columns:
    # Fallback to positional if names differ slightly but order is known from preview
    temp = df.iloc[:, 1]
    weight = df.iloc[:, 2]
    dtg = df.iloc[:, 3]
else:
    temp = df[temp_col]
    weight = df[weight_col]
    dtg = df[dtg_col]

# Clean data: remove NaNs
mask = ~(np.isnan(temp) | np.isnan(weight) | np.isnan(dtg))
temp = temp[mask]
weight = weight[mask]
dtg = dtg[mask]

# Invert DTG so peaks point up
dtg_inv = -dtg

# Find Residue at ~800°C
idx_800 = np.argmin(np.abs(temp - 800))
residue_temp = temp.iloc[idx_800]
residue_weight = weight.iloc[idx_800]

# Find Peak temperature of highest DTG peak above 150°C
mask_above_150 = temp > 150
if mask_above_150.any():
    temp_above_150 = temp[mask_above_150]
    dtg_inv_above_150 = dtg_inv[mask_above_150]
    
    idx_peak_local = np.argmax(dtg_inv_above_150)
    peak_temp = temp_above_150.iloc[idx_peak_local]
    peak_dtg_val = dtg_inv_above_150.iloc[idx_peak_local]
else:
    peak_temp = None
    peak_dtg_val = None

# Plotting
fig, ax1 = plt.subplots(figsize=(10, 6))

# Font size setting
font_size = 16

# Left axis: Weight (%)
color_w = '#1f4e9c'
ax1.plot(temp, weight, color=color_w, linewidth=2, label='Weight')
ax1.set_xlabel('Temperature (°C)', fontsize=font_size)
ax1.set_ylabel('Weight (%)', color=color_w, fontsize=font_size)
ax1.tick_params(axis='x', labelsize=font_size-2)
ax1.tick_params(axis='y', labelcolor=color_w, labelsize=font_size-2)

# Right axis: DTG
ax2 = ax1.twinx()
color_d = '#c0392b'
ax2.plot(temp, dtg_inv, color=color_d, linewidth=2, linestyle='--', label='DTG')
ax2.set_ylabel('DTG (%/°C)', color=color_d, fontsize=font_size)
ax2.tick_params(axis='y', labelcolor=color_d, labelsize=font_size-2)

# Title
sheet_name = "PVC"
ax1.set_title(sheet_name, fontsize=font_size+2, pad=10)

# Legend
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
legend = ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left', bbox_to_anchor=(0.02, 0.5), fontsize=font_size-2)

# Annotations: Residue
ax1.plot(residue_temp, residue_weight, 'o', color=color_w, markersize=8)
ax1.annotate(f'Residue {residue_weight:.1f}%', 
             xy=(residue_temp, residue_weight), 
             xytext=(residue_temp - 100, residue_weight + 15),
             fontsize=font_size-2, color=color_w,
             arrowprops=dict(facecolor=color_w, shrink=0.05, width=1.5, headwidth=6))

# Annotations: Peak
if peak_temp is not None:
    # Get the weight value at the peak temperature for annotation positioning reference if needed, 
    # but usually just marking the DTG peak on its own axis scale visually is enough or pointing to it.
    # We plot a marker on the DTG curve (right axis)
    ax2.plot(peak_temp, peak_dtg_val, 'v', color=color_d, markersize=8)
    ax2.annotate(f'{peak_temp:.0f} °C', 
                 xy=(peak_temp, peak_dtg_val), 
                 xytext=(peak_temp + 20, peak_dtg_val * 1.1),
                 fontsize=font_size-2, color=color_d,
                 arrowprops=dict(facecolor=color_d, shrink=0.05, width=1.5, headwidth=6))

plt.tight_layout()

out_path = os.path.join(OUT_DIR, f"{sheet_name}.png")
plt.savefig(out_path, dpi=300, bbox_inches='tight')
plt.close()
```
