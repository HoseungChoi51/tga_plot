```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import matplotlib

# Set Korean font
matplotlib.rcParams['font.family'] = 'Noto Sans CJK KR'
matplotlib.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/01_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets from the Excel file
xl_file = pd.ExcelFile(DATA_FILE)
sheet_names = xl_file.sheet_names

for sheet_name in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)
    
    # Identify columns based on typical TGA structure
    # Assume: col B = Temperature, col C = Weight(%), col D = DTG (D.W %/C)
    # We'll try to find them by name or position
    
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        col_lower = str(col).lower()
        if 'temperature' in col_lower or 'temp' in col_lower:
            temp_col = col
        elif 'weight' in col_lower and '%' in col_lower:
            weight_col = col
        elif 'd.w' in col_lower or 'dtg' in col_lower or ('derivative' in col_lower):
            dtg_col = col
    
    # Fallback to positional if names not found clearly
    if temp_col is None and len(df.columns) >= 2:
        temp_col = df.columns[1]  # Column B
    if weight_col is None and len(df.columns) >= 3:
        weight_col = df.columns[2]  # Column C
    if dtg_col is None and len(df.columns) >= 4:
        dtg_col = df.columns[3]  # Column D
    
    # Extract data
    temperature = df[temp_col].values
    weight = df[weight_col].values
    dtg = df[dtg_col].values
    
    # Create figure
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot Weight (%) on left y-axis
    color_weight = '#1f4e9c'
    ax1.plot(temperature, weight, color=color_weight, label='Weight (%)', linewidth=1.5)
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color_weight, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color_weight)
    ax1.set_xlim(min(temperature), max(temperature))
    
    # Plot DTG on right y-axis, multiplied by -1 so peaks point up
    ax2 = ax1.twinx()
    color_dtg = '#c0392b'
    ax2.plot(temperature, -dtg, color=color_dtg, label='DTG (%/°C)', linewidth=1.5)
    ax2.set_ylabel('DTG (%/°C)', color=color_dtg, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color_dtg)
    
    # Title
    ax1.set_title(sheet_name, fontsize=14, pad=10)
    
    # Legend at center left
    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left', bbox_to_anchor=(0.02, 0.5), fontsize=10)
    
    # Find residue at nearest 800°C
    idx_800 = np.argmin(np.abs(temperature - 800))
    residue_value = weight[idx_800]
    residue_temp = temperature[idx_800]
    
    # Mark residue point
    ax1.plot(residue_temp, residue_value, 'o', color=color_weight, markersize=6)
    ax1.annotate(f'Residue {residue_value:.1f}%', 
                 xy=(residue_temp, residue_value),
                 xytext=(10, 10), textcoords='offset points',
                 fontsize=10, color=color_weight,
                 arrowprops=dict(arrowstyle='->', color=color_weight, lw=1))
    
    # Find highest DTG peak above 150°C
    # Since we plotted -dtg, the highest peak corresponds to the most negative dtg value
    mask_above_150 = temperature > 150
    if np.any(mask_above_150):
        dtg_neg = -dtg[mask_above_150]
        temp_filtered = temperature[mask_above_150]
        
        peak_idx_local = np.argmax(dtg_neg)
        peak_temp = temp_filtered[peak_idx_local]
        peak_dtg_val = dtg_neg[peak_idx_local]
        
        # Mark peak on DTG curve (right axis)
        ax2.plot(peak_temp, peak_dtg_val, 'o', color=color_dtg, markersize=6)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_dtg_val),
                     xytext=(10, 10), textcoords='offset points',
                     fontsize=10, color=color_dtg,
                     arrowprops=dict(arrowstyle='->', color=color_dtg, lw=1))
    
    # Adjust layout
    plt.tight_layout()
    
    # Save figure
    safe_sheet_name = sheet_name.replace('/', '_').replace('\\', '_')
    output_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    plt.savefig(output_path, dpi=300, bbox_inches='tight')
    plt.close(fig)
```
