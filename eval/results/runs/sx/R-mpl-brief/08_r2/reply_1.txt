```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/08_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets from the Excel file
xl_file = pd.ExcelFile(DATA_FILE)
sheet_names = xl_file.sheet_names

for sheet_name in sheet_names:
    # Read data for the current sheet
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    
    # Identify columns based on standard TGA structure provided in preview
    # x: Temperature (col B), y1: Weight(%) (col C), y2: D.W(%/C) (col D)
    temp_col = None
    weight_col = None
    dtg_col = None
    
    for col in df.columns:
        col_lower = str(col).lower()
        if 'temperature' in col_lower or 'temp' in col_lower:
            temp_col = col
        elif 'weight' in col_lower and '%' in col_lower:
            weight_col = col
        elif 'd.w' in col_lower or 'derivative' in col_lower or 'dtg' in col_lower:
            dtg_col = col
            
    # Fallback to column indices if names are not found (based on preview: A=Time, B=Temp, C=Weight, D=DTG)
    if temp_col is None: temp_col = df.columns[1]
    if weight_col is None: weight_col = df.columns[2]
    if dtg_col is None: dtg_col = df.columns[3]
    
    # Extract data
    x_data = df[temp_col].dropna().values
    y1_data = df[weight_col].dropna().values
    y2_data = df[dtg_col].dropna().values
    
    # Align lengths just in case of NaNs dropping differently
    min_len = min(len(x_data), len(y1_data), len(y2_data))
    x_data = x_data[:min_len]
    y1_data = y1_data[:min_len]
    y2_data = y2_data[:min_len]
    
    # Create figure and axis
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot Weight (%) on left axis (y1)
    color1 = '#1f4e9c'
    line1, = ax1.plot(x_data, y1_data, color=color1, label='무게 (%)')
    ax1.set_xlabel('온도 (°C)', fontsize=12)
    ax1.set_ylabel('무게 (%)', color=color1, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color1)
    
    # Create second axis sharing the same x-axis
    ax2 = ax1.twinx()
    
    # Plot DTG on right axis (y2), multiplied by -1 so peaks point up
    color2 = '#c0392b'
    # Filter out non-finite values for plotting safety if needed, but usually raw is fine
    y2_plot = -y2_data 
    line2, = ax2.plot(x_data, y2_plot, color=color2, label='미분 무게 (%/°C)')
    ax2.set_ylabel('미분 무게 (%/°C)', color=color2, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color2)
    
    # Title: Sheet name
    ax1.set_title(sheet_name, fontsize=14, pad=15)
    
    # Legend
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='center left', bbox_to_anchor=(0.05, 0.5))
    
    # Calculate Residue at ~800°C
    # Find index closest to 800
    idx_800 = np.argmin(np.abs(x_data - 800))
    residue_val = y1_data[idx_800]
    residue_temp = x_data[idx_800]
    
    # Add Residue marker and text
    ax1.plot(residue_temp, residue_val, 'o', color=color1, markersize=5)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(residue_temp, residue_val), 
                 xytext=(10, 10), 
                 textcoords='offset points',
                 fontsize=10, color=color1,
                 arrowprops=dict(arrowstyle='->', color=color1, lw=0.5))
                 
    # Calculate Peak temperature above 150°C
    # We look for the maximum of the inverted DTG (which corresponds to the most negative original DTG)
    # Only consider temperatures > 150
    mask_peak = x_data > 150
    if np.any(mask_peak):
        y2_filtered = y2_plot[mask_peak]
        x_filtered = x_data[mask_peak]
        
        peak_idx_local = np.argmax(y2_filtered)
        peak_temp = x_filtered[peak_idx_local]
        peak_val_inv = y2_filtered[peak_idx_local] # This is positive value of inverted DTG
        
        # Plot marker on the DTG curve (right axis)
        ax2.plot(peak_temp, peak_val_inv, 'o', color=color2, markersize=5)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_val_inv), 
                     xytext=(0, 10), 
                     textcoords='offset points',
                     ha='center',
                     fontsize=10, color=color2,
                     arrowprops=dict(arrowstyle='->', color=color2, lw=0.5))
    
    # Adjust layout to prevent clipping
    plt.tight_layout()
    
    # Save figure
    safe_sheet_name = sheet_name.replace('/', '_').replace('\\', '_')
    output_filename = f"{safe_sheet_name}.png"
    output_path = os.path.join(OUT_DIR, output_filename)
    
    plt.savefig(output_path, dpi=300, bbox_inches='tight')
    plt.close(fig)

print("Plots generated successfully.")
```
