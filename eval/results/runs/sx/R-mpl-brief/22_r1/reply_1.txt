```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/22_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

for sheet in sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet)
    
    # Identify columns based on the preview structure
    # Assuming standard TGA structure from preview: Time, Temperature, Weight(%), D.W(%/C)
    x_col = None
    y1_col = None
    y2_col = None
    
    cols = df.columns.tolist()
    for c in cols:
        cl = str(c).lower()
        if 'temperature' in cl or 'temp' in cl:
            x_col = c
        elif 'weight' in cl and '%' in cl:
            y1_col = c
        elif 'd.w' in cl or 'dtg' in cl or ('derivative' in cl and 'weight' in cl):
            y2_col = c
            
    if not all([x_col, y1_col, y2_col]):
        # Fallback to positional if names don't match strictly but structure is known
        # From preview: B=Temp, C=Weight, D=D.W
        if len(cols) >= 4:
             x_col = cols[1]
             y1_col = cols[2]
             y2_col = cols[3]
        else:
            continue

    x = df[x_col].values
    y1 = df[y1_col].values
    y2 = df[y2_col].values
    
    # Filter out NaNs
    mask = ~(np.isnan(x) | np.isnan(y1) | np.isnan(y2))
    x = x[mask]
    y1 = y1[mask]
    y2 = y2[mask]
    
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot y1 (Weight %) - Left axis
    color_y1 = '#1f4e9c'
    line1, = ax1.plot(x, y1, color=color_y1, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color_y1, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color_y1)
    
    # Create twin axis for y2 (DTG)
    ax2 = ax1.twinx()
    
    # User request: y2 line color green, y2 axis label color green
    color_y2 = 'green'
    # Multiply by -1 so peaks point up
    y2_inv = -y2
    line2, = ax2.plot(x, y2_inv, color=color_y2, linestyle='--', label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color=color_y2, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color_y2)
    
    # Title
    ax1.set_title(sheet, fontsize=14)
    
    # Legend at center left
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='center left', bbox_to_anchor=(0.05, 0.5))
    
    # Residue calculation (nearest to 800°C)
    target_temp_res = 800.0
    idx_res = np.argmin(np.abs(x - target_temp_res))
    res_weight = y1[idx_res]
    res_temp = x[idx_res]
    
    # Peak calculation (highest DTG peak above 150°C)
    # Note: We look for the minimum of original y2 (most negative) or maximum of y2_inv
    mask_peak = x > 150.0
    if np.any(mask_peak):
        y2_filtered = y2_inv[mask_peak]
        x_filtered = x[mask_peak]
        max_idx = np.argmax(y2_filtered)
        peak_temp = x_filtered[max_idx]
        peak_val = y2_filtered[max_idx]
        
        # Add marker and text for peak
        ax2.plot(peak_temp, peak_val, 'o', color=color_y2, markersize=6)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_val), 
                     xytext=(peak_temp + 20, peak_val + 0.05),
                     arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                     fontsize=10, color='black')
    else:
        peak_temp = None
        
    # Add marker and text for residue
    ax1.plot(res_temp, res_weight, 'o', color=color_y1, markersize=6)
    ax1.annotate(f'Residue {res_weight:.1f}%', 
                 xy=(res_temp, res_weight), 
                 xytext=(res_temp - 50, res_weight + 5),
                 arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                 fontsize=10, color='black')
                 
    plt.tight_layout()
    
    safe_sheet_name = "".join([i if i.isalnum() or i == '-' or i == '_' else "_" for i in sheet])
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    plt.savefig(out_path, dpi=300)
    plt.close(fig)
```
