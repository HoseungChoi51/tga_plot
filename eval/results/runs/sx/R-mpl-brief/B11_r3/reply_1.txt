```python
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import os

# 데이터 파일 경로 및 출력 디렉토리 설정
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B11_r3/out"

# 한글 폰트 설정
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# 시트 이름 정의
sheet_name = "PET"

# 데이터 읽기
try:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
except Exception as e:
    print(f"Error reading file: {e}")
    exit(1)

# 열 이름 확인 및 매핑 (데이터 미리보기 기준)
# x: Temperature (°C), y1: Weight (%), y2: Deriv. Weight (%/°C)
x_col = "Temperature (°C)"
y1_col = "Weight (%)"
y2_col = "Deriv. Weight (%/°C)"

# 데이터 정제: 결측치 제거 및 타입 변환
df[x_col] = pd.to_numeric(df[x_col], errors='coerce')
df[y1_col] = pd.to_numeric(df[y1_col], errors='coerce')
df[y2_col] = pd.to_numeric(df[y2_col], errors='coerce')
df.dropna(subset=[x_col, y1_col, y2_col], inplace=True)

# 그래프 그리기
fig, ax1 = plt.subplots(figsize=(8, 6))

# y1 축 (Weight %) - 왼쪽 축
color_y1 = '#1f4e9c'
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color_y1, fontsize=12)
line1 = ax1.plot(df[x_col], df[y1_col], color=color_y1, label='Weight')
ax1.tick_params(axis='y', labelcolor=color_y1)

# y2 축 (DTG) - 오른쪽 축
ax2 = ax1.twinx()
color_y2 = '#c0392b'
ax2.set_ylabel('DTG (%/°C)', color=color_y2, fontsize=12)
# DTG 값에 -1을 곱하여 피크가 위로 향하도록 함
dtg_neg = df[y2_col] * -1
line2 = ax2.plot(df[x_col], dtg_neg, color=color_y2, label='DTG')
ax2.tick_params(axis='y', labelcolor=color_y2)

# 제목 설정
ax1.set_title(sheet_name + ' 열분해', fontsize=14) # 요청사항: 제목은 'PET 열분해' -> 시트명 PET이므로 'PET 열분해'

# 격자 켜기
ax1.grid(True, linestyle='--', alpha=0.7)

# 잔류량 계산 및 표시 (Residue at ~800°C)
# 800도에 가장 가까운 데이터 포인트 찾기
idx_800 = np.abs(df[x_col] - 800).argmin()
residue_val = df.iloc[idx_800][y1_col]
temp_residue = df.iloc[idx_800][x_col]

# 소수 둘째 자리까지 포맷팅
residue_str = f"Residue {residue_val:.2f}%"

# 잔류량 마커 및 텍스트 추가
ax1.scatter(temp_residue, residue_val, color=color_y1, zorder=5)
ax1.annotate(residue_str, 
             xy=(temp_residue, residue_val), 
             xytext=(10, 10), textcoords='offset points',
             fontsize=10, color=color_y1,
             bbox=dict(boxstyle="round,pad=0.3", fc="white", ec=color_y1, lw=1, alpha=0.8))

# 피크 온도 계산 및 표시 (Highest DTG peak above 150°C)
# DTG는 음수값이므로 절대값이 가장 큰 곳, 또는 역전환된 값(dtg_neg)의 최대값
# 조건: 온도 > 150도
mask_temp_gt_150 = df[x_col] > 150
df_filtered = df[mask_temp_gt_150].copy()
df_filtered['dtg_neg'] = df_filtered[y2_col] * -1

if not df_filtered.empty:
    idx_peak = df_filtered['dtg_neg'].idxmax()
    peak_temp = df.loc[idx_peak, x_col]
    peak_dtg_val = df.loc[idx_peak, y2_col] # 원본 DTG 값
    
    peak_str = f"{peak_temp:.0f} °C"
    
    # 피크 마커 및 텍스트 추가 (y2 축 기준)
    ax2.scatter(peak_temp, peak_dtg_val * -1, color=color_y2, zorder=5)
    ax2.annotate(peak_str, 
                 xy=(peak_temp, peak_dtg_val * -1), 
                 xytext=(10, 10), textcoords='offset points',
                 fontsize=10, color=color_y2,
                 bbox=dict(boxstyle="round,pad=0.3", fc="white", ec=color_y2, lw=1, alpha=0.8))

# 범례 설정 (왼쪽 아래)
lines = line1 + line2
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='lower left', fontsize=10)

# 레이아웃 조정
plt.tight_layout()

# 저장
output_filename = f"{sheet_name}_TGA.png"
output_path = os.path.join(OUT_DIR, output_filename)

# 디렉토리가 없으면 생성
os.makedirs(OUT_DIR, exist_ok=True)

plt.savefig(output_path, dpi=300)
plt.close()

print(f"Plot saved to {output_path}")
```
