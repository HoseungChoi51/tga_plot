```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import matplotlib

# Set Korean font
matplotlib.rcParams['font.family'] = 'Noto Sans CJK KR'
matplotlib.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/09_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

xls = pd.ExcelFile(DATA_FILE)

for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name, header=0)
    
    # Identify columns based on the preview structure
    # Columns are typically: Time, Temperature, Weight(%), D.W(%/C)
    cols = df.columns.tolist()
    
    # Find x (Temperature), y1 (Weight), y2 (DTG/D.W)
    x_col = None
    y1_col = None
    y2_col = None
    
    for c in cols:
        if 'temp' in str(c).lower():
            x_col = c
        elif 'weight' in str(c).lower() and '%' in str(c):
            y1_col = c
        elif 'd.w' in str(c).lower() or 'dtg' in str(c).lower():
            y2_col = c
            
    if x_col is None or y1_col is None or y2_col is None:
        continue
        
    data = df[[x_col, y1_col, y2_col]].dropna()
    
    temp = data[x_col].values
    weight = data[y1_col].values
    dtg = data[y2_col].values
    
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot Weight on left axis
    color_y1 = '#1f4e9c'
    line1 = ax1.plot(temp, weight, color=color_y1, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color_y1, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color_y1, direction='out', labelsize=9)
    ax1.tick_params(axis='x', direction='out', labelsize=9)
    
    # Plot DTG on right axis
    ax2 = ax1.twinx()
    color_y2 = '#c0392b'
    # Multiply by -1 so peaks point up
    dtg_inv = -1 * dtg
    line2 = ax2.plot(temp, dtg_inv, color=color_y2, linestyle='--', label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color=color_y2, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color_y2, direction='out', labelsize=9)
    
    # Title
    ax1.set_title(sheet_name, fontsize=14)
    
    # Legend
    lines = line1 + line2
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='center left', bbox_to_anchor=(0.5, 0.5), frameon=False, fontsize=10)
    
    # Calculate Residue at ~800°C
    idx_800 = np.argmin(np.abs(temp - 800))
    residue_val = weight[idx_800]
    temp_800 = temp[idx_800]
    
    # Mark residue
    ax1.plot(temp_800, residue_val, 'o', color=color_y1, markersize=8)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(temp_800, residue_val), 
                 xytext=(temp_800 - 100, residue_val + 5),
                 arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                 fontsize=9, color=color_y1)
                 
    # Calculate Peak temperature above 150°C
    mask = temp > 150
    if np.any(mask):
        dtg_masked = dtg_inv[mask]
        temp_masked = temp[mask]
        peak_idx = np.argmax(dtg_masked)
        peak_temp = temp_masked[peak_idx]
        peak_dtg_val = dtg_masked[peak_idx]
        
        # Mark peak on ax2
        ax2.plot(peak_temp, peak_dtg_val, 'v', color=color_y2, markersize=8)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_dtg_val), 
                     xytext=(peak_temp + 20, peak_dtg_val + 0.1),
                     arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                     fontsize=9, color=color_y2)
    
    # Adjust layout to prevent clipping
    plt.tight_layout()
    
    # Save figure
    safe_sheet_name = "".join([c for c in sheet_name if c.isalnum() or c in (' ', '-', '_')]).rstrip()
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    plt.savefig(out_path, dpi=300, bbox_inches='tight')
    plt.close(fig)
```
