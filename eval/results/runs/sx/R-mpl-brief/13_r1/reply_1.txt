```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/13_r1/out"

os.makedirs(OUT_DIR, exist_ok=True)

xls = pd.ExcelFile(DATA_FILE)

for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name)
    
    # Identify columns based on the preview and user request
    # x = Temperature (°C) -> Column B
    # y1 = Weight (mg) -> Column D (User requested this instead of default Weight %)
    # y2 = Deriv. Weight (%/°C) -> Column G
    
    temp_col = None
    weight_mg_col = None
    dtg_col = None
    
    for col in df.columns:
        col_lower = str(col).lower()
        if 'temperature' in col_lower:
            temp_col = col
        elif 'weight (mg)' in col_lower or ('weight' in col_lower and 'mg' in col_lower):
            weight_mg_col = col
        elif 'deriv' in col_lower or 'dtg' in col_lower:
            dtg_col = col
            
    # Fallback to indices if names don't match exactly but structure is known from preview
    if temp_col is None: temp_col = df.columns[1] # B
    if weight_mg_col is None: weight_mg_col = df.columns[3] # D
    if dtg_col is None: dtg_col = df.columns[6] # G
    
    # Clean data
    df_clean = df.dropna(subset=[temp_col, weight_mg_col, dtg_col])
    
    x = df_clean[temp_col].values
    y1 = df_clean[weight_mg_col].values
    y2_raw = df_clean[dtg_col].values
    
    # Invert DTG so peaks point up
    y2 = -y2_raw
    
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot y1 (Weight mg)
    color_y1 = '#1f4e9c'
    line1, = ax1.plot(x, y1, color=color_y1, label='Weight (mg)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (mg)', fontsize=12, color=color_y1)
    ax1.tick_params(axis='y', labelcolor=color_y1)
    
    # Create twin axis for y2 (DTG)
    ax2 = ax1.twinx()
    color_y2 = '#c0392b'
    line2, = ax2.plot(x, y2, color=color_y2, label='DTG (-1 * %/°C)')
    ax2.set_ylabel('Derivative Weight (×-1 %/°C)', fontsize=12, color=color_y2)
    ax2.tick_params(axis='y', labelcolor=color_y2)
    
    # Title
    ax1.set_title(sheet_name, fontsize=14)
    
    # Find Residue at ~800°C
    idx_800 = np.argmin(np.abs(x - 800))
    residue_val = y1[idx_800]
    temp_800 = x[idx_800]
    
    # Mark residue on curve
    ax1.plot(temp_800, residue_val, 'o', color=color_y1, markersize=5)
    ax1.annotate(f'Residue {residue_val:.2f} mg', 
                 xy=(temp_800, residue_val), 
                 xytext=(temp_800 + 50, residue_val + (np.max(y1)-np.min(y1))*0.1),
                 arrowprops=dict(facecolor=color_y1, shrink=0.05, width=1, headwidth=5),
                 fontsize=10, color=color_y1)
                 
    # Find Peak in DTG above 150°C
    mask_peak = x > 150
    if np.any(mask_peak):
        x_peak_range = x[mask_peak]
        y2_peak_range = y2[mask_peak]
        
        max_idx_local = np.argmax(y2_peak_range)
        peak_temp = x_peak_range[max_idx_local]
        peak_val = y2_peak_range[max_idx_local]
        
        # Mark peak on curve
        ax2.plot(peak_temp, peak_val, 'v', color=color_y2, markersize=7)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_val), 
                     xytext=(peak_temp + 20, peak_val + (np.max(y2)-np.min(y2))*0.1),
                     arrowprops=dict(facecolor=color_y2, shrink=0.05, width=1, headwidth=5),
                     fontsize=10, color=color_y2)
                     
    # Legend
    lines = [line1, line2]
    labels = [l.get_label() for l in lines]
    ax1.legend(lines, labels, loc='center left', bbox_to_anchor=(0.05, 0.5))
    
    # Grid
    ax1.grid(True, linestyle='--', alpha=0.5)
    
    plt.tight_layout()
    
    out_path = os.path.join(OUT_DIR, f"{sheet_name}.png")
    plt.savefig(out_path, dpi=300)
    plt.close(fig)
```
