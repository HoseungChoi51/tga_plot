```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from pathlib import Path

# Set up Korean font support
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

# Define paths
DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B22_r3/out"

# Ensure output directory exists
Path(OUT_DIR).mkdir(parents=True, exist_ok=True)

# Read the 'Raw' sheet
df = pd.read_excel(DATA_FILE, sheet_name='Raw')

# Extract columns based on the preview info
# Col B: Temperature (°C), Col C: Weight (%), Col D: Deriv. Weight (%/°C)
temp_col = 'Temperature (°C)'
weight_col = 'Weight (%)'
dtg_col = 'Deriv. Weight (%/°C)'

temps = df[temp_col].values
weights = df[weight_col].values
dtgs = df[dtg_col].values

# Recalculate weight normalized to 100% at 150 °C
# Find the index closest to 150 °C
idx_150 = np.argmin(np.abs(temps - 150))
weight_at_150 = weights[idx_150]

if weight_at_150 == 0:
    # Avoid division by zero, though unlikely in TGA
    normalized_weights = weights 
else:
    normalized_weights = (weights / weight_at_150) * 100

# Prepare for plotting
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight (left axis)
color_w = '#1f4e9c'
ax1.plot(temps, normalized_weights, color=color_w, label='Weight (%)')
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color_w, fontsize=12)
ax1.tick_params(axis='y', labelcolor=color_w)
ax1.grid(True, linestyle='--', alpha=0.5)

# Create second y-axis for DTG
ax2 = ax1.twinx()
color_d = '#c0392b'
# Multiply DTG by -1 so peaks point up
ax2.plot(temps, dtgs * -1, color=color_d, label='DTG (-dW/dT)')
ax2.set_ylabel('DTG (%/°C)', color=color_d, fontsize=12)
ax2.tick_params(axis='y', labelcolor=color_d)

# Title
ax1.set_title('Raw', fontsize=14, pad=10)

# Legend
lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()
ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='center left')

# Calculate Residue at 800 °C
idx_800 = np.argmin(np.abs(temps - 800))
residue_weight = normalized_weights[idx_800]
residue_temp = temps[idx_800]

# Add Residue marker and label
ax1.plot(residue_temp, residue_weight, 'o', color=color_w, markersize=8)
ax1.annotate(f'Residue {residue_weight:.1f}%', 
             xy=(residue_temp, residue_weight), 
             xytext=(10, 10), textcoords='offset points',
             arrowprops=dict(arrowstyle='->', color=color_w),
             fontsize=10, color=color_w, ha='left')

# Find Peak temperature (highest DTG peak above 150 °C)
# We look for the minimum of original DTG (which becomes maximum after *-1) or simply max of inverted DTG
mask_above_150 = temps > 150
if np.any(mask_above_150):
    inverted_dtg = dtgs * -1
    # Find index of max inverted DTG within the mask
    masked_inverted_dtg = np.where(mask_above_150, inverted_dtg, -np.inf)
    idx_peak = np.argmax(masked_inverted_dtg)
    
    peak_temp = temps[idx_peak]
    peak_val_inv = inverted_dtg[idx_peak]
    
    # Add Peak marker and label on the right axis
    ax2.plot(peak_temp, peak_val_inv, 'v', color=color_d, markersize=8)
    ax2.annotate(f'{peak_temp:.0f} °C', 
                 xy=(peak_temp, peak_val_inv), 
                 xytext=(10, -15), textcoords='offset points',
                 arrowprops=dict(arrowstyle='->', color=color_d),
                 fontsize=10, color=color_d, ha='left')

# Save figure
output_path = Path(OUT_DIR) / "Raw_normalized.png"
plt.savefig(output_path, dpi=300, bbox_inches='tight')
plt.close()
```
