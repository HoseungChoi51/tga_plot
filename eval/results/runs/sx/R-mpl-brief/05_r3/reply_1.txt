```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Font settings for Korean text
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/05_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

xls = pd.ExcelFile(DATA_FILE)

for sheet_name in xls.sheet_names:
    df = pd.read_excel(xls, sheet_name=sheet_name, header=0)
    
    # Identify columns based on the preview structure
    # Col B: Temperature, Col C: Weight(%), Col D: D.W(%/C)
    try:
        temp_col = df.columns[1]  # Temperature
        weight_col = df.columns[2]  # Weight(%)
        dtg_col = df.columns[3]   # D.W(%/C)
    except IndexError:
        continue
        
    x = df[temp_col].values
    y1 = df[weight_col].values
    y2 = df[dtg_col].values
    
    # Create figure and axis
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot Weight (y1) - Black line, linewidth 2.5
    color_y1 = '#000000'
    ax1.plot(x, y1, color=color_y1, linewidth=2.5, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color_y1, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color_y1)
    
    # Plot DTG (y2) - Dashed line
    # Multiply by -1 so peaks point up
    y2_inverted = y2 * -1
    color_y2 = '#c0392b'
    ax2 = ax1.twinx()
    ax2.plot(x, y2_inverted, color=color_y2, linestyle='--', linewidth=1.5, label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color=color_y2, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color_y2)
    
    # Title is the sheet name
    ax1.set_title(sheet_name, fontsize=14)
    
    # Calculate Residue at ~800°C
    mask_800 = (x >= 790) & (x <= 810)
    if np.any(mask_800):
        idx_res = np.argmin(np.abs(x[mask_800] - 800))
        res_temp = x[mask_800][idx_res]
        res_weight = y1[mask_800][idx_res]
        
        # Mark residue on plot
        ax1.plot(res_temp, res_weight, 'o', color=color_y1, markersize=5)
        ax1.annotate(f'Residue {res_weight:.1f}%', 
                     xy=(res_temp, res_weight), 
                     xytext=(res_temp + 20, res_weight + 5),
                     arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                     fontsize=10, color='black')
    
    # Calculate Peak temperature (>150°C) from inverted DTG
    mask_peak = x > 150
    if np.any(mask_peak):
        y2_inv_filtered = y2_inverted[mask_peak]
        x_filtered = x[mask_peak]
        
        max_idx = np.argmax(y2_inv_filtered)
        peak_temp = x_filtered[max_idx]
        peak_val = y2_inv_filtered[max_idx]
        
        # Mark peak on plot
        ax2.plot(peak_temp, peak_val, 'v', color=color_y2, markersize=6)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_val), 
                     xytext=(peak_temp - 50, peak_val + 0.1),
                     arrowprops=dict(facecolor='red', shrink=0.05, width=1, headwidth=5),
                     fontsize=10, color='red')

    # Legend
    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left', bbox_to_anchor=(0.05, 0.5))
    
    plt.tight_layout()
    
    # Save file
    safe_sheet_name = sheet_name.replace('/', '_').replace('\\', '_')
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    plt.savefig(out_path, dpi=300)
    plt.close(fig)

print("Plots saved successfully.")
```
