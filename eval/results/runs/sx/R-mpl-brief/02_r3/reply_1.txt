```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from matplotlib import font_manager

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/02_r3/out"
os.makedirs(OUT_DIR, exist_ok=True)

# Read the specific sheet
sheet_name = "1006R1-V normal"
df = pd.read_excel(DATA_FILE, sheet_name=sheet_name, header=0)

# Identify columns based on user request: B column (index 1) -> X, C column (index 2) -> Y1, D column (index 3) -> Y2
# Based on preview: Col A=Time, B=Temperature, C=Weight(%), D=D.W(%/C)
col_x = df.columns[1]  # Temperature
col_y1 = df.columns[2] # Weight(%)
col_y2 = df.columns[3] # D.W(%/C) which is DTG

x_data = df[col_x].values
y1_data = df[col_y1].values
y2_data = df[col_y2].values

# Create figure and axis
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot Weight (%) on left y-axis (y1)
color1 = '#1f4e9c'
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
line1, = ax1.plot(x_data, y1_data, color=color1, label='Weight')
ax1.tick_params(axis='y', labelcolor=color1)
ax1.grid(True, linestyle='--', alpha=0.5)

# Create twin axis for DTG (right y-axis, y2)
ax2 = ax1.twinx()
color2 = '#c0392b'
ax2.set_ylabel('Deriv. Weight (%/°C)', color=color2, fontsize=12)
# Multiply DTG by -1 so peaks point up
dtg_plot = y2_data * -1
line2, = ax2.plot(x_data, dtg_plot, color=color2, label='DTG')
ax2.tick_params(axis='y', labelcolor=color2)

# Set title to sheet name
ax1.set_title(sheet_name, fontsize=14)

# Combine legends
lines = [line1, line2]
labels = [l.get_label() for l in lines]
ax1.legend(lines, labels, loc='center left')

# --- Tick Settings ---
# X-axis (Temperature): start from 0, step 200
max_temp = max(x_data)
min_temp = min(x_data)
# User requested temperature starts from 0 with step 200
xticks = np.arange(0, int(max_temp) + 200, 200)
ax1.set_xticks(xticks)
ax1.set_xlim(left=0, right=int(max_temp) + 50) # Add slight padding

# Y1-axis (Weight): step 20
min_w = min(y1_data)
max_w = max(y1_data)
yticks_y1 = np.arange(int(min_w // 20) * 20, int(max_w // 20) * 20 + 20, 20)
ax1.set_yticks(yticks_y1)
ax1.set_ylim(bottom=min_w - 5, top=max_w + 5)

# Y2-axis (DTG inverted): step 0.2
min_dtg_inv = min(dtg_plot)
max_dtg_inv = max(dtg_plot)
# Calculate ticks for the right axis
yticks_y2 = np.arange(np.floor(min_dtg_inv / 0.2) * 0.2, np.ceil(max_dtg_inv / 0.2) * 0.2 + 0.2, 0.2)
ax2.set_yticks(yticks_y2)
ax2.set_ylim(bottom=min_dtg_inv - 0.1, top=max_dtg_inv + 0.1)

# --- Annotations ---

# 1. Peak Label: Highest DTG peak above 150°C
# Find indices where temperature > 150
mask_peak = x_data > 150
if np.any(mask_peak):
    # We look for the highest value in the inverted DTG plot (which corresponds to the most negative original DTG)
    # Since we plotted -DTG, the "highest" visual peak is the max of dtg_plot within mask
    dtg_inverted_subset = dtg_plot[mask_peak]
    temps_subset = x_data[mask_peak]
    
    idx_max = np.argmax(dtg_inverted_subset)
    peak_temp = temps_subset[idx_max]
    peak_val_display = dtg_inverted_subset[idx_max]
    
    # Mark on plot
    ax2.plot(peak_temp, peak_val_display, 'o', color=color2, markersize=8)
    ax2.annotate(f'{peak_temp:.0f} °C', 
                 xy=(peak_temp, peak_val_display), 
                 xytext=(peak_temp + 20, peak_val_display + 0.05),
                 arrowprops=dict(facecolor='black', shrink=0.05),
                 fontsize=10, color='black')

# 2. Residue Label: Weight at temperature nearest 800°C
target_temp = 800
idx_residue = np.argmin(np.abs(x_data - target_temp))
residue_temp = x_data[idx_residue]
residue_weight = y1_data[idx_residue]

# Mark on plot
ax1.plot(residue_temp, residue_weight, 'o', color=color1, markersize=8)
ax1.annotate(f'Residue {residue_weight:.1f}%', 
             xy=(residue_temp, residue_weight), 
             xytext=(residue_temp - 100, residue_weight + 10),
             arrowprops=dict(facecolor='black', shrink=0.05),
             fontsize=10, color='black')

plt.tight_layout()

# Save figure
out_path = os.path.join(OUT_DIR, f"{sheet_name.replace(' ', '_')}.png")
plt.savefig(out_path, dpi=300, bbox_inches='tight')
plt.close()
```
