```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from matplotlib import rcParams

# Set Korean font
rcParams['font.family'] = 'sans-serif'
rcParams['font.sans-serif'] = ['Noto Sans CJK KR', 'Arial Unicode MS', 'SimHei']
rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/17_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read all sheets from the Excel file
xl_file = pd.ExcelFile(DATA_FILE)
sheet_names = xl_file.sheet_names

for sheet_name in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    
    # Identify columns based on typical TGA structure
    # Assuming columns: Time, Temperature, Weight(%), D.W(%/C)
    cols = df.columns.tolist()
    
    x_col = None
    y1_col = None
    y2_col = None
    
    for col in cols:
        col_lower = str(col).lower()
        if 'temperature' in col_lower or 'temp' in col_lower:
            x_col = col
        elif 'weight' in col_lower and '%' in col_lower:
            y1_col = col
        elif 'd.w' in col_lower or 'derivative' in col_lower or 'dtg' in col_lower:
            y2_col = col
            
    # Fallback if names don't match perfectly but positions are standard (B, C, D)
    if x_col is None and len(cols) >= 2:
        x_col = cols[1] # B column usually
    if y1_col is None and len(cols) >= 3:
        y1_col = cols[2] # C column usually
    if y2_col is None and len(cols) >= 4:
        y2_col = cols[3] # D column usually
        
    if x_col is None or y1_col is None or y2_col is None:
        continue
        
    temp = df[x_col].values
    weight = df[y1_col].values
    dtg = df[y2_col].values
    
    # Clean data: remove NaNs
    mask = ~np.isnan(temp) & ~np.isnan(weight) & ~np.isnan(dtg)
    temp = temp[mask]
    weight = weight[mask]
    dtg = dtg[mask]
    
    if len(temp) == 0:
        continue

    fig, ax1 = plt.subplots(figsize=(8, 5))
    
    # Plot Weight (%) on left axis
    color1 = '#1f4e9c'
    ax1.plot(temp, weight, color=color1, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=10)
    ax1.set_ylabel('Weight (%)', color=color1, fontsize=10)
    ax1.tick_params(axis='y', labelcolor=color1)
    
    # Create second y-axis for DTG
    ax2 = ax1.twinx()
    color2 = '#c0392b'
    # Multiply by -1 to make peaks point up
    dtg_inverted = -dtg
    ax2.plot(temp, dtg_inverted, color=color2, label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color=color2, fontsize=10)
    ax2.tick_params(axis='y', labelcolor=color2)
    
    # Title
    ax1.set_title(sheet_name, fontsize=12)
    
    # Find Residue at nearest 800°C
    target_temp = 800
    idx_residue = np.argmin(np.abs(temp - target_temp))
    residue_val = weight[idx_residue]
    residue_temp = temp[idx_residue]
    
    # Mark residue
    ax1.plot(residue_temp, residue_val, 'o', color=color1, markersize=6)
    ax1.annotate(f'Residue {residue_val:.1f}%', 
                 xy=(residue_temp, residue_val), 
                 xytext=(10, -10), textcoords='offset points',
                 color=color1, fontsize=9,
                 arrowprops=dict(arrowstyle='->', color=color1))
                 
    # Find Peak temperature of highest DTG peak above 150°C
    mask_peak = temp > 150
    if np.any(mask_peak):
        temp_filtered = temp[mask_peak]
        dtg_inverted_filtered = dtg_inverted[mask_peak]
        
        idx_max_dtg = np.argmax(dtg_inverted_filtered)
        peak_temp = temp_filtered[idx_max_dtg]
        peak_val = dtg_inverted[idx_max_dtg] # This is the inverted value used for plotting
        
        # Mark peak on DTG curve (ax2)
        ax2.plot(peak_temp, peak_val, 'o', color=color2, markersize=6)
        ax2.annotate(f'{peak_temp:.0f} °C', 
                     xy=(peak_temp, peak_val), 
                     xytext=(-10, 10), textcoords='offset points',
                     color=color2, fontsize=9, ha='right',
                     arrowprops=dict(arrowstyle='->', color=color2))
                     
    # Legend
    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left', bbox_to_anchor=(0.02, 0.5), fontsize=9)
    
    # Grid
    ax1.grid(True, linestyle='--', alpha=0.5)
    
    # Save figure
    safe_sheet_name = "".join([c for c in sheet_name if c.isalnum() or c in (' ', '-', '_')]).rstrip()
    out_path = os.path.join(OUT_DIR, f"{safe_sheet_name}.png")
    
    plt.tight_layout()
    plt.savefig(out_path, dpi=600)
    plt.close(fig)

print("Plots saved.")
```
