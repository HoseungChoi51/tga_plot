```python
import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.font_manager as fm
import numpy as np
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B21_r2/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read the Raw sheet
df = pd.read_excel(DATA_FILE, sheet_name="Raw")

# Extract columns based on preview info
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"
dtg_col = "Deriv. Weight (%/°C)"

temps = df[temp_col].values
weights = df[weight_col].values
dtgs = df[dtg_col].values

# Create figure and axis
fig, y1 = plt.subplots(figsize=(8, 6))

# Plot weight (%) on left y-axis
line1, = y1.plot(temps, weights, color='#1f4e9c', label='Weight (%)')
y1.set_xlabel('Temperature (°C)', fontsize=12)
y1.set_ylabel('Weight (%)', color='#1f4e9c', fontsize=12)
y1.tick_params(axis='y', labelcolor='#1f4e9c')

# Invert x-axis so high temperature is on the left
y1.invert_xaxis()

# Create right y-axis for DTG
y2 = y1.twinx()
# Multiply DTG by -1 to make peaks point up
dtg_negated = dtgs * -1
line2, = y2.plot(temps, dtg_negated, color='#c0392b', label='DTG (%/°C)')
y2.set_ylabel('DTG (%/°C)', color='#c0392b', fontsize=12)
y2.tick_params(axis='y', labelcolor='#c0392b')

# Title from sheet name
fig.suptitle('Raw', fontsize=14)

# Find residue at ~800°C
idx_800 = np.argmin(np.abs(temps - 800))
residue_weight = weights[idx_800]
residue_temp = temps[idx_800]

# Add residue marker and label
y1.plot(residue_temp, residue_weight, 'o', color='#1f4e9c', markersize=6)
y1.annotate(f'Residue {residue_weight:.1f}%', 
            xy=(residue_temp, residue_weight),
            xytext=(residue_temp + 50, residue_weight + 5),
            arrowprops=dict(arrowstyle='->', color='#1f4e9c'),
            fontsize=10, color='#1f4e9c')

# Find peak: highest DTG peak above 150°C
mask_above_150 = temps > 150
if np.any(mask_above_150):
    dtg_for_peak = dtgs[mask_above_150]
    temps_for_peak = temps[mask_above_150]
    # Since we want the most negative DTG (largest mass loss rate), find min of original DTG
    idx_peak_local = np.argmin(dtg_for_peak)
    peak_temp = temps_for_peak[idx_peak_local]
    peak_dtg_val = dtg_for_peak[idx_peak_local]
    
    # Marker on the negated DTG curve
    y2.plot(peak_temp, -peak_dtg_val, 'o', color='#c0392b', markersize=6)
    y2.annotate(f'{peak_temp:.0f} °C', 
                xy=(peak_temp, -peak_dtg_val),
                xytext=(peak_temp + 50, -peak_dtg_val + 0.1),
                arrowprops=dict(arrowstyle='->', color='#c0392b'),
                fontsize=10, color='#c0392b')

# Legend at center left
lines = [line1, line2]
labels = [l.get_label() for l in lines]
y1.legend(lines, labels, loc='center left', bbox_to_anchor=(0.05, 0.5), fontsize=10)

plt.tight_layout()

# Save figure
out_path = os.path.join(OUT_DIR, "Raw.png")
plt.savefig(out_path, dpi=300, bbox_inches='tight')
plt.close()
```
