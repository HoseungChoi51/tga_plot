```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Configure fonts for Korean support
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B23_r1/out"

if not os.path.exists(OUT_DIR):
    os.makedirs(OUT_DIR)

# Read all sheets from the Excel file
xls = pd.ExcelFile(DATA_FILE)
sheet_names = xls.sheet_names

# Define colors as requested
colors = {
    "Raw": "black",
    "Acid-washed": "blue",
    "Annealed": "red"
}

fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx()

# Plot each sheet
for sheet in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    
    # Identify columns based on standard TGA structure provided in preview
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    x = df[temp_col]
    y1 = df[weight_col]
    y2 = df[dtg_col] * -1  # Invert DTG so peaks point up
    
    color = colors.get(sheet, "gray")
    
    # Plot Weight (left axis)
    line1, = ax1.plot(x, y1, label=f"{sheet} Weight", color=color, linestyle='-')
    
    # Plot DTG (right axis)
    line2, = ax2.plot(x, y2, label=f"{sheet} DTG", color=color, linestyle='--', alpha=0.7)

# Set labels and titles
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("DTG (-% / °C)")
ax1.set_title("TGA Analysis of Samples")

# Combine legends from both axes
lines_labels = [ax1.get_legend_handles_labels(), ax2.get_legend_handles_labels()]
lines = [l for ll in lines_labels for l in ll[0]]
labels = [l for ll in lines_labels for l in ll[1]]
ax1.legend(lines, labels, loc='center left')

# Calculate and annotate Residue at ~800°C and Peak Temperature > 150°C for each sample
annotations = []
for sheet in sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    temp_col = "Temperature (°C)"
    weight_col = "Weight (%)"
    dtg_col = "Deriv. Weight (%/°C)"
    
    x = df[temp_col].values
    y1 = df[weight_col].values
    y2_raw = df[dtg_col].values
    
    # Residue at nearest 800°C
    idx_800 = np.argmin(np.abs(x - 800))
    residue_val = y1[idx_800]
    residue_temp = x[idx_800]
    
    # Peak temperature of highest DTG peak above 150°C
    mask = x > 150
    if np.any(mask):
        # Find index of minimum original DTG (which is maximum negative value, i.e., steepest loss)
        # Since we plotted -DTG, the peak is the max of -DTG where x > 150
        y2_plot_vals = y2_raw[mask] * -1
        x_masked = x[mask]
        
        peak_idx_local = np.argmax(y2_plot_vals)
        peak_temp = x_masked[peak_idx_local]
        peak_dtg_val = y2_plot_vals[peak_idx_local]
        
        annotations.append((sheet, residue_temp, residue_val, peak_temp, peak_dtg_val))

# Add text annotations to the plot
for sheet, r_temp, r_val, p_temp, p_dtg in annotations:
    color = colors.get(sheet, "gray")
    # Annotate Residue
    ax1.annotate(f"Residue {r_val:.1f}%\n@{r_temp:.0f}°C", 
                 xy=(r_temp, r_val), 
                 xytext=(r_temp + 20, r_val + 5),
                 arrowprops=dict(facecolor=color, shrink=0.05),
                 fontsize=8, color=color)
    
    # Annotate Peak
    # Need to convert back to right axis coordinates for annotation position if needed, 
    # but usually easier to just place near the top or use ax2 for positioning logic
    # Let's find the corresponding y-value on the right axis for the peak
    ax2.annotate(f"Peak\n{p_temp:.0f}°C", 
                 xy=(p_temp, p_dtg), 
                 xytext=(p_temp + 20, p_dtg + 0.2),
                 arrowprops=dict(facecolor=color, shrink=0.05, linestyle='--'),
                 fontsize=8, color=color)

plt.tight_layout()
output_path = os.path.join(OUT_DIR, "tga_overlay.png")
plt.savefig(output_path, dpi=300)
plt.close()
```
