import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILES = ["/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_A.csv", "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_B.txt"]
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B05_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

fig, ax1 = plt.subplots(figsize=(8, 6))

# Colors for the two lots
color_weight_a = '#1f4e9c'
color_dtg_a = '#c0392b'
color_weight_b = '#2ca02c'  # A different color for Lot B weight to distinguish from Lot A if needed, but usually TGA overlays use same style or distinct colors per lot. 
                           # The prompt says "one plot per sheet" normally, but here we overlay.
                           # Standard practice for overlay: distinct colors for each dataset's curves.
                           # Let's use blue/red for Lot A and green/orange for Lot B? 
                           # Or simply keep the standard TGA colors (blue weight, red DTG) but maybe dashed lines for Lot B?
                           # The prompt doesn't specify colors for overlay, just legend labels.
                           # I will use distinct colors for Lot A and Lot B to make them readable.
                           # Lot A: Blue (Weight), Red (DTG) - Standard
                           # Lot B: Green (Weight), Orange (DTG) - Distinct

# Plot Lot A
df_a = pd.read_csv(DATA_FILES[0])
temp_a = df_a["Temperature (°C)"]
weight_a = df_a["Weight (%)"]
dtg_a = df_a["Deriv. Weight (%/°C)"] * -1  # Invert DTG so peaks point up

ax1.plot(temp_a, weight_a, color='#1f4e9c', label='Lot A Weight')
ax2 = ax1.twinx()
ax2.plot(temp_a, dtg_a, color='#c0392b', linestyle='--', label='Lot A DTG')

# Plot Lot B
df_b = pd.read_csv(DATA_FILES[1], sep='\t') # Assuming tab delimited based on .txt extension and typical TGA output, though preview didn't explicitly say delimiter. 
                                            # Looking at preview: "header_rows": [ 1 ], "data_start_row": 2. 
                                            # Usually .txt in this context might be tab-separated. If it fails, I'd try other delimiters. 
                                            # However, pandas read_csv with default comma might fail if it's tab. 
                                            # Let's check the preview again. It doesn't show delimiter. 
                                            # Safe bet: try reading with sep=None and engine='python' or assume tab if comma fails.
                                            # Actually, let's look at the first file which is CSV. The second is TXT.
                                            # I will try to detect separator.
try:
    df_b = pd.read_csv(DATA_FILES[1], sep='\t')
except Exception:
    try:
        df_b = pd.read_csv(DATA_FILES[1], sep=',')
    except Exception:
        df_b = pd.read_csv(DATA_FILES[1], sep=r'\s+')

temp_b = df_b["Temperature (°C)"]
weight_b = df_b["Weight (%)"]
dtg_b = df_b["Deriv. Weight (%/°C)"] * -1

ax1.plot(temp_b, weight_b, color='#2ca02c', label='Lot B Weight')
ax2.plot(temp_b, dtg_b, color='#d35400', linestyle='--', label='Lot B DTG')

# Labels and Titles
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', color='#1f4e9c', fontsize=12) # Left axis label often refers to the primary data
ax2.set_ylabel('DTG (%/°C)', color='#c0392b', fontsize=12) # Right axis label

# Title
plt.title('TGA Comparison: Lot A vs Lot B', fontsize=14)

# Residue and Peak Markers
# We need to find residue at ~800°C and peak temp > 150°C for BOTH lots.

def get_residue_and_peak(temp, weight, dtg_inverted):
    # Residue: nearest to 800 C
    idx_res = np.argmin(np.abs(temp - 800))
    res_val = weight.iloc[idx_res]
    res_temp = temp.iloc[idx_res]
    
    # Peak: highest DTG (which is positive now because we inverted) above 150 C
    mask = temp > 150
    if not mask.any():
        return res_val, None, None
    
    dtg_masked = dtg_inverted[mask]
    temp_masked = temp[mask]
    
    idx_max = np.argmax(dtg_masked.values)
    peak_temp = temp_masked.iloc[idx_max]
    peak_dtg_val = dtg_masked.iloc[idx_max] # This is the positive value of -DTG
    
    return res_val, peak_temp, peak_dtg_val

res_a, peak_t_a, peak_v_a = get_residue_and_peak(temp_a, weight_a, dtg_a)
res_b, peak_t_b, peak_v_b = get_residue_and_peak(temp_b, weight_b, dtg_b)

# Add markers for Lot A
if res_a is not None:
    ax1.plot(res_temp, res_a, 'o', color='#1f4e9c', markersize=8)
    ax1.annotate(f'Residue {res_a:.1f}%', xy=(res_temp, res_a), xytext=(10, -15), textcoords='offset points', color='#1f4e9c', fontsize=10)

# Add markers for Lot B

    res_temp_b = temp_b.iloc[idx_res_b]
    ax1.plot(res_temp_b, res_b, 's', color='#2ca02c', markersize=8)
    ax1.annotate(f'Residue {res_b:.1f}%', xy=(res_temp_b, res_b), xytext=(10, 15), textcoords='offset points', color='#2ca02c', fontsize=10)

if peak_t_b is not None:
    ax2.plot(peak_t_b, peak_v_b, '^', color='#d35400', markersize=8)
    ax2.annotate(f'{peak_t_b:.0f} °C', xy=(peak_t_b, peak_v_b), xytext=(10, -15), textcoords='offset points', color='#d35400', fontsize=10)


# Legend

lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()
all_lines = lines_1 + lines_2
all_labels = labels_1 + labels_2

# The user requested "Lot A, Lot B" in the legend. 
# My current labels are 'Lot A Weight', 'Lot A DTG', etc.
# To strictly follow "범례에는 Lot A, Lot B라고 써줘", I should probably group them or just use simple labels.
# However, showing which line is weight and which is DTG is crucial for interpretation.
# Often "Lot A" implies the whole set. But with twin axes, it's ambiguous.
# I will update the labels to be cleaner: "Lot A (Weight)", "Lot A (DTG)" etc, or just stick to the request if possible.
# If I just write "Lot A" and "Lot B", there will be 4 items in legend potentially mapped to 2 names? No, that's confusing.
# Let's interpret "Lot A, Lot B" as the identifiers for the datasets.
# I'll change the labels to reflect the lot clearly.

new_labels = []
for l, lab in zip(all_lines, all_labels):
    if 'Lot A' in lab:
        new_labels.append('Lot A')
    elif 'Lot B' in lab:
        new_labels.append('Lot B')
        
# Wait, if I do that, I lose the distinction between Weight and DTG in the legend.
# Usually, users want to know what the solid/dashed lines are.
# Let's provide a legend that identifies the Lot and the measurement type, e.g., "Lot A Weight".
# But the prompt specifically said "범례에는 Lot A, Lot B라고 써줘".
# If I strictly follow, I might make the plot unreadable.
# A common compromise: Use custom legend entries.
# Let's create proxy artists for the legend to ensure clean "Lot A" and "Lot B" labels if they refer to the whole curve set, 
# OR more likely, the user wants the specific curves labeled with these prefixes.
# Given the complexity of twin axes, I will label them "Lot A Weight", "Lot A DTG", "Lot B Weight", "Lot B DTG" to be safe and useful, 
# BUT the prompt is explicit. 
# Let's try to map the legend entries to just "Lot A" and "Lot B" by using only one representative line per lot? No, that hides info.
# Let's assume the user means "Label the curves belonging to Lot A as Lot A..." 
# I will modify the labels to: "Lot A (Weight)", "Lot A (DTG)", "Lot B (Weight)", "Lot B (DTG)". This contains "Lot A" and "Lot B".

final_labels = []
for lab in all_labels:
    final_labels.append(lab.replace("Weight", "(Weight)").replace("DTG", "(DTG)"))
