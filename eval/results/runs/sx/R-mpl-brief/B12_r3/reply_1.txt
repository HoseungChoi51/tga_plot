```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'
plt.rcParams['axes.unicode_minus'] = False

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B12_r3/out"

os.makedirs(OUT_DIR, exist_ok=True)

# Read data
df = pd.read_csv(DATA_FILE)

# Identify columns based on preview
temp_col = "Temperature (°C)"
weight_col = "Weight (%)"

# Filter data for x range 100 to 900
mask = (df[temp_col] >= 100) & (df[temp_col] <= 900)
df_plot = df[mask].copy()

# Create plot
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot weight curve
ax1.plot(df_plot[temp_col], df_plot[weight_col], color='#1f4e9c', linewidth=1.5, label='Weight (%)')

# Set x axis range and labels
ax1.set_xlim(100, 900)
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12)

# Find residue at nearest point to 900 °C within the plotted range
target_temp = 900
idx_residue = np.argmin(np.abs(df_plot[temp_col] - target_temp))
residue_temp = df_plot.iloc[idx_residue][temp_col]
residue_weight = df_plot.iloc[idx_residue][weight_col]

# Mark residue point
ax1.plot(residue_temp, residue_weight, 'o', color='#1f4e9c', markersize=6)
label_text = f'Residue {residue_weight:.1f}%'
ax1.annotate(label_text, xy=(residue_temp, residue_weight), 
             xytext=(10, -15), textcoords='offset points',
             fontsize=10, color='#1f4e9c',
             arrowprops=dict(arrowstyle='->', color='#1f4e9c'))

# Title from sheet name (file name without extension)
sheet_name = os.path.splitext(os.path.basename(DATA_FILE))[0]
ax1.set_title(sheet_name, fontsize=14)

# Legend
ax1.legend(loc='center left', fontsize=10)

# Grid
ax1.grid(True, linestyle='--', alpha=0.5)

plt.tight_layout()

# Save figure
out_path = os.path.join(OUT_DIR, f"{sheet_name}.png")
plt.savefig(out_path, dpi=300)
plt.close()
```
