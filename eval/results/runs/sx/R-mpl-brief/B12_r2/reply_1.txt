```python
import pandas as pd
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/B12_r2/out"

# Read data
df = pd.read_csv(DATA_FILE)

# Extract columns based on the preview info
# x: Temperature (°C), y1: Weight (%)
x_col = "Temperature (°C)"
y1_col = "Weight (%)"

# Filter data for x range 100 to 900
mask = (df[x_col] >= 100) & (df[x_col] <= 900)
df_filtered = df[mask].copy()

# Find residue at 900 °C (nearest point to 900 within the filtered data or original if needed, but request says label at 900)
# We'll find the point in the original data closest to 900 °C to get the accurate residue value and position
idx_residue = (df[x_col] - 900).abs().idxmin()
residue_temp = df.loc[idx_residue, x_col]
residue_weight = df.loc[idx_residue, y1_col]

# Create plot
fig, ax1 = plt.subplots(figsize=(8, 6))

# Plot weight curve
ax1.plot(df_filtered[x_col], df_filtered[y1_col], color='#1f4e9c', label='Weight (%)')

# Set axis labels and title
ax1.set_xlabel('Temperature (°C)', fontsize=12)
ax1.set_ylabel('Weight (%)', fontsize=12)
ax1.set_title('caox', fontsize=14) # Using sheet name from preview

# Set x-axis range
ax1.set_xlim(100, 900)

# Add residue marker and label
ax1.plot(residue_temp, residue_weight, 'o', color='black', markersize=5)
ax1.annotate(f'Residue {residue_weight:.1f}%', 
             xy=(residue_temp, residue_weight), 
             xytext=(10, 10), textcoords='offset points',
             fontsize=10, ha='left', va='bottom',
             bbox=dict(boxstyle="round,pad=0.3", fc="yellow", ec="k", lw=1, alpha=0.8))

# Legend
ax1.legend(loc='center left', fontsize=10)

# Grid
ax1.grid(True, linestyle='--', alpha=0.5)

# Save figure
os.makedirs(OUT_DIR, exist_ok=True)
output_path = os.path.join(OUT_DIR, "caox.png")
plt.tight_layout()
plt.savefig(output_path, dpi=300)
plt.close()
```
