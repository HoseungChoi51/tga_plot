```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# Set Korean font
plt.rcParams['font.family'] = 'Noto Sans CJK KR'

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/sx/R-mpl-brief/23_r3/out"

def read_tga_data(filepath):
    """Read TGA data from text file with no header, skipping first 10 rows."""
    # Use sep=r'\s+' to handle whitespace separated values robustly
    try:
        df = pd.read_csv(filepath, sep=r'\s+', header=None, skiprows=10)
    except Exception:
        # Fallback if regex fails or different whitespace
        df = pd.read_csv(filepath, sep=None, engine='python', header=None, skiprows=10)
    
    # Assign column names based on preview: A=time/index, B=temp, C=weight, D=DTG
    # We expect at least 4 columns. If more, we take the first 4 relevant ones based on standard structure.
    if df.shape[1] >= 4:
        df.columns = ['time', 'temperature', 'weight', 'dtg'] + [f'extra_{i}' for i in range(df.shape[1]-4)]
        df = df[['time', 'temperature', 'weight', 'dtg']]
    elif df.shape[1] == 3:
        # Assume Temp, Weight, DTG if Time is missing
        df.columns = ['temperature', 'weight', 'dtg']
    else:
        raise ValueError(f"Unexpected number of columns: {df.shape[1]}")
        
    return df

def plot_tga(df, sheet_name="instrument_export"):
    fig, ax1 = plt.subplots(figsize=(8, 6))
    
    # Plot Weight (%) on left y-axis
    color1 = '#1f4e9c'
    ax1.plot(df['temperature'], df['weight'], color=color1, label='Weight (%)')
    ax1.set_xlabel('Temperature (°C)', fontsize=12)
    ax1.set_ylabel('Weight (%)', color=color1, fontsize=12)
    ax1.tick_params(axis='y', labelcolor=color1)
    
    # Create twin axis for DTG
    ax2 = ax1.twinx()
    color2 = '#c0392b'
    # Multiply DTG by -1 so peaks point up
    dtg_inverted = df['dtg'] * -1
    ax2.plot(df['temperature'], dtg_inverted, color=color2, linestyle='--', label='DTG (%/°C)')
    ax2.set_ylabel('DTG (%/°C)', color=color2, fontsize=12)
    ax2.tick_params(axis='y', labelcolor=color2)
    
    # Find residue at ~800°C
    target_temp = 800.0
    idx_residue = np.argmin(np.abs(df['temperature'] - target_temp))
    residue_weight = df.iloc[idx_residue]['weight']
    residue_temp = df.iloc[idx_residue]['temperature']
    
    # Mark residue on the weight curve
    ax1.plot(residue_temp, residue_weight, 'o', color=color1, markersize=8)
    # Custom format string requested: 'Residue: {}%'
    residue_label = f'Residue: {residue_weight:.1f}%'
    ax1.annotate(residue_label, xy=(residue_temp, residue_weight), 
                 xytext=(residue_temp + 50, residue_weight + 5),
                 arrowprops=dict(facecolor='black', shrink=0.05),
                 fontsize=10, color=color1)
    
    # Find peak temperature above 150°C from inverted DTG
    mask = df['temperature'] > 150
    if not mask.empty:
        max_dtg_idx = df.loc[mask, 'dtg'].idxmin()  # min of original dtg is max of inverted
        peak_temp = df.loc[max_dtg_idx, 'temperature']
        peak_val_inv = df.loc[max_dtg_idx, 'dtg'] * -1
        
        # Mark peak on the DTG curve
        ax2.plot(peak_temp, peak_val_inv, 'v', color=color2, markersize=8)
        ax2.annotate(f'{peak_temp:.0f} °C', xy=(peak_temp, peak_val_inv),
                     xytext=(peak_temp - 100, peak_val_inv + 0.05),
                     arrowprops=dict(facecolor='black', shrink=0.05),
                     fontsize=10, color=color2)
    
    # Title and Legend
    ax1.set_title(sheet_name, fontsize=14)
    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(lines1 + lines2, labels1 + labels2, loc='center left', bbox_to_anchor=(0.05, 0.5))
    
    # Grid
    ax1.grid(True, linestyle=':', alpha=0.7)
    
    # Save figure
    os.makedirs(OUT_DIR, exist_ok=True)
    out_path = os.path.join(OUT_DIR, "instrument_export.png")
    plt.tight_layout()
    plt.savefig(out_path, dpi=300)
    plt.close(fig)

if __name__ == "__main__":
    try:
        df = read_tga_data(DATA_FILE)
        plot_tga(df, sheet_name="instrument_export")
    except Exception as e:
        import traceback
        print(f"Error processing file: {e}")
        traceback.print_exc()
```
