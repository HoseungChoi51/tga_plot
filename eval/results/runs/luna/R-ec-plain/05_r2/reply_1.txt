```python
import json
import re
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"


def clean_name(value):
    return re.sub(r"\s+", " ", str(value)).strip().lower()


def find_column(columns, candidates):
    normalized = {clean_name(col): col for col in columns}
    for candidate in candidates:
        candidate = clean_name(candidate)
        for key, original in normalized.items():
            if candidate == key or candidate in key:
                return original
    return None


def sheet_option(sheet_name, frame):
    x_col = find_column(frame.columns, ["Temperature", "Temp"])
    y1_col = find_column(frame.columns, ["Weight(%)", "Weight"])
    y2_col = find_column(frame.columns, ["D.W(%/C)", "Derivative", "D.W"])

    if x_col is None or y1_col is None or y2_col is None:
        raise ValueError(f"필요한 TGA 열을 찾을 수 없습니다: {sheet_name}")

    data = frame[[x_col, y1_col, y2_col]].copy()
    for col in data.columns:
        data[col] = pd.to_numeric(data[col], errors="coerce")
    data = data.replace([np.inf, -np.inf], np.nan).dropna()

    x_values = data[x_col].to_numpy(dtype=float)
    y1_values = data[y1_col].to_numpy(dtype=float)
    y2_values = data[y2_col].to_numpy(dtype=float)

    return {
        "title": {
            "text": sheet_name,
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
        },
        "tooltip": {"trigger": "axis"},
        "legend": {"top": 32},
        "grid": {"left": 76, "right": 76, "top": 82, "bottom": 66},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 42,
            "min": float(np.nanmin(x_values)),
            "max": float(np.nanmax(x_values)),
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 52,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            },
            {
                "type": "value",
                "name": "D.W (%/°C)",
                "nameLocation": "middle",
                "nameGap": 52,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            },
        ],
        "series": [
            {
                "name": str(y1_col),
                "type": "line",
                "yAxisIndex": 0,
                "showSymbol": False,
                "lineStyle": {"color": "#000000", "width": 2.5},
                "data": [[float(x), float(y)] for x, y in zip(x_values, y1_values)],
            },
            {
                "name": str(y2_col),
                "type": "line",
                "yAxisIndex": 1,
                "showSymbol": False,
                "lineStyle": {"type": "dashed"},
                "data": [[float(x), float(y)] for x, y in zip(x_values, y2_values)],
            },
        ],
    }


def main():
    workbook = pd.ExcelFile(DATA_FILE)
    figures = []
    for sheet_name in workbook.sheet_names:
        frame = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        figures.append(
            {
                "name": re.sub(r"[^A-Za-z0-9_-]+", "_", sheet_name).strip("_") or "plot",
                "option": sheet_option(sheet_name, frame),
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        )
    json.dump({"figures": figures}, sys.stdout, ensure_ascii=False, allow_nan=False)


if __name__ == "__main__":
    main()
```
