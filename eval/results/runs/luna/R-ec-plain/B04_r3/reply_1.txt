```python
import json
import os

import numpy as np
import pandas as pd
from scipy.signal import find_peaks


DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/xrd.csv"

df = pd.read_csv(DATA_FILE)
x = pd.to_numeric(df.iloc[:, 0], errors="coerce").to_numpy()
y = pd.to_numeric(df.iloc[:, 1], errors="coerce").to_numpy()

valid = np.isfinite(x) & np.isfinite(y)
x, y = x[valid], y[valid]
order = np.argsort(x)
x, y = x[order], y[order]

# Select the five most intense local maxima; fall back to the strongest
# measured points if the data contain fewer than five local maxima.
peak_indices, _ = find_peaks(y)
if len(peak_indices) < 5:
    peak_indices = np.arange(len(y))
strongest = peak_indices[np.argsort(y[peak_indices])[-5:]]
strongest = strongest[np.argsort(x[strongest])]

peak_data = [
    {
        "value": [float(x[i]), float(y[i])],
        "symbolSize": 8,
        "itemStyle": {"color": "#d62728"},
        "label": {
            "show": True,
            "position": "top",
            "formatter": f"{x[i]:.2f}°",
            "fontSize": 11,
            "color": "#b22222",
            "backgroundColor": "rgba(255,255,255,0.85)",
            "padding": [2, 3],
        },
    }
    for i in strongest
]

option = {
    "animation": False,
    "tooltip": {"trigger": "axis"},
    "grid": {"left": 72, "right": 28, "top": 52, "bottom": 66},
    "xAxis": {
        "type": "value",
        "name": "2θ (deg)",
        "nameLocation": "middle",
        "nameGap": 38,
        "min": float(np.min(x)),
        "max": float(np.max(x)),
        "axisLine": {"lineStyle": {"color": "#444"}},
        "axisLabel": {"color": "#333"},
        "splitLine": {"show": False},
    },
    "yAxis": {
        "type": "value",
        "name": "Intensity (counts)",
        "nameLocation": "middle",
        "nameGap": 52,
        "min": 0,
        "axisLine": {"show": True, "lineStyle": {"color": "#444"}},
        "axisLabel": {"color": "#333"},
        "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
    },
    "series": [
        {
            "name": "XRD pattern",
            "type": "line",
            "data": [[float(a), float(b)] for a, b in zip(x, y)],
            "showSymbol": False,
            "lineStyle": {"color": "#245a8d", "width": 1.5},
            "emphasis": {"disabled": True},
        },
        {
            "name": "Strong peaks",
            "type": "scatter",
            "data": peak_data,
            "symbol": "circle",
            "z": 3,
        },
    ],
}

print(
    json.dumps(
        {
            "figures": [
                {
                    "name": "xrd_pattern",
                    "option": option,
                    "width": 614,
                    "height": 442,
                    "dpi": 300,
                    "formats": ["png"],
                }
            ]
        },
        ensure_ascii=False,
        allow_nan=False,
    )
)
```
