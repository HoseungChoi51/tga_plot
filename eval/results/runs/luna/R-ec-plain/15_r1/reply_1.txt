```python
import json
import os

import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

# Load all sheets and build overlay series for weight and derivative weight.
sheets = pd.read_excel(DATA_FILE, sheet_name=None)
palette = ["#2563EB", "#DC2626", "#059669", "#9333EA", "#D97706", "#0891B2"]

series = []
for idx, (sheet_name, df) in enumerate(sheets.items()):
    df.columns = [str(c).strip() for c in df.columns]
    temperature_col = next((c for c in df.columns if "temperature" in c.lower()), None)
    weight_col = next((c for c in df.columns if "weight" in c.lower() and "d.w" not in c.lower()), None)
    derivative_col = next((c for c in df.columns if "d.w" in c.lower()), None)

    if not temperature_col or not weight_col:
        continue

    color = palette[idx % len(palette)]
    x = pd.to_numeric(df[temperature_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    valid = x.notna() & weight.notna()

    series.append({
        "name": f"{sheet_name} Weight",
        "type": "line",
        "xAxisIndex": 0,
        "yAxisIndex": 0,
        "data": [[float(a), float(b)] for a, b in zip(x[valid], weight[valid])],
        "showSymbol": False,
        "symbol": "none",
        "lineStyle": {"width": 2, "color": color},
        "itemStyle": {"color": color},
        "emphasis": {"focus": "series"},
    })

    if derivative_col:
        derivative = pd.to_numeric(df[derivative_col], errors="coerce")
        valid_d = x.notna() & derivative.notna()
        series.append({
            "name": f"{sheet_name} D.W",
            "type": "line",
            "xAxisIndex": 0,
            "yAxisIndex": 1,
            "data": [[float(a), float(b)] for a, b in zip(x[valid_d], derivative[valid_d])],
            "showSymbol": False,
            "symbol": "none",
            "lineStyle": {"width": 1.6, "type": "dashed", "color": color, "opacity": 0.8},
            "itemStyle": {"color": color},
            "emphasis": {"focus": "series"},
        })

option = {
    "animation": False,
    "title": {
        "text": "TGA 비교",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 18},
    },
    "tooltip": {"trigger": "axis", "axisPointer": {"type": "cross"}},
    "legend": {
        "type": "scroll",
        "top": 32,
        "left": 72,
        "right": 72,
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 10},
    },
    "grid": {"left": 78, "right": 82, "top": 100, "bottom": 66},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 36,
        "min": 30,
        "max": 900,
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 52,
            "min": 0,
            "max": 105,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        {
            "type": "value",
            "name": "D.W (%/°C)",
            "nameLocation": "middle",
            "nameGap": 56,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": False},
        },
    ],
    "series": series,
}

print(json.dumps({
    "figures": [{
        "name": "tga_all_sheets_overlay",
        "option": option,
        "width": 1100,
        "height": 700,
        "dpi": 300,
        "formats": ["png"],
    }]
}, ensure_ascii=False, allow_nan=False))
```
