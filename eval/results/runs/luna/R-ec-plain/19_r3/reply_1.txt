```python
import json
import re
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

TEXT_COLOR = "#808080"
FONT_SIZE = 13


def read_sheets(path):
    workbook = pd.ExcelFile(path)
    sheets = []
    for sheet_name in workbook.sheet_names:
        frame = pd.read_excel(path, sheet_name=sheet_name)
        frame.columns = [str(c).strip() for c in frame.columns]
        sheets.append((sheet_name, frame))
    return sheets


def numeric_column(frame, patterns):
    for pattern in patterns:
        for column in frame.columns:
            if re.search(pattern, str(column), re.IGNORECASE):
                values = pd.to_numeric(frame[column], errors="coerce")
                if values.notna().sum() > 1:
                    return column, values
    return None, None


def clean_xy(x_values, y_values):
    data = pd.DataFrame({"x": x_values, "y": y_values}).dropna()
    data = data.replace([np.inf, -np.inf], np.nan).dropna()
    return [[float(x), float(y)] for x, y in data[["x", "y"]].to_numpy()]


def make_option(sheet_name, frame):
    x_col, x = numeric_column(frame, [r"temperature", r"temp"])
    if x is None:
        x_col, x = numeric_column(frame, [r"time"])
    if x is None:
        numeric = [c for c in frame.columns if pd.to_numeric(frame[c], errors="coerce").notna().sum() > 1]
        if len(numeric) < 2:
            return None
        x_col = numeric[0]
        x = pd.to_numeric(frame[x_col], errors="coerce")

    weight_col, weight = numeric_column(frame, [r"weight"])
    derivative_col, derivative = numeric_column(frame, [r"d\.?\s*w", r"derivative", r"dtg"])

    if weight is None:
        numeric = [c for c in frame.columns if c != x_col and pd.to_numeric(frame[c], errors="coerce").notna().sum() > 1]
        if not numeric:
            return None
        weight_col = numeric[0]
        weight = pd.to_numeric(frame[weight_col], errors="coerce")

    series = [{
        "name": str(weight_col),
        "type": "line",
        "showSymbol": False,
        "smooth": False,
        "lineStyle": {"width": 2},
        "data": clean_xy(x, weight),
    }]

    if derivative is not None:
        series.append({
            "name": str(derivative_col),
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "smooth": False,
            "lineStyle": {"width": 2},
            "data": clean_xy(x, derivative),
        })

    option = {
        "animation": False,
        "backgroundColor": "#ffffff",
        "textStyle": {
            "color": TEXT_COLOR,
            "fontSize": FONT_SIZE,
            "fontFamily": "Noto Sans CJK KR",
        },
        "title": {
            "text": str(sheet_name),
            "left": "center",
            "top": 10,
            "textStyle": {"color": TEXT_COLOR, "fontSize": FONT_SIZE, "fontWeight": "normal"},
        },
        "tooltip": {"trigger": "axis"},
        "legend": {
            "top": 38,
            "textStyle": {"color": TEXT_COLOR, "fontSize": FONT_SIZE},
        },
        "grid": {"left": 78, "right": 78 if derivative is not None else 35, "top": 86, "bottom": 65},
        "xAxis": {
            "type": "value",
            "name": str(x_col),
            "nameLocation": "middle",
            "nameGap": 38,
            "axisLine": {"lineStyle": {"color": TEXT_COLOR}},
            "axisLabel": {"color": TEXT_COLOR, "fontSize": FONT_SIZE},
            "nameTextStyle": {"color": TEXT_COLOR, "fontSize": FONT_SIZE},
            "splitLine": {"lineStyle": {"color": "#e6e6e6"}},
        },
        "yAxis": [
            {
                "type": "value",
                "name": str(weight_col),
                "nameLocation": "middle",
                "nameGap": 52,
                "axisLine": {"show": True, "lineStyle": {"color": TEXT_COLOR}},
                "axisLabel": {"color": TEXT_COLOR, "fontSize": FONT_SIZE},
                "nameTextStyle": {"color": TEXT_COLOR, "fontSize": FONT_SIZE},
                "splitLine": {"lineStyle": {"color": "#e6e6e6"}},
            }
        ],
        "series": series,
    }

    if derivative is not None:
        option["yAxis"].append({
            "type": "value",
            "name": str(derivative_col),
            "nameLocation": "middle",
            "nameGap": 50,
            "axisLine": {"show": True, "lineStyle": {"color": TEXT_COLOR}},
            "axisLabel": {"color": TEXT_COLOR, "fontSize": FONT_SIZE},
            "nameTextStyle": {"color": TEXT_COLOR, "fontSize": FONT_SIZE},
            "splitLine": {"show": False},
        })

    return option


def main():
    figures = []
    for sheet_name, frame in read_sheets(DATA_FILE):
        option = make_option(sheet_name, frame)
        if option is None:
            continue
        safe_name = re.sub(r"[^A-Za-z0-9_-]+", "_", sheet_name).strip("_") or f"plot{len(figures) + 1}"
        figures.append({
            "name": safe_name,
            "option": option,
            "width": 614,
            "height": 442,
            "dpi": 300,
            "formats": ["png"],
        })
    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
