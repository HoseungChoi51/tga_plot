```python
import json
import re
from pathlib import Path

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt"

# The instrument export has metadata before the four-column numeric data table.
# Locate the first numeric row and read the remaining whitespace/comma-delimited data.
path = Path(DATA_FILE)
lines = path.read_text(encoding="utf-8", errors="replace").splitlines()

start = None
for i, line in enumerate(lines):
    parts = re.split(r"[\s,;\t]+", line.strip())
    if len(parts) >= 4:
        try:
            [float(v) for v in parts[:4]]
            start = i
            break
        except ValueError:
            continue

if start is None:
    raise ValueError(f"No four-column numeric data found in {path}")

rows = []
for line in lines[start:]:
    parts = re.split(r"[\s,;\t]+", line.strip())
    if len(parts) < 4:
        continue
    try:
        rows.append([float(v) for v in parts[:4]])
    except ValueError:
        continue

data = np.asarray(rows, dtype=float)
if data.ndim != 2 or data.shape[1] < 4 or len(data) == 0:
    raise ValueError(f"Could not parse numeric data from {path}")

# Columns: elapsed time (min), temperature (°C), mass (%), derivative.
temperature = data[:, 1]
mass = data[:, 2]

# Keep the mass curve in acquisition order and use the final measured mass as residue.
residue = float(mass[-1])
residue_text = f"Residue: {residue:.2f}%"

option = {
    "animation": False,
    "backgroundColor": "#ffffff",
    "textStyle": {
        "fontFamily": "Noto Sans CJK KR, sans-serif",
        "color": "#222222",
    },
    "title": {
        "text": "TGA",
        "left": "center",
        "top": 12,
        "textStyle": {"fontSize": 17, "fontWeight": "normal"},
    },
    "grid": {"left": 74, "right": 34, "top": 62, "bottom": 68},
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "line"},
        "valueFormatter": "{value}",
    },
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 42,
        "min": float(np.nanmin(temperature)),
        "max": float(np.nanmax(temperature)),
        "axisLine": {"lineStyle": {"color": "#555555"}},
        "axisLabel": {"color": "#444444"},
        "splitLine": {"show": False},
    },
    "yAxis": {
        "type": "value",
        "name": "Mass (%)",
        "nameLocation": "middle",
        "nameGap": 52,
        "min": 0,
        "max": 105,
        "axisLine": {"lineStyle": {"color": "#555555"}},
        "axisLabel": {"color": "#444444"},
        "splitLine": {"lineStyle": {"color": "#e5e5e5"}},
    },
    "graphic": [
        {
            "type": "text",
            "right": 36,
            "top": 38,
            "style": {
                "text": residue_text,
                "font": "14px Noto Sans CJK KR",
                "fill": "#333333",
            },
        }
    ],
    "series": [
        {
            "name": "Mass",
            "type": "line",
            "showSymbol": False,
            "smooth": False,
            "lineStyle": {"width": 2, "color": "#2166ac"},
            "data": [
                [round(float(x), 6), round(float(y), 6)]
                for x, y in zip(temperature, mass)
                if np.isfinite(x) and np.isfinite(y)
            ],
        }
    ],
}

print(
    json.dumps(
        {
            "figures": [
                {
                    "name": "tga",
                    "option": option,
                    "width": 614,
                    "height": 442,
                    "dpi": 300,
                    "formats": ["png"],
                }
            ]
        },
        ensure_ascii=False,
        allow_nan=False,
    )
)
```
