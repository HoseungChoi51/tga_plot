```python
import json
import re
import sys

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"


def clean_sheet_name(name):
    name = re.sub(r"[^A-Za-z0-9가-힣_-]+", "_", str(name)).strip("_")
    return name or "plot"


def main():
    workbook = pd.ExcelFile(DATA_FILE)
    figures = []

    for sheet in workbook.sheet_names:
        df = pd.read_excel(workbook, sheet_name=sheet)
        df.columns = [str(col).strip() for col in df.columns]

        temperature_col = next(
            (col for col in df.columns if "temperature" in col.lower()),
            df.columns[1],
        )
        weight_col = next(
            (col for col in df.columns if "weight" in col.lower()),
            df.columns[2],
        )
        derivative_col = next(
            (col for col in df.columns if "d.w" in col.lower() or "dw" in col.lower()),
            df.columns[3],
        )

        data = df[[temperature_col, weight_col, derivative_col]].apply(
            pd.to_numeric, errors="coerce"
        ).dropna()

        x = data[temperature_col].to_numpy(dtype=float)
        weight = data[weight_col].to_numpy(dtype=float)
        derivative = data[derivative_col].to_numpy(dtype=float)

        option = {
            "animation": False,
            "title": {
                "text": str(sheet),
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
            },
            "tooltip": {"trigger": "axis"},
            "legend": {
                "data": [weight_col, derivative_col],
                "top": 32,
                "textStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
            "grid": {"left": 74, "right": 76, "top": 82, "bottom": 64},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 38,
                "min": float(np.nanmin(x)),
                "max": float(np.nanmax(x)),
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                },
                {
                    "type": "value",
                    "name": "D.W. (%/°C)",
                    "nameLocation": "middle",
                    "nameGap": 54,
                    "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
                },
            ],
            "series": [
                {
                    "name": weight_col,
                    "type": "line",
                    "yAxisIndex": 0,
                    "showSymbol": False,
                    "data": [[float(a), float(b)] for a, b in zip(x, weight)],
                    "lineStyle": {"color": "#000000", "width": 2.5},
                    "itemStyle": {"color": "#000000"},
                },
                {
                    "name": derivative_col,
                    "type": "line",
                    "yAxisIndex": 1,
                    "showSymbol": False,
                    "data": [[float(a), float(b)] for a, b in zip(x, derivative)],
                    "lineStyle": {"type": "dashed", "width": 1.5},
                    "itemStyle": {"color": "#d62728"},
                },
            ],
        }

        figures.append(
            {
                "name": clean_sheet_name(sheet),
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        )

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
