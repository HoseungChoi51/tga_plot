import json
import re
from pathlib import Path

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/instrument_export.txt"

def read_data(path):
    try:
        df = pd.read_csv(path, sep=None, engine="python", header=None, comment="#")
        if df.shape[1] < 2:
            raise ValueError("Not enough columns")
        return df
    except Exception:
        return pd.read_csv(path, delim_whitespace=True, header=None, comment="#")

df = read_data(DATA_FILE)

# Instrument export starts data at row 11 (1-based); select numeric rows.
numeric = df.apply(pd.to_numeric, errors="coerce")
numeric = numeric.dropna(how="all").reset_index(drop=True)
if numeric.shape[1] < 3:
    raise ValueError("Expected at least three numeric columns")

# Columns: elapsed time (A), temperature (B), mass/residue (C), derivative (D).
# The preview indicates column B is the temperature axis (30–900), C is mass percent.
x = numeric.iloc[:, 1].to_numpy(dtype=float)
y = numeric.iloc[:, 2].to_numpy(dtype=float)
valid = np.isfinite(x) & np.isfinite(y)
x, y = x[valid], y[valid]

# Find final residue from the final finite mass value.
residue = float(y[-1]) if len(y) else 0.0
residue_text = f"Residue: {residue:.2f}%"

option = {
    "animation": False,
    "textStyle": {"fontFamily": "Noto Sans CJK KR"},
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "cross"},
        "valueFormatter": "{value}"
    },
    "grid": {"left": 78, "right": 32, "top": 42, "bottom": 68},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 38,
        "axisLine": {"show": True},
        "splitLine": {"show": False}
    },
    "yAxis": {
        "type": "value",
        "name": "Mass (%)",
        "nameLocation": "middle",
        "nameGap": 52,
        "min": 0,
        "max": 105,
        "axisLine": {"show": True},
        "splitLine": {"show": True, "lineStyle": {"color": "#e6e6e6"}}
    },
    "series": [
        {
            "name": "Mass",
            "type": "line",
            "showSymbol": False,
            "lineStyle": {"width": 2, "color": "#276FBF"},
            "data": [[float(a), float(b)] for a, b in zip(x, y)]
        }
    ],
    "graphic": [
        {
            "type": "text",
            "right": 38,
            "top": 12,
            "style": {
                "text": residue_text,
                "fill": "#333333",
                "font": "bold 15px Noto Sans CJK KR"
            }
        }
    ]
}

print(json.dumps({
    "figures": [{
        "name": "tga_plot",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"]
    }]
}, ensure_ascii=False, allow_nan=False))
