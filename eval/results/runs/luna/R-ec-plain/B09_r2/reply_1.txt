import json
import os
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

df = pd.read_excel(DATA_FILE, sheet_name="PE")
df.columns = [str(c).strip() for c in df.columns]

x_col = next(c for c in df.columns if "Temperature" in c)
y_col = next(c for c in df.columns if "Weight" in c and "Deriv" not in c)
dy_col = next(c for c in df.columns if "Deriv" in c)

data = []
for x, y in zip(pd.to_numeric(df[x_col], errors="coerce"),
                pd.to_numeric(df[y_col], errors="coerce")):
    if pd.notna(x) and pd.notna(y):
        data.append([float(x), float(y)])

deriv_data = []
for x, y in zip(pd.to_numeric(df[x_col], errors="coerce"),
                pd.to_numeric(df[dy_col], errors="coerce")):
    if pd.notna(x) and pd.notna(y):
        deriv_data.append([float(x), float(y)])

option = {
