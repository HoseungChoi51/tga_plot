```python
import json
import os
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

sheets = pd.read_excel(DATA_FILE, sheet_name=None)
figures = []

colors = ["#2878B5", "#D9534F", "#3A923A"]

for index, (sheet_name, df) in enumerate(sheets.items()):
    df.columns = [str(column).strip() for column in df.columns]

    temperature_col = next((c for c in df.columns if "temperature" in c.lower()), df.columns[1])
    weight_col = next((c for c in df.columns if "weight" in c.lower() and "d.w" not in c.lower()), df.columns[2])
    derivative_col = next((c for c in df.columns if "d.w" in c.lower() or "dw(" in c.lower()), df.columns[3])

    temperature = pd.to_numeric(df[temperature_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    derivative = pd.to_numeric(df[derivative_col], errors="coerce")

    weight_data = [
        [float(x), float(y)]
        for x, y in zip(temperature, weight)
        if pd.notna(x) and pd.notna(y)
    ]
    derivative_data = [
        [float(x), float(y)]
        for x, y in zip(temperature, derivative)
        if pd.notna(x) and pd.notna(y)
    ]

    x_min = float(temperature.min())
    x_max = float(temperature.max())
    x_min_tick = int((x_min // 100) * 100)
    x_max_tick = int(((x_max + 99) // 100) * 100)

    option = {
        "animation": False,
        "title": {
            "text": sheet_name,
            "left": "center",
            "top": 12,
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"},
        },
        "legend": {
            "top": 42,
            "data": ["Weight(%)", "D.W(%/C)"],
            "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "grid": {"left": 72, "right": 76, "top": 84, "bottom": 62},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 34,
            "min": x_min_tick,
            "max": x_max_tick,
            "interval": 100,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
            {
                "type": "value",
                "name": "D.W (%/°C)",
                "nameLocation": "middle",
                "nameGap": 52,
                "min": -0.05,
                "max": 1.2,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
        ],
        "series": [
            {
                "name": "Weight(%)",
                "type": "line",
                "data": weight_data,
                "yAxisIndex": 0,
                "showSymbol": False,
                "lineStyle": {"width": 2, "color": colors[index % len(colors)]},
            },
            {
                "name": "D.W(%/C)",
                "type": "line",
                "data": derivative_data,
                "yAxisIndex": 1,
                "showSymbol": False,
                "lineStyle": {"width": 1.6, "color": "#D9534F"},
            },
        ],
    }

    figures.append(
        {
            "name": f"tga_{index + 1}",
            "option": option,
            "width": 800,
            "height": 520,
            "dpi": 300,
            "formats": ["png"],
        }
    )

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
```
