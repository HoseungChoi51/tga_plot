```python
import json
import re
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"


def clean_number(value):
    if pd.isna(value):
        return None
    try:
        number = float(value)
        return number if np.isfinite(number) else None
    except (TypeError, ValueError):
        return None


def safe_name(text):
    name = re.sub(r"[^\w가-힣.-]+", "_", str(text), flags=re.UNICODE).strip("_")
    return name or "tga"


def main():
    workbook = pd.ExcelFile(DATA_FILE)
    figures = []

    for sheet_name in workbook.sheet_names:
        df = pd.read_excel(workbook, sheet_name=sheet_name)
        df.columns = [str(column).strip() for column in df.columns]

        temperature_col = next(
            (col for col in df.columns if "temperature" in col.lower() or "온도" in col),
            None,
        )
        weight_col = next(
            (col for col in df.columns if "weight" in col.lower() or "중량" in col),
            None,
        )
        derivative_col = next(
            (
                col
                for col in df.columns
                if "d.w" in col.lower()
                or "dw" in col.lower()
                or "derivative" in col.lower()
            ),
            None,
        )

        if temperature_col is None or weight_col is None:
            numeric = [col for col in df.columns if pd.to_numeric(df[col], errors="coerce").notna().any()]
            if len(numeric) < 2:
                continue
            temperature_col, weight_col = numeric[:2]
            if len(numeric) >= 3 and derivative_col is None:
                derivative_col = numeric[2]

        columns = [temperature_col, weight_col]
        if derivative_col is not None:
            columns.append(derivative_col)

        data = df[columns].copy()
        for column in columns:
            data[column] = pd.to_numeric(data[column], errors="coerce")
        data = data.dropna(subset=[temperature_col, weight_col]).sort_values(temperature_col)

        if data.empty:
            continue

        temperatures = [clean_number(v) for v in data[temperature_col]]
        weights = [clean_number(v) for v in data[weight_col]]
        weight_points = [
            [x, y] for x, y in zip(temperatures, weights) if x is not None and y is not None
        ]

        series = [
            {
                "name": weight_col,
                "type": "line",
                "showSymbol": False,
                "smooth": False,
                "lineStyle": {"width": 2, "color": "#2563eb"},
                "itemStyle": {"color": "#2563eb"},
                "data": weight_points,
            }
        ]

        if derivative_col is not None:
            derivative_points = []
            for x, y in zip(temperatures, data[derivative_col]):
                y = clean_number(y)
                if x is not None and y is not None:
                    derivative_points.append([x, y])
            if derivative_points:
                series.append(
                    {
                        "name": derivative_col,
                        "type": "line",
                        "yAxisIndex": 1,
                        "showSymbol": False,
                        "smooth": False,
                        "lineStyle": {"width": 1.8, "color": "#dc2626"},
                        "itemStyle": {"color": "#dc2626"},
                        "data": derivative_points,
                    }
                )

        option = {
            "animation": False,
            "backgroundColor": "#ffffff",
            "title": {
                "text": str(sheet_name),
                "left": "center",
                "top": 12,
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 17, "fontWeight": "bold"},
            },
            "tooltip": {"trigger": "axis", "axisPointer": {"type": "cross"}},
            "legend": {
                "data": [item["name"] for item in series],
                "top": 42,
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11},
            },
            "grid": {"left": 76, "right": 78 if derivative_col else 32, "top": 82, "bottom": 64},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 38,
                "min": 30,
                "max": 900 if data[temperature_col].max() > 750 else None,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 12},
                "splitLine": {"show": True, "lineStyle": {"color": "#e5e7eb"}},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "nameLocation": "middle",
                    "nameGap": 52,
                    "min": 0,
                    "max": 105,
                    "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 12},
                    "splitLine": {"show": True, "lineStyle": {"color": "#e5e7eb"}},
                }
            ],
            "series": series,
        }

        if derivative_col is not None and len(series) > 1:
            option["yAxis"].append(
                {
                    "type": "value",
                    "name": derivative_col,
                    "nameLocation": "middle",
                    "nameGap": 52,
                    "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 12},
                    "splitLine": {"show": False},
                }
            )

        figures.append(
            {
                "name": safe_name(sheet_name),
                "option": option,
                "width": 900,
                "height": 560,
                "dpi": 300,
                "formats": ["png"],
            }
        )

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
