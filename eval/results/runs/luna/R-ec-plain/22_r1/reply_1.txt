import json
import re
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

figures = []

for sheet_name in ["1006R1-V normal", "1006R1-V acid", "1006R1-V 750C"]:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    df.columns = [str(c).strip() for c in df.columns]

    x_col = next(c for c in df.columns if "temperature" in c.lower())
    y1_col = next(c for c in df.columns if "weight" in c.lower())
    y2_col = next(c for c in df.columns if "d.w" in c.lower() or "dw" in c.lower())

    data = df[[x_col, y1_col, y2_col]].apply(pd.to_numeric, errors="coerce").dropna()
    x = data[x_col].tolist()
    y1 = data[y1_col].tolist()
    y2 = data[y2_col].tolist()

    safe_name = re.sub(r"[^A-Za-z0-9_-]+", "_", sheet_name).strip("_").lower()

    option = {
        "animation": False,
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "cross"},
        },
        "legend": {
            "data": [y1_col, y2_col],
            "top": 10,
            "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "grid": {"left": 72, "right": 78, "top": 55, "bottom": 65},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 42,
            "axisLine": {"lineStyle": {"color": "#555"}},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": False},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 52,
                "min": 0,
                "max": 105,
                "axisLine": {"show": True, "lineStyle": {"color": "#5470C6"}},
                "axisLabel": {
                    "color": "#5470C6",
                    "fontFamily": "Noto Sans CJK KR",
                },
                "nameTextStyle": {
                    "color": "#5470C6",
                    "fontFamily": "Noto Sans CJK KR",
                },
                "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
            },
            {
                "type": "value",
                "name": "D.W (%/°C)",
                "nameLocation": "middle",
                "nameGap": 55,
                "axisLine": {"show": True, "lineStyle": {"color": "green"}},
                "axisLabel": {
                    "color": "green",
                    "fontFamily": "Noto Sans CJK KR",
                },
                "nameTextStyle": {
                    "color": "green",
                    "fontFamily": "Noto Sans CJK KR",
                },
                "splitLine": {"show": False},
            },
        ],
        "series": [
            {
                "name": y1_col,
                "type": "line",
                "yAxisIndex": 0,
                "showSymbol": False,
                "data": [[float(a), float(b)] for a, b in zip(x, y1)],
                "lineStyle": {"width": 2, "color": "#5470C6"},
            },
            {
                "name": y2_col,
                "type": "line",
                "yAxisIndex": 1,
                "showSymbol": False,
                "data": [[float(a), float(b)] for a, b in zip(x, y2)],
                "lineStyle": {"width": 2, "color": "green"},
            },
        ],
    }

    figures.append({
        "name": safe_name,
        "option": option,
        "width": 900,
        "height": 600,
        "dpi": 300,
        "formats": ["png"],
    })

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
