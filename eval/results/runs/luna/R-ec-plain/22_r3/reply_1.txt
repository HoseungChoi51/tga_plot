```python
import json
import re
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def clean_name(value):
    name = re.sub(r"[^A-Za-z0-9가-힣]+", "_", str(value).strip()).strip("_")
    return name or "plot"

def main():
    workbook = pd.ExcelFile(DATA_FILE)
    figures = []

    for sheet in workbook.sheet_names:
        df = pd.read_excel(DATA_FILE, sheet_name=sheet)
        df.columns = [str(c).strip() for c in df.columns]

        temperature = pd.to_numeric(df["Temperature"], errors="coerce")
        weight = pd.to_numeric(df["Weight(%)"], errors="coerce")
        derivative = pd.to_numeric(df["D.W(%/C)"], errors="coerce")
        valid = temperature.notna() & weight.notna() & derivative.notna()

        t = temperature[valid].tolist()
        w = weight[valid].tolist()
        dw = derivative[valid].tolist()

        option = {
            "animation": False,
            "title": {
                "text": sheet,
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16}
            },
            "tooltip": {"trigger": "axis"},
            "legend": {
                "data": ["Weight(%)", "D.W(%/C)"],
                "top": 32,
                "textStyle": {"fontFamily": "Noto Sans CJK KR"}
            },
            "grid": {"left": 72, "right": 78, "top": 78, "bottom": 64},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 38,
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"}
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "min": 0,
                    "max": 105,
                    "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"}
                },
                {
                    "type": "value",
                    "name": "D.W (%/°C)",
                    "nameLocation": "middle",
                    "nameGap": 52,
                    "position": "right",
                    "axisLabel": {"color": "#008000", "fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"color": "#008000", "fontFamily": "Noto Sans CJK KR"},
                    "axisLine": {"show": True, "lineStyle": {"color": "#008000"} },
                    "splitLine": {"show": False}
                }
            ],
            "series": [
                {
                    "name": "Weight(%)",
                    "type": "line",
                    "showSymbol": False,
                    "symbol": "none",
                    "xAxisIndex": 0,
                    "yAxisIndex": 0,
                    "data": [[x, y] for x, y in zip(t, w)],
                    "lineStyle": {"width": 2, "color": "#1f77b4"}
                },
                {
                    "name": "D.W(%/C)",
                    "type": "line",
                    "showSymbol": False,
                    "symbol": "none",
                    "xAxisIndex": 0,
                    "yAxisIndex": 1,
                    "data": [[x, y] for x, y in zip(t, dw)],
                    "lineStyle": {"width": 2, "color": "#008000"}
                }
            ]
        }

        figures.append({
            "name": clean_name(sheet),
            "option": option,
            "width": 800,
            "height": 520,
            "dpi": 300,
            "formats": ["png"]
        })

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))

if __name__ == "__main__":
    main()
```
