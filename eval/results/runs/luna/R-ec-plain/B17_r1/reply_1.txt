```python
import json
import os
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

colors = {
    "PE": "#2563eb",
    "PVC": "#dc2626",
    "PET": "#16a34a",
}

series = []
for sheet in ["PE", "PVC", "PET"]:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
    weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
    valid = temperature.notna() & weight.notna()
    temperature = temperature[valid].to_numpy(dtype=float)
    weight = weight[valid].to_numpy(dtype=float)

    # Vertically separate the curves by 20 percentage points.
    offset = {"PE": 40, "PVC": 20, "PET": 0}[sheet]
    data = [
        [round(float(x), 4), round(float(y + offset), 4)]
        for x, y in zip(temperature, weight)
        if np.isfinite(x) and np.isfinite(y)
    ]
    series.append({
        "name": sheet,
        "type": "line",
        "showSymbol": False,
        "smooth": False,
        "data": data,
        "lineStyle": {"width": 2, "color": colors[sheet]},
        "itemStyle": {"color": colors[sheet]},
    })

option = {
    "backgroundColor": "#ffffff",
    "animation": False,
    "title": {
        "text": "TGA 무게 곡선",
        "left": "center",
        "top": 12,
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 18, "fontWeight": "bold"},
    },
    "tooltip": {
        "trigger": "axis",
        "axisPointer": {"type": "cross"},
    },
    "legend": {
        "data": ["PE", "PVC", "PET"],
        "top": 42,
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 12},
    },
    "grid": {"left": 78, "right": 30, "top": 88, "bottom": 66},
    "xAxis": {
        "type": "value",
        "name": "온도 (°C)",
        "nameLocation": "middle",
        "nameGap": 38,
        "min": 30,
        "max": 900,
        "axisLine": {"lineStyle": {"color": "#555"}},
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        "splitLine": {"show": True, "lineStyle": {"color": "#e5e7eb"}},
    },
    "yAxis": {
        "type": "value",
        "name": "무게 (%) — 곡선 간격 20%",
        "nameLocation": "middle",
        "nameGap": 54,
        "min": 0,
        "max": 160,
        "interval": 20,
        "axisLine": {"lineStyle": {"color": "#555"}},
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        "splitLine": {"show": True, "lineStyle": {"color": "#e5e7eb"}},
    },
    "series": series,
}

figure = {
    "name": "polymer_tga_offset",
    "option": option,
    "width": 900,
    "height": 600,
    "dpi": 300,
    "formats": ["png"],
}

print(json.dumps({"figures": [figure]}, ensure_ascii=False, allow_nan=False))
```
