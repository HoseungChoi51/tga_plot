import pandas as pd

df = pd.read_csv(DATA_FILE, sep=None, engine="python", comment="#")

temp_col = next(c for c in df.columns if "temperature" in c.lower())
weight_col = next(c for c in df.columns if "weight" in c.lower())

df = df.dropna(subset=[time_col, temp_col, weight_col]).sort_values(time_col)

times = df[time_col].tolist()
temps = df[temp_col].tolist()
weights = df[weight_col].tolist()

target_time = 140.0
target_weight = float(df.loc[df[time_col].sub(target_time).abs().idxmin(), weight_col])
