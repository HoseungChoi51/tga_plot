```python
import json
import re
from pathlib import Path

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

TEXT_COLOR = "#808080"
FONT_SIZE = 13
PALETTE = ["#5470C6", "#91CC75", "#FAC858", "#EE6666"]


def safe_name(value):
    name = re.sub(r"[^A-Za-z0-9가-힣_-]+", "_", str(value)).strip("_")
    return name or "plot"


def clean_number(value):
    value = float(value)
    if not np.isfinite(value):
        return None
    return round(value, 8)


def build_figure(sheet_name, frame):
    columns = {str(col).strip().lower(): col for col in frame.columns}
    x_col = next((col for key, col in columns.items() if "temperature" in key), None)
    weight_col = next((col for key, col in columns.items() if "weight" in key), None)
    derivative_col = next(
        (col for key, col in columns.items() if "d.w" in key or "dw(" in key or "derivative" in key),
        None,
    )

    if x_col is None:
        x_col = frame.columns[1] if len(frame.columns) > 1 else frame.columns[0]
    if weight_col is None:
        weight_col = frame.columns[2] if len(frame.columns) > 2 else frame.columns[-1]

    work = frame.copy()
    for col in (x_col, weight_col, derivative_col):
        if col is not None:
            work[col] = pd.to_numeric(work[col], errors="coerce")
    work = work.dropna(subset=[x_col, weight_col]).sort_values(x_col)

    x_values = work[x_col].to_numpy(dtype=float)
    weight_values = work[weight_col].to_numpy(dtype=float)
    weight_data = [
        [clean_number(x), clean_number(y)]
        for x, y in zip(x_values, weight_values)
        if np.isfinite(x) and np.isfinite(y)
    ]

    series = [
        {
            "name": str(weight_col),
            "type": "line",
            "data": weight_data,
            "showSymbol": False,
            "smooth": False,
            "yAxisIndex": 0,
            "lineStyle": {"width": 2},
        }
    ]

    y_axes = [
        {
            "type": "value",
            "name": str(weight_col),
            "nameLocation": "middle",
            "nameGap": 48,
            "min": 0,
            "max": 105,
            "axisLabel": {"formatter": "{value}%"},
            "splitLine": {"lineStyle": {"color": "#e8e8e8"}},
        }
    ]

    if derivative_col is not None:
        derivative_work = work.dropna(subset=[derivative_col])
        derivative_data = [
            [clean_number(x), clean_number(y)]
            for x, y in zip(
                derivative_work[x_col].to_numpy(dtype=float),
                derivative_work[derivative_col].to_numpy(dtype=float),
            )
            if np.isfinite(x) and np.isfinite(y)
        ]
        series.append(
            {
                "name": str(derivative_col),
                "type": "line",
                "data": derivative_data,
                "showSymbol": False,
                "smooth": False,
                "yAxisIndex": 1,
                "lineStyle": {"width": 1.8, "type": "dashed"},
            }
        )
        y_axes.append(
            {
                "type": "value",
                "name": str(derivative_col),
                "nameLocation": "middle",
                "nameGap": 50,
                "position": "right",
                "splitLine": {"show": False},
                "axisLabel": {"formatter": "{value}"},
            }
        )

    for index, item in enumerate(series):
        item["itemStyle"] = {"color": PALETTE[index % len(PALETTE)]}
        item["lineStyle"]["color"] = PALETTE[index % len(PALETTE)]

    option = {
        "animation": False,
        "backgroundColor": "#ffffff",
        "textStyle": {
            "color": TEXT_COLOR,
            "fontSize": FONT_SIZE,
            "fontFamily": "Noto Sans CJK KR, sans-serif",
        },
        "title": {
            "text": str(sheet_name),
            "left": "center",
            "top": 12,
            "textStyle": {"color": TEXT_COLOR, "fontSize": FONT_SIZE, "fontWeight": "normal"},
        },
        "tooltip": {"trigger": "axis"},
        "legend": {
            "top": 42,
            "textStyle": {"color": TEXT_COLOR, "fontSize": FONT_SIZE},
        },
        "grid": {
            "left": 78,
            "right": 82 if derivative_col is not None else 35,
            "top": 88,
            "bottom": 62,
            "containLabel": True,
        },
        "xAxis": {
            "type": "value",
            "name": str(x_col) + " (°C)",
            "nameLocation": "middle",
            "nameGap": 38,
            "min": float(np.nanmin(x_values)) if len(x_values) else 0,
            "max": float(np.nanmax(x_values)) if len(x_values) else 900,
            "axisLine": {"lineStyle": {"color": TEXT_COLOR}},
            "axisTick": {"lineStyle": {"color": TEXT_COLOR}},
            "axisLabel": {"color": TEXT_COLOR, "fontSize": FONT_SIZE},
        },
        "yAxis": y_axes,
        "series": series,
    }

    for axis in option["yAxis"]:
        axis["axisLine"] = {"show": True, "lineStyle": {"color": TEXT_COLOR}}
        axis["axisTick"] = {"lineStyle": {"color": TEXT_COLOR}}
        axis["axisLabel"] = {"color": TEXT_COLOR, "fontSize": FONT_SIZE}
        axis["nameTextStyle"] = {"color": TEXT_COLOR, "fontSize": FONT_SIZE}

    option["xAxis"]["nameTextStyle"] = {"color": TEXT_COLOR, "fontSize": FONT_SIZE}
    option["xAxis"]["splitLine"] = {"lineStyle": {"color": "#e8e8e8"}}

    return {
        "name": safe_name(sheet_name),
        "option": option,
        "width": 900,
        "height": 560,
        "dpi": 300,
        "formats": ["png"],
    }


path = Path(DATA_FILE)
figures = []
excel = pd.ExcelFile(path)
for sheet_name in excel.sheet_names:
    frame = pd.read_excel(path, sheet_name=sheet_name)
    figures.append(build_figure(sheet_name, frame))

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
```
