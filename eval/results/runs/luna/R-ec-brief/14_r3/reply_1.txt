import json
import os
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/korean_excel.csv"

df = pd.read_csv(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

temperature = pd.to_numeric(df["온도(°C)"], errors="coerce").to_numpy()
weight = pd.to_numeric(df["무게(%)"], errors="coerce").to_numpy()
dtg = pd.to_numeric(df["미분무게(%/°C)"], errors="coerce").to_numpy()

valid = np.isfinite(temperature) & np.isfinite(weight) & np.isfinite(dtg)
temperature = temperature[valid]
weight = weight[valid]
dtg = dtg[valid]

residue_idx = int(np.argmin(np.abs(temperature - 800)))
residue_temp = float(temperature[residue_idx])
residue_weight = float(weight[residue_idx])

peak_mask = temperature > 150
if np.any(peak_mask):
    peak_candidates = np.flatnonzero(peak_mask)
    peak_idx = int(peak_candidates[np.argmax(-dtg[peak_mask])])
else:
    peak_idx = int(np.argmax(-dtg))
peak_temp = float(temperature[peak_idx])
peak_value = float(-dtg[peak_idx])

option = {
