```python
import json
import os

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

COLORS = ["#1f4e9c", "#c0392b", "#2e8b57"]
SHEETS = ["PE", "PVC", "PET"]
OFFSET_STEP = 20.0


def clean_number(value):
    if pd.isna(value):
        return None
    value = float(value)
    return value if np.isfinite(value) else None


def main():
    workbook = pd.ExcelFile(DATA_FILE)
    selected_sheets = [name for name in SHEETS if name in workbook.sheet_names]
    if not selected_sheets:
        selected_sheets = workbook.sheet_names

    series = []
    all_temperatures = []

    for index, sheet_name in enumerate(selected_sheets):
        df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        normalized = {str(col).strip().lower(): col for col in df.columns}

        temperature_col = next(
            (col for key, col in normalized.items() if "temperature" in key or "온도" in key),
            df.columns[1] if len(df.columns) > 1 else df.columns[0],
        )
        weight_col = next(
            (col for key, col in normalized.items() if "weight" in key and "deriv" not in key),
            df.columns[2] if len(df.columns) > 2 else df.columns[-1],
        )

        data = pd.DataFrame(
            {
                "temperature": pd.to_numeric(df[temperature_col], errors="coerce"),
                "weight": pd.to_numeric(df[weight_col], errors="coerce"),
            }
        ).dropna()

        if data.empty:
            continue

        offset = index * OFFSET_STEP
        temperatures = data["temperature"].to_numpy(dtype=float)
        weights = data["weight"].to_numpy(dtype=float) + offset
        all_temperatures.extend(temperatures.tolist())

        series.append(
            {
                "name": f"{sheet_name} (+{offset:g}%)" if offset else sheet_name,
                "type": "line",
                "showSymbol": False,
                "smooth": False,
                "connectNulls": False,
                "lineStyle": {"width": 2, "color": COLORS[index % len(COLORS)]},
                "data": [
                    [clean_number(x), clean_number(y)]
                    for x, y in zip(temperatures, weights)
                ],
            }
        )

    if not series:
        raise ValueError("선택한 시트에서 온도와 무게 데이터를 찾지 못했습니다.")

    x_min = float(np.nanmin(all_temperatures))
    x_max = float(np.nanmax(all_temperatures))

    option = {
        "animation": False,
        "title": {
            "text": "PE, PVC, PET 무게 곡선 (20% offset)",
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
        },
        "tooltip": {
            "trigger": "axis",
            "axisPointer": {"type": "line"},
        },
        "legend": {
            "data": [item["name"] for item in series],
            "left": "center",
            "bottom": 8,
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 11},
        },
        "grid": {"left": 72, "right": 28, "top": 62, "bottom": 78},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 32,
            "min": x_min,
            "max": x_max,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        },
        "yAxis": {
            "type": "value",
            "name": "Weight (%) + offset",
            "nameLocation": "middle",
            "nameGap": 48,
            "min": 0,
            "max": 160,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
        },
        "series": series,
    }

    result = {
        "figures": [
            {
                "name": "polymer_weight_offset",
                "option": option,
                "width": 800,
                "height": 560,
                "dpi": 300,
                "formats": ["png"],
            }
        ]
    }
    print(json.dumps(result, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
