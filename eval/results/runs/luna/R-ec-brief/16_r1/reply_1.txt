```python
import json
import os
import sys

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
SHEET_NAME = "1006R1-V normal"

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
df.columns = [str(c).strip() for c in df.columns]

temperature_col = next(
    (c for c in df.columns if "temperature" in c.lower() or "온도" in c),
    df.columns[1],
)
weight_col = next(
    (c for c in df.columns if "weight" in c.lower() or "무게" in c),
    df.columns[2],
)

data = df[[temperature_col, weight_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[temperature_col].to_numpy(dtype=float)
weight = data[weight_col].to_numpy(dtype=float)

series_data = [
    [round(float(x), 6), round(float(y), 6)]
    for x, y in zip(temperature, weight)
    if np.isfinite(x) and np.isfinite(y)
]

option = {
    "title": {
        "text": SHEET_NAME,
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR"},
    },
    "tooltip": {"trigger": "axis"},
    "grid": {"left": 72, "right": 30, "top": 65, "bottom": 65},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 38,
        "axisLine": {"show": True},
        "axisTick": {"show": True},
        "axisLabel": {"show": True},
        "splitLine": {"show": False},
    },
    "yAxis": {
        "type": "value",
        "name": "Weight (%)",
        "nameLocation": "middle",
        "nameGap": 52,
        "axisLine": {"show": True},
        "axisTick": {"show": True},
        "axisLabel": {"show": True},
        "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
    },
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "data": series_data,
            "showSymbol": False,
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "itemStyle": {"color": "#1f4e9c"},
        }
    ],
}

print(
    json.dumps(
        {
            "figures": [
                {
                    "name": "1006R1-V_normal",
                    "option": option,
                    "width": 614,
                    "height": 442,
                    "dpi": 300,
                    "formats": ["png"],
                }
            ]
        },
        ensure_ascii=False,
        allow_nan=False,
    )
)
```
