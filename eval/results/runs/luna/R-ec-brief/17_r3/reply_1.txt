```python
import json
import math
import re
from pathlib import Path

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

WIDTH_IN = 8
HEIGHT_IN = 5
DPI = 600
WIDTH = round(WIDTH_IN * 96)
HEIGHT = round(HEIGHT_IN * 96)


def clean_name(value):
    text = re.sub(r"[^\w가-힣.-]+", "_", str(value).strip())
    return text.strip("_") or "plot"


def finite_pairs(x, y):
    result = []
    for xv, yv in zip(x, y):
        if pd.isna(xv) or pd.isna(yv):
            continue
        xv, yv = float(xv), float(yv)
        if math.isfinite(xv) and math.isfinite(yv):
            result.append([xv, yv])
    return result


def make_figure(sheet_name, frame):
    columns = {str(c).strip().lower(): c for c in frame.columns}

    def find_column(terms):
        for term in terms:
            for key, original in columns.items():
                if term in key:
                    return original
        return None

    x_col = find_column(["temperature", "temp", "온도"]) or frame.columns[1]
    weight_col = find_column(["weight", "mass", "중량", "무게"]) or frame.columns[2]
    dtg_col = find_column(["d.w", "dtg", "derivative", "미분"]) or frame.columns[3]

    x = pd.to_numeric(frame[x_col], errors="coerce")
    weight = pd.to_numeric(frame[weight_col], errors="coerce")
    dtg = pd.to_numeric(frame[dtg_col], errors="coerce")
    valid = x.notna() & weight.notna()
    x_values = x[valid].to_numpy(dtype=float)
    weight_values = weight[valid].to_numpy(dtype=float)

    weight_data = finite_pairs(x_values, weight_values)
    dtg_valid = x.notna() & dtg.notna()
    dtg_x = x[dtg_valid].to_numpy(dtype=float)
    dtg_y = -dtg[dtg_valid].to_numpy(dtype=float)
    dtg_data = finite_pairs(dtg_x, dtg_y)

    residue_marker = None
    if len(x_values):
        idx = int(np.argmin(np.abs(x_values - 800.0)))
        residue_marker = [float(x_values[idx]), float(weight_values[idx])]
        residue_text = f"Residue {weight_values[idx]:.1f}%"
    else:
        residue_text = "Residue"

    peak_marker = None
    peak_candidates = np.where(dtg_x > 150.0)[0] if len(dtg_x) else []
    if len(peak_candidates):
        peak_idx = int(peak_candidates[np.argmax(dtg_y[peak_candidates])])
        peak_marker = [float(dtg_x[peak_idx]), float(dtg_y[peak_idx])]
        peak_text = f"{dtg_x[peak_idx]:.0f} °C"
    else:
        peak_text = ""

    series = [
        {
            "name": "Weight (%)",
            "type": "line",
            "yAxisIndex": 0,
            "showSymbol": False,
            "connectNulls": False,
            "data": weight_data,
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "itemStyle": {"color": "#1f4e9c"},
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 9,
                "data": [
                    {
                        "coord": residue_marker,
                        "name": residue_text,
                        "label": {
                            "show": True,
                            "formatter": residue_text,
                            "position": "top",
                            "color": "#1f4e9c",
                        },
                        "itemStyle": {"color": "#1f4e9c"},
                    }
                ],
            } if residue_marker else {},
        },
        {
            "name": "-DTG (%/°C)",
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "connectNulls": False,
            "data": dtg_data,
            "lineStyle": {"color": "#c0392b", "width": 2},
            "itemStyle": {"color": "#c0392b"},
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 9,
                "data": [
                    {
                        "coord": peak_marker,
                        "name": peak_text,
                        "label": {
                            "show": True,
                            "formatter": peak_text,
                            "position": "top",
                            "color": "#c0392b",
                        },
                        "itemStyle": {"color": "#c0392b"},
                    }
                ],
            } if peak_marker else {},
        },
    ]

    option = {
        "animation": False,
        "backgroundColor": "#ffffff",
        "title": {
            "text": str(sheet_name),
            "left": "center",
            "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
        },
        "tooltip": {"trigger": "axis"},
        "legend": {
            "data": ["Weight (%)", "-DTG (%/°C)"],
            "left": "center",
            "bottom": 8,
            "textStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "grid": {"left": 72, "right": 72, "top": 60, "bottom": 60},
        "xAxis": {
            "type": "value",
            "name": "Temperature (°C)",
            "nameLocation": "middle",
            "nameGap": 34,
            "axisLine": {"show": True},
            "axisTick": {"show": True},
            "splitLine": {"show": False},
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
        },
        "yAxis": [
            {
                "type": "value",
                "name": "Weight (%)",
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR", "color": "#1f4e9c"},
                "splitLine": {"show": True, "lineStyle": {"color": "#e5e5e5"}},
            },
            {
                "type": "value",
                "name": "-DTG (%/°C)",
                "nameLocation": "middle",
                "nameGap": 48,
                "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR", "color": "#c0392b"},
                "splitLine": {"show": False},
            },
        ],
        "series": series,
    }

    return {
        "name": clean_name(sheet_name),
        "option": option,
        "width": WIDTH,
        "height": HEIGHT,
        "dpi": DPI,
        "formats": ["png"],
    }


def main():
    path = Path(DATA_FILE)
    workbook = pd.ExcelFile(path)
    figures = []
    for sheet_name in workbook.sheet_names:
        frame = pd.read_excel(path, sheet_name=sheet_name)
        figures.append(make_figure(sheet_name, frame))
    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
