import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"

df = pd.read_excel(DATA_FILE, sheet_name="Annealed")
df.columns = [str(c).strip() for c in df.columns]

def find_col(pattern):
    for col in df.columns:
        if re.search(pattern, col, re.IGNORECASE):
            return col
    raise ValueError(f"Required column not found: {pattern}")

x_col = find_col(r"Temperature")
weight_col = find_col(r"Weight\s*\(%")
dtg_col = find_col(r"Deriv")

x = pd.to_numeric(df[x_col], errors="coerce").to_numpy()
weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy()
dtg = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy()
valid = np.isfinite(x) & np.isfinite(weight) & np.isfinite(dtg)
x, weight, dtg = x[valid], weight[valid], dtg[valid]

# Mark the measured weight nearest 800 °C.
residue_idx = int(np.argmin(np.abs(x - 800)))
residue_x = float(x[residue_idx])
residue_y = float(weight[residue_idx])

# The upward-pointing DTG curve is -DTG; select its highest peak above 150 °C.
peak_candidates = np.flatnonzero(x > 150)
peak_idx = int(peak_candidates[np.argmax(-dtg[peak_candidates])])
peak_x = float(x[peak_idx])
peak_y = float(-dtg[peak_idx])
