```python
import json
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"

df = pd.read_excel(DATA_FILE, sheet_name="PVC")
df.columns = [str(c).strip() for c in df.columns]

temp_col = next(c for c in df.columns if "temperature" in c.lower() or "온도" in c)
weight_col = next(c for c in df.columns if "weight" in c.lower() or "무게" in c or "질량" in c)
dtg_col = next((c for c in df.columns if "deriv" in c.lower() or "dtg" in c.lower()), None)

temp_all = pd.to_numeric(df[temp_col], errors="coerce").to_numpy(dtype=float)
weight_all = pd.to_numeric(df[weight_col], errors="coerce").to_numpy(dtype=float)
valid = np.isfinite(temp_all) & np.isfinite(weight_all)
temp, weight = temp_all[valid], weight_all[valid]

i200 = int(np.argmin(np.abs(temp - 200)))
i400 = int(np.argmin(np.abs(temp - 400)))
i800 = int(np.argmin(np.abs(temp - 800)))
mass_loss = float(weight[i200] - weight[i400])

series = [{
    "name": "Weight (%)",
    "type": "line",
    "showSymbol": False,
    "lineStyle": {"color": "#1f4e9c", "width": 2},
    "data": [[float(x), float(y)] for x, y in zip(temp, weight)],
    "markPoint": {
        "symbol": "circle",
        "symbolSize": 8,
        "itemStyle": {"color": "#1f4e9c"},
        "label": {
            "show": True,
            "formatter": f"Residue {weight[i800]:.1f}%",
            "position": "right",
        },
        "data": [{"coord": [float(temp[i800]), float(weight[i800])]}],
    },
    "markLine": {
        "symbol": ["none", "none"],
        "lineStyle": {"color": "#666666", "type": "dashed", "width": 1.5},
        "label": {
            "show": True,
            "formatter": f"200–400 °C 질량 감소량: {mass_loss:.2f}%",
            "position": "insideEndTop",
        },
        "data": [[
            {"coord": [float(temp[i200]), float(weight[i200])]},
            {"coord": [float(temp[i400]), float(weight[i400])]},
        ]],
    },
}]

legend_data = ["Weight (%)"]
if dtg_col:
    dtg_all = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy(dtype=float)
    dtg = dtg_all[valid]
    peak_mask = np.isfinite(dtg) & (temp > 150)
    if peak_mask.any():
        candidates = np.flatnonzero(peak_mask)
        peak_idx = int(candidates[np.argmax(-dtg[candidates])])
        peak_temp = float(temp[peak_idx])
        series.append({
            "name": "-DTG (%/°C)",
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "lineStyle": {"color": "#c0392b", "width": 1.5},
            "data": [
                [float(x), float(-y) if np.isfinite(y) else None]
                for x, y in zip(temp, dtg)
            ],
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#c0392b"},
                "label": {
                    "show": True,
                    "formatter": f"{peak_temp:.0f} °C",
                    "position": "top",
                },
                "data": [{"coord": [peak_temp, float(-dtg[peak_idx])]}],
            },
        })
        legend_data.append("-DTG (%/°C)")

option = {
    "title": {
        "text": "PVC",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR"},
    },
    "tooltip": {"trigger": "axis"},
    "legend": {"data": legend_data, "left": "center"},
    "grid": {"left": 65, "right": 65, "top": 75, "bottom": 55},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "min": 30,
        "max": 900,
        "nameLocation": "middle",
        "nameGap": 32,
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "min": 0,
            "max": 105,
            "nameLocation": "middle",
            "nameGap": 45,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        },
        {
            "type": "value",
            "name": "-DTG (%/°C)",
            "nameLocation": "middle",
            "nameGap": 45,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
        },
    ],
    "series": series,
}

print(json.dumps({
    "figures": [{
        "name": "PVC",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"],
    }]
}, ensure_ascii=False, allow_nan=False))
```
