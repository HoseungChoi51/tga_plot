```python
import json
import os
import pandas as pd
import numpy as np

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/caox.csv"

df = pd.read_csv(DATA_FILE)
df.columns = [str(c).strip() for c in df.columns]

def find_col(tokens):
    for col in df.columns:
        if all(token.lower() in col.lower() for token in tokens):
            return col
    return None

x_col = find_col(["temperature"]) or df.columns[1]
weight_col = find_col(["weight"]) or df.columns[2]
dtg_col = find_col(["deriv"]) or df.columns[3]

data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
x = data[x_col].to_numpy()
weight = data[weight_col].to_numpy()
dtg = data[dtg_col].to_numpy()

residue_idx = int(np.argmin(np.abs(x - 800)))
residue_x = float(x[residue_idx])
residue_y = float(weight[residue_idx])

peak_mask = x > 150
if peak_mask.any():
    peak_indices = np.where(peak_mask)[0]
    peak_idx = int(peak_indices[np.argmax(-dtg[peak_mask])])
    peak_x = float(x[peak_idx])
    peak_y = float(-dtg[peak_idx])
else:
    peak_x = float(x[np.argmax(-dtg)])
    peak_y = float(np.max(-dtg))

series_data = [[float(xi), float(yi)] for xi, yi in zip(x, weight)]
dtg_data = [[float(xi), float(-yi)] for xi, yi in zip(x, dtg)]

bands = [
    (120, 250, "H2O", "rgba(52, 152, 219, 0.13)"),
    (430, 580, "CO", "rgba(46, 204, 113, 0.13)"),
    (640, 830, "CO2", "rgba(243, 156, 18, 0.13)"),
]

mark_area_data = [
    [
        {"xAxis": start, "itemStyle": {"color": color}},
        {"xAxis": end},
    ]
    for start, end, _, color in bands
]

mark_labels = [
    {
        "coord": [(start + end) / 2, 101.0],
        "value": label,
        "label": {
            "show": True,
            "position": "top",
            "color": "#333333",
            "fontFamily": "Noto Sans CJK KR",
            "fontSize": 14,
            "fontWeight": "bold",
        },
        "symbol": "none",
    }
    for start, end, label, _ in bands
]

option = {
    "animation": False,
    "title": {
        "text": "caox",
        "left": "center",
        "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 16},
    },
    "tooltip": {"trigger": "axis"},
    "legend": {"data": ["Weight (%)", "-Deriv. Weight (%/°C)"], "left": "center", "top": 30},
    "grid": {"left": 75, "right": 75, "top": 80, "bottom": 65},
    "xAxis": {
        "type": "value",
        "name": "Temperature (°C)",
        "nameLocation": "middle",
        "nameGap": 35,
        "min": float(np.nanmin(x)),
        "max": float(np.nanmax(x)),
        "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
    },
    "yAxis": [
        {
            "type": "value",
            "name": "Weight (%)",
            "nameLocation": "middle",
            "nameGap": 50,
            "min": 0,
            "max": 110,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
        },
        {
            "type": "value",
            "name": "-Deriv. Weight (%/°C)",
            "nameLocation": "middle",
            "nameGap": 55,
            "inverse": False,
            "axisLabel": {"fontFamily": "Noto Sans CJK KR"},
            "splitLine": {"show": False},
        },
    ],
    "series": [
        {
            "name": "Weight (%)",
            "type": "line",
            "showSymbol": False,
            "lineStyle": {"color": "#1f4e9c", "width": 2},
            "data": series_data,
            "markArea": {"silent": True, "data": mark_area_data},
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#1f4e9c"},
                "data": [
                    {
                        "coord": [residue_x, residue_y],
                        "value": f"Residue {residue_y:.1f}%",
                        "label": {
                            "show": True,
                            "formatter": f"Residue {residue_y:.1f}%",
                            "position": "right",
                            "color": "#1f4e9c",
                        },
                    },
                    *mark_labels,
                ],
            },
        },
        {
            "name": "-Deriv. Weight (%/°C)",
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "lineStyle": {"color": "#c0392b", "width": 1.5},
            "data": dtg_data,
            "markPoint": {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#c0392b"},
                "data": [
                    {
                        "coord": [peak_x, peak_y],
                        "value": f"{peak_x:.0f} °C",
                        "label": {
                            "show": True,
                            "formatter": f"{peak_x:.0f} °C",
                            "position": "top",
                            "color": "#c0392b",
                        },
                    }
                ],
            },
        },
    ],
}

print(json.dumps({
    "figures": [{
        "name": "plot1",
        "option": option,
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"],
    }]
}, ensure_ascii=False, allow_nan=False))
```
