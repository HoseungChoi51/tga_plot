```python
import json
import re
from pathlib import Path

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def clean_name(value):
    name = re.sub(r'[\\/:*?"<>|]+', "_", str(value)).strip()
    return name or "plot"

def number(value):
    try:
        result = float(value)
        return result if np.isfinite(result) else None
    except (TypeError, ValueError):
        return None

figures = []

book = pd.ExcelFile(DATA_FILE)
for sheet in book.sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    if df.empty:
        continue

    columns = list(df.columns)

    def find_column(pattern):
        for col in columns:
            if re.search(pattern, str(col), re.IGNORECASE):
                return col
        return None

    xcol = find_column(r"temperature|temp|온도")
    y1col = find_column(r"weight|mass|무게|중량")
    y2col = find_column(r"d\.?\s*w|dtg|derivative|미분")

    if xcol is None or y1col is None:
        numeric = [c for c in columns if pd.to_numeric(df[c], errors="coerce").notna().sum() > 0]
        if len(numeric) < 2:
            continue
        xcol = xcol or numeric[0]
        y1col = y1col or numeric[1]
        y2col = y2col or (numeric[2] if len(numeric) > 2 else None)

    x = pd.to_numeric(df[xcol], errors="coerce")
    y1 = pd.to_numeric(df[y1col], errors="coerce")
    y2 = pd.to_numeric(df[y2col], errors="coerce") if y2col is not None else pd.Series(np.nan, index=df.index)

    valid = x.notna() & y1.notna()
    if not valid.any():
        continue
    x_values = x[valid].to_numpy(dtype=float)
    y1_values = y1[valid].to_numpy(dtype=float)

    order = np.argsort(x_values)
    x_values = x_values[order]
    y1_values = y1_values[order]

    valid2 = x.notna() & y2.notna()
    x2 = x[valid2].to_numpy(dtype=float)
    y2_values = -y2[valid2].to_numpy(dtype=float)
    order2 = np.argsort(x2)
    x2 = x2[order2]
    y2_values = y2_values[order2]

    series = [
        {
            "name": str(y1col),
            "type": "line",
            "yAxisIndex": 0,
            "showSymbol": False,
            "connectNulls": True,
            "lineStyle": {"width": 2, "color": "#1f4e9c"},
            "itemStyle": {"color": "#1f4e9c"},
            "data": [[number(a), number(b)] for a, b in zip(x_values, y1_values)],
        }
    ]

    if len(x2):
        series.append({
            "name": "- " + str(y2col),
            "type": "line",
            "yAxisIndex": 1,
            "showSymbol": False,
            "connectNulls": True,
            "lineStyle": {"width": 2, "color": "#c0392b"},
            "itemStyle": {"color": "#c0392b"},
            "data": [[number(a), number(b)] for a, b in zip(x2, y2_values)],
        })

    # 450 °C 세로 보조선과 라벨
    series.append({
        "name": "산화 시작",
        "type": "line",
        "yAxisIndex": 0,
        "silent": True,
        "symbol": "none",
        "lineStyle": {"opacity": 0},
        "data": [[450, float(np.nanmin(y1_values))], [450, float(np.nanmax(y1_values))]],
        "markLine": {
            "symbol": "none",
            "silent": True,
            "lineStyle": {"color": "#555555", "width": 1.5, "type": "dashed"},
            "label": {
                "show": True,
                "formatter": "산화 시작",
                "position": "insideEndTop",
                "color": "#333333",
                "fontFamily": "Noto Sans CJK KR",
                "fontSize": 12,
            },
            "data": [{"xAxis": 450}],
        },
    })

    figures.append({
        "name": clean_name(sheet),
        "width": 614,
        "height": 442,
        "dpi": 300,
        "formats": ["png"],
        "option": {
            "title": {
                "text": str(sheet),
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
            "animation": False,
            "textStyle": {"fontFamily": "Noto Sans CJK KR"},
            "tooltip": {"trigger": "axis"},
            "legend": {"left": "center", "bottom": 8},
            "grid": {"left": 70, "right": 70, "top": 55, "bottom": 65},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 35,
                "min": float(np.nanmin(x_values)),
                "max": float(np.nanmax(x_values)),
                "axisLine": {"show": True},
                "axisTick": {"show": True},
                "axisLabel": {"show": True},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": str(y1col),
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                    "axisLabel": {"color": "#1f4e9c"},
                    "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
                },
                {
                    "type": "value",
                    "name": "-" + str(y2col) if y2col is not None else "DTG",
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                    "axisLabel": {"color": "#c0392b"},
                    "splitLine": {"show": False},
                },
            ],
            "series": series,
        },
    })

print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))
```
