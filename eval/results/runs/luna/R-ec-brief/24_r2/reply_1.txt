```python
import json
import re
import sys

import numpy as np
import pandas as pd


DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/extra_columns.xlsx"


def clean_number(value):
    if pd.isna(value):
        return None
    value = float(value)
    return value if np.isfinite(value) else None


def main():
    book = pd.ExcelFile(DATA_FILE)
    figures = []

    for sheet_name in book.sheet_names:
        df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
        columns = {str(c).strip().lower(): c for c in df.columns}

        def find_column(predicate):
            for normalized, original in columns.items():
                if predicate(normalized):
                    return original
            return None

        x_col = find_column(lambda s: "temperature" in s or "temp" in s)
        weight_col = find_column(lambda s: "weight" in s and "%" in s and "deriv" not in s)
        heat_col = find_column(lambda s: "heat flow" in s)

        if x_col is None or weight_col is None or heat_col is None:
            continue

        x = pd.to_numeric(df[x_col], errors="coerce")
        weight = pd.to_numeric(df[weight_col], errors="coerce")
        heat = pd.to_numeric(df[heat_col], errors="coerce")

        valid = x.notna() & weight.notna() & heat.notna()
        points = [
            [clean_number(a), clean_number(b), clean_number(c)]
            for a, b, c in zip(x[valid], weight[valid], heat[valid])
        ]

        series_data_weight = [[p[0], p[1]] for p in points]
        series_data_heat = [[p[0], p[2]] for p in points]

        # Preserve the standard residue annotation at the measured temperature
        # nearest 800 °C; the requested DTG peak annotation is omitted.
        residue = None
        if points:
            nearest = min(points, key=lambda p: abs(p[0] - 800.0))
            residue = {"temperature": nearest[0], "weight": nearest[1]}

        series = [
            {
                "name": str(weight_col),
                "type": "line",
                "showSymbol": False,
                "yAxisIndex": 0,
                "data": series_data_weight,
                "lineStyle": {"color": "#1f4e9c", "width": 2},
                "itemStyle": {"color": "#1f4e9c"},
            },
            {
                "name": str(heat_col),
                "type": "line",
                "showSymbol": False,
                "yAxisIndex": 1,
                "data": series_data_heat,
                "lineStyle": {"color": "#c0392b", "width": 2},
                "itemStyle": {"color": "#c0392b"},
            },
        ]

        if residue is not None:
            series[0]["markPoint"] = {
                "symbol": "circle",
                "symbolSize": 8,
                "itemStyle": {"color": "#1f4e9c"},
                "label": {
                    "show": True,
                    "formatter": "Residue " + format(residue["weight"], ".1f") + "%",
                    "position": "top",
                },
                "data": [{"coord": [residue["temperature"], residue["weight"]]}],
            }

        option = {
            "animation": False,
            "title": {
                "text": str(sheet_name),
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
            "tooltip": {"trigger": "axis"},
            "legend": {"left": "center", "bottom": 8},
            "grid": {"left": 72, "right": 78, "top": 58, "bottom": 66},
            "xAxis": {
                "type": "value",
                "name": "Temperature (°C)",
                "nameLocation": "middle",
                "nameGap": 38,
                "axisTick": {"show": True},
                "axisLabel": {"show": True, "fontFamily": "Noto Sans CJK KR"},
                "nameTextStyle": {"fontFamily": "Noto Sans CJK KR"},
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": "Weight (%)",
                    "nameLocation": "middle",
                    "nameGap": 52,
                    "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                    "axisLabel": {"color": "#1f4e9c", "fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"color": "#1f4e9c", "fontFamily": "Noto Sans CJK KR"},
                    "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
                },
                {
                    "type": "value",
                    "name": "Heat flow (W/g)",
                    "nameLocation": "middle",
                    "nameGap": 58,
                    "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                    "axisLabel": {"color": "#c0392b", "fontFamily": "Noto Sans CJK KR"},
                    "nameTextStyle": {"color": "#c0392b", "fontFamily": "Noto Sans CJK KR"},
                    "splitLine": {"show": False},
                },
            ],
            "series": series,
        }

        safe_name = re.sub(r"[^A-Za-z0-9_-]+", "_", str(sheet_name)).strip("_") or "plot"
        figures.append(
            {
                "name": safe_name,
                "option": option,
                "width": 614,
                "height": 442,
                "dpi": 300,
                "formats": ["png"],
            }
        )

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))


if __name__ == "__main__":
    main()
```
