```python
import json
import re
import sys

import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"

def clean_name(value):
    name = re.sub(r"[^A-Za-z0-9_-]+", "_", str(value).strip()).strip("_")
    return name or "plot"

def main():
    sheets = pd.read_excel(DATA_FILE, sheet_name=None)
    figures = []

    for sheet_name, df in sheets.items():
        if df.empty or len(df.columns) < 4:
            continue

        columns = list(df.columns)

        def find_column(pattern, fallback):
            for col in columns:
                if re.search(pattern, str(col), flags=re.I):
                    return col
            return columns[fallback]

        x_col = find_column(r"temperature|temp|온도", 1)
        weight_col = find_column(r"weight|mass|중량|무게", 2)
        dtg_col = find_column(r"d\.?\s*w|dtg|derivative", 3)

        data = df[[x_col, weight_col, dtg_col]].copy()
        for col in data.columns:
            data[col] = pd.to_numeric(data[col], errors="coerce")
        data = data.replace([np.inf, -np.inf], np.nan).dropna()

        if data.empty:
            continue

        x = data[x_col].to_numpy(dtype=float)
        weight = data[weight_col].to_numpy(dtype=float)
        dtg = data[dtg_col].to_numpy(dtype=float)

        # The input DTG is negative for mass loss; invert it so peaks point upward.
        dtg_plot = -dtg
        residue_target = 800.0 if np.nanmin(x) <= 800 <= np.nanmax(x) else float(x[-1])
        residue_idx = int(np.nanargmin(np.abs(x - residue_target)))

        eligible = np.where(x > 150)[0]
        peak_idx = int(eligible[np.nanargmax(dtg_plot[eligible])]) if len(eligible) else int(np.nanargmax(dtg_plot))

        option = {
            "animation": False,
            "textStyle": {"fontFamily": "Noto Sans CJK KR"},
            "title": {
                "text": str(sheet_name),
                "left": "center",
                "textStyle": {"fontFamily": "Noto Sans CJK KR", "fontSize": 15},
            },
            "tooltip": {"trigger": "axis"},
            "legend": {
                "data": ["Weight(%)", "DTG"],
                "left": "center",
                "top": "middle",
                "orient": "vertical",
            },
            "grid": {"left": 70, "right": 70, "top": 65, "bottom": 65},
            "xAxis": {
                "type": "value",
                "name": str(x_col),
                "nameLocation": "middle",
                "nameGap": 38,
                "min": float(np.nanmin(x)),
                "max": float(np.nanmax(x)),
            },
            "yAxis": [
                {
                    "type": "value",
                    "name": str(weight_col),
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "axisLine": {"show": True, "lineStyle": {"color": "#1f4e9c"}},
                    "axisLabel": {"color": "#1f4e9c"},
                    "splitLine": {"show": True, "lineStyle": {"color": "#dddddd"}},
                },
                {
                    "type": "value",
                    "name": "- " + str(dtg_col),
                    "nameLocation": "middle",
                    "nameGap": 48,
                    "axisLine": {"show": True, "lineStyle": {"color": "#c0392b"}},
                    "axisLabel": {"color": "#c0392b"},
                    "splitLine": {"show": False},
                },
            ],
            "series": [
                {
                    "name": "Weight(%)",
                    "type": "line",
                    "showSymbol": False,
                    "yAxisIndex": 0,
                    "lineStyle": {"color": "#1f4e9c", "width": 2},
                    "data": [[float(a), float(b)] for a, b in zip(x, weight)],
                    "markPoint": {
                        "symbol": "circle",
                        "symbolSize": 9,
                        "itemStyle": {"color": "#1f4e9c"},
                        "label": {
                            "show": True,
                            "formatter": f"Residue {weight[residue_idx]:.1f}%",
                            "position": "right",
                        },
                        "data": [{"coord": [float(x[residue_idx]), float(weight[residue_idx])]}],
                    },
                },
                {
                    "name": "DTG",
                    "type": "line",
                    "showSymbol": False,
                    "yAxisIndex": 1,
                    "lineStyle": {"color": "#c0392b", "width": 2},
                    "data": [[float(a), float(b)] for a, b in zip(x, dtg_plot)],
                    "markPoint": {
                        "symbol": "circle",
                        "symbolSize": 9,
                        "itemStyle": {"color": "#c0392b"},
                        "label": {
                            "show": True,
                            "formatter": f"{x[peak_idx]:.0f} °C",
                            "position": "top",
                        },
                        "data": [{"coord": [float(x[peak_idx]), float(dtg_plot[peak_idx])]}],
                    },
                },
            ],
        }

        figures.append({
            "name": clean_name(sheet_name),
            "option": option,
            "width": 768,
            "height": 480,
            "dpi": 600,
            "formats": ["png"],
        })

    print(json.dumps({"figures": figures}, ensure_ascii=False, allow_nan=False))

if __name__ == "__main__":
    main()
```
