from pathlib import Path

import pandas as pd
import matplotlib.pyplot as plt


DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/rheology.csv")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B03_r1/out")

# Read the CSV and use the columns identified in the data preview.
df = pd.read_csv(DATA_FILE)
x = pd.to_numeric(df["Angular frequency (rad/s)"], errors="coerce")
g_storage = pd.to_numeric(df["Storage modulus G' (Pa)"], errors="coerce")
g_loss = pd.to_numeric(df["Loss modulus G'' (Pa)"], errors="coerce")

valid = x.notna() & g_storage.notna() & g_loss.notna()
valid &= (x > 0) & (g_storage > 0) & (g_loss > 0)
x, g_storage, g_loss = x[valid], g_storage[valid], g_loss[valid]

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax1 = plt.subplots(figsize=(8, 6))
ax2 = ax1.twinx()

line1, = ax1.plot(
    x, g_storage, color="#1f4e9c", linewidth=2,
    marker="o", markersize=4, label="저장 탄성률 G' (Pa)"
)
line2, = ax2.plot(
    x, g_loss, color="#c0392b", linewidth=2,
    marker="s", markersize=4, label="손실 탄성률 G'' (Pa)"
)

ax1.set_xscale("log")
ax1.set_yscale("log")
ax2.set_yscale("log")

ax1.set_xlabel("각주파수 (rad/s)")
ax1.set_ylabel("저장 탄성률 G' (Pa)", color="#1f4e9c")
ax2.set_ylabel("손실 탄성률 G'' (Pa)", color="#c0392b")
ax1.tick_params(axis="y", colors="#1f4e9c")
ax2.tick_params(axis="y", colors="#c0392b")
ax1.set_title("주파수에 따른 저장 및 손실 탄성률")
ax1.grid(True, which="both", linestyle="--", alpha=0.35)

ax1.legend(
    [line1, line2],
    [line1.get_label(), line2.get_label()],
    loc="center left",
    frameon=True,
)

fig.tight_layout()
OUT_DIR.mkdir(parents=True, exist_ok=True)
fig.savefig(OUT_DIR / "rheology.png", dpi=300, bbox_inches="tight")
plt.close(fig)
