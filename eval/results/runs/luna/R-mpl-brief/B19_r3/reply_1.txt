```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B19_r3/out"

FONT = "Noto Sans CJK KR"
X_COL = "Temperature (°C)"
WEIGHT_COL = "Weight (%)"
DTG_COL = "Deriv. Weight (%/°C)"
LOWER_TEMP = 200.0
UPPER_TEMP = 400.0


def interpolate_weight(temperature, weight, target):
    """Interpolate weight at a target temperature, sorting x if needed."""
    order = np.argsort(temperature)
    return float(np.interp(target, temperature[order], weight[order]))


def main():
    out_dir = Path(OUT_DIR)
    out_dir.mkdir(parents=True, exist_ok=True)

    frame = pd.read_excel(DATA_FILE, sheet_name="PVC")
    frame.columns = [str(column).strip() for column in frame.columns]

    temperature = pd.to_numeric(frame[X_COL], errors="coerce").to_numpy()
    weight = pd.to_numeric(frame[WEIGHT_COL], errors="coerce").to_numpy()
    dtg = pd.to_numeric(frame[DTG_COL], errors="coerce").to_numpy()

    valid = np.isfinite(temperature) & np.isfinite(weight) & np.isfinite(dtg)
    temperature, weight, dtg = temperature[valid], weight[valid], dtg[valid]
    if temperature.size == 0:
        raise ValueError("PVC 시트에서 유효한 온도, 질량 및 DTG 데이터를 찾지 못했습니다.")

    weight_200 = interpolate_weight(temperature, weight, LOWER_TEMP)
    weight_400 = interpolate_weight(temperature, weight, UPPER_TEMP)
    mass_loss = weight_200 - weight_400

    matplotlib.rcParams["font.family"] = FONT
    matplotlib.rcParams["axes.unicode_minus"] = False

    fig, ax1 = plt.subplots(figsize=(10, 6))
    ax2 = ax1.twinx()

    ax1.plot(temperature, weight, color="#1f4e9c", linewidth=2, label="Weight")
    ax2.plot(temperature, -dtg, color="#c0392b", linewidth=1.8, label="−DTG")

    # Mark interpolated weights at the requested temperature bounds.
    ax1.scatter(
        [LOWER_TEMP, UPPER_TEMP], [weight_200, weight_400],
        color="#1f4e9c", edgecolor="white", linewidth=0.8, zorder=5
    )
    ax1.axvline(LOWER_TEMP, color="0.45", linestyle="--", linewidth=1)
    ax1.axvline(UPPER_TEMP, color="0.45", linestyle="--", linewidth=1)

    summary = (
        f"{LOWER_TEMP:.0f} °C: {weight_200:.2f}%\n"
        f"{UPPER_TEMP:.0f} °C: {weight_400:.2f}%\n"
        f"질량 감소량: {mass_loss:.2f}%"
    )
    ax1.text(
        0.03, 0.05, summary,
        transform=ax1.transAxes, ha="left", va="bottom", fontsize=11,
        bbox=dict(boxstyle="round,pad=0.45", facecolor="white", edgecolor="0.7", alpha=0.95)
    )

    # Standard TGA annotations: residue nearest 800 °C and highest DTG peak above 150 °C.
    residue_index = int(np.argmin(np.abs(temperature - 800.0)))
    ax1.scatter(
        temperature[residue_index], weight[residue_index],
        color="#1f4e9c", marker="o", edgecolor="white", linewidth=0.8, zorder=6
    )
    ax1.annotate(
        f"Residue {weight[residue_index]:.1f}%",
        (temperature[residue_index], weight[residue_index]),
        xytext=(7, 8), textcoords="offset points", color="#1f4e9c", fontsize=9
    )

    peak_candidates = np.flatnonzero(temperature > 150.0)
    if peak_candidates.size:
        peak_index = int(peak_candidates[np.argmax(-dtg[peak_candidates])])
        ax2.scatter(
            temperature[peak_index], -dtg[peak_index],
            color="#c0392b", marker="o", edgecolor="white", linewidth=0.8, zorder=6
        )
        ax2.annotate(
            f"{temperature[peak_index]:.0f} °C",
            (temperature[peak_index], -dtg[peak_index]),
            xytext=(7, 8), textcoords="offset points", color="#c0392b", fontsize=9
        )

    ax1.set_title("PVC")
    ax1.set_xlabel(X_COL)
    ax1.set_ylabel(WEIGHT_COL, color="#1f4e9c")
    ax2.set_ylabel("−" + DTG_COL, color="#c0392b")
    ax1.tick_params(axis="y", colors="#1f4e9c")
    ax2.tick_params(axis="y", colors="#c0392b")
    ax1.grid(True, linestyle=":", alpha=0.45)

    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(lines1 + lines2, labels1 + labels2, loc="center left")

    fig.tight_layout()
    fig.savefig(out_dir / "PVC_mass_loss_200_400C.png", dpi=300, bbox_inches="tight")
    plt.close(fig)


if __name__ == "__main__":
    main()
```
