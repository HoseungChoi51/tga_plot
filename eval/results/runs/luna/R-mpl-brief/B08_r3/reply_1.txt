```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib.ticker import AutoMinorLocator

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B08_r3/out"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

df = pd.read_excel(DATA_FILE, sheet_name="Raw")
df.columns = [str(c).strip() for c in df.columns]

def find_column(columns, token):
    matches = [c for c in columns if token.lower() in c.lower()]
    if not matches:
        raise ValueError(f"Required column containing {token!r} was not found.")
    return matches[0]

x_col = find_column(df.columns, "Temperature")
weight_col = find_column(df.columns, "Weight")
dtg_col = find_column(df.columns, "Deriv")

data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
x = data[x_col].to_numpy()
weight = data[weight_col].to_numpy()
dtg_up = -data[dtg_col].to_numpy()

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(x, weight, color="#1f4e9c", lw=2, label="Weight")
dtg_line, = ax2.plot(x, dtg_up, color="#c0392b", lw=1.8, label="DTG")

# 요청한 무게 50% 위치의 수평 보조선
ax1.axhline(50, color="#666666", linestyle="--", linewidth=1.2, zorder=0)
ax1.text(
    0.015, 50, "50%",
    transform=ax1.get_yaxis_transform(),
    ha="left", va="bottom", color="#555555", fontsize=9,
    bbox=dict(facecolor="white", edgecolor="none", alpha=0.75, pad=1.5),
)

# 표준 TGA 잔류량 표시: 800 °C에 가장 가까운 측정점
residue_idx = int(np.nanargmin(np.abs(x - 800)))
residue_temp = x[residue_idx]
residue = weight[residue_idx]
ax1.plot(residue_temp, residue, marker="o", color="#1f4e9c", ms=6, zorder=4)
ax1.annotate(
    f"Residue {residue:.1f}%",
    (residue_temp, residue),
    xytext=(8, 10), textcoords="offset points",
    color="#1f4e9c", fontsize=9,
)

# 표준 TGA 피크 표시: 150 °C 초과 구간에서 가장 높은 -DTG 피크
peak_mask = x > 150
if np.any(peak_mask):
    peak_candidates = np.flatnonzero(peak_mask)
    peak_idx = int(peak_candidates[np.nanargmax(dtg_up[peak_mask])])
    peak_temp = x[peak_idx]
    peak_value = dtg_up[peak_idx]
    ax2.plot(peak_temp, peak_value, marker="o", color="#c0392b", ms=6, zorder=4)
    ax2.annotate(
        f"{peak_temp:.0f} °C",
        (peak_temp, peak_value),
        xytext=(8, 10), textcoords="offset points",
        color="#c0392b", fontsize=9,
    )

ax1.set_title("Raw")
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.xaxis.set_minor_locator(AutoMinorLocator())
ax1.yaxis.set_minor_locator(AutoMinorLocator())
ax2.yaxis.set_minor_locator(AutoMinorLocator())
ax1.grid(True, which="major", color="#d9d9d9", linewidth=0.7, alpha=0.7)

ax1.legend(
    [weight_line, dtg_line], ["Weight", "DTG"],
    loc="center left", frameon=True,
)

# 요청한 메모를 오른쪽 위에 배치
ax1.text(
    0.98, 0.97, "공기, 10 °C/min",
    transform=ax1.transAxes,
    ha="right", va="top", fontsize=11,
    bbox=dict(facecolor="white", edgecolor="#cccccc", alpha=0.9, pad=4),
)

fig.tight_layout()
fig.savefig(out_dir / "Raw.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
