```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/16_r2/out"
SHEET_NAME = "1006R1-V normal"

# Korean font configuration
plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
df.columns = [str(col).strip() for col in df.columns]

def find_column(candidates):
    for col in df.columns:
        if any(candidate in col.lower() for candidate in candidates):
            return col
    return None

temperature_col = find_column(["temperature", "temp", "온도"])
weight_col = find_column(["weight", "무게", "중량"])

# Fall back to the columns shown in the provided workbook preview.
if temperature_col is None:
    temperature_col = df.columns[1]
if weight_col is None:
    weight_col = df.columns[2]

data = df[[temperature_col, weight_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[temperature_col]
weight = data[weight_col]

fig, ax = plt.subplots(figsize=(9, 6))
ax.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight")

# Mark the measured weight nearest 800 °C, as in the standard TGA plot.
residue_idx = (temperature - 800).abs().idxmin()
residue_temp = temperature.loc[residue_idx]
residue_weight = weight.loc[residue_idx]
ax.scatter([residue_temp], [residue_weight], color="#1f4e9c", edgecolor="white",
           linewidth=0.8, zorder=3)
ax.annotate(
    f"Residue {residue_weight:.1f}%",
    xy=(residue_temp, residue_weight),
    xytext=(8, 10),
    textcoords="offset points",
    color="#1f4e9c",
    fontsize=10,
)

ax.set_title(SHEET_NAME)
ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%)")
ax.grid(True, alpha=0.25)
ax.legend(loc="center left")
fig.tight_layout()

fig.savefig(out_dir / "1006R1-V normal.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
