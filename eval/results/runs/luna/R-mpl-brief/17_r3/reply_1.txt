from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/17_r3/out")

FIGSIZE = (8, 5)
DPI = 600

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

OUT_DIR.mkdir(parents=True, exist_ok=True)


def find_column(columns, candidates):
    """Find a column by matching its normalized name against candidate terms."""
    normalized = {str(col).strip().lower().replace(" ", ""): col for col in columns}
    for candidate in candidates:
        key = candidate.strip().lower().replace(" ", "")
        for normalized_name, original_name in normalized.items():
            if key in normalized_name:
                return original_name
    return None


sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df = df.dropna(how="all")
    if df.empty:
        continue

    temp_col = find_column(df.columns, ["temperature", "temp"])
    weight_col = find_column(df.columns, ["weight(%)", "weight", "mass"])
    dtg_col = find_column(df.columns, ["d.w", "dtg", "derivative"])

    # TGA workbooks typically store temperature, weight, and DTG in columns B, C, and D.
    if temp_col is None or weight_col is None or dtg_col is None:
        if len(df.columns) < 4:
            continue
        temp_col, weight_col, dtg_col = df.columns[1], df.columns[2], df.columns[3]

    temperature = pd.to_numeric(df[temp_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    dtg = pd.to_numeric(df[dtg_col], errors="coerce")

    data = pd.DataFrame(
        {"temperature": temperature, "weight": weight, "dtg": dtg}
    ).dropna()
    if data.empty:
        continue

    x = data["temperature"].to_numpy()
    y1 = data["weight"].to_numpy()
    y2 = -data["dtg"].to_numpy()

    fig, ax1 = plt.subplots(figsize=FIGSIZE)
    ax2 = ax1.twinx()

    line1, = ax1.plot(x, y1, color="#1f4e9c", label="Weight (%)")
    line2, = ax2.plot(x, y2, color="#c0392b", label="DTG (%/°C)")

    residue_idx = int(np.argmin(np.abs(x - 800)))
    residue_temp = x[residue_idx]
    residue_weight = y1[residue_idx]
    ax1.scatter(
        residue_temp, residue_weight, color="#1f4e9c", edgecolor="white",
        linewidth=0.7, zorder=5
    )
    ax1.annotate(
        f"Residue {residue_weight:.1f}%",
        (residue_temp, residue_weight),
        xytext=(7, 9),
        textcoords="offset points",
        color="#1f4e9c",
        fontsize=9,
    )

    peak_mask = x > 150
    if peak_mask.any():
        peak_candidates = np.flatnonzero(peak_mask)
        peak_idx = peak_candidates[int(np.argmax(y2[peak_mask]))]
        peak_temp = x[peak_idx]
        peak_dtg = y2[peak_idx]
        ax2.scatter(
            peak_temp, peak_dtg, color="#c0392b", edgecolor="white",
            linewidth=0.7, zorder=5
        )
        ax2.annotate(
            f"{peak_temp:.0f} °C",
            (peak_temp, peak_dtg),
            xytext=(7, 9),
            textcoords="offset points",
            color="#c0392b",
            fontsize=9,
        )

    ax1.set_title(str(sheet_name))
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="#1f4e9c")
    ax2.set_ylabel("DTG (%/°C)", color="#c0392b")
    ax1.tick_params(axis="y", colors="#1f4e9c")
    ax2.tick_params(axis="y", colors="#c0392b")
    ax1.grid(True, alpha=0.25)

    ax1.legend(
        [line1, line2],
        [line1.get_label(), line2.get_label()],
        loc="center left",
    )

    fig.tight_layout()
    safe_sheet_name = "".join(
        char if char.isalnum() or char in "-_ ." else "_" for char in str(sheet_name)
    ).strip()
    fig.savefig(OUT_DIR / f"{safe_sheet_name or 'TGA'}.png", dpi=DPI, bbox_inches="tight")
    plt.close(fig)
