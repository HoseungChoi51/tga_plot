```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/22_r1/out"

Y2_COLOR = "green"
RESIDUE_TEMP = 800.0
PEAK_MIN_TEMP = 150.0

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

excel = pd.ExcelFile(DATA_FILE)

for sheet_name in excel.sheet_names:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet_name)
    df.columns = [str(col).strip() for col in df.columns]

    # Identify TGA columns by header names, with positional fallbacks.
    def find_column(keywords, fallback_index):
        for col in df.columns:
            if any(word in col.lower() for word in keywords):
                return col
        return df.columns[fallback_index]

    x_col = find_column(["temperature", "temp"], 1)
    weight_col = find_column(["weight", "mass"], 2)
    dtg_col = find_column(["d.w", "dtg", "derivative"], 3)

    data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
    if data.empty:
        continue

    x = data[x_col].to_numpy()
    weight = data[weight_col].to_numpy()
    dtg = data[dtg_col].to_numpy()
    dtg_up = -dtg

    fig, ax1 = plt.subplots(figsize=(10, 6))
    ax2 = ax1.twinx()

    weight_line, = ax1.plot(x, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)")
    dtg_line, = ax2.plot(x, dtg_up, color=Y2_COLOR, linewidth=1.8, label="DTG")

    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)")
    ax2.set_ylabel("DTG (%/°C)", color=Y2_COLOR)
    ax2.tick_params(axis="y", colors=Y2_COLOR)
    ax2.spines["right"].set_color(Y2_COLOR)
    ax1.set_title(sheet_name)

    # Residue at the measured point nearest 800 °C.
    residue_idx = int(np.nanargmin(np.abs(x - RESIDUE_TEMP)))
    ax1.plot(x[residue_idx], weight[residue_idx], "o", color="#1f4e9c", markersize=5)
    ax1.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        (x[residue_idx], weight[residue_idx]),
        xytext=(8, 8),
        textcoords="offset points",
        color="#1f4e9c",
    )

    # Highest upward DTG peak above 150 °C.
    eligible = np.flatnonzero(x > PEAK_MIN_TEMP)
    if eligible.size:
        peak_idx = int(eligible[np.nanargmax(dtg_up[eligible])])
        ax2.plot(x[peak_idx], dtg_up[peak_idx], "o", color=Y2_COLOR, markersize=5)
        ax2.annotate(
            f"{x[peak_idx]:.0f} °C",
            (x[peak_idx], dtg_up[peak_idx]),
            xytext=(8, 8),
            textcoords="offset points",
            color=Y2_COLOR,
        )

    ax1.legend([weight_line, dtg_line], [weight_line.get_label(), dtg_line.get_label()],
               loc="center left")
    ax1.grid(True, alpha=0.25)
    fig.tight_layout()

    safe_name = "".join(c if c.isalnum() or c in "-_ ." else "_" for c in sheet_name).strip()
    fig.savefig(out_dir / f"{safe_name or 'plot'}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
