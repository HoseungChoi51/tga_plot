```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/caox.csv")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B12_r1/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

# Read the CSV and identify temperature and weight columns from their headers.
df = pd.read_csv(DATA_FILE)
temperature_col = next(col for col in df.columns if "temperature" in str(col).lower())
weight_col = next(col for col in df.columns if "weight" in str(col).lower() and "deriv" not in str(col).lower())

temperature = pd.to_numeric(df[temperature_col], errors="coerce")
weight = pd.to_numeric(df[weight_col], errors="coerce")
valid = temperature.notna() & weight.notna()

temperature = temperature[valid]
weight = weight[valid]

# Label the measured weight at the point nearest 900 °C.
residue_idx = (temperature - 900).abs().idxmin()
residue_temp = temperature.loc[residue_idx]
residue_weight = weight.loc[residue_idx]

plt.rcParams["font.family"] = "Noto Sans CJK KR"
fig, ax = plt.subplots(figsize=(8, 5))

ax.plot(temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight")
ax.set_xlim(100, 900)
ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%)")
ax.set_title(DATA_FILE.stem)
ax.grid(True, alpha=0.3)

ax.scatter([residue_temp], [residue_weight], color="#1f4e9c", zorder=3)
ax.annotate(
    f"Residue {residue_weight:.1f}%",
    xy=(residue_temp, residue_weight),
    xytext=(-8, 12),
    textcoords="offset points",
    ha="right",
    color="#1f4e9c",
)

ax.legend(loc="center left")
fig.tight_layout()
fig.savefig(OUT_DIR / f"{DATA_FILE.stem}_weight.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
