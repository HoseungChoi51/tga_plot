from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt


DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/08_r2/out")

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def find_column(columns, keywords):
    """Find a column whose name contains one of the supplied keywords."""
    normalized = {col: str(col).strip().lower() for col in columns}
    for keyword in keywords:
        for col, name in normalized.items():
            if keyword in name:
                return col
    return None


def safe_filename(name):
    cleaned = "".join(ch if ch.isalnum() or ch in "-_ ." else "_" for ch in str(name))
    return cleaned.strip().replace(" ", "_") or "sheet"


def plot_sheet(sheet_name, data):
    columns = data.columns
    x_col = find_column(columns, ["temperature", "temp", "온도"])
    weight_col = find_column(columns, ["weight", "무게", "mass"])
    dtg_col = find_column(columns, ["d.w", "dtg", "derivative", "미분"])

    if x_col is None or weight_col is None or dtg_col is None:
        raise ValueError(
            f"시트 '{sheet_name}'에서 온도, 무게, DTG 열을 찾을 수 없습니다. "
            f"열 이름: {list(columns)}"
        )

    x = pd.to_numeric(data[x_col], errors="coerce")
    weight = pd.to_numeric(data[weight_col], errors="coerce")
    dtg = pd.to_numeric(data[dtg_col], errors="coerce")
    valid = x.notna() & weight.notna() & dtg.notna()
    x = x[valid].to_numpy()
    weight = weight[valid].to_numpy()
    dtg = dtg[valid].to_numpy()

    if x.size == 0:
        raise ValueError(f"시트 '{sheet_name}'에 유효한 데이터가 없습니다.")

    order = np.argsort(x)
    x, weight, dtg = x[order], weight[order], dtg[order]
    dtg_up = -dtg

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax2 = ax1.twinx()

    weight_line, = ax1.plot(
        x, weight, color="#1f4e9c", linewidth=1.8, label="Weight"
    )
    dtg_line, = ax2.plot(
        x, dtg_up, color="#c0392b", linewidth=1.5, label="DTG"
    )

    # Residue: measured weight at the point nearest 800 °C.
    residue_idx = int(np.argmin(np.abs(x - 800)))
    ax1.scatter(
        x[residue_idx], weight[residue_idx], color="#1f4e9c",
        edgecolor="white", linewidth=0.7, s=45, zorder=5
    )
    ax1.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        (x[residue_idx], weight[residue_idx]),
        xytext=(8, 10), textcoords="offset points",
        color="#1f4e9c", fontsize=9
    )

    # Peak: highest upward-pointing DTG peak above 150 °C.
    peak_candidates = np.flatnonzero(x > 150)
    if peak_candidates.size:
        peak_idx = int(peak_candidates[np.argmax(dtg_up[peak_candidates])])
        ax2.scatter(
            x[peak_idx], dtg_up[peak_idx], color="#c0392b",
            edgecolor="white", linewidth=0.7, s=45, zorder=5
        )
        ax2.annotate(
            f"{x[peak_idx]:.0f} °C",
            (x[peak_idx], dtg_up[peak_idx]),
            xytext=(8, -16), textcoords="offset points",
            color="#c0392b", fontsize=9
        )

    ax1.set_xlabel("온도 (°C)")
    ax1.set_ylabel("무게 (%)", color="#1f4e9c")
    ax2.set_ylabel("미분 무게 (%/°C)", color="#c0392b")
    ax1.tick_params(axis="y", colors="#1f4e9c")
    ax2.tick_params(axis="y", colors="#c0392b")
    ax1.set_title(str(sheet_name))
    ax1.grid(True, alpha=0.25)
    ax1.legend(
        [weight_line, dtg_line], ["Weight", "DTG"],
        loc="center left", frameon=True
    )

    fig.tight_layout()
    OUT_DIR.mkdir(parents=True, exist_ok=True)
    fig.savefig(OUT_DIR / f"{safe_filename(sheet_name)}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)


def main():
    OUT_DIR.mkdir(parents=True, exist_ok=True)
    sheets = pd.read_excel(DATA_FILE, sheet_name=None)
    for sheet_name, data in sheets.items():
        plot_sheet(sheet_name, data)


if __name__ == "__main__":
    main()
