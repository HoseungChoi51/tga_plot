from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from matplotlib import font_manager

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B14_r2/out")
SHEET_NAME = "PVC"

OUT_DIR.mkdir(parents=True, exist_ok=True)

# 한글 표시를 위한 폰트 설정
available_fonts = {font.name for font in font_manager.fontManager.ttflist}
if "Noto Sans CJK KR" in available_fonts:
    plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False
plt.rcParams["font.size"] = 16

df = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)

# 열 이름으로 온도, 무게, DTG 데이터 선택
def find_column(columns, keyword):
    matches = [col for col in columns if keyword.lower() in str(col).lower()]
    if not matches:
        raise ValueError(f"'{keyword}'에 해당하는 열을 찾을 수 없습니다: {list(columns)}")
    return matches[0]

temperature_col = find_column(df.columns, "temperature")
weight_col = find_column(df.columns, "weight (%)")
dtg_col = find_column(df.columns, "deriv")

data = df[[temperature_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
temperature = data[temperature_col].to_numpy()
weight = data[weight_col].to_numpy()
dtg_up = -data[dtg_col].to_numpy()

fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx()

weight_line, = ax1.plot(
    temperature, weight, color="#1f4e9c", linewidth=2.2, label="Weight (%)"
)
dtg_line, = ax2.plot(
    temperature, dtg_up, color="#c0392b", linewidth=2.0, label="DTG (%/°C)"
)

ax1.set_title(SHEET_NAME, fontsize=16)
ax1.set_xlabel("Temperature (°C)", fontsize=16)
ax1.set_ylabel("Weight (%)", fontsize=16)
ax2.set_ylabel("DTG (%/°C)", fontsize=16)

# 800 °C에 가장 가까운 측정점의 잔류량 표시
residue_idx = int(np.argmin(np.abs(temperature - 800)))
residue_temp = temperature[residue_idx]
residue_weight = weight[residue_idx]
ax1.scatter(
    residue_temp, residue_weight, color="#1f4e9c", edgecolor="white",
    linewidth=0.8, s=75, zorder=5
)
ax1.annotate(
    f"Residue {residue_weight:.1f}%",
    xy=(residue_temp, residue_weight),
    xytext=(-12, 14),
    textcoords="offset points",
    ha="right",
    color="#1f4e9c",
    fontsize=16,
)

# 150 °C 초과 구간에서 가장 높은 DTG 피크 표시
peak_candidates = np.flatnonzero(temperature > 150)
if peak_candidates.size:
    peak_idx = peak_candidates[np.argmax(dtg_up[peak_candidates])]
    peak_temp = temperature[peak_idx]
    peak_value = dtg_up[peak_idx]
    ax2.scatter(
        peak_temp, peak_value, color="#c0392b", edgecolor="white",
        linewidth=0.8, s=75, zorder=5
    )
    ax2.annotate(
        f"{peak_temp:.0f} °C",
        xy=(peak_temp, peak_value),
        xytext=(10, 10),
        textcoords="offset points",
        ha="left",
        color="#c0392b",
        fontsize=16,
    )

ax1.legend(
    [weight_line, dtg_line],
    [weight_line.get_label(), dtg_line.get_label()],
    loc="center left",
    fontsize=16,
    frameon=True,
)
ax1.tick_params(axis="both", labelsize=16)
ax2.tick_params(axis="y", labelsize=16)
ax1.grid(True, alpha=0.25)
fig.tight_layout()

output_file = OUT_DIR / "PVC_TGA.png"
fig.savefig(output_file, dpi=300, bbox_inches="tight")
plt.close(fig)
