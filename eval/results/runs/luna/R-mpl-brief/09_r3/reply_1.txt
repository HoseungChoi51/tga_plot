```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/09_r3/out"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df.columns = [str(col).strip() for col in df.columns]

    x_col = next((col for col in df.columns if "temperature" in col.lower()), df.columns[1])
    weight_col = next((col for col in df.columns if "weight" in col.lower()), df.columns[2])
    dtg_col = next((col for col in df.columns if "d.w" in col.lower() or "dtg" in col.lower()), df.columns[3])

    data = df[[x_col, weight_col, dtg_col]].apply(pd.to_numeric, errors="coerce").dropna()
    x = data[x_col].to_numpy()
    weight = data[weight_col].to_numpy()
    dtg = data[dtg_col].to_numpy()

    fig, ax1 = plt.subplots(figsize=(8, 5))
    ax2 = ax1.twinx()

    ax1.plot(x, weight, color="#1f4e9c", label="Weight (%)")
    ax2.plot(x, -dtg, color="#c0392b", label="DTG")

    residue_idx = int(np.argmin(np.abs(x - 800)))
    ax1.plot(x[residue_idx], weight[residue_idx], "o", color="#1f4e9c")
    ax1.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        xy=(x[residue_idx], weight[residue_idx]),
        xytext=(8, 10),
        textcoords="offset points",
        color="#1f4e9c",
    )

    peak_mask = x > 150
    if peak_mask.any():
        peak_indices = np.flatnonzero(peak_mask)
        peak_idx = peak_indices[np.argmax(-dtg[peak_mask])]
        ax2.plot(x[peak_idx], -dtg[peak_idx], "o", color="#c0392b")
        ax2.annotate(
            f"{x[peak_idx]:.0f} °C",
            xy=(x[peak_idx], -dtg[peak_idx]),
            xytext=(8, 10),
            textcoords="offset points",
            color="#c0392b",
        )

    ax1.set_title(sheet_name)
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)")
    ax2.set_ylabel("DTG (%/°C)")

    ax1.tick_params(axis="both", direction="out", labelsize=9)
    ax2.tick_params(axis="y", direction="out", labelsize=9)

    lines = ax1.get_lines()[:1] + ax2.get_lines()[:1]
    ax1.legend(lines, [line.get_label() for line in lines], loc="center left")

    fig.tight_layout()
    safe_name = "".join(ch if ch.isalnum() or ch in "-_ ." else "_" for ch in sheet_name).strip()
    fig.savefig(out_dir / f"{safe_name}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
