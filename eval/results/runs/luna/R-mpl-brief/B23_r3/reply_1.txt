```python
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib import font_manager

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B23_r3/out"

SHEETS = ["Raw", "Acid-washed", "Annealed"]
COLORS = {
    "Raw": "black",
    "Acid-washed": "blue",
    "Annealed": "red",
}

# Korean-capable font
available_fonts = {font.name for font in font_manager.fontManager.ttflist}
if "Noto Sans CJK KR" in available_fonts:
    plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)

sheets = pd.read_excel(DATA_FILE, sheet_name=SHEETS)
fig, ax1 = plt.subplots(figsize=(10, 6))
ax2 = ax1.twinx()

for sheet_name in SHEETS:
    df = sheets[sheet_name].copy()
    df.columns = [str(c).strip() for c in df.columns]

    temp_col = next(c for c in df.columns if "temperature" in c.lower())
    weight_col = next(c for c in df.columns if "weight" in c.lower() and "deriv" not in c.lower())
    dtg_col = next(c for c in df.columns if "deriv" in c.lower())

    temp = pd.to_numeric(df[temp_col], errors="coerce").to_numpy()
    weight = pd.to_numeric(df[weight_col], errors="coerce").to_numpy()
    dtg = pd.to_numeric(df[dtg_col], errors="coerce").to_numpy()
    valid = np.isfinite(temp) & np.isfinite(weight) & np.isfinite(dtg)
    temp, weight, dtg = temp[valid], weight[valid], dtg[valid]

    color = COLORS[sheet_name]
    ax1.plot(temp, weight, color=color, lw=1.8, label=f"{sheet_name} Weight")
    ax2.plot(temp, -dtg, color=color, lw=1.4, ls="--", label=f"{sheet_name} DTG")

ax1.set_title("TGA Comparison")
ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("DTG (-%/°C)")
ax1.set_xlim(30, 900)
ax1.grid(True, alpha=0.25)

handles1, labels1 = ax1.get_legend_handles_labels()
handles2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(handles1 + handles2, labels1 + labels2, loc="center left", fontsize=9)

fig.tight_layout()
fig.savefig(out_dir / "tga_overlaid.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
