```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

DATA_FILES = [
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_A.csv",
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_B.txt",
]
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B05_r2/out"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def read_data(path):
    """Read a CSV or delimited text file, trying common separators."""
    suffix = Path(path).suffix.lower()
    if suffix == ".csv":
        return pd.read_csv(path)

    for sep in (None, "\t", ",", ";", r"\s+"):
        try:
            frame = pd.read_csv(path, sep=sep, engine="python")
            if frame.shape[1] >= 2:
                return frame
        except Exception:
            continue
    raise ValueError(f"Could not read delimited data file: {path}")


def find_column(frame, candidates):
    normalized = {str(col).strip().lower(): col for col in frame.columns}
    for candidate in candidates:
        for name, original in normalized.items():
            if candidate in name:
                return original
    raise KeyError(f"Could not find a column matching any of: {candidates}")


def numeric_column(frame, column):
    return pd.to_numeric(frame[column], errors="coerce")


datasets = []
for path, label in zip(DATA_FILES, ("Lot A", "Lot B")):
    data = read_data(path)
    x_col = find_column(data, ("temperature", "temp"))
    weight_col = find_column(data, ("weight", "mass"))
    dtg_col = find_column(data, ("deriv", "dtg"))

    x = numeric_column(data, x_col)
    weight = numeric_column(data, weight_col)
    dtg = numeric_column(data, dtg_col)
    valid = x.notna() & weight.notna() & dtg.notna()
    datasets.append((label, x[valid].to_numpy(), weight[valid].to_numpy(), dtg[valid].to_numpy()))

if not datasets:
    raise ValueError("No data files were provided.")

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()
colors = ("#1f4e9c", "#4472c4")
styles = ("-", "--")

for (label, x, weight, dtg), color, style in zip(datasets, colors, styles):
    order = np.argsort(x)
    x, weight, dtg = x[order], weight[order], dtg[order]

    ax1.plot(x, weight, color=color, linestyle=style, linewidth=1.8, label=label)
    ax2.plot(x, -dtg, color=color, linestyle=style, linewidth=1.5, alpha=0.72,
             label=f"{label} DTG")

    # Standard residue and peak annotations for each run.
    residue_idx = int(np.nanargmin(np.abs(x - 800)))
    ax1.plot(x[residue_idx], weight[residue_idx], marker="o", color=color,
             markersize=5, linestyle="None")
    ax1.annotate(
        f"{label} residue {weight[residue_idx]:.1f}%",
        (x[residue_idx], weight[residue_idx]),
        xytext=(6, 7), textcoords="offset points", color=color, fontsize=9,
    )

    peak_mask = x > 150
    if np.any(peak_mask):
        peak_indices = np.flatnonzero(peak_mask)
        peak_idx = peak_indices[int(np.nanargmax((-dtg)[peak_mask]))]
        ax2.plot(x[peak_idx], -dtg[peak_idx], marker="^", color=color,
                 markersize=6, linestyle="None")
        ax2.annotate(
            f"{x[peak_idx]:.0f} °C",
            (x[peak_idx], -dtg[peak_idx]),
            xytext=(6, -13), textcoords="offset points", color=color, fontsize=9,
        )

ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)")
ax2.set_ylabel("DTG (-dWeight/dT, %/°C)")
ax1.set_title("Lot A and Lot B")
ax1.grid(True, alpha=0.25)

handles1, labels1 = ax1.get_legend_handles_labels()
handles2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(handles1 + handles2, labels1 + labels2, loc="center left", frameon=True)

fig.tight_layout()
out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)
fig.savefig(out_dir / "cnt_lot_A_vs_B.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
