from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILES = [
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_A.csv",
    "/home/chs/Work/tga_plot/eval/inputs_b/cnt_lot_B.txt",
]
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/B05_r1/out"

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def load_data(path):
    path = Path(path)
    if path.suffix.lower() == ".csv":
        return pd.read_csv(path)

    # Text exports may use tabs, spaces, or semicolons, and may have
    # non-data preamble lines. Try pandas' delimiter detection first.
    attempts = [
        dict(sep=None, engine="python", comment="#"),
        dict(sep=r"\s+", engine="python", comment="#"),
        dict(sep="\t", engine="python", comment="#"),
        dict(sep=";", engine="python", comment="#"),
        dict(sep=",", engine="python", comment="#"),
    ]
    best = None
    best_score = -1
    for kwargs in attempts:
        try:
            frame = pd.read_csv(path, **kwargs)
            score = frame.shape[1]
            if score > best_score:
                best, best_score = frame, score
        except Exception:
            continue
    if best is None:
        raise ValueError(f"Unable to parse text file: {path}")
    return best


def numeric_matrix(path):
    """Fallback: extract numeric columns from a text file without relying on headers."""
    raw = pd.read_csv(path, sep=None, engine="python", header=None, comment="#")
    numeric = raw.apply(pd.to_numeric, errors="coerce")
    numeric = numeric.dropna(axis=1, how="all").dropna(axis=0, how="all")
    return numeric


def columns_from_frame(frame, path):
    lower_names = {str(col).strip().lower(): col for col in frame.columns}

    def match(words):
        for name, col in lower_names.items():
            if any(word in name for word in words):
                return col
        return None

    x_col = match(["temperature", "temp"])
    weight_col = match(["weight"])
    dtg_col = match(["deriv", "dtg"])

    if x_col is not None and weight_col is not None and dtg_col is not None:
        data = pd.DataFrame(
            {
                "x": pd.to_numeric(frame[x_col], errors="coerce"),
                "weight": pd.to_numeric(frame[weight_col], errors="coerce"),
                "dtg": pd.to_numeric(frame[dtg_col], errors="coerce"),
            }
        ).dropna()
        if not data.empty:
            return data

    # Headerless / irregular text fallback: expected columns are time,
    # temperature, weight, derivative weight (A-D).
    numeric = numeric_matrix(path)
    if numeric.shape[1] >= 4:
        numeric = numeric.iloc[:, :4]
        data = pd.DataFrame(
            {
                "x": numeric.iloc[:, 1],
                "weight": numeric.iloc[:, 2],
                "dtg": numeric.iloc[:, 3],
            }
        ).dropna()
        if not data.empty:
            return data

    # Also allow files containing only temperature, weight, and derivative.
    if numeric.shape[1] >= 3:
        data = pd.DataFrame(
            {
                "x": numeric.iloc[:, 0],
                "weight": numeric.iloc[:, 1],
                "dtg": numeric.iloc[:, 2],
            }
        ).dropna()
        if not data.empty:
            return data

    raise ValueError(f"No usable numeric TGA rows found in {path}")


datasets = []
for path, label in zip(DATA_FILES, ["Lot A", "Lot B"]):
    frame = load_data(path)
    data = columns_from_frame(frame, path).sort_values("x")
    datasets.append(
        (
            label,
            data["x"].to_numpy(dtype=float),
            data["weight"].to_numpy(dtype=float),
            data["dtg"].to_numpy(dtype=float),
        )
    )

fig, ax1 = plt.subplots(figsize=(9, 6))
ax2 = ax1.twinx()
colors = ["#1f4e9c", "#c0392b"]

for (label, x, weight, dtg), color in zip(datasets, colors):
    ax1.plot(x, weight, color=color, linewidth=2, label=label)
    ax2.plot(x, -dtg, color=color, linewidth=1.7, linestyle="--", alpha=0.85)

    residue_idx = int(np.argmin(np.abs(x - 800)))
    ax1.plot(x[residue_idx], weight[residue_idx], "o", color=color, markersize=5)
    ax1.annotate(
        f"Residue {weight[residue_idx]:.1f}%",
        (x[residue_idx], weight[residue_idx]),
        xytext=(7, 8),
        textcoords="offset points",
        color=color,
        fontsize=9,
    )

    peak_candidates = np.flatnonzero(x > 150)
    if peak_candidates.size:
        peak_idx = int(peak_candidates[np.argmax(-dtg[peak_candidates])])
        ax2.plot(x[peak_idx], -dtg[peak_idx], "s", color=color, markersize=5)
        ax2.annotate(
            f"{x[peak_idx]:.0f} °C",
            (x[peak_idx], -dtg[peak_idx]),
            xytext=(7, 8),
            textcoords="offset points",
            color=color,
            fontsize=9,
        )

ax1.set_xlabel("Temperature (°C)")
ax1.set_ylabel("Weight (%)", color="#1f4e9c")
ax2.set_ylabel("−DTG (%/°C)", color="#c0392b")
ax1.tick_params(axis="y", labelcolor="#1f4e9c")
ax2.tick_params(axis="y", labelcolor="#c0392b")
ax1.set_title("Lot A vs Lot B")
ax1.grid(True, alpha=0.25)
ax1.legend(loc="center left")
fig.tight_layout()

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)
fig.savefig(out_dir / "lot_A_vs_lot_B.png", dpi=300, bbox_inches="tight")
plt.close(fig)
