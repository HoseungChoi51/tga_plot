from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-brief/20_r3/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False


def find_column(columns, keyword):
    for column in columns:
        if keyword in str(column).lower():
            return column
    return None


def safe_filename(name):
    return "".join(c if c.isalnum() or c in "-_ ." else "_" for c in str(name)).strip()


excel = pd.ExcelFile(DATA_FILE)

for sheet_name in excel.sheet_names:
    data = pd.read_excel(excel, sheet_name=sheet_name)

    x_col = find_column(data.columns, "temperature")
    weight_col = find_column(data.columns, "weight")
    dtg_col = find_column(data.columns, "d.w")

    # Fall back to the standard TGA column order if headers differ.
    if x_col is None:
        x_col = data.columns[1]
    if weight_col is None:
        weight_col = data.columns[2]
    if dtg_col is None:
        dtg_col = data.columns[3]

    plot_data = data[[x_col, weight_col, dtg_col]].apply(
        pd.to_numeric, errors="coerce"
    ).dropna()
    if plot_data.empty:
        continue

    temperature = plot_data[x_col].to_numpy()
    weight = plot_data[weight_col].to_numpy()
    dtg_up = -plot_data[dtg_col].to_numpy()

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax1.axvspan(300, 450, color="lightgray", alpha=0.55, zorder=0)

    line1, = ax1.plot(
        temperature, weight, color="#1f4e9c", linewidth=1.8, label="Weight (%)"
    )
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="#1f4e9c")
    ax1.tick_params(axis="y", labelcolor="#1f4e9c")
    ax1.set_title(str(sheet_name))
    ax1.grid(True, alpha=0.25)

    ax2 = ax1.twinx()
    line2, = ax2.plot(
        temperature, dtg_up, color="#c0392b", linewidth=1.5, label="DTG"
    )
    ax2.set_ylabel("-DTG (%/°C)", color="#c0392b")
    ax2.tick_params(axis="y", labelcolor="#c0392b")

    residue_index = int(np.argmin(np.abs(temperature - 800)))
    residue_temp = temperature[residue_index]
    residue_value = weight[residue_index]
    ax1.scatter(
        [residue_temp], [residue_value], color="#1f4e9c", edgecolor="white",
        linewidth=0.7, zorder=5
    )
    ax1.annotate(
        f"Residue {residue_value:.1f}%",
        (residue_temp, residue_value),
        xytext=(8, 10),
        textcoords="offset points",
        color="#1f4e9c",
        fontsize=9,
    )

    peak_candidates = np.flatnonzero(temperature > 150)
    if peak_candidates.size:
        peak_index = peak_candidates[np.argmax(dtg_up[peak_candidates])]
        peak_temp = temperature[peak_index]
        peak_value = dtg_up[peak_index]
        ax2.scatter(
            [peak_temp], [peak_value], color="#c0392b", edgecolor="white",
            linewidth=0.7, zorder=5
        )
        ax2.annotate(
            f"{peak_temp:.0f} °C",
            (peak_temp, peak_value),
            xytext=(8, 10),
            textcoords="offset points",
            color="#c0392b",
            fontsize=9,
        )

    ax1.legend(
        [line1, line2], [line1.get_label(), line2.get_label()],
        loc="center left"
    )
    fig.tight_layout()
    fig.savefig(OUT_DIR / f"{safe_filename(sheet_name)}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
