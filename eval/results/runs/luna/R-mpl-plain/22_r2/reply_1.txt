```python
from pathlib import Path

import pandas as pd
import matplotlib.pyplot as plt
from matplotlib import font_manager

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/22_r2/out")

# 한글 표시를 위한 글꼴 설정
available_fonts = {font.name for font in font_manager.fontManager.ttflist}
if "Noto Sans CJK KR" in available_fonts:
    plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

OUT_DIR.mkdir(parents=True, exist_ok=True)

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

for sheet_name, df in sheets.items():
    df.columns = [str(column).strip() for column in df.columns]

    temperature_col = next(
        (col for col in df.columns if "temperature" in col.lower()), None
    )
    weight_col = next(
        (col for col in df.columns if "weight" in col.lower()), None
    )
    derivative_col = next(
        (col for col in df.columns if "d.w" in col.lower() or "dw" in col.lower()),
        None,
    )

    if not all((temperature_col, weight_col, derivative_col)):
        raise ValueError(f"시트 '{sheet_name}'에서 필요한 열을 찾지 못했습니다.")

    x = pd.to_numeric(df[temperature_col], errors="coerce")
    weight = pd.to_numeric(df[weight_col], errors="coerce")
    derivative = pd.to_numeric(df[derivative_col], errors="coerce")
    valid = x.notna() & weight.notna() & derivative.notna()

    fig, ax1 = plt.subplots(figsize=(9, 6))
    ax1.plot(x[valid], weight[valid], color="tab:blue", linewidth=1.8)
    ax1.set_xlabel("Temperature (°C)")
    ax1.set_ylabel("Weight (%)", color="tab:blue")
    ax1.tick_params(axis="y", labelcolor="tab:blue")
    ax1.grid(True, linestyle="--", alpha=0.35)

    ax2 = ax1.twinx()
    ax2.plot(x[valid], derivative[valid], color="green", linewidth=1.8)
    ax2.set_ylabel("D.W. (%/°C)", color="green")
    ax2.tick_params(axis="y", labelcolor="green")

    ax1.set_title(sheet_name)
    fig.tight_layout()

    safe_name = "".join(
        character if character.isalnum() or character in " -_." else "_"
        for character in sheet_name
    ).strip()
    fig.savefig(OUT_DIR / f"{safe_name}.png", dpi=300, bbox_inches="tight")
    plt.close(fig)
```
