```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib.ticker import AutoMinorLocator
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B20_r3/out")
OUT_DIR.mkdir(parents=True, exist_ok=True)

# Read the Acid-washed worksheet and select columns by their headers.
df = pd.read_excel(DATA_FILE, sheet_name="Acid-washed")
df.columns = [str(column).strip() for column in df.columns]

temperature = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
derivative = pd.to_numeric(df["Deriv. Weight (%/°C)"], errors="coerce")
valid = temperature.notna() & weight.notna() & derivative.notna()

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax_weight = plt.subplots(figsize=(9, 6))
ax_derivative = ax_weight.twinx()

ax_weight.plot(
    temperature[valid], weight[valid],
    color="tab:blue", linewidth=1.8, label="Weight"
)
ax_derivative.plot(
    temperature[valid], derivative[valid],
    color="tab:red", linewidth=1.5, label="Deriv. Weight"
)

ax_weight.set_xlabel("Temperature (°C)")
ax_weight.set_ylabel("Weight (%)", color="tab:blue")
ax_derivative.set_ylabel("Deriv. Weight (%/°C)", color="tab:red")
ax_weight.tick_params(axis="y", labelcolor="tab:blue")
ax_derivative.tick_params(axis="y", labelcolor="tab:red")
ax_weight.set_xlim(30, 900)

# Add x-axis minor ticks and display them.
ax_weight.xaxis.set_minor_locator(AutoMinorLocator())
ax_weight.tick_params(axis="x", which="minor", length=3)
ax_weight.tick_params(axis="x", which="major", length=6)

ax_weight.grid(True, which="major", axis="both", alpha=0.25)
ax_weight.grid(True, which="minor", axis="x", alpha=0.12)

lines = ax_weight.get_lines() + ax_derivative.get_lines()
ax_weight.legend(lines, [line.get_label() for line in lines], loc="best")
ax_weight.set_title("Acid-washed TGA")
fig.tight_layout()

fig.savefig(OUT_DIR / "acid_washed_tga.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
