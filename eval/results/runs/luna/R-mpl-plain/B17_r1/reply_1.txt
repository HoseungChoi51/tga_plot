from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B17_r1/out"

SHEETS = ["PE", "PVC", "PET"]
OFFSET_STEP = 20  # percentage points


def main():
    out_dir = Path(OUT_DIR)
    out_dir.mkdir(parents=True, exist_ok=True)

    plt.rcParams["font.family"] = "Noto Sans CJK KR"
    plt.rcParams["axes.unicode_minus"] = False

    fig, ax = plt.subplots(figsize=(9, 6), constrained_layout=True)
    colors = plt.rcParams["axes.prop_cycle"].by_key()["color"]

    for i, sheet in enumerate(SHEETS):
        df = pd.read_excel(DATA_FILE, sheet_name=sheet)
        df.columns = [str(col).strip() for col in df.columns]

        temperature_col = next(
            col for col in df.columns
            if "temperature" in col.lower() or "온도" in col
        )
        weight_col = next(
            col for col in df.columns
            if "weight" in col.lower() or "무게" in col
        )

        data = df[[temperature_col, weight_col]].apply(pd.to_numeric, errors="coerce").dropna()
        offset = i * OFFSET_STEP
        ax.plot(
            data[temperature_col],
            data[weight_col] + offset,
            label=f"{sheet} (+{offset}%)" if offset else sheet,
            color=colors[i % len(colors)],
            linewidth=1.8,
        )

    ax.set_xlabel("Temperature (°C)")
    ax.set_ylabel("Weight (%) + vertical offset (percentage points)")
    ax.set_title("TGA weight curves with vertical offsets")
    ax.grid(True, alpha=0.3)
    ax.legend(frameon=False)
    fig.savefig(out_dir / "polymer_weight_curves_offset.png", dpi=300, bbox_inches="tight")
    plt.close(fig)


if __name__ == "__main__":
    main()
