from pathlib import Path

import pandas as pd
import matplotlib.pyplot as plt

DATA_FILE = "/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/15_r3/out"

def main():
    out_dir = Path(OUT_DIR)
    out_dir.mkdir(parents=True, exist_ok=True)

    plt.rcParams["font.family"] = "Noto Sans CJK KR"
    plt.rcParams["axes.unicode_minus"] = False

    xlsx = pd.ExcelFile(DATA_FILE)
    fig, ax = plt.subplots(figsize=(10, 6))

    for sheet in xlsx.sheet_names:
        df = pd.read_excel(xlsx, sheet_name=sheet)
        df.columns = [str(c).strip() for c in df.columns]

        temp_col = next((c for c in df.columns if "temperature" in c.lower()), None)
        weight_col = next((c for c in df.columns if "weight" in c.lower()), None)
        if temp_col is None or weight_col is None:
            continue

        data = df[[temp_col, weight_col]].apply(pd.to_numeric, errors="coerce").dropna()
        ax.plot(data[temp_col], data[weight_col], linewidth=1.8, label=sheet)

    ax.set_xlabel("Temperature (°C)")
    ax.set_ylabel("Weight (%)")
    ax.set_title("TGA — All Sheets")
    ax.grid(True, alpha=0.3)
    ax.legend(title="Sheet", frameon=True)
    fig.tight_layout()

    fig.savefig(out_dir / "tga_all_sheets.png", dpi=300, bbox_inches="tight")
    plt.close(fig)

if __name__ == "__main__":
    main()
