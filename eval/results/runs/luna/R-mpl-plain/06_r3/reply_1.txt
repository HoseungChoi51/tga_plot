```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/tests/inputs/TGA.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/06_r3/out")

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

OUT_DIR.mkdir(parents=True, exist_ok=True)

sheets = pd.read_excel(DATA_FILE, sheet_name=None)

fig, ax = plt.subplots(figsize=(9, 6))

for sheet_name, df in sheets.items():
    df.columns = [str(column).strip() for column in df.columns]
    temperature = pd.to_numeric(df["Temperature"], errors="coerce")
    weight = pd.to_numeric(df["Weight(%)"], errors="coerce")
    valid = temperature.notna() & weight.notna()
    ax.plot(temperature[valid], weight[valid], linewidth=1.8, label=sheet_name)

ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%)")
ax.set_title("TGA")
ax.grid(True, linestyle="--", linewidth=0.7, alpha=0.65)
# 범례는 표시하지 않습니다.
fig.tight_layout()
fig.savefig(OUT_DIR / "TGA.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
