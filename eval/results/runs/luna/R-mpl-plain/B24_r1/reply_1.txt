We need script only. Plot PE temperature K, weight likely TGA curve. Read xlsx, convert C+273.15. Label. Save. Korean font.```python
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/polymers.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B24_r1/out")

OUT_DIR.mkdir(parents=True, exist_ok=True)

# PE 시트의 온도(°C)를 켈빈으로 변환하고 무게(%)를 온도에 따라 표시
df = pd.read_excel(DATA_FILE, sheet_name="PE")
temperature_c = pd.to_numeric(df["Temperature (°C)"], errors="coerce")
weight = pd.to_numeric(df["Weight (%)"], errors="coerce")
temperature_k = temperature_c + 273.15

valid = temperature_k.notna() & weight.notna()

plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

fig, ax = plt.subplots(figsize=(8, 5.5))
ax.plot(temperature_k[valid], weight[valid], color="tab:blue", linewidth=1.8)

ax.set_xlabel("온도 (K)")
ax.set_ylabel("무게 (%)")
ax.set_title("PE 열중량 분석 (TGA)")
ax.grid(True, linestyle="--", alpha=0.4)
ax.margins(x=0.02, y=0.05)

fig.tight_layout()
fig.savefig(OUT_DIR / "PE_TGA_temperature_K.png", dpi=300, bbox_inches="tight")
plt.close(fig)
