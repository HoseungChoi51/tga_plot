```python
from pathlib import Path

import pandas as pd
import matplotlib.pyplot as plt
from matplotlib import font_manager

DATA_FILE = "/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx"
OUT_DIR = "/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B23_r1/out"

# 한글 표시 설정
available_fonts = {font.name for font in font_manager.fontManager.ttflist}
if "Noto Sans CJK KR" in available_fonts:
    plt.rcParams["font.family"] = "Noto Sans CJK KR"
plt.rcParams["axes.unicode_minus"] = False

sheets = [
    ("Raw", "black"),
    ("Acid-washed", "blue"),
    ("Annealed", "red"),
]

fig, ax = plt.subplots(figsize=(9, 6))
for sheet, color in sheets:
    df = pd.read_excel(DATA_FILE, sheet_name=sheet)
    df.columns = [str(c).strip() for c in df.columns]

    x_col = next(c for c in df.columns if "Temperature" in c)
    y_col = next(c for c in df.columns if "Weight" in c and "Deriv" not in c)

    data = df[[x_col, y_col]].apply(pd.to_numeric, errors="coerce").dropna()
    ax.plot(data[x_col], data[y_col], color=color, linewidth=1.8, label=sheet)

ax.set_xlabel("Temperature (°C)")
ax.set_ylabel("Weight (%)")
ax.set_title("TGA — Weight vs. Temperature")
ax.legend(frameon=False)
ax.grid(True, alpha=0.25)
fig.tight_layout()

out_dir = Path(OUT_DIR)
out_dir.mkdir(parents=True, exist_ok=True)
fig.savefig(out_dir / "overlaid_tga_weight.png", dpi=300, bbox_inches="tight")
plt.close(fig)
```
