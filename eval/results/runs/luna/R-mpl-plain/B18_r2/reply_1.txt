from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib import font_manager

DATA_FILE = Path("/home/chs/Work/tga_plot/eval/inputs_b/cnt_batches.xlsx")
OUT_DIR = Path("/home/chs/Work/tga_plot/eval/results/runs/luna/R-mpl-plain/B18_r2/out")
SHEET_NAME = "Annealed"


def first_derivative_peak_segment(temperature, weight):
    """Find the principal mass-loss step and return its peak and linear segment."""
    temperature = np.asarray(temperature, dtype=float)
    weight = np.asarray(weight, dtype=float)
    valid = np.isfinite(temperature) & np.isfinite(weight)
    temperature, weight = temperature[valid], weight[valid]

    order = np.argsort(temperature)
    temperature, weight = temperature[order], weight[order]
    derivative = np.gradient(weight, temperature)

    # A mass-loss peak is the most negative derivative.
    peak = int(np.argmin(derivative))

    # Fit a line to the steep central portion of this loss step. Use a
    # derivative-based region around the peak, with a minimum number of points.
    peak_slope = derivative[peak]
    threshold = 0.5 * peak_slope
    left = peak
    while left > 0 and derivative[left - 1] <= threshold:
        left -= 1
    right = peak
    while right < len(derivative) - 1 and derivative[right + 1] <= threshold:
        right += 1

    minimum_points = min(7, len(temperature))
    if right - left + 1 < minimum_points:
        left = max(0, peak - minimum_points // 2)
        right = min(len(temperature) - 1, left + minimum_points - 1)
        left = max(0, right - minimum_points + 1)

    # Least-squares line through the steep section.
    slope, intercept = np.polyfit(temperature[left:right + 1], weight[left:right + 1], 1)

    # Extrapolated onset: intersection of that tangent with the initial
    # stable-mass baseline, estimated from the first 10% of the temperature span.
    initial_end = max(2, int(np.ceil(0.10 * len(temperature))))
    baseline = float(np.nanmedian(weight[:initial_end]))
    onset = (baseline - intercept) / slope if slope != 0 else np.nan

    return temperature, weight, derivative, peak, left, right, baseline, onset, slope, intercept


def main():
    OUT_DIR.mkdir(parents=True, exist_ok=True)

    # Korean-compatible font when available.
    available_fonts = {font.name for font in font_manager.fontManager.ttflist}
    if "Noto Sans CJK KR" in available_fonts:
        plt.rcParams["font.family"] = "Noto Sans CJK KR"
    plt.rcParams["axes.unicode_minus"] = False

    data = pd.read_excel(DATA_FILE, sheet_name=SHEET_NAME)
    temperature_col = next(c for c in data.columns if "Temperature" in str(c))
    weight_col = next(c for c in data.columns if "Weight" in str(c) and "Deriv" not in str(c))

    result = first_derivative_peak_segment(data[temperature_col], data[weight_col])
    temperature, weight, derivative, peak, left, right, baseline, onset, slope, intercept = result

    fig, ax = plt.subplots(figsize=(9, 6))
    ax.plot(temperature, weight, color="#2468a2", lw=2, label="Weight")
    ax.axhline(baseline, color="0.45", ls="--", lw=1.2, label=f"Initial baseline ({baseline:.1f}%)")

    tangent_x = np.linspace(max(temperature.min(), onset - 25), min(temperature.max(), temperature[right] + 15), 150)
    ax.plot(tangent_x, slope * tangent_x + intercept, color="#e07a1f", ls="--", lw=1.8, label="Steep-region tangent")

    if np.isfinite(onset):
        ax.axvline(onset, color="#c83232", ls=":", lw=1.8)
        ax.scatter([onset], [baseline], color="#c83232", zorder=5)
        ax.annotate(
            f"Extrapolated onset\n{onset:.1f} °C",
            xy=(onset, baseline),
            xytext=(12, 18),
            textcoords="offset points",
            color="#a51f1f",
            arrowprops={"arrowstyle": "->", "color": "#a51f1f"},
        )

    ax.scatter(temperature[peak], weight[peak], color="#e07a1f", zorder=5, label="Maximum mass-loss rate")
    ax.set_title("Annealed TGA — Extrapolated Onset")
    ax.set_xlabel("Temperature (°C)")
    ax.set_ylabel("Weight (%)")
    ax.set_xlim(temperature.min(), temperature.max())
    ax.grid(True, alpha=0.25)
    ax.legend(frameon=False, loc="best")
    fig.tight_layout()

    output_file = OUT_DIR / "annealed_extrapolated_onset.png"
    fig.savefig(output_file, dpi=300, bbox_inches="tight")
    plt.close(fig)

    print(f"Extrapolated onset: {onset:.2f} °C")
    print(f"Figure saved to: {output_file}")


if __name__ == "__main__":
    main()
